/** TEMP-HARNESS (not shipped): v0.16.7 Quirk core screenshots (HUD, ERASED, costume layer). Enabled by run/qc.flag. */
public final class QuirkCoreHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static Vec3 origin;
	private static ServerPlayer a, b;

	private QuirkCoreHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("qc.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(QuirkCoreHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(420_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "qc-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (x, y) -> () -> {
			x.run();
			y.run();
		});
	}

	private static void shot(int t, String name) {
		at(t, () -> {
			Minecraft mc = Minecraft.getInstance();
			Screenshot.grab(mc.gameDirectory, "qc_" + name + ".png", mc.getMainRenderTarget(), m -> {
			});
			System.out.println("[HARNESS] shot " + name);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(QuirkCoreHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = false;
			mc.options.guiScale().set(2);
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.PEACEFUL, true);
				p.setGameMode(GameType.SURVIVAL);
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			server(mc, Quirks::tick);
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("QC_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false,
				Difficulty.PEACEFUL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	private static java.util.UUID skinUuid(boolean slim) {
		for (int i = 0; i < 1000; i++) {
			java.util.UUID id = java.util.UUID.randomUUID();
			boolean s = net.minecraft.client.resources.DefaultPlayerSkin.get(id).model() == net.minecraft.client.resources.PlayerSkin.Model.SLIM;
			if (s == slim) {
				return id;
			}
		}
		return java.util.UUID.randomUUID();
	}

	private static ServerPlayer fake(ServerPlayer near, String name, Vec3 at, java.util.UUID id) {
		ServerLevel level = near.serverLevel();
		GameProfile profile = new GameProfile(id, name);
		CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
		ServerPlayer f = new ServerPlayer(level.getServer(), level, profile, cookie.clientInformation());
		Connection conn = new Connection(PacketFlow.SERVERBOUND);
		new io.netty.channel.embedded.EmbeddedChannel(conn);
		level.getServer().getPlayerList().placeNewPlayer(conn, f, cookie);
		f.setGameMode(GameType.SURVIVAL);
		f.teleportTo(level, at.x, at.y, at.z, 0f, 0f);
		f.setYHeadRot(0f);
		f.yBodyRot = 0f;
		return f;
	}

	private static void camera(ServerPlayer p, Vec3 eye, Vec3 target) {
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		p.teleportTo(p.serverLevel(), eye.x, eye.y - 1.62, eye.z, yaw, pit);
	}

	/** A stand-in kit (the real Hellflame kit is another agent's) so the HUD has moves, a cooldown and a lit key. */
	private static void demoKit() {
		QuirkKit real = QuirkKits.of(Quirk.HELLFLAME);
		QuirkKits.register(Quirk.HELLFLAME, new QuirkKit() {
			@Override
			public void handle(ServerPlayer player, AbilitySlot slot, boolean pressed, boolean shift) {
			}

			@Override
			public String[] idsOf(AbilitySlot slot) {
				String[] r = real.idsOf(slot);
				return r.length > 0 ? r : new String[] { "demo_" + slot.ordinal(), "demo_shift_" + slot.ordinal() };
			}

			@Override
			public int maxCooldown(String abilityId) {
				return 400;
			}

			@Override
			public boolean active(net.minecraft.world.entity.player.Player player, AbilitySlot slot) {
				return slot == AbilitySlot.SLOT_3;
			}

			@Override
			public boolean usableWhileOverloaded(AbilitySlot slot, boolean shift) {
				return slot == AbilitySlot.SLOT_6 && shift;
			}
		});
	}

	private static void script(Minecraft mc) {
		int t = 10;
		at(t, () -> server(mc, p -> {
			demoKit();
			a = fake(p, "Wide", origin.add(-0.9, 0, 3), skinUuid(false));
			b = fake(p, "Slim", origin.add(0.9, 0, 3), skinUuid(true));
			Quirks.grant(a, Quirk.HELLFLAME);
			Quirks.grant(b, Quirk.HELLFLAME);
			a.teleportTo(a.serverLevel(), a.getX(), a.getY(), a.getZ(), 180f, 0f);
			b.teleportTo(b.serverLevel(), b.getX(), b.getY(), b.getZ(), 180f, 0f);
			a.setYHeadRot(180f);
			b.setYHeadRot(180f);
			a.yBodyRot = 180f;
			b.yBodyRot = 180f;
			Quirks.grant(p, Quirk.HELLFLAME);
			Quirks.setMeter(p, 63f);
			String[] g = QuirkKits.of(Quirk.HELLFLAME).idsOf(AbilitySlot.SLOT_2);
			Quirks.setCooldown(p, g[0], 260);
			String[] v = QuirkKits.of(Quirk.HELLFLAME).idsOf(AbilitySlot.SLOT_5);
			Quirks.setCooldown(p, v[1], 340);
			camera(p, origin.add(0, 1.62, 0.2), origin.add(0, 1.2, 3));
		}));
		shot(t + 20, "hud_meter_costumes");
		at(t + 22, () -> com.projecthero.mod.client.quirk.QuirkHud.forceShift = true);
		shot(t + 26, "hud_shift");
		at(t + 28, () -> {
			com.projecthero.mod.client.quirk.QuirkHud.forceShift = false;
			server(mc, p -> {
				Quirks.setMeter(p, 100f);
				Quirks.overload(p, 160);
			});
		});
		shot(t + 36, "hud_overloaded");
		at(t + 38, () -> com.projecthero.mod.client.gui.MoveListOverlay.forceHeld = true);
		shot(t + 42, "hud_movelist");
		at(t + 44, () -> {
			com.projecthero.mod.client.gui.MoveListOverlay.forceHeld = false;
			server(mc, p -> {
				Quirks.edit(p, n -> n.overloadUntil = 0L);
				Quirks.setMeter(p, 40f);
				PowerSuppression.suppress(p, 600);
			});
		});
		shot(t + 56, "hud_erased");
		at(t + 58, () -> mc.setScreen(new InventoryScreen(mc.player)));
		shot(t + 66, "erased_effect_inventory");
		at(t + 68, () -> {
			mc.setScreen(null);
			mc.options.hideGui = true;
			server(mc, p -> {
				PowerSuppression.release(p);
				camera(p, origin.add(0, 1.5, 0.9), origin.add(0, 1.0, 3));
			});
		});
		shot(t + 78, "costume_front_close");
		at(t + 80, () -> server(mc, p -> camera(p, origin.add(0, 1.8, 6.0), origin.add(0, 1.0, 3))));
		shot(t + 88, "costume_back");
		at(t + 90, () -> server(mc, p -> {
			mc.options.hideGui = false;
			camera(p, origin.add(-5, 1.62, -5), origin.add(-5, 1.3, -9));
		}));
		shot(t + 100, "costume_firstperson");
		at(t + 104, () -> {
			System.out.println("[HARNESS] DONE");
			Runtime.getRuntime().halt(0);
		});
	}
}
