/**
 * TEMP-HARNESS (not shipped, v0.17.5 beam clash rework): the local player (Kryptonian Heat Vision, aimed 15 degrees off) against a
 * fake player "Rival" (Saiyan Kamehameha, 1.5 s beam). Scene 1, side camera: the clash forms at the imperfect angle, the Kamehameha
 * stays out, both mash, the local player wins. Scene 2, first person: the HUD, then Escape forfeits (no pause menu). Enabled by
 * run/beamclash2.flag = "<out dir>"; frames under <out>/screenshots.
 */
public final class BeamClashHarness2 implements ClientModInitializer {
	public static volatile double[] camPose;
	private static File out;
	private static boolean createdWorld;
	private static int waitTicks, t = -1;
	private static ServerPlayer rival;
	private static Vec3 base;
	private static int scene, phaseT, lockT = -1, endT = -1, escT = -1, frame;
	private static String film;
	private static boolean sawLock;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("beamclash2.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File("beamclash2_out");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(BeamClashHarness2::tick);
		ClientTickEvents.START_CLIENT_TICK.register(BeamClashHarness2::aim);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(540_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "beamclash2-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				for (ServerPlayer p : srv.getPlayerList().getPlayers()) {
					if (p != rival) {
						try {
							fn.accept(p);
						} catch (Throwable e) {
							System.out.println("[HARNESS] error: " + e);
							e.printStackTrace();
						}
						return;
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	/** The local player's look: 15 degrees right of the rival, level with his eyes. */
	private static void aim(Minecraft mc) {
		if (mc.player == null || base == null || scene == 0) {
			return;
		}
		Vec3 eye = mc.player.getEyePosition();
		Vec3 to = base.add(0, 1.62, 10).subtract(eye);
		float yaw = (float) (Math.toDegrees(Math.atan2(to.z, to.x)) - 90.0) + 15f;
		float pitch = (float) -Math.toDegrees(Math.atan2(to.y, Math.sqrt(to.x * to.x + to.z * to.z)));
		mc.player.setYRot(yaw);
		mc.player.setXRot(pitch);
		mc.player.yRotO = yaw;
		mc.player.xRotO = pitch;
		mc.player.setYHeadRot(yaw);
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(BeamClashHarness2::createWorld);
			}
			return;
		}
		t++;
		if (t == 80) {
			mc.options.hideGui = false;
			mc.options.fov().set(70);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.EASY, true);
				p.setGameMode(GameType.SURVIVAL);
				base = new Vec3(Math.floor(p.getX()) + 0.5, Math.floor(p.getY()), Math.floor(p.getZ()) + 0.5);
				p.teleportTo(level, base.x, base.y, base.z, 0f, 0f);
				Kryptonian.grant(p);
				rival = fake(p, "Rival", base.add(0, 0, 10));
				Saiyan.grant(rival);
				rival.setGameMode(GameType.SURVIVAL);
				System.out.println("[HARNESS] set up at " + base);
			});
		}
		if (t == 120) {
			startScene(mc, 1);
		}
		if (scene > 0) {
			runScene(mc);
		}
	}

	private static void startScene(Minecraft mc, int s) {
		scene = s;
		phaseT = t;
		lockT = -1;
		endT = -1;
		escT = -1;
		sawLock = false;
		frame = 0;
		film = s == 1 ? "film_win" : "film_forfeit";
		Vec3 b = base;
		camPose = s == 1 ? camera(b.add(9.5, 2.6, 5.0), b.add(0, 1.4, 5.0)) : null;
		server(mc, p -> {
			p.setHealth(p.getMaxHealth());
			p.teleportTo(p.serverLevel(), b.x, b.y, b.z, p.getYRot(), p.getXRot());
			rival.getAttribute(Attributes.MAX_HEALTH).setBaseValue(200);
			rival.setHealth(200);
			rival.teleportTo(rival.serverLevel(), b.x, b.y, b.z + 10, 180f, 0f);
			rival.setYHeadRot(180f);
			rival.yBodyRot = 180f;
			SaiyanKit.resetForTest(rival);
			SaiyanKit.overrideForTest(rival, SaiyanForm.BASE, 100f);
			KryptonianAbilities.stopHeatVision(p);
		});
		System.out.println("[HARNESS] scene " + s);
	}

	private static void runScene(Minecraft mc) {
		int dt = t - phaseT;
		if (dt % 5 == 0) {
			mc.gui.getChat().clearMessages(false);
			mc.getToasts().clear();
			mc.gui.clear();
		}
		if (film != null && dt % 2 == 0) {
			new File(out, "screenshots/" + film).mkdirs();
			Screenshot.grab(out, film + "/f" + String.format("%03d", frame++) + ".png", mc.getMainRenderTarget(), m -> {
			});
		}
		if (dt == 6) {
			server(mc, KryptonianAbilities::startHeatVision);
		}
		if (dt == 14) {
			server(mc, p -> SaiyanKit.press(rival, SaiyanKit.BEAM, true));
		}
		if (dt == 24) {
			server(mc, p -> SaiyanKit.press(rival, SaiyanKit.BEAM, false));
		}
		boolean in = BeamClashClient.inClash();
		if (in && lockT < 0) {
			lockT = t;
			sawLock = true;
			System.out.println("[HARNESS] locked at dt=" + dt);
		}
		if (!in && sawLock && endT < 0) {
			endT = t;
			System.out.println("[HARNESS] clash over at dt=" + dt + " screen=" + mc.screen + " hp=" + mc.player.getHealth());
		}
		if (lockT >= 0) {
			int lt = t - lockT;
			if (scene == 1) {
				if (lt == 3) {
					shot(mc, "01_start_imperfect_angle");
				}
				if (lt == 30) {
					shot(mc, "02_clash_hud_side");
				}
				if (lt == 60) {
					shot(mc, "03_saiyan_short_beam_held");
				}
				if (endT < 0 && lt >= 20 && lt % 4 == 0) {
					ClientPlayNetworking.send(new BeamClashMashPayload());
				}
				if (endT < 0 && lt >= 20 && lt < 110 && lt % 5 == 0) {
					server(mc, p -> com.projecthero.mod.beamclash.BeamClash.mash(rival));
				}
			} else {
				if (lt == 25) {
					shot(mc, "04_clash_hud_first_person");
				}
				if (lt == 50 && escT < 0) {
					escT = t;
					System.out.println("[HARNESS] pressing Escape");
					mc.keyboardHandler.keyPress(mc.getWindow().getWindow(), GLFW.GLFW_KEY_ESCAPE, 0, GLFW.GLFW_PRESS, 0);
					mc.keyboardHandler.keyPress(mc.getWindow().getWindow(), GLFW.GLFW_KEY_ESCAPE, 0, GLFW.GLFW_RELEASE, 0);
				}
				if (escT >= 0 && t - escT == 4) {
					shot(mc, "05_after_escape_forfeit");
					System.out.println("[HARNESS] after escape: screen=" + mc.screen + " inClash=" + BeamClashClient.inClash());
				}
			}
		}
		if (endT >= 0 && t - endT == 12) {
			shot(mc, scene == 1 ? "03b_win_result" : "06_forfeit_result");
		}
		if (endT >= 0 && t - endT == 40) {
			film = null;
			server(mc, KryptonianAbilities::stopHeatVision);
			if (scene == 1) {
				phaseT = t;
				scene = -2; // wait out the pair cooldown
			} else {
				System.out.println("[HARNESS] done");
				Runtime.getRuntime().halt(0);
			}
		}
		if (dt > 700) {
			System.out.println("[HARNESS] scene " + scene + " timed out");
			Runtime.getRuntime().halt(4);
		}
	}

	static {
		ClientTickEvents.END_CLIENT_TICK.register(mc -> {
			if (scene == -2 && t - phaseT >= 80) {
				startScene(mc, 2);
			}
		});
	}

	private static double[] camera(Vec3 eye, Vec3 target) {
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		double y = Math.toDegrees(Math.atan2(dz, dx)) - 90.0;
		double pit = -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		return new double[] { eye.x, eye.y, eye.z, y, pit };
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("BC_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.EASY, true,
				new GameRules(), net.minecraft.world.level.WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}

	static ServerPlayer fake(ServerPlayer near, String name, Vec3 at) {
		ServerLevel level = near.serverLevel();
		GameProfile profile = new GameProfile(java.util.UUID.nameUUIDFromBytes(("harness-" + name).getBytes()), name);
		CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
		ServerPlayer f = new ServerPlayer(level.getServer(), level, profile, cookie.clientInformation());
		Connection conn = new Connection(PacketFlow.SERVERBOUND);
		new io.netty.channel.embedded.EmbeddedChannel(conn);
		level.getServer().getPlayerList().placeNewPlayer(conn, f, cookie);
		f.teleportTo(level, at.x, at.y, at.z, 180f, 0f);
		return f;
	}
}

package com.projecthero.mod.client.mixin;

import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.Shadow;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

import net.minecraft.client.Camera;
import net.minecraft.world.entity.Entity;
import net.minecraft.world.level.BlockGetter;

/** TEMP-HARNESS (not shipped): the BeamClashHarness2's fixed camera. */
@Mixin(Camera.class)
public abstract class BeamClashHarness2CameraMixin {
	@Shadow
	private boolean detached;

	@Shadow
	protected abstract void setRotation(float yaw, float pitch);

	@Shadow
	protected abstract void setPosition(double x, double y, double z);

	@Inject(method = "setup", at = @At("TAIL"))
	private void projecthero$harnessCam(BlockGetter level, Entity entity, boolean detached, boolean mirrored, float partial, CallbackInfo ci) {
		double[] c = com.projecthero.mod.client.BeamClashHarness2.camPose;
		if (c != null) {
			this.detached = true;
			setPosition(c[0], c[1], c[2]);
			setRotation((float) c[3], (float) c[4]);
		}
	}
}
