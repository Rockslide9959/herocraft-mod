// TEMP-HARNESS copy (v0.17.7 Black Panther comics suit + cape + high collar). To use:
//   node scratchpad/v0177/bp_suit_harness.js install
//   write run/bpsuit.flag = "bpsuit_out"
//   node scratchpad/low_graphics_options.js run/options.txt ; ./gradlew runClient
//   node scratchpad/v0177/bp_suit_film.js
//   node scratchpad/v0177/bp_suit_harness.js remove
// Real client: the power through BlackPanther.grant, the suit by wearing the four pieces (BlackPanther.syncSuit runs the build),
// walking / sprinting / jumping through the real movement keys. The camera follows the player at a fixed offset (camera mixin).

// ===== BlackPantherSuitHarness.java =====
package com.projecthero.mod.client;

import java.nio.file.Files;
import java.nio.file.Path;

import java.util.function.Consumer;

/** TEMP-HARNESS (not shipped): stills and films of the v0.17.7 Black Panther suit. Enabled by run/bpsuit.flag = "<out dir>". */
public final class BlackPantherSuitHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static BlockPos base;
	private static File out;
	private static final java.util.Map<Integer, List<Runnable>> AT = new java.util.HashMap<>();
	private static int film0 = -1, filmEnd = -1;
	private static String filmDir = "film";
	private static float pitch = 8f, headYaw = 0f;
	private static boolean walk, sprint, jump, sneak;

	/** Camera offset from the player's (interpolated) feet: dx, dy, dz, yaw, pitch; null = the normal camera. */
	public static volatile double[] camPose;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("bpsuit.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(BlackPantherSuitHarness::tick);
		ClientTickEvents.START_CLIENT_TICK.register(BlackPantherSuitHarness::keys);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(480_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "bpsuit-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		new File(out, "screenshots/" + name + ".png").getParentFile().mkdirs();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
	}

	private static void at(int t, Runnable r) {
		AT.computeIfAbsent(t, k -> new ArrayList<>()).add(r);
	}

	/** Camera {@code dist} from the player at {@code angle} (0 = in front of him, 180 = behind), {@code h} up, looking at {@code lookH}. */
	private static void cam(float angle, float dist, float h, float lookH) {
		double r = Math.toRadians(angle);
		double dx = Math.sin(r) * dist, dz = Math.cos(r) * dist;
		double ldx = -dx, ldy = lookH - h, ldz = -dz;
		float yaw = (float) Math.toDegrees(Math.atan2(-ldx, ldz));
		float pt = (float) -Math.toDegrees(Math.atan2(ldy, Math.sqrt(ldx * ldx + ldz * ldz)));
		camPose = new double[] { dx, h, dz, yaw, pt };
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = true;
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
	}

	private static void keys(Minecraft mc) {
		if (mc.player == null) {
			return;
		}
		mc.options.keyUp.setDown(walk);
		mc.options.keySprint.setDown(sprint);
		mc.options.keyJump.setDown(jump);
		mc.options.keyShift.setDown(sneak);
		if (sprint && walk) {
			mc.player.setSprinting(true);
		}
	}

	private static void face(Minecraft mc) {
		var p = mc.player;
		p.setYRot(headYaw);
		p.yRotO = headYaw;
		p.setXRot(pitch);
		p.xRotO = pitch;
		p.yHeadRot = headYaw;
		p.yHeadRotO = headYaw;
		p.yBodyRot = 0f;
		p.yBodyRotO = 0f;
	}

	private static void suit(ServerPlayer p, boolean on) {
		p.setItemSlot(EquipmentSlot.HEAD, on ? new ItemStack(BlackPantherItems.SUIT_HELMET) : ItemStack.EMPTY);
		p.setItemSlot(EquipmentSlot.CHEST, on ? new ItemStack(BlackPantherItems.SUIT_CHESTPLATE) : ItemStack.EMPTY);
		p.setItemSlot(EquipmentSlot.LEGS, on ? new ItemStack(BlackPantherItems.SUIT_LEGGINGS) : ItemStack.EMPTY);
		p.setItemSlot(EquipmentSlot.FEET, on ? new ItemStack(BlackPantherItems.SUIT_BOOTS) : ItemStack.EMPTY);
	}

	private static void stage(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		level.setDayTime(6000L);
		level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
		p.setGameMode(GameType.SURVIVAL);
		base = p.blockPosition();
		p.getInventory().clearContent();
		BlackPanther.grant(p);
		System.out.println("[HARNESS] power: " + BlackPanther.hasPower(p));
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, 8f);
	}

	private static void home(ServerPlayer p) {
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, pitch);
	}

	private static void film(int from, int to, String dir) {
		at(from, () -> {
			film0 = from;
			filmEnd = to;
			filmDir = dir;
		});
	}

	private static void plan() {
		Minecraft mc = Minecraft.getInstance();
		int t = 0;
		// ---- the suit-up from behind: the texel build, the collar rising, the cape unrolling
		at(t, () -> cam(150f, 4.0f, 1.8f, 1.1f));
		film(t + 2, t + 90, "film_suitup");
		at(t + 6, () -> server(mc, p -> suit(p, true)));
		t += 96;
		// ---- stills standing
		at(t, () -> server(mc, p -> System.out.println("[HARNESS] suited: " + BlackPanther.suited(p))));
		at(t + 2, () -> cam(0f, 3.2f, 1.3f, 1.0f));
		at(t + 8, () -> shot(mc, "bp_front"));
		at(t + 10, () -> cam(180f, 3.2f, 1.4f, 1.0f));
		at(t + 16, () -> shot(mc, "bp_back"));
		at(t + 18, () -> cam(90f, 3.2f, 1.3f, 1.0f));
		at(t + 24, () -> shot(mc, "bp_side"));
		at(t + 26, () -> cam(-140f, 3.0f, 1.7f, 1.1f));
		at(t + 32, () -> shot(mc, "bp_back_three_quarter"));
		at(t + 34, () -> cam(35f, 3.0f, 1.4f, 1.0f));
		at(t + 40, () -> shot(mc, "bp_front_three_quarter"));
		at(t + 42, () -> cam(150f, 1.6f, 2.0f, 1.55f));
		at(t + 48, () -> shot(mc, "bp_collar_closeup_back"));
		at(t + 50, () -> cam(50f, 1.5f, 1.8f, 1.55f));
		at(t + 56, () -> shot(mc, "bp_collar_closeup_front"));
		// head turned / looking up: the collar turning with it
		at(t + 58, () -> {
			cam(-150f, 2.2f, 1.9f, 1.4f);
			headYaw = 50f;
		});
		at(t + 66, () -> shot(mc, "bp_head_turned"));
		at(t + 68, () -> {
			headYaw = 0f;
			pitch = -60f;
		});
		at(t + 76, () -> shot(mc, "bp_looking_up"));
		at(t + 78, () -> {
			pitch = 8f;
			sneak = true;
			cam(120f, 3.0f, 1.4f, 0.9f);
		});
		at(t + 86, () -> shot(mc, "bp_crouching"));
		at(t + 88, () -> sneak = false);
		// the Kinetic glow at full bar
		at(t + 90, () -> server(mc, p -> BlackPanther.setKinetic(p, 100f)));
		at(t + 92, () -> cam(-160f, 3.0f, 1.4f, 1.0f));
		at(t + 98, () -> shot(mc, "bp_kinetic_glow_back"));
		at(t + 100, () -> cam(20f, 3.0f, 1.4f, 1.0f));
		at(t + 106, () -> shot(mc, "bp_kinetic_glow_front"));
		at(t + 108, () -> server(mc, p -> BlackPanther.setKinetic(p, 0f)));
		t += 112;
		// ---- walking (camera following, side-back)
		at(t, () -> server(mc, BlackPantherSuitHarness::home));
		at(t + 2, () -> cam(125f, 4.0f, 1.5f, 1.0f));
		at(t + 6, () -> walk = true);
		film(t + 10, t + 70, "film_walk");
		at(t + 40, () -> shot(mc, "bp_walking"));
		at(t + 70, () -> walk = false);
		t += 76;
		// ---- sprinting and jumping
		at(t, () -> server(mc, BlackPantherSuitHarness::home));
		at(t + 2, () -> cam(100f, 4.4f, 1.5f, 1.0f));
		at(t + 6, () -> {
			walk = true;
			sprint = true;
		});
		film(t + 10, t + 90, "film_sprint_jump");
		at(t + 30, () -> shot(mc, "bp_sprinting"));
		at(t + 40, () -> jump = true);
		at(t + 47, () -> shot(mc, "bp_sprint_jump"));
		at(t + 90, () -> {
			walk = false;
			sprint = false;
			jump = false;
		});
		t += 96;
		// ---- a drop from 14 blocks: the cape streams up
		at(t, () -> server(mc, p -> p.connection.teleport(base.getX() + 0.5, base.getY() + 14, base.getZ() + 0.5, 0f, pitch)));
		at(t + 1, () -> cam(130f, 4.5f, 1.2f, 1.0f));
		film(t + 2, t + 40, "film_drop");
		at(t + 16, () -> shot(mc, "bp_falling"));
		t += 44;
		// ---- the suit taken off: cape and collar go first, front three-quarters
		at(t, () -> server(mc, BlackPantherSuitHarness::home));
		at(t + 4, () -> cam(35f, 4.2f, 1.6f, 1.1f));
		film(t + 6, t + 60, "film_removal");
		at(t + 8, () -> server(mc, p -> suit(p, false)));
		at(t + 62, () -> shot(mc, "no_suit_after_removal"));
		t += 66;
		final int end = t;
		at(end, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(BlackPantherSuitHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 20) {
			GLFW.glfwSetWindowSize(mc.getWindow().getWindow(), 854, 480);
		}
		if (t == 40) {
			mc.options.fov().set(70);
			mc.options.fovEffectScale().set(0.0);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			mc.getToasts().clear();
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, BlackPantherSuitHarness::stage);
		}
		if (t < 140 || base == null) {
			return;
		}
		face(mc);
		int s = t - 140;
		if (s == 0) {
			plan();
		}
		if (s % 10 == 0) {
			mc.gui.getChat().clearMessages(false);
		}
		List<Runnable> todo = AT.remove(s);
		if (todo != null) {
			todo.forEach(Runnable::run);
		}
		if (s >= film0 && s < filmEnd) {
			shot(mc, String.format(filmDir + "/f%03d", s - film0));
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(GameRules.RULE_DOMOBSPAWNING).set(false, null);
		LevelSettings settings = new LevelSettings("BPSUIT_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.PEACEFUL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}

package com.projecthero.mod.client.mixin;

import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.Shadow;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

import net.minecraft.client.Camera;
import net.minecraft.world.entity.Entity;
import net.minecraft.world.level.BlockGetter;
import net.minecraft.world.phys.Vec3;

/** TEMP-HARNESS (not shipped): the BlackPantherSuitHarness's camera, a fixed offset from the player's interpolated position. */
@Mixin(Camera.class)
public abstract class BlackPantherSuitHarnessCameraMixin {
	@Shadow
	private boolean detached;

	@Shadow
	protected abstract void setRotation(float yaw, float pitch);

	@Shadow
	protected abstract void setPosition(double x, double y, double z);

	@Inject(method = "setup", at = @At("TAIL"))
	private void projecthero$harnessCam(BlockGetter level, Entity entity, boolean detached, boolean mirrored, float partial, CallbackInfo ci) {
		double[] c = com.projecthero.mod.client.BlackPantherSuitHarness.camPose;
		if (c != null && entity != null) {
			Vec3 p = entity.getPosition(partial);
			this.detached = true;
			setPosition(p.x + c[0], p.y + c[1], p.z + c[2]);
			setRotation((float) c[3], (float) c[4]);
		}
	}
}
