/** TEMP-HARNESS (not shipped). Kept as scratchpad/GuideHarness.v0177.java.txt. */
public final class GuideHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks, planTick;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static File out;
	private static File film;
	private static int filmFrom = -1, filmTo = -1, frame;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("guide.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		film = new File(out, "film_book");
		film.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(GuideHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(300_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "guide-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		mc.getToasts().clear();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			try {
				prev.run();
				r.run();
			} catch (Throwable e) {
				System.out.println("[HARNESS] step failed: " + e);
				e.printStackTrace();
			}
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(GuideHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.guiScale().set(2);
			mc.resizeDisplay();
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
		}
		if (t == 46) {
			buildPlan(mc);
		}
		if (t < 50) {
			return;
		}
		if (planTick < PLAN.size()) {
			int p = planTick;
			PLAN.get(planTick++).run();
			if (filmFrom >= 0 && p >= filmFrom && p <= filmTo && (p - filmFrom) % 2 == 0) {
				Screenshot.grab(film, String.format("f%03d.png", frame++), mc.getMainRenderTarget(), m -> {
				});
			}
		} else if (PLAN.size() > 0) {
			Runtime.getRuntime().halt(0);
		}
	}

	// ---- reflection into the screen

	private static Object call(Object target, String name, Object... args) {
		try {
			for (Method m : target.getClass().getDeclaredMethods()) {
				if (m.getName().equals(name) && m.getParameterCount() == args.length) {
					m.setAccessible(true);
					return m.invoke(target, args);
				}
			}
			System.out.println("[HARNESS] no method " + name);
		} catch (Exception e) {
			System.out.println("[HARNESS] " + name + " failed: " + e);
			e.printStackTrace();
		}
		return null;
	}

	private static Object get(Object target, String name) {
		try {
			Field f = target.getClass().getDeclaredField(name);
			f.setAccessible(true);
			return f.get(target);
		} catch (Exception e) {
			System.out.println("[HARNESS] field " + name + " failed: " + e);
			return null;
		}
	}

	private static int rec(Object record, String component) {
		return (int) call(record, component);
	}

	/** Puts the (virtual) mouse at GUI coordinates (x, y), so hover highlights draw. */
	private static void mouse(Minecraft mc, double x, double y) {
		try {
			double s = mc.getWindow().getGuiScale();
			Field fx = net.minecraft.client.MouseHandler.class.getDeclaredField("xpos");
			Field fy = net.minecraft.client.MouseHandler.class.getDeclaredField("ypos");
			fx.setAccessible(true);
			fy.setAccessible(true);
			fx.setDouble(mc.mouseHandler, x * s);
			fy.setDouble(mc.mouseHandler, y * s);
		} catch (Exception e) {
			System.out.println("[HARNESS] mouse failed: " + e);
		}
	}

	private static void hoverTile(Minecraft mc, HeroPackGuide.Category c) {
		if (!(mc.screen instanceof HeroPackGuideScreen s)) {
			return;
		}
		for (Object t : (List<?>) call(s, "homeLayout")) {
			if (call(t, "category") == c) {
				mouse(mc, rec(t, "x") + rec(t, "w") / 2.0, rec(t, "y") + rec(t, "h") / 2.0);
			}
		}
	}

	private static void clickChip(Minecraft mc, String filter) {
		if (!(mc.screen instanceof HeroPackGuideScreen s)) {
			return;
		}
		for (Object c : (List<?>) call(s, "chips")) {
			if (call(c, "filter").toString().equals(filter)) {
				double x = rec(c, "x") + rec(c, "w") / 2.0, y = rec(c, "y") + 7;
				mouse(mc, x, y);
				s.mouseClicked(x, y, 0);
			}
		}
	}

	/** Hovers the chapter row that links to {@code target} on the open spread. */
	private static void hoverLink(Minecraft mc, int target) {
		if (!(mc.screen instanceof HeroPackGuideScreen s)) {
			return;
		}
		@SuppressWarnings("unchecked")
		List<Integer> links = (List<Integer>) get(s, "rowLink");
		@SuppressWarnings("unchecked")
		List<int[]> pages = (List<int[]>) get(s, "pages");
		int spread = (int) get(s, "spread"), columns = (int) get(s, "columns"), colW = (int) get(s, "colW"), bx = (int) get(s, "bx");
		int top = (int) call(s, "pageTop");
		for (int c = 0; c < columns; c++) {
			int p = spread * columns + c;
			if (p >= pages.size()) {
				continue;
			}
			for (int r = pages.get(p)[0]; r < pages.get(p)[1]; r++) {
				if (links.get(r) == target) {
					mouse(mc, bx + 2 + c * (colW + 18) + 40, top + (r - pages.get(p)[0]) * 10 + 4);
					return;
				}
			}
		}
	}

	private static void type(Minecraft mc, String text) {
		for (char ch : text.toCharArray()) {
			mc.screen.charTyped(ch, 0);
		}
	}

	private static HeroPackGuideScreen book(Minecraft mc) {
		return mc.screen instanceof HeroPackGuideScreen s ? s : null;
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("GUIDE_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.PEACEFUL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17051L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
