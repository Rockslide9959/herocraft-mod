/** TEMP-HARNESS (not shipped): films of the v0.17.7 Mk I moveset. Enabled by run/immk1.flag = "<out dir>". */
public final class IronManMk1MovesHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static BlockPos base;
	private static File out;
	private static final java.util.Map<Integer, List<Runnable>> AT = new java.util.HashMap<>();
	private static float pitch = 5f, headYaw = 0f;

	/** Relative camera {dx, dy, dz, yaw, pitch} off the player, or an absolute one {x, y, z, yaw, pitch}. */
	public static volatile double[] camPose;
	public static volatile double[] camAbs;

	private static final String[] KEYS = { "e1mk1_iron_fist", "e1mk1_gauntlet_smash", "e1mk1_flamethrower", "e1mk1_arm_rocket",
			"e1mk1_rocket_jump", "e1mk1_rocket_leap", "e1mk1_scrap_shotgun", "e1mk1_fire_nova", "e1mk1_iron_shell", "e1mk1_scrap_repair",
			"e1mk1_smoke_screen", "e1mk1_reactor_overload" };

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("immk1.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File("screenshots");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(IronManMk1MovesHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(540_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "immk1-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
	}

	private static void at(int t, Runnable r) {
		AT.computeIfAbsent(t, k -> new ArrayList<>()).add(r);
	}

	/** A camera orbiting the player: {@code angle} degrees round from in front (+Z), {@code dist} out, {@code h} up, looking at {@code lookH}. */
	private static void cam(float angle, float dist, float h, float lookH) {
		double r = Math.toRadians(angle);
		double dx = Math.sin(r) * dist, dz = Math.cos(r) * dist;
		double ldx = -dx, ldy = lookH - h, ldz = -dz;
		float yaw = (float) Math.toDegrees(Math.atan2(-ldx, ldz));
		float pt = (float) -Math.toDegrees(Math.atan2(ldy, Math.sqrt(ldx * ldx + ldz * ldz)));
		camAbs = null;
		camPose = new double[] { dx, h, dz, yaw, pt };
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = true;
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
	}

	/** A fixed camera at base + (x, y, z) looking at base + (lx, ly, lz). */
	private static void camFixed(double x, double y, double z, double lx, double ly, double lz) {
		double ldx = lx - x, ldy = ly - y, ldz = lz - z;
		float yaw = (float) Math.toDegrees(Math.atan2(-ldx, ldz));
		float pt = (float) -Math.toDegrees(Math.atan2(ldy, Math.sqrt(ldx * ldx + ldz * ldz)));
		camPose = null;
		camAbs = new double[] { base.getX() + 0.5 + x, base.getY() + y, base.getZ() + 0.5 + z, yaw, pt };
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = true;
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
	}

	private static void firstPerson() {
		camPose = null;
		camAbs = null;
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = false;
		mc.options.setCameraType(CameraType.FIRST_PERSON);
	}

	private static void face(Minecraft mc) {
		var p = mc.player;
		p.setYRot(headYaw);
		p.yRotO = headYaw;
		p.setXRot(pitch);
		p.xRotO = pitch;
		p.yHeadRot = headYaw;
		p.yHeadRotO = headYaw;
		p.yBodyRot = headYaw;
		p.yBodyRotO = headYaw;
	}

	private static void wear(ServerPlayer p, String suit) {
		p.setItemSlot(EquipmentSlot.HEAD, new ItemStack(IronManItems.armor(suit, ArmorItem.Type.HELMET)));
		p.setItemSlot(EquipmentSlot.CHEST, new ItemStack(IronManItems.armor(suit, ArmorItem.Type.CHESTPLATE)));
		p.setItemSlot(EquipmentSlot.LEGS, new ItemStack(IronManItems.armor(suit, ArmorItem.Type.LEGGINGS)));
		p.setItemSlot(EquipmentSlot.FEET, new ItemStack(IronManItems.armor(suit, ArmorItem.Type.BOOTS)));
		TonyStark.setActiveSuit(p, suit);
	}

	private static void stage(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		level.setDayTime(6000L);
		level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
		p.server.setDifficulty(Difficulty.EASY, true);
		p.setGameMode(GameType.SURVIVAL);
		base = p.blockPosition();
		p.getInventory().clearContent();
		TonyStark.grant(p);
		TonyStark.unlockTech(p, 99);
		p.getInventory().clearContent(); // empty hands (the grant hands out a blueprint)
		wear(p, "mark_1");
		System.out.println("[HARNESS] tony: " + TonyStark.hasPower(p) + " suit " + com.projecthero.mod.ironman.IronManArmor.wornSuitId(p));
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, 5f);
	}

	/** Before each move: back to the start, mobs cleared, full reactor and integrity, every cooldown ready. */
	private static void reset(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		for (var e : level.getEntitiesOfClass(net.minecraft.world.entity.Entity.class, new AABB(base).inflate(24), e -> e != p)) {
			e.discard();
		}
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, headYaw, pitch);
		p.setDeltaMovement(net.minecraft.world.phys.Vec3.ZERO);
		p.clearFire();
		p.setHealth(p.getMaxHealth());
		IronManEnergy.setEnergy(p, "mark_1", IronManEnergy.capacity("mark_1"));
		IronManEnergy.setIntegrity(p, "mark_1", IronManEnergy.maxIntegrity("mark_1"));
		for (String k : KEYS) {
			TonyStark.triggerCooldown(p, "mark_1", k, 1);
		}
	}

	private static void husk(ServerPlayer p, double x, double z) {
		ServerLevel level = p.serverLevel();
		Husk h = EntityType.HUSK.create(level);
		h.setNoAi(true);
		h.getAttribute(Attributes.MAX_HEALTH).setBaseValue(400);
		h.setHealth(400);
		h.setPersistenceRequired();
		h.moveTo(base.getX() + 0.5 + x, base.getY(), base.getZ() + 0.5 + z, 180f, 0f);
		h.setYHeadRot(180f);
		h.setYBodyRot(180f);
		level.addFreshEntity(h);
	}

	/** One key edge through the real router (Shift set on the server for the press). */
	private static void key(Minecraft mc, String slot, boolean shift, boolean pressed) {
		server(mc, p -> {
			p.setShiftKeyDown(shift);
			IronManComicRouter.trigger(p, IronManSuits.byId("mark_1"), slot, pressed);
			p.setShiftKeyDown(false);
		});
	}

	/** Frames {@code from}..{@code to} (inclusive) named {@code tag}_NNN. */
	private static void film(Minecraft mc, String tag, int from, int to) {
		for (int s = from; s <= to; s++) {
			final int n = s - from;
			at(s, () -> shot(mc, String.format("mk1_%s_%03d", tag, n)));
		}
	}

	/** One move filmed; returns the next free tick. {@code fp} = first person. */
	private static int move(Minecraft mc, int t, String name, boolean fp) {
		String tag = (fp ? "fp_" : "") + name;
		switch (name) {
			case "gauntlet_smash" -> {
				at(t, () -> {
					pitch = 18f;
					server(mc, p -> {
						reset(p);
						husk(p, 0, 2.3);
						husk(p, 2.0, 3.6);
						husk(p, -1.8, 3.6);
					});
				});
				at(t + 6, () -> {
					if (fp) firstPerson(); else cam(-62f, 5.4f, 2.4f, 1.0f);
				});
				at(t + 20, () -> key(mc, "comic_r", true, true));
				at(t + 46, () -> key(mc, "comic_r", true, false));
				film(mc, tag, t + 14, t + 72);
				return t + 76;
			}
			case "arm_rocket" -> {
				at(t, () -> {
					pitch = 4f;
					server(mc, p -> {
						reset(p);
						husk(p, 0, 9);
						husk(p, 1.9, 9.3);
						husk(p, -1.7, 8.6);
					});
				});
				at(t + 6, () -> {
					if (fp) firstPerson(); else camFixed(7.5, 3.0, 4.0, 0, 1.2, 5.0);
				});
				at(t + 20, () -> key(mc, "comic_g", true, true));
				film(mc, tag, t + 14, t + 44);
				return t + 48;
			}
			case "scrap_shotgun" -> {
				at(t, () -> {
					pitch = 8f;
					server(mc, p -> {
						reset(p);
						husk(p, 0, 3.2);
						husk(p, 0.9, 4.2);
						husk(p, -0.9, 4.0);
					});
				});
				at(t + 6, () -> {
					if (fp) firstPerson(); else cam(60f, 4.4f, 2.0f, 1.1f);
				});
				at(t + 20, () -> key(mc, "comic_z", false, true));
				film(mc, tag, t + 14, t + 44);
				return t + 48;
			}
			case "fire_nova" -> {
				at(t, () -> {
					pitch = 10f;
					server(mc, p -> {
						reset(p);
						husk(p, 0, 3);
						husk(p, 3, 0.5);
						husk(p, -2.6, -1.5);
						husk(p, 1.2, -3.2);
						husk(p, -6.5, 1.5); // outside the 5 blocks
					});
				});
				at(t + 6, () -> {
					if (fp) firstPerson(); else cam(35f, 8.5f, 5.0f, 0.8f);
				});
				at(t + 20, () -> key(mc, "comic_z", true, true));
				film(mc, tag, t + 14, t + 50);
				return t + 54;
			}
			case "rocket_leap" -> {
				at(t, () -> {
					pitch = -22f;
					server(mc, MkHelp::reset);
				});
				at(t + 6, () -> camFixed(13.0, 4.0, 7.0, 0, 3.5, 7.0));
				at(t + 20, () -> key(mc, "comic_x", true, true));
				film(mc, tag, t + 16, t + 76);
				at(t + 78, () -> pitch = 5f);
				return t + 82;
			}
			case "iron_shell" -> {
				at(t, () -> {
					pitch = 5f;
					server(mc, MkHelp::reset);
				});
				at(t + 6, () -> {
					if (fp) firstPerson(); else cam(80f, 4.6f, 1.9f, 1.1f);
				});
				at(t + 20, () -> key(mc, "comic_v", false, true));
				for (int k = 0; k < 4; k++) {
					final int kk = k;
					at(t + 26 + k * 6, () -> server(mc, p -> {
						Arrow a = new Arrow(p.serverLevel(), p.getX() + (kk % 2 == 0 ? 0.3 : -0.3), p.getY() + 1.3, p.getZ() + 7.0,
								new ItemStack(Items.ARROW), null);
						a.setDeltaMovement(0, 0.02, -1.4);
						p.serverLevel().addFreshEntity(a);
					}));
				}
				film(mc, tag, t + 14, t + 66);
				return t + 110; // the shell runs 4 s: nothing after it overlaps
			}
			case "scrap_repair" -> {
				at(t, () -> {
					pitch = 20f;
					server(mc, p -> {
						reset(p);
						IronManEnergy.setIntegrity(p, "mark_1", IronManEnergy.maxIntegrity("mark_1") * 0.5f);
						p.getInventory().setItem(20, new ItemStack(Items.IRON_INGOT, 3)); // off the hotbar: the hand stays empty
					});
				});
				at(t + 6, () -> {
					if (fp) firstPerson(); else cam(30f, 3.0f, 1.8f, 1.2f);
				});
				at(t + 20, () -> key(mc, "comic_v", true, true));
				film(mc, tag, t + 14, t + 50);
				return t + 54;
			}
			case "overload" -> {
				at(t, () -> {
					pitch = 5f;
					server(mc, MkHelp::reset);
				});
				at(t + 6, () -> {
					if (fp) firstPerson(); else cam(25f, 3.4f, 1.8f, 1.2f);
				});
				at(t + 20, () -> key(mc, "comic_c", false, true));
				at(t + 48, () -> key(mc, "comic_c", false, false));
				film(mc, tag, t + 16, t + 56);
				return t + 60;
			}
			default -> {
				return t;
			}
		}
	}

	/** Static helpers usable as method references from the switch. */
	static final class MkHelp {
		static void reset(ServerPlayer p) {
			IronManMk1MovesHarness.reset(p);
		}
	}

	private static void plan() {
		Minecraft mc = Minecraft.getInstance();
		int t = 0;
		for (String m : new String[] { "gauntlet_smash", "arm_rocket", "scrap_shotgun", "fire_nova", "rocket_leap", "scrap_repair",
				"overload", "iron_shell" }) {
			t = move(mc, t, m, false);
		}
		for (String m : new String[] { "gauntlet_smash", "arm_rocket", "scrap_shotgun", "fire_nova", "scrap_repair", "overload", "iron_shell" }) {
			t = move(mc, t, m, true);
		}
		final int end = t + 4;
		at(end, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(IronManMk1MovesHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 20) {
			GLFW.glfwSetWindowSize(mc.getWindow().getWindow(), 854, 480);
		}
		if (t == 40) {
			mc.options.fov().set(70);
			mc.options.fovEffectScale().set(0.0);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			mc.getToasts().clear();
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, IronManMk1MovesHarness::stage);
		}
		if (t < 160 || base == null) {
			return;
		}
		face(mc);
		int s = t - 160;
		if (s == 0) {
			plan();
		}
		if (s % 10 == 0) {
			mc.gui.getChat().clearMessages(false);
		}
		List<Runnable> todo = AT.remove(s);
		if (todo != null) {
			todo.forEach(Runnable::run);
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(GameRules.RULE_DOMOBSPAWNING).set(false, null);
		LevelSettings settings = new LevelSettings("IMMK1_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.EASY, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}

package com.projecthero.mod.client.mixin;

import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.Shadow;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

import net.minecraft.client.Camera;
import net.minecraft.world.entity.Entity;
import net.minecraft.world.level.BlockGetter;
import net.minecraft.world.phys.Vec3;

/** TEMP-HARNESS (not shipped): the IronManMk1MovesHarness's camera -- an offset off the player, or a fixed point. */
@Mixin(Camera.class)
public abstract class IronManMk1MovesHarnessCameraMixin {
	@Shadow
	private boolean detached;

	@Shadow
	protected abstract void setRotation(float yaw, float pitch);

	@Shadow
	protected abstract void setPosition(double x, double y, double z);

	@Inject(method = "setup", at = @At("TAIL"))
	private void projecthero$harnessCam(BlockGetter level, Entity entity, boolean detached, boolean mirrored, float partial, CallbackInfo ci) {
		double[] a = com.projecthero.mod.client.IronManMk1MovesHarness.camAbs;
		if (a != null) {
			this.detached = true;
			setPosition(a[0], a[1], a[2]);
			setRotation((float) a[3], (float) a[4]);
			return;
		}
		double[] c = com.projecthero.mod.client.IronManMk1MovesHarness.camPose;
		if (c != null && entity != null) {
			Vec3 p = entity.getPosition(partial);
			this.detached = true;
			setPosition(p.x + c[0], p.y + c[1], p.z + c[2]);
			setRotation((float) c[3], (float) c[4]);
		}
	}
}
