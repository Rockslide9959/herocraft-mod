/** TEMP-HARNESS (not shipped): v0.17.1 Speed Force Speedster screenshots. Enabled by run/speedforce.flag. */
public final class SpeedForceHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static Vec3 origin;
	private static boolean holdForward;
	private static boolean holdSprint;
	/** Held on the server every tick while >= 0. */
	private static float lockMomentum = -1f;
	public static Vec3 camOffset;
	public static Vec3 camLook;
	private static String phase = "";

	private SpeedForceHarness() {
	}

	private static void orbit(int t, Vec3 offset, Vec3 look) {
		at(t, () -> {
			camOffset = offset;
			camLook = look;
		});
	}

	public static void init() {
		if (!Files.exists(Path.of("speedforce.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(SpeedForceHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(480_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "sf-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (x, y) -> () -> {
			x.run();
			y.run();
		});
	}

	private static void shot(int t, String name) {
		at(t, () -> {
			Minecraft mc = Minecraft.getInstance();
			Screenshot.grab(mc.gameDirectory, "speedforce_" + name + ".png", mc.getMainRenderTarget(), m -> {
			});
			System.out.println("[HARNESS] shot " + name + " pos=" + mc.player.position() + " m=" + SpeedForceSpeedster.momentum(mc.player));
		});
	}

	private static void husks(ServerPlayer p, Vec3 base, int n, double dist, double side) {
		ServerLevel level = p.serverLevel();
		for (int i = 0; i < n; i++) {
			Husk h = EntityType.HUSK.create(level);
			double a = (i - (n - 1) / 2.0) * 0.45;
			h.moveTo(base.x + Math.sin(a) * dist + side, base.y, base.z + Math.cos(a) * dist, 180f, 0f);
			h.setNoAi(true);
			h.getAttribute(Attributes.MAX_HEALTH).setBaseValue(200);
			h.setHealth(200);
			h.setPersistenceRequired();
			level.addFreshEntity(h);
		}
	}

	private static void reset(ServerPlayer p, float yaw) {
		p.serverLevel().getEntitiesOfClass(Husk.class, p.getBoundingBox().inflate(80)).forEach(net.minecraft.world.entity.Entity::discard);
		p.connection.teleport(origin.x, origin.y, origin.z, yaw, 0f);
		var n = SpeedForceSpeedster.state(p).copy();
		n.abilityReadyAt.clear();
		n.meter = 100f;
		p.setAttached(SpeedForceAttachments.STATE, n);
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(SpeedForceHarness::createWorld);
			}
			return;
		}
		mc.options.keyUp.setDown(holdForward);
		mc.options.keySprint.setDown(holdSprint);
		if (holdSprint) {
			mc.player.setSprinting(true);
		}
		if (inWorldTicks % 5 == 0) {
			server(mc, p -> p.serverLevel().getEntitiesOfClass(net.minecraft.world.entity.monster.Slime.class, p.getBoundingBox().inflate(120)).forEach(net.minecraft.world.entity.Entity::discard));
		}
		if (lockMomentum >= 0f) {
			float lm = lockMomentum;
			server(mc, p -> SpeedForceSpeedster.setMomentum(p, lm));
		}
		inWorldTicks++;
		if (t0 >= 0 && ((inWorldTicks - t0) % 10 == 0 || phase.equals("full_suit") && inWorldTicks - t0 < 192)) {
			double dx = mc.player.getX() - mc.player.xo;
			double dz = mc.player.getZ() - mc.player.zo;
			System.out.println("[HARNESS] t=" + (inWorldTicks - t0) + " " + phase + " bps=" + Math.round(Math.sqrt(dx * dx + dz * dz) * 20)
					+ " m=" + SpeedForceSpeedster.momentum(mc.player) + " ground=" + mc.player.onGround() + " y=" + Math.round(mc.player.getY() * 10) / 10.0
					+ " sf=" + SpeedForceSpeedster.hasPower(mc.player) + " hc=" + mc.player.horizontalCollision + " chunkAhead=" + mc.level.hasChunk(((int) Math.floor(mc.player.getX())) >> 4, ((int) Math.floor(mc.player.getZ() + 96)) >> 4) + " below=" + mc.level.getBlockState(mc.player.blockPosition().below()).getBlock() + " empty=" + (mc.level.getChunk(mc.player.blockPosition()) instanceof net.minecraft.world.level.chunk.EmptyLevelChunk) + " vel=" + mc.player.getDeltaMovement() + " here=" + mc.level.hasChunk(((int) Math.floor(mc.player.getX())) >> 4, ((int) Math.floor(mc.player.getZ())) >> 4));
		}
		if (inWorldTicks == 100) {
			mc.options.hideGui = false;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.bobView().set(false);
			mc.options.fovEffectScale().set(0.0);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.EASY, true);
				p.setGameMode(GameType.SURVIVAL);
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
				p.connection.teleport(origin.x, origin.y, origin.z, 0f, 5f);
				p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 99999, 4));
				// the plain Speedster, ready for the awakening: Speed, a Tachyon Device, Overdrive -- and a plate ahead
				SpeedForce.grant(p);
				p.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SPEED, 2400, 1));
				p.setItemInHand(InteractionHand.OFF_HAND, new ItemStack(SpeedForceMod.TACHYON_DEVICE));
				BatchA.set(p, SuperSpeedHandlers.KEY, SuperSpeedHandlers.OVERDRIVE_UNTIL, level.getGameTime() + 600, 1e12f);
				BlockPos plate = BlockPos.containing(origin.x, origin.y - 1, origin.z + 30);
				for (int dx = -2; dx <= 2; dx++) {
					level.setBlock(plate.offset(dx, 0, 0), ModDevices.CHARGED_COPPER_PLATES.defaultBlockState(), 3);
				}
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0 && inWorldTicks - t0 > 10) {
			mc.gui.getChat().clearMessages(false);
			mc.gui.clear();
			mc.getToasts().clear();
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	private static void suit(ServerPlayer p, boolean on) {
		p.setItemSlot(EquipmentSlot.HEAD, on ? new ItemStack(FlashSuit.HELMET) : ItemStack.EMPTY);
		p.setItemSlot(EquipmentSlot.CHEST, on ? new ItemStack(FlashSuit.CHESTPLATE) : ItemStack.EMPTY);
		p.setItemSlot(EquipmentSlot.LEGS, on ? new ItemStack(FlashSuit.LEGGINGS) : ItemStack.EMPTY);
		p.setItemSlot(EquipmentSlot.FEET, on ? new ItemStack(FlashSuit.BOOTS) : ItemStack.EMPTY);
	}

	private static void script(Minecraft mc) {
		// 1. the awakening: a plain Speedster in Overdrive runs over the plates 30 blocks ahead
		at(6, () -> {
			phase = "awaken";
			holdForward = true;
			holdSprint = true;
		});
		orbit(6, new Vec3(-4.0, 1.8, 3.0), new Vec3(0, 1.0, 0));
		for (int k = 10; k <= 30; k += 4) {
			shot(k, "awaken_" + k);
		}
		at(40, () -> {
			holdForward = false;
			holdSprint = false;
			System.out.println("[HARNESS] awakened=" + SpeedForceSpeedster.hasPower(mc.player));
		});
		// 2. the skate run without the suit at 35% momentum (~51 b/s): side, front, back
		at(60, () -> server(mc, p -> {
			reset(p, 0f);
			p.removeAllEffects();
			p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 99999, 4));
		}));
		at(62, () -> {
			phase = "skate35";
			lockMomentum = 0.35f;
			holdForward = true;
			holdSprint = true;
		});
		orbit(62, new Vec3(-3.4, 1.1, 0.0), new Vec3(0, 0.9, 0));
		for (int k = 80; k <= 90; k++) {
			shot(k, "side_" + k);
		}
		orbit(91, new Vec3(0.5, 1.2, 3.4), new Vec3(0, 0.9, 0));
		for (int k = 96; k <= 101; k++) {
			shot(k, "front_" + k);
		}
		orbit(102, new Vec3(2.4, 2.3, -3.4), new Vec3(0, 0.9, 0));
		for (int k = 107; k <= 112; k++) {
			shot(k, "back_" + k);
		}
		// 3. full momentum in the full Flash Suit: 250 b/s along the ground, HUD visible
		at(115, () -> {
			holdForward = false;
			holdSprint = false;
			lockMomentum = -1f;
		});
		at(120, () -> server(mc, p -> {
			reset(p, 0f);
			suit(p, true);
		}));
		at(123, () -> {
			phase = "full_suit";
			lockMomentum = 1.0f;
			holdForward = true;
			holdSprint = true;
		});
		orbit(123, new Vec3(-4.5, 1.6, -2.5), new Vec3(0, 0.9, 1.0));
		shot(150, "full_side");
		shot(152, "full_side_b");
		at(160, () -> camOffset = null);
		shot(166, "full_hud");
		at(175, () -> mc.options.setCameraType(CameraType.FIRST_PERSON));
		shot(180, "full_firstperson");
		at(190, () -> {
			holdForward = false;
			holdSprint = false;
			lockMomentum = -1f;
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
		});
		// 4. moves
		at(205, () -> server(mc, p -> {
			reset(p, 0f);
			suit(p, false);
			SpeedForceSpeedster.setMomentum(p, 0.6f);
			husks(p, origin, 3, 8.0, 0.0);
		}));
		orbit(206, new Vec3(-5.5, 2.4, 2.0), new Vec3(0, 1.0, 4.0));
		at(212, () -> server(mc, SpeedForceAbilities::lightningThrow));
		shot(221, "lightning_throw");
		shot(223, "lightning_throw_b");
		at(232, () -> server(mc, p -> {
			reset(p, 0f);
			SpeedForceSpeedster.setMomentum(p, 0.6f);
			husks(p, origin, 4, 4.0, 0.0);
		}));
		orbit(233, new Vec3(-7.0, 5.0, -4.0), new Vec3(0, 1.0, 1.0));
		at(236, () -> server(mc, SpeedForceAbilities::speedTornado));
		shot(250, "tornado");
		at(275, () -> server(mc, p -> {
			reset(p, 0f);
			SpeedForceSpeedster.setMomentum(p, 1.0f);
			husks(p, origin, 1, 2.5, 0.0);
		}));
		orbit(276, new Vec3(-4.0, 1.6, 1.2), new Vec3(0, 1.0, 1.2));
		at(280, () -> server(mc, SpeedForceAbilities::infiniteMassPunch));
		shot(283, "mass_punch");
		at(295, () -> server(mc, p -> {
			reset(p, 0f);
			SpeedForceSpeedster.setMomentum(p, 0.2f);
		}));
		orbit(296, new Vec3(-3.2, 1.6, 2.6), new Vec3(0, 1.0, 0));
		at(300, () -> server(mc, SpeedForceAbilities::surge));
		shot(304, "surge");
		shot(312, "surge_b");
		at(330, () -> server(mc, p -> {
			reset(p, 0f);
			SpeedForceSpeedster.setMomentum(p, 0f);
		}));
		at(334, () -> {
			camOffset = null;
			mc.options.setCameraType(CameraType.FIRST_PERSON);
		});
		shot(340, "hud_idle");
		at(342, () -> mc.setScreen(new com.projecthero.mod.client.gui.PowerInfoScreen()));
		shot(350, "imenu");
		at(352, () -> mc.setScreen(null));
		at(360, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("SF_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
				Difficulty.EASY, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
