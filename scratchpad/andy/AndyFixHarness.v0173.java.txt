/** TEMP-HARNESS (not shipped): films Andy's Suit and the tester fixes. Enabled by run/andyfix.flag. */
public final class AndyFixHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static String recording;
	private static int recFrame;
	private static int recUntil;
	private static Vec3 origin;
	private static Vec3 base;

	public static volatile Vec3 camPos;
	public static volatile Vec3 camLook;

	private AndyFixHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("andyfix.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(AndyFixHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(600_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "andyfix-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(AndyFixHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			if (recording != null) {
				Screenshot.grab(mc.gameDirectory, String.format("af_%s_%03d.png", recording, recFrame++), mc.getMainRenderTarget(), c -> {
				});
				if (inWorldTicks - t0 >= recUntil) {
					System.out.println("[HARNESS] recorded " + recording + " frames=" + recFrame);
					recording = null;
				}
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("AF_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false,
				Difficulty.PEACEFUL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	private static void record(int start, String name, int ticks) {
		at(start, () -> {
			recording = name;
			recFrame = 0;
			recUntil = start + ticks;
		});
	}

	private static void still(int t, String name) {
		at(t, () -> {
			Screenshot.grab(Minecraft.getInstance().gameDirectory, "af_" + name + ".png", Minecraft.getInstance().getMainRenderTarget(), c -> {
			});
			System.out.println("[HARNESS] still " + name);
		});
	}

	private static void suit(ServerPlayer p, boolean on) {
		p.setItemSlot(EquipmentSlot.HEAD, on ? new ItemStack(AndyItems.SUIT_HELMET) : ItemStack.EMPTY);
		p.setItemSlot(EquipmentSlot.CHEST, on ? new ItemStack(AndyItems.SUIT_CHESTPLATE) : ItemStack.EMPTY);
		p.setItemSlot(EquipmentSlot.LEGS, on ? new ItemStack(AndyItems.SUIT_LEGGINGS) : ItemStack.EMPTY);
		p.setItemSlot(EquipmentSlot.FEET, on ? new ItemStack(AndyItems.SUIT_BOOTS) : ItemStack.EMPTY);
	}

	private static void reset(ServerPlayer p, float yaw, float pitch) {
		p.serverLevel().getEntitiesOfClass(net.minecraft.world.entity.Entity.class, p.getBoundingBox().inflate(64),
				e -> e instanceof AndyPartEntity || e instanceof com.projecthero.mod.andy.entity.AndyCloneEntity).forEach(e -> e.discard());
		Andy.restore(p);
		AndyState n = Andy.state(p).copy();
		n.abilityReadyAt.clear();
		n.animId = 0;
		n.regenSpeed = AndyConfig.SPEED_NORMAL;
		n.setFlag(AndyState.FLAG_REGEN_OFF, false);
		p.setAttached(ModAttachments.ANDY_STATE, n);
		p.setItemInHand(InteractionHand.MAIN_HAND, ItemStack.EMPTY);
		p.setDeltaMovement(Vec3.ZERO);
		p.teleportTo(p.serverLevel(), base.x, base.y, base.z, yaw, pitch);
		p.setYHeadRot(yaw);
		p.yBodyRot = yaw;
		p.getFoodData().setFoodLevel(20);
		p.getFoodData().setSaturation(20f);
		p.setHealth(p.getMaxHealth());
	}

	private static void scene(Minecraft mc, int t, int index, CameraType type, Vec3 cam, Vec3 look, float yaw, float pitch, boolean gui,
			java.util.function.Consumer<ServerPlayer> setup) {
		at(t, () -> {
			base = origin.add(index * 30, 0, 0);
			mc.options.setCameraType(type);
			mc.gui.clear();
			mc.options.hideGui = !gui;
			camPos = cam == null ? null : base.add(cam);
			camLook = look == null ? null : base.add(look);
			server(mc, p -> {
				reset(p, yaw, pitch);
				setup.accept(p);
			});
		});
	}

	private static void srv(Minecraft mc, int t, java.util.function.Consumer<ServerPlayer> fn) {
		at(t, () -> server(mc, fn));
	}

	private static void script(Minecraft mc) {
		int t = 10;
		srv(mc, t, p -> {
			base = origin;
			p.setGameMode(GameType.SURVIVAL);
			Andy.grant(p);
		});
		t += 30;
		int idx = 1;
		Vec3 look = new Vec3(0, 1.0, 0);
		// --- the suit, worn: front, back, side, three quarter
		scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(0, 1.3, 3.2), look, 0f, 0f, false, p -> suit(p, true));
		still(t + 30, "suit_front");
		at(t + 32, () -> camPos = base.add(new Vec3(0, 1.3, -3.2)));
		still(t + 40, "suit_back");
		at(t + 42, () -> camPos = base.add(new Vec3(3.2, 1.3, 0)));
		still(t + 50, "suit_side");
		at(t + 52, () -> camPos = base.add(new Vec3(2.3, 1.6, 2.3)));
		still(t + 60, "suit_three_quarter");
		// beside it, no suit, for comparison
		at(t + 62, () -> server(mc, p -> suit(p, false)));
		still(t + 72, "nosuit_three_quarter");
		t += 80;
		// --- a limb regrowing with the suit on (Fast), side three quarter
		scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(2.0, 1.4, 2.0), new Vec3(0, 1.1, 0), 0f, 0f, false, p -> suit(p, true));
		srv(mc, t + 15, p -> {
			Andy.setSpeed(p, AndyConfig.SPEED_FAST);
			Andy.breakPart(p, AndyState.LEFT_ARM, AndyState.KIND_BURST);
			Andy.breakPart(p, AndyState.RIGHT_LEG, AndyState.KIND_BURST);
		});
		record(t + 15, "suit_regrow", 120);
		t += 145;
		// --- head only with the suit: the pieces go with the parts
		scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(1.8, 1.2, 1.8), new Vec3(0, 0.5, 0), 0f, 0f, false, p -> suit(p, true));
		srv(mc, t + 10, p -> {
			Andy.setSpeed(p, AndyConfig.SPEED_BURST);
			for (int i = AndyState.BODY; i < AndyState.PARTS; i++) {
				Andy.breakPart(p, i, AndyState.KIND_BURST);
			}
		});
		record(t + 10, "suit_headonly_regrow", 140);
		t += 160;
		// --- the clones (Dead Quardead): skin, blood and bones
		scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(0, 1.6, 5.0), new Vec3(0, 1.0, -0.5), 0f, 0f, false, p -> {
		});
		srv(mc, t + 10, AndyAbilities::quardead);
		still(t + 40, "clones_front");
		at(t + 42, () -> camPos = base.add(new Vec3(3.5, 1.6, 3.5)));
		still(t + 52, "clones_three_quarter");
		at(t + 54, () -> camPos = base.add(new Vec3(0, 1.6, -4.5)));
		still(t + 64, "clones_back");
		t += 75;
		// --- alternating finger shots and arm throws, front view
		scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(1.5, 1.5, 3.0), new Vec3(0, 1.3, 0), 0f, 0f, false, p -> {
		});
		for (int k = 0; k < 6; k++) {
			srv(mc, t + 10 + k * 6, AndyAbilities::partsBullet);
		}
		srv(mc, t + 50, p -> {
			AndyAbilities.startPunch(p);
		});
		srv(mc, t + 66, AndyAbilities::throwPunch);
		srv(mc, t + 72, p -> {
			Andy.state(p).abilityReadyAt.clear();
			AndyAbilities.startPunch(p);
		});
		srv(mc, t + 88, AndyAbilities::throwPunch);
		record(t + 8, "alternating", 95);
		t += 110;
		// --- the thrown head picked up (right-click), third person
		scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(3.5, 1.8, 2.5), new Vec3(0, 0.8, 2.0), 0f, 10f, false, p -> {
		});
		srv(mc, t + 10, AndyAbilities::headBullet);
		srv(mc, t + 60, p -> {
			for (AndyPartEntity e : p.serverLevel().getEntitiesOfClass(AndyPartEntity.class, new AABB(p.blockPosition()).inflate(32),
					e -> e.ownerIs(p) && e.bodyPart() == AndyState.HEAD)) {
				e.interact(p, InteractionHand.MAIN_HAND);
				System.out.println("[HARNESS] head picked up at " + e.position());
			}
		});
		record(t + 8, "head_pickup", 80);
		t += 95;
		// --- Off: the HUD says so and the arm stays gone (first person, GUI on), then Normal brings it back
		scene(mc, t, idx++, CameraType.FIRST_PERSON, null, null, 0f, 10f, true, p -> {
			p.getInventory().clearContent();
			p.getInventory().setItem(0, new ItemStack(AndyItems.SUIT_HELMET));
			p.getInventory().setItem(1, new ItemStack(AndyItems.SUIT_CHESTPLATE));
			p.getInventory().setItem(2, new ItemStack(AndyItems.SUIT_LEGGINGS));
			p.getInventory().setItem(3, new ItemStack(AndyItems.SUIT_BOOTS));
			p.getInventory().selected = 8;
			Andy.setRegenOff(p, true);
			Andy.breakPart(p, AndyState.LEFT_ARM, AndyState.KIND_BURST);
		});
		still(t + 40, "off_hud");
		srv(mc, t + 60, p -> {
			Andy.setSpeed(p, AndyConfig.SPEED_BURST);
		});
		still(t + 120, "back_on_hud");
		t += 130;
		at(t, () -> {
			System.out.println("[HARNESS] DONE");
			Minecraft.getInstance().stop();
		});
	}
}
