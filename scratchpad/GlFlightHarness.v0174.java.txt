/** TEMP-HARNESS (not shipped): v0.17.4 Green Lantern flight films. Flag run/glflight.flag (text = frames dir). */
public final class GlFlightHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static String recording;
	private static int recFrame;
	private static int recUntil;
	private static int recEvery = 2;
	private static File outDir;

	static ServerPlayer flier;
	static Vec3 origin;
	static int pathT = -1;
	static Vec3 vel = Vec3.ZERO;
	static float yaw, pitch;
	static String kind = "lantern";
	static String view;
	static boolean firstPerson;

	private GlFlightHarness() {
	}

	public static void init() {
		Path flag = Path.of("glflight.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			outDir = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			outDir = new File(".");
		}
		ClientTickEvents.END_CLIENT_TICK.register(GlFlightHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(900_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "glf-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(GlFlightHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.PEACEFUL, true);
				p.setGameMode(GameType.SPECTATOR);
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY() + 6, Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			if (recording != null) {
				if ((inWorldTicks - t0) % recEvery == 0) {
					String name = String.format("glf_%s_%03d.png", recording, recFrame++);
					Screenshot.grab(outDir, name, mc.getMainRenderTarget(), c -> {
					});
				}
				if (inWorldTicks - t0 >= recUntil) {
					System.out.println("[HARNESS] recorded " + recording + " frames=" + recFrame);
					recording = null;
				}
			}
			server(mc, GlFlightHarness::fly);
			if (!firstPerson) {
				chase(mc);
			}
		}
	}

	static final int PATH = 200;

	/** Hover, cruise, sprint (a turn, a climb, a dive), cruise, hover. */
	private static void fly(ServerPlayer cam) {
		if (flier == null || pathT < 0) {
			return;
		}
		int t = pathT++;
		double speed = 0.0;
		boolean sprint = false;
		if (t < 30) {
			speed = 0.0;
		} else if (t < 70) {
			speed = 0.6;
		} else if (t < 140) {
			speed = 1.5;
			sprint = true;
		} else if (t < 170) {
			speed = 0.6;
		}
		if (t >= 90 && t < 110) {
			yaw += 3.0f;
		}
		if (t >= 112 && t < 122) {
			pitch = Math.max(-25f, pitch - 4f);
		} else if (t >= 122 && t < 136) {
			pitch = Math.min(22f, pitch + 4f);
		} else if (t >= 136) {
			pitch *= 0.8f;
		}
		flier.setSprinting(sprint);
		Vec3 look = Vec3.directionFromRotation(pitch, yaw);
		Vec3 want = look.scale(speed);
		vel = vel.add(want.subtract(vel).scale(0.2));
		if (t >= PATH) {
			vel = Vec3.ZERO;
		}
		flier.move(MoverType.SELF, vel);
		flier.setDeltaMovement(vel);
		flier.setYRot(yaw);
		flier.setXRot(pitch);
		flier.setYHeadRot(yaw);
		flier.yBodyRot = yaw;
		flier.hurtMarked = true;
		keepFlying();
	}

	private static void chase(Minecraft mc) {
		net.minecraft.world.entity.player.Player r = null;
		for (net.minecraft.world.entity.player.Player pl : mc.level.players()) {
			if (pl != mc.player && "Flier".equals(pl.getGameProfile().getName())) {
				r = pl;
			}
		}
		if (r == null || pathT < 0) {
			return;
		}
		double yr = Math.toRadians(r.getYHeadRot());
		Vec3 fwd = new Vec3(-Math.sin(yr), 0, Math.cos(yr));
		Vec3 left = new Vec3(Math.cos(yr), 0, Math.sin(yr));
		Vec3 eye;
		if ("side".equals(view)) {
			// from his right (the ring side)
			eye = r.position().add(left.scale(-3.4)).add(fwd.scale(0.3)).add(0, 0.9, 0);
		} else if ("front".equals(view)) {
			eye = r.position().add(fwd.scale(3.4)).add(left.scale(-1.4)).add(0, 1.3, 0);
		} else {
			eye = r.position().add(fwd.scale(-2.8)).add(left.scale(-1.4)).add(0, 1.3, 0);
		}
		Vec3 target = r.position().add(0, 0.8, 0);
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float y = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		mc.player.setPos(eye.x, eye.y - mc.player.getEyeHeight(), eye.z);
		mc.player.xo = mc.player.getX();
		mc.player.yo = mc.player.getY();
		mc.player.zo = mc.player.getZ();
		mc.player.setYRot(y);
		mc.player.setXRot(pit);
		mc.player.yRotO = y;
		mc.player.xRotO = pit;
		mc.player.setDeltaMovement(Vec3.ZERO);
	}

	private static void keepFlying() {
		if ("lantern".equals(kind)) {
			if (!flier.getAttachedOrElse(ModAttachments.GREEN_LANTERN_FLYING, false)) {
				flier.setAttached(ModAttachments.GREEN_LANTERN_FLYING, true);
			}
		} else if (!flier.getAttachedOrElse(ModAttachments.MAX_STEEL_FLYING, false)) {
			flier.setAttached(ModAttachments.MAX_STEEL_FLYING, true);
		}
		flier.resetFallDistance();
	}

	static void script(Minecraft mc) {
		int t = 10;
		at(t, () -> server(mc, p -> {
			flier = fake(p, "Flier", origin);
			flier.setNoGravity(true);
			GreenLantern.bond(flier);
			GreenLanternSuitArmor.equip(flier);
			GreenLanternState s = GreenLantern.state(flier).copy();
			s.suited = true;
			GreenLantern.save(flier, s);
		}));
		t += 30;
		// another flight through the same shared fix (Max Steel, no Lantern pose): before / after, from the side
		at(t, () -> server(mc, p -> {
			flier.setAttached(ModAttachments.GREEN_LANTERN_FLYING, false);
			com.projecthero.mod.greenlantern.GreenLanternSuit.forceSuitDown(flier);
			GreenLanternState s = GreenLantern.state(flier).copy();
			s.hasPower = false;
			s.suited = false;
			GreenLantern.save(flier, s);
			for (var slot : new net.minecraft.world.entity.EquipmentSlot[] { net.minecraft.world.entity.EquipmentSlot.HEAD,
					net.minecraft.world.entity.EquipmentSlot.CHEST, net.minecraft.world.entity.EquipmentSlot.LEGS,
					net.minecraft.world.entity.EquipmentSlot.FEET }) {
				flier.setItemSlot(slot, net.minecraft.world.item.ItemStack.EMPTY);
			}
		}));
		t += 10;
		t = run(mc, t, "maxsteel", true, "warmup");
		t = run(mc, t, "maxsteel", false, "front");
		t = run(mc, t, "maxsteel", true, "front");
		// first person: the local player as a flying Lantern, the sprint arm forced in and out
		at(t, () -> server(mc, p -> {
			flier.setAttached(ModAttachments.MAX_STEEL_FLYING, false);
			flier.teleportTo(flier.serverLevel(), origin.x + 40, origin.y, origin.z, 0f, 0f);
			p.setGameMode(GameType.CREATIVE);
			p.getAbilities().flying = true;
			p.onUpdateAbilities();
			GreenLantern.bond(p);
			GreenLanternSuitArmor.equip(p);
			GreenLanternState s = GreenLantern.state(p).copy();
			s.suited = true;
			GreenLantern.save(p, s);
			p.setAttached(ModAttachments.GREEN_LANTERN_FLYING, true);
			p.connection.teleport(origin.x, origin.y, origin.z, 0f, 0f);
		}));
		at(t + 1, () -> {
			firstPerson = true;
			pathT = -1;
		});
		at(t + 30, () -> {
			mc.options.hideGui = false;
			mc.gui.getChat().clearMessages(false);
			mc.getToasts().clear();
			GreenLanternFlightPose.debugForce = new float[] { 1f, 0f, 0f };
		});
		at(t + 35, () -> shot(mc, "fp_hover"));
		for (int i = 0; i <= 30; i++) {
			final float w = i <= 15 ? i / 15f : (30 - i) / 15f;
			at(t + 40 + i, () -> GreenLanternFlightPose.debugForce = new float[] { 1f, w, w });
		}
		record(t + 40, "fp_sprint_in_out", 32, 1);
		at(t + 55, () -> shot(mc, "fp_sprint"));
		done(t + 90);
	}

	/** The fake flier is never ticked, so its equipment never syncs on its own. */
	static void sendEquipment(ServerPlayer p) {
		java.util.List<com.mojang.datafixers.util.Pair<net.minecraft.world.entity.EquipmentSlot, net.minecraft.world.item.ItemStack>> l = new java.util.ArrayList<>();
		for (var slot : net.minecraft.world.entity.EquipmentSlot.values()) {
			l.add(com.mojang.datafixers.util.Pair.of(slot, flier.getItemBySlot(slot).copy()));
		}
		p.connection.send(new net.minecraft.network.protocol.game.ClientboundSetEquipmentPacket(flier.getId(), l));
	}

	static void shot(Minecraft mc, String name) {
		Screenshot.grab(outDir, "glf_" + name + ".png", mc.getMainRenderTarget(), c -> {
		});
	}

	static int run(Minecraft mc, int t, String k, boolean after, String v) {
		at(t, () -> server(mc, p -> {
			FlightPoseHelper.WALK_FIX_ENABLED = after;
			GreenLanternFlightPose.ENABLED = after;
			kind = k;
			view = v;
			flier.setAttached(ModAttachments.GREEN_LANTERN_FLYING, false);
			flier.setAttached(ModAttachments.MAX_STEEL_FLYING, false);
			flier.teleportTo(flier.serverLevel(), origin.x, origin.y, origin.z, 0f, 0f);
			p.teleportTo(p.serverLevel(), origin.x - 3, origin.y + 1, origin.z - 3, 0f, 0f);
			yaw = 0f;
			pitch = 0f;
			vel = Vec3.ZERO;
			keepFlying();
		}));
		at(t + 5, () -> server(mc, p -> sendEquipment(p)));
		at(t + 8, () -> server(mc, p -> pathT = 0));
		record(t + 8, k + "_" + v + (after ? "_after" : "_before"), PATH, 2);
		return t + PATH + 20;
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("GLF_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
				Difficulty.PEACEFUL, true, new GameRules(), net.minecraft.world.level.WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	static ServerPlayer fake(ServerPlayer near, String name, Vec3 at) {
		ServerLevel level = near.serverLevel();
		GameProfile profile = new GameProfile(java.util.UUID.nameUUIDFromBytes(("harness-" + name).getBytes()), name);
		CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
		ServerPlayer f = new ServerPlayer(level.getServer(), level, profile, cookie.clientInformation());
		Connection conn = new Connection(PacketFlow.SERVERBOUND);
		new io.netty.channel.embedded.EmbeddedChannel(conn);
		level.getServer().getPlayerList().placeNewPlayer(conn, f, cookie);
		f.setGameMode(GameType.CREATIVE);
		f.teleportTo(level, at.x, at.y, at.z, 0f, 0f);
		return f;
	}

	static void record(int start, String name, int ticks, int every) {
		at(start, () -> {
			recording = name;
			recFrame = 0;
			recUntil = start + ticks;
			recEvery = every;
		});
	}

	static void done(int t) {
		at(t, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}
}
