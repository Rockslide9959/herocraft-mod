/**
 * TEMP-HARNESS (not shipped, v0.17.5 Symbiote as a Secondary power): a Sentry with the Symbiote bonded -- the Sentry's HUD, the suit-up
 * (the same TOGGLE_SYMBIOTE packet Shift+H sends for a Primary that uses H), the Symbiote column taking the corner, a Spike Shot (G
 * through the real router), the suit coming off and the Sentry HUD back; then a mutation user with the Symbiote holding H (the real
 * key mapping held down) until the power wheel opens. Third person from behind, HUD on. Enabled by run/symsec.flag = "<out dir>".
 * Kept as scratchpad/SymbioteSecondaryHarness.v0175.java.txt.
 */
public final class SymbioteSecondaryHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static File out;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static int planTick;
	private static BlockPos site;
	private static Husk dummy;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("symsec.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readAllLines(flag).get(0).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(SymbioteSecondaryHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(420_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "symsec-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		mc.getToasts().clear();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		if (!name.startsWith("film_")) {
			System.out.println("[HARNESS] shot " + name + " active=" + Symbiote.isActive(mc.player) + " stacked=" + Symbiote.stacked(mc.player)
					+ " owns=" + Symbiote.ownsKeysOverPrimary(mc.player) + " screen=" + mc.screen);
		}
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			prev.run();
			r.run();
		});
	}

	private static void toggle() {
		ClientPlayNetworking.send(new com.projecthero.mod.network.SpiderActionPayload(
				com.projecthero.mod.network.SpiderActionPayload.Action.TOGGLE_SYMBIOTE));
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(SymbioteSecondaryHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(70);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(com.projecthero.mod.loot.finds.SupplyDrops.RULE).set(false, p.server);
				BlockPos sp = p.blockPosition();
				int y = level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, sp.getX(), sp.getZ());
				site = new BlockPos(sp.getX(), y - 1, sp.getZ());
				for (int x = -10; x <= 10; x++) {
					for (int z = -10; z <= 10; z++) {
						for (int yy = -2; yy <= 12; yy++) {
							BlockState st = yy < 0 ? Blocks.DIRT.defaultBlockState() : yy == 0 ? Blocks.GRASS_BLOCK.defaultBlockState() : Blocks.AIR.defaultBlockState();
							level.setBlock(site.offset(x, yy, z), st, 2 | 16);
						}
					}
				}
				p.setGameMode(GameType.SURVIVAL);
				p.connection.teleport(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5, 180f, 8f);
				boolean sentry = com.projecthero.mod.sentry.Sentry.grant(p);
				Symbiote.grant(p);
				SymbioteSuitColours.setPreset(p, SymbioteColourPreset.CLASSIC);
				p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 20 * 60 * 30, 4, false, false));
				p.addEffect(new MobEffectInstance(MobEffects.SATURATION, 20 * 60 * 30, 4, false, false));
				Husk h = new Husk(EntityType.HUSK, level);
				h.moveTo(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5 - 9, 0f, 0f);
				h.setNoAi(true);
				h.setPersistenceRequired();
				h.getAttribute(Attributes.MAX_HEALTH).setBaseValue(1000.0);
				h.setHealth(1000f);
				level.addFreshEntity(h);
				dummy = h;
				System.out.println("[HARNESS] sentry=" + sentry + " stackedNow=" + Symbiote.stackedNow(p));
			});
		}
		if (t == 46) {
			buildPlan(mc);
		}
		if (t < 50 || site == null) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		}
	}

	private static void buildPlan(Minecraft mc) {
		int t = 10;
		at(t, () -> {
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.hideGui = false;
		});
		at(t + 20, () -> shot(mc, "01_sentry_hud_symbiote_dormant"));
		// FILM: suit on (the Shift+H packet), the HUD swap, a Spike Shot, suit off, the Sentry HUD back
		int f0 = t + 22;
		for (int f = 0; f < 230; f++) {
			final int n = f;
			at(f0 + f, () -> shot(mc, String.format("film_secondary_%03d", n)));
		}
		at(f0 + 10, SymbioteSecondaryHarness::toggle);
		at(f0 + 85, () -> shot(mc, "02_suited_symbiote_owns_hud"));
		at(f0 + 95, () -> server(mc, p -> {
			p.connection.teleport(p.getX(), p.getY(), p.getZ(), 180f, 4f);
			com.projecthero.mod.hero.AbilityRouter.handleInput(p, AbilitySlot.SLOT_2.number(), true);
		}));
		at(f0 + 96, () -> server(mc, p -> com.projecthero.mod.hero.AbilityRouter.handleInput(p, AbilitySlot.SLOT_2.number(), false)));
		at(f0 + 101, () -> shot(mc, "03_spike_shot_through_the_router"));
		at(f0 + 140, SymbioteSecondaryHarness::toggle);
		at(f0 + 229, () -> shot(mc, "04_suit_off_sentry_hud_back"));
		t = f0 + 236;
		// hold H with a mutation that uses H (and a second one so the wheel has something to switch to)
		at(t, () -> server(mc, p -> {
			com.projecthero.mod.sentry.Sentry.revoke(p);
			System.out.println("[HARNESS] mutations=" + Powers.mutations().size());
			// Super Strength (the wheel's second entry), then Crystalkinesis (H = Resonance Spire), selected last
			for (String key : new String[] { "power_01_super_strength", "power_06_crystalkinesis" }) {
				Power pw = Powers.byKey(key);
				boolean ok = pw != null && com.projecthero.mod.hero.PowerGrants.grantExperimental(p, pw);
				System.out.println("[HARNESS] mutation " + key + " ok=" + ok + " h=" + (pw != null && pw.hasSlot(AbilitySlot.SLOT_7))
						+ " enabled=" + (pw != null && pw.enabled()));
			}
			com.projecthero.mod.hero.data.ExperimentalState es = ExperimentalPowers.state(p);
			System.out.println("[HARNESS] owned=" + es.ownedPowers + " active=" + es.activePower + " symbiote=" + Symbiote.hasSymbiote(p));
			Symbiote.refreshPrimary(p);
		}));
		at(t + 30, () -> shot(mc, "05_mutation_hud_before_hold"));
		KeyMapping h = ModKeyBindings.POWER_SELECT;
		at(t + 32, () -> h.setDown(true));
		at(t + 37, () -> shot(mc, "06_holding_h_5_ticks_no_wheel_yet"));
		at(t + 46, () -> shot(mc, "07_hold_h_opens_the_power_wheel"));
		at(t + 48, () -> h.setDown(false));
		at(t + 52, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("SYMSEC_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17051L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
