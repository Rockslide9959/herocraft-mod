/** TEMP-HARNESS (not shipped): films of the v0.17.7 Thor x Super Soldier lend, kits and flight. Enabled by run/worthy.flag. */
public final class WorthyFilmHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static BlockPos base;
	private static File out;
	private static final java.util.Map<Integer, List<Runnable>> AT = new java.util.HashMap<>();
	private static int film0 = -1, filmEnd = -1;
	private static String filmDir = "film";
	private static float camAngle, camDist, camH, camLookAhead, camLookH = 1.1f;
	private static boolean camOn, camFollow;
	private static float faceYaw = 0f, facePitch = 8f;
	private static FakePlayer thor;
	private static com.projecthero.mod.squad.Squad squad;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("worthy.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim().split("\\|")[0]);
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(WorthyFilmHarness::tick);
		ClientTickEvents.START_CLIENT_TICK.register(mc -> placeCam(mc));
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(600_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "worthy-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		new File(out, "screenshots/" + name + ".png").getParentFile().mkdirs();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
	}

	private static void at(int t, Runnable r) {
		AT.computeIfAbsent(t, k -> new ArrayList<>()).add(r);
	}

	private static Item item(String id) {
		return BuiltInRegistries.ITEM.get(ResourceLocation.parse(id));
	}

	private static void cam3(float angle, float dist, float h, float lookAhead, float lookH, boolean follow) {
		camAngle = angle;
		camDist = dist;
		camH = h;
		camLookAhead = lookAhead;
		camLookH = lookH;
		camOn = true;
		camFollow = follow;
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = false;
		placeCam(mc);
	}

	private static void fp() {
		camOn = false;
		camPose = null;
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = false;
		mc.options.setCameraType(CameraType.FIRST_PERSON);
	}

	public static volatile double[] camPose;

	private static void placeCam(Minecraft mc) {
		if (!camOn || mc.player == null || mc.level == null || base == null) {
			camPose = null;
			return;
		}
		Vec3 c = camFollow ? mc.player.position() : new Vec3(base.getX() + 0.5, base.getY(), base.getZ() + 0.5);
		double r = Math.toRadians(camAngle);
		Vec3 eye = c.add(Math.sin(r) * camDist, camH, Math.cos(r) * camDist);
		Vec3 look = c.add(0, camLookH, camLookAhead);
		Vec3 d = look.subtract(eye);
		float yaw = (float) (Math.toDegrees(Math.atan2(-d.x, d.z)));
		float pitch = (float) (-Math.toDegrees(Math.atan2(d.y, d.horizontalDistance())));
		camPose = new double[] { eye.x, eye.y, eye.z, yaw, pitch };
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
	}

	private static void face(Minecraft mc) {
		var p = mc.player;
		p.setYRot(faceYaw);
		p.yRotO = faceYaw;
		p.setXRot(facePitch);
		p.xRotO = facePitch;
		p.yHeadRot = faceYaw;
		p.yHeadRotO = faceYaw;
	}

	private static void stage(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		level.setDayTime(6000L);
		level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
		p.server.setDifficulty(Difficulty.EASY, true);
		p.setGameMode(GameType.SURVIVAL);
		base = p.blockPosition();
		for (int x = -16; x <= 16; x++) {
			for (int z = -10; z <= 40; z++) {
				level.setBlock(base.offset(x, -1, z), (x + z) % 2 == 0 ? Blocks.SMOOTH_STONE.defaultBlockState() : Blocks.POLISHED_ANDESITE.defaultBlockState(), 3);
			}
		}
		p.getInventory().clearContent();
		System.out.println("[HARNESS] grant: " + com.projecthero.mod.hero.PowerGrants.grantHero(p, "super_soldier").message().getString());
		thor = FakePlayer.get(level, new GameProfile(UUID.fromString("7a1c0de0-0000-4000-8000-000000000001"), "Thor"));
		thor.moveTo(base.getX() + 3.5, base.getY(), base.getZ() + 1.5, 0f, 0f);
		com.projecthero.mod.worthiness.Worthiness.setScore(thor, com.projecthero.mod.worthiness.Worthiness.TEST_WORTHY_SCORE);
		var squads = com.projecthero.mod.squad.SquadManager.get(p.server);
		squad = squads.create("Avengers", thor.getUUID());
		squads.addMember(squad, p.getUUID());
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, 8f);
	}

	/** Back to the start: no loans, fresh state, hands as given, every old mob gone. */
	private static void reset(ServerPlayer p, Item main, Item off) {
		ServerLevel level = p.serverLevel();
		for (Mob m : level.getEntitiesOfClass(Mob.class, new AABB(base).inflate(64))) {
			m.discard();
		}
		var loans = com.projecthero.mod.worthy.WorthyLoans.get(p.server);
		for (var loan : loans.involving(p.getUUID())) {
			loans.remove(loan.hammerId());
		}
		if (com.projecthero.mod.worthy.WorthyFlight.isFlying(p)) {
			com.projecthero.mod.worthy.WorthyFlight.stop(p);
		}
		p.setAttached(com.projecthero.mod.attachment.ModAttachments.THOR_THROWN, com.projecthero.mod.power.ThorThrown.EMPTY);
		p.setAttached(com.projecthero.mod.worthy.WorthyAttachments.STATE, new com.projecthero.mod.worthy.WorthyState());
		p.removeAllEffects();
		p.setHealth(p.getMaxHealth());
		p.getInventory().clearContent();
		p.getInventory().selected = 0;
		p.getInventory().setItem(0, main == null ? ItemStack.EMPTY : new ItemStack(main));
		p.setItemInHand(InteractionHand.OFF_HAND, off == null ? ItemStack.EMPTY : new ItemStack(off));
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, 8f);
		com.projecthero.mod.worthy.WorthyLending.refreshView(p.server, p.getUUID());
	}

	/** Thor gets a fresh bound weapon and lends it to the Super Soldier through the real lend. */
	private static void lend(ServerPlayer p, boolean axe) {
		ItemStack w = new ItemStack(axe ? com.projecthero.mod.item.ModItems.STORMBREAKER : com.projecthero.mod.item.ModItems.MJOLNIR);
		w.set(com.projecthero.mod.item.ModDataComponents.HAMMER_ID, UUID.randomUUID());
		w.set(com.projecthero.mod.item.ModDataComponents.BOUND_OWNER, thor.getUUID());
		w.set(com.projecthero.mod.item.ModDataComponents.BOUND_OWNER_NAME, "Thor");
		thor.setItemInHand(InteractionHand.MAIN_HAND, w);
		System.out.println("[HARNESS] lend " + (axe ? "axe" : "hammer") + ": " + com.projecthero.mod.worthy.WorthyLending.lend(thor, p)
				+ " kit=" + com.projecthero.mod.worthy.WorthyKit.kit(p));
	}

	private static Husk husk(ServerPlayer p, double x, double z) {
		ServerLevel level = p.serverLevel();
		Husk h = new Husk(EntityType.HUSK, level);
		h.setNoAi(true);
		h.getAttribute(Attributes.MAX_HEALTH).setBaseValue(250);
		h.setHealth(250);
		h.setPersistenceRequired();
		h.moveTo(base.getX() + 0.5 + x, base.getY(), base.getZ() + 0.5 + z, 180f, 0f);
		h.setYHeadRot(180f);
		level.addFreshEntity(h);
		return h;
	}

	private static void group(ServerPlayer p) {
		husk(p, 0.0, 5.0);
		husk(p, 2.2, 7.0);
		husk(p, -2.4, 7.5);
		husk(p, 0.8, 10.0);
		husk(p, -1.0, 12.5);
	}

	private static void key(ServerPlayer p, int slot, boolean shift) {
		p.setShiftKeyDown(shift);
		com.projecthero.mod.hero.AbilityRouter.handleInput(p, slot, true);
		p.setShiftKeyDown(false);
	}

	private static void film(int from, int to, String dir) {
		at(from, () -> {
			film0 = from;
			filmEnd = to;
			filmDir = dir;
		});
	}

	private static void report(Minecraft mc, String what) {
		server(mc, p -> {
			StringBuilder b = new StringBuilder("[HARNESS] " + what + ": kit=" + com.projecthero.mod.worthy.WorthyKit.kit(p) + " flying="
					+ com.projecthero.mod.worthy.WorthyFlight.isFlying(p) + " main=" + p.getMainHandItem() + " husks:");
			for (Husk h : p.serverLevel().getEntitiesOfClass(Husk.class, new AABB(base).inflate(48))) {
				b.append(String.format(" (%.1f,%.1f)=%.0f", h.getX() - base.getX() - 0.5, h.getZ() - base.getZ() - 0.5, 250 - h.getHealth()));
			}
			System.out.println(b);
		});
	}

	private static void plan() {
		Minecraft mc = Minecraft.getInstance();
		final Item shield = item("projecthero:adamantium_shield");
		int t = 0;
		// ---- 1. the lend: Thor (off camera, to his right) lends Mjolnir; the torch steps into the pack
		at(t, () -> {
			faceYaw = 0f;
			facePitch = 8f;
			server(mc, p -> reset(p, Items.TORCH, null));
		});
		at(t + 2, () -> cam3(140f, 4.2f, 1.6f, 0.0f, 1.2f, false));
		film(t + 6, t + 56, "film_lend");
		at(t + 14, () -> server(mc, p -> lend(p, false)));
		at(t + 54, () -> report(mc, "lend"));
		t += 60;
		// ---- 2. Mjolnir only: Hammer Throw (C) through the group, then the Worthy Storm (Shift+Z)
		at(t, () -> {
			facePitch = 4f;
			server(mc, p -> {
				reset(p, null, null);
				lend(p, false);
				group(p);
			});
		});
		at(t + 2, () -> cam3(155f, 9.0f, 5.5f, 6.0f, 0.8f, false));
		film(t + 6, t + 100, "film_mjolnir_throw_storm");
		at(t + 10, () -> server(mc, p -> key(p, 6, false)));
		at(t + 38, () -> server(mc, p -> key(p, 4, true)));
		at(t + 96, () -> report(mc, "mjolnir"));
		t += 104;
		// ---- 3. Mjolnir only, first person: Hammer Combo (R) on the near husk, Call Lightning (G)
		at(t, () -> {
			facePitch = 12f;
			server(mc, p -> {
				reset(p, null, null);
				lend(p, false);
				husk(p, 0.0, 2.6);
				husk(p, 0.0, 9.0);
			});
		});
		at(t + 2, () -> fp());
		film(t + 6, t + 60, "film_mjolnir_combo_fp");
		at(t + 10, () -> server(mc, p -> key(p, 1, false)));
		at(t + 34, () -> server(mc, p -> {
			p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, 4f);
			key(p, 2, false);
		}));
		at(t + 58, () -> report(mc, "combo fp"));
		t += 64;
		// ---- 4. Stormbreaker only: Lightning Cleave (G), then Stormbreaker's Verdict (Shift+Z)
		at(t, () -> {
			facePitch = 6f;
			server(mc, p -> {
				reset(p, null, null);
				lend(p, true);
				group(p);
			});
		});
		at(t + 2, () -> cam3(125f, 9.0f, 4.5f, 6.0f, 0.8f, false));
		film(t + 6, t + 90, "film_stormbreaker_cleave_verdict");
		at(t + 10, () -> server(mc, p -> key(p, 2, false)));
		at(t + 36, () -> server(mc, p -> key(p, 4, true)));
		at(t + 86, () -> report(mc, "stormbreaker"));
		t += 94;
		// ---- 5. shield + Mjolnir: Shield and Weapon (R), then the Thunder Clap (Shift+Z)
		at(t, () -> {
			facePitch = 6f;
			server(mc, p -> {
				reset(p, null, shield);
				lend(p, false);
				husk(p, 0.0, 2.6);
				husk(p, 2.0, 6.5);
				husk(p, -2.0, 7.0);
				husk(p, 0.5, 9.0);
			});
		});
		at(t + 2, () -> cam3(120f, 7.5f, 3.0f, 3.0f, 1.0f, false));
		film(t + 6, t + 80, "film_shield_mjolnir_clap");
		at(t + 10, () -> server(mc, p -> key(p, 1, false)));
		at(t + 34, () -> server(mc, p -> key(p, 4, true)));
		at(t + 76, () -> report(mc, "shield + mjolnir"));
		t += 84;
		// ---- 6. flight: a jump, the double-tap (whirl spin-up), then flying forward, sprinting
		at(t, () -> {
			facePitch = -8f;
			server(mc, p -> {
				reset(p, null, null);
				lend(p, false);
			});
		});
		at(t + 2, () -> cam3(130f, 6.0f, 2.0f, 0.0f, 1.2f, true));
		film(t + 6, t + 86, "film_flight");
		at(t + 10, () -> server(mc, p -> {
			p.connection.teleport(p.getX(), p.getY() + 1.5, p.getZ(), 0f, -8f);
			System.out.println("[HARNESS] flight toggle: " + com.projecthero.mod.worthy.WorthyFlight.toggle(p));
		}));
		at(t + 26, () -> {
			mc.options.keyUp.setDown(true);
			mc.options.keySprint.setDown(true);
		});
		at(t + 80, () -> {
			mc.options.keyUp.setDown(false);
			mc.options.keySprint.setDown(false);
			report(mc, "flight");
		});
		t += 90;
		final int end = t;
		at(end, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(WorthyFilmHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 20) {
			GLFW.glfwSetWindowSize(mc.getWindow().getWindow(), 854, 480);
		}
		if (t == 40) {
			mc.options.fov().set(70);
			mc.options.fovEffectScale().set(0.0);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			mc.getToasts().clear();
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, WorthyFilmHarness::stage);
		}
		if (t < 140 || base == null) {
			return;
		}
		face(mc);
		int s = t - 140;
		if (s == 0) {
			plan();
		}
		if (s % 10 == 0) {
			server(mc, p -> p.setHealth(p.getMaxHealth()));
		}
		List<Runnable> todo = AT.remove(s);
		if (todo != null) {
			todo.forEach(Runnable::run);
		}
		if (s >= film0 && s < filmEnd) {
			shot(mc, String.format(filmDir + "/f%03d", s - film0));
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(GameRules.RULE_DOMOBSPAWNING).set(false, null);
		LevelSettings settings = new LevelSettings("WORTHY_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.EASY, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}

package com.projecthero.mod.client.mixin;

import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.Shadow;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

import net.minecraft.client.Camera;
import net.minecraft.world.entity.Entity;
import net.minecraft.world.level.BlockGetter;

/** TEMP-HARNESS (not shipped): the WorthyFilmHarness's fixed camera. */
@Mixin(Camera.class)
public abstract class WorthyFilmHarnessCameraMixin {
	@Shadow
	private boolean detached;

	@Shadow
	protected abstract void setRotation(float yaw, float pitch);

	@Shadow
	protected abstract void setPosition(double x, double y, double z);

	@Inject(method = "setup", at = @At("TAIL"))
	private void projecthero$harnessCam(BlockGetter level, Entity entity, boolean detached, boolean mirrored, float partial, CallbackInfo ci) {
		double[] c = com.projecthero.mod.client.WorthyFilmHarness.camPose;
		if (c != null) {
			this.detached = true;
			setPosition(c[0], c[1], c[2]);
			setRotation((float) c[3], (float) c[4]);
		}
	}
}
