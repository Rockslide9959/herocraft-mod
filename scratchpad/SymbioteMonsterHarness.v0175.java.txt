/**
 * TEMP-HARNESS (not shipped, v0.17.5 Symbiote Monster form): films the Shift+Z transformation (real time, every rendered frame, and the
 * cocoon again in slow motion), turntable stills on Classic and Carnage, a face close-up, the HUD + Left-Alt list, every monster move on
 * iron golems / zombies, first person, the melt-back, and a side-by-side with the Titan boss. Enabled by run/symmon.flag = "<out dir>".
 * Kept as scratchpad/SymbioteMonsterHarness.v0175.java.txt (+ SymMonHarnessCameraMixin / SymMonHarnessGuiMixin .v0175.java.txt).
 */
public final class SymbioteMonsterHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static File out;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static int planTick;
	private static BlockPos site;
	public static volatile net.minecraft.world.phys.Vec3 camOffset;
	public static volatile double camLookY = 1.2;
	public static volatile boolean hideHud;
	private static volatile String frameFilm;
	private static volatile long frameUntil;
	private static int frameN;

	private static void cam(double x, double y, double z, double lookY, int fov) {
		camOffset = new net.minecraft.world.phys.Vec3(x, y, z);
		camLookY = lookY;
		Minecraft.getInstance().options.fov().set(fov);
	}

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("symmon.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readAllLines(flag).get(0).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(SymbioteMonsterHarness::tick);
		WorldRenderEvents.END.register(ctx -> {
			String f = frameFilm;
			if (f == null) {
				return;
			}
			if (System.currentTimeMillis() > frameUntil) {
				frameFilm = null;
				System.out.println("[HARNESS] film " + f + " frames=" + frameN);
				return;
			}
			Screenshot.grab(out, String.format("film_%s_%04d.png", f, frameN++), Minecraft.getInstance().getMainRenderTarget(), m -> {
			});
		});
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(560_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "symmon-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void rate(Minecraft mc, float tps) {
		server(mc, p -> p.server.tickRateManager().setTickRate(tps));
	}

	private static void frames(String name, long ms) {
		frameN = 0;
		frameUntil = System.currentTimeMillis() + ms;
		frameFilm = name;
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		mc.getToasts().clear();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		System.out.println("[HARNESS] shot " + name + " phase=" + SymbioteMonster.state(mc.player).phase() + " h=" + mc.player.getBbHeight());
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			prev.run();
			r.run();
		});
	}

	private static void toggle() {
		ClientPlayNetworking.send(new com.projecthero.mod.network.SpiderActionPayload(
				com.projecthero.mod.network.SpiderActionPayload.Action.TOGGLE_SYMBIOTE));
	}

	private static void face(Minecraft mc, float yaw, float pitch) {
		server(mc, p -> p.connection.teleport(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5, yaw, pitch));
		faceYaw = yaw;
		facePitch = pitch;
		faceTicks = 6;
	}

	private static float faceYaw, facePitch;
	private static int faceTicks;

	/** The body turns lazily toward the head on the client: square it up with the look for a few ticks after each turn. */
	private static void holdFacing(Minecraft mc) {
		if (faceTicks > 0 && mc.player != null) {
			faceTicks--;
			mc.player.setYRot(faceYaw);
			mc.player.setXRot(facePitch);
			mc.player.setYHeadRot(faceYaw);
			mc.player.yHeadRotO = faceYaw;
			mc.player.yBodyRot = faceYaw;
			mc.player.yBodyRotO = faceYaw;
		}
	}

	/** A server-side press of a monster key (no Shift). */
	private static void move(Minecraft mc, AbilitySlot slot) {
		server(mc, p -> {
			p.setShiftKeyDown(false);
			SymbioteAbilityManager.handle(p, slot, true);
			SymbioteAbilityManager.handle(p, slot, false);
		});
	}

	private static void charge(Minecraft mc) {
		server(mc, p -> {
			SymbioteVitalsManager.restoreBiomass(p);
			SymbioteState c = Symbiote.state(p).copy();
			c.abilityCooldowns.set(SymbioteMonster.CDI_MONSTER, 0L);
			p.setAttached(ModAttachments.SYMBIOTE_STATE, c);
			SymbioteMonster.handleCharge(p, true, p.level().getGameTime());
		});
	}

	/** Mobs in front of the host (north, -z), relative to the site; golems unless {@code zombie}. */
	private static void mob(Minecraft mc, double dx, double dz, boolean zombie) {
		server(mc, p -> {
			ServerLevel level = p.serverLevel();
			Mob m = zombie ? EntityType.ZOMBIE.create(level) : EntityType.IRON_GOLEM.create(level);
			if (m == null) {
				return;
			}
			m.moveTo(site.getX() + 0.5 + dx, site.getY() + 1, site.getZ() + 0.5 + dz, 180f + (float) Math.toDegrees(Math.atan2(dx, -dz)), 0f);
			m.setNoAi(true);
			m.setPersistenceRequired();
			m.setYHeadRot(0f);
			if (zombie) {
				m.addEffect(new MobEffectInstance(MobEffects.FIRE_RESISTANCE, 20 * 600, 0, false, false));
			}
			level.addFreshEntity(m);
		});
	}

	private static void clearMobs(Minecraft mc) {
		server(mc, p -> p.serverLevel().getEntitiesOfClass(Mob.class, p.getBoundingBox().inflate(40)).forEach(e -> e.discard()));
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(SymbioteMonsterHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(70);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(com.projecthero.mod.loot.finds.SupplyDrops.RULE).set(false, p.server);
				BlockPos sp = p.blockPosition();
				int y = level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, sp.getX(), sp.getZ());
				site = new BlockPos(sp.getX(), y - 1, sp.getZ());
				for (int x = -18; x <= 30; x++) {
					for (int z = -22; z <= 14; z++) {
						for (int yy = -2; yy <= 22; yy++) {
							BlockState st = yy < 0 ? Blocks.DIRT.defaultBlockState() : yy == 0 ? Blocks.GRASS_BLOCK.defaultBlockState() : Blocks.AIR.defaultBlockState();
							level.setBlock(site.offset(x, yy, z), st, 2 | 16);
						}
					}
				}
				p.setGameMode(GameType.SURVIVAL);
				p.connection.teleport(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5, 180f, 8f);
				Symbiote.grant(p);
				SymbioteSuitColours.setPreset(p, SymbioteColourPreset.CLASSIC);
				p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 20 * 60 * 30, 4, false, false));
				p.addEffect(new MobEffectInstance(MobEffects.SATURATION, 20 * 60 * 30, 4, false, false));
			});
		}
		if (t == 46) {
			buildPlan(mc);
		}
		if (t < 50 || site == null) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		}
		holdFacing(mc);
	}

	private static void preset(Minecraft mc, SymbioteColourPreset c) {
		server(mc, p -> SymbioteSuitColours.setPreset(p, c));
	}

	/** Eight stills round the monster + three face close-ups, from {@code t}; returns the next free tick. */
	private static int turntable(Minecraft mc, int t, String prefix) {
		String[] names = { "1_front", "2_front_right", "3_right", "4_back_right", "5_back", "6_back_left", "7_left", "8_front_left" };
		for (int k = 0; k < 8; k++) {
			final double a = Math.toRadians(k * 45);
			final String n = names[k];
			at(t + k * 6, () -> {
				face(mc, 180f, 8f);
				cam(Math.sin(a) * 11.5, 4.2, -Math.cos(a) * 11.5, 3.4, 55);
			});
			at(t + k * 6 + 5, () -> shot(mc, prefix + "_" + n));
		}
		int f = t + 48;
		at(f, () -> cam(2.2, 5.2, -5.6, 4.6, 40));
		at(f + 5, () -> shot(mc, prefix + "_face_three_quarter"));
		at(f + 6, () -> cam(0, 5.0, -6.0, 4.6, 40));
		at(f + 11, () -> shot(mc, prefix + "_face_front"));
		at(f + 12, () -> cam(6.2, 4.6, -1.0, 4.2, 45));
		at(f + 17, () -> shot(mc, prefix + "_face_side_hunch"));
		return f + 20;
	}

	private static void buildPlan(Minecraft mc) {
		int t = 10;
		at(t, () -> {
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.hideGui = true;
		});
		at(t + 5, SymbioteMonsterHarness::toggle);
		at(t + 85, () -> {
			face(mc, 180f, 8f);
			cam(6.0, 3.2, -8.0, 2.8, 60);
		});
		at(t + 90, () -> shot(mc, "00_suited"));

		// ---- the transformation, real time (every rendered frame)
		int c = t + 95;
		at(c, () -> frames("transform", 10000));
		at(c + 1, () -> charge(mc));
		at(c + 30, () -> shot(mc, "01_charging"));
		at(c + 61 + 8, () -> shot(mc, "02_cocoon_tendrils_erupt"));
		at(c + 61 + 20, () -> shot(mc, "03_cocoon_weaving"));
		at(c + 61 + 38, () -> shot(mc, "04_cocoon_swollen_heartbeat"));
		at(c + 61 + 48, () -> shot(mc, "05_cocoon_straining"));
		at(c + 61 + 52, () -> shot(mc, "06_burst"));
		at(c + 61 + 62, () -> shot(mc, "07_emerge_roar"));
		int burst = c + 61 + 50;

		// ---- turntables, Classic then Carnage
		int s = burst + 36;
		s = turntable(mc, s, "turntable_classic");
		at(s, () -> preset(mc, SymbioteColourPreset.CARNAGE));
		s = turntable(mc, s + 4, "turntable_carnage");
		at(s, () -> preset(mc, SymbioteColourPreset.CLASSIC));
		// the HUD and the Left-Alt move list
		at(s + 2, () -> {
			mc.options.hideGui = false;
			cam(6.0, 3.6, -8.0, 3.0, 60);
		});
		at(s + 8, () -> shot(mc, "hud_monster"));
		at(s + 9, () -> com.projecthero.mod.client.gui.MoveListOverlay.forceHeld = true);
		at(s + 13, () -> shot(mc, "hud_move_list"));
		at(s + 14, () -> {
			com.projecthero.mod.client.gui.MoveListOverlay.forceHeld = false;
			mc.options.hideGui = true;
		});

		// ---- every move, real time
		int a = s + 20;
		at(a, () -> {
			face(mc, 180f, 8f);
			cam(10.0, 4.6, -7.0, 2.6, 62);
			mob(mc, -2.0, -5.0, false);
			mob(mc, 2.2, -5.2, false);
			mob(mc, 0.0, -7.5, false);
		});
		at(a + 6, () -> frames("attacks", 17500));
		at(a + 8, () -> move(mc, AbilitySlot.SLOT_1));          // R swipe
		at(a + 8 + 7, () -> shot(mc, "move_1_swipe"));
		at(a + 34, () -> move(mc, AbilitySlot.SLOT_2));         // G pound
		at(a + 34 + 6, () -> shot(mc, "move_2_pound_raised"));
		at(a + 34 + 11, () -> shot(mc, "move_2_pound_slam"));
		at(a + 70, () -> move(mc, AbilitySlot.SLOT_3));         // X leap
		at(a + 70 + 12, () -> shot(mc, "move_3_leap_air"));
		at(a + 118, () -> {
			clearMobs(mc);
			face(mc, 180f, 8f);
		});
		at(a + 122, () -> {
			mob(mc, -2.5, -6.0, false);
			mob(mc, 2.5, -6.0, false);
			mob(mc, 0.0, -9.0, false);
		});
		at(a + 130, () -> move(mc, AbilitySlot.SLOT_4));        // Z roar
		at(a + 130 + 10, () -> shot(mc, "move_4_roar"));
		at(a + 168, () -> {
			clearMobs(mc);
			mob(mc, -3.0, -9.0, false);
			mob(mc, 3.0, -9.5, false);
			mob(mc, 0.0, -10.5, true);
		});
		at(a + 176, () -> move(mc, AbilitySlot.SLOT_5));        // V pull
		at(a + 176 + 6, () -> shot(mc, "move_5_pull"));
		at(a + 214, () -> {
			clearMobs(mc);
			mob(mc, 0.0, -4.6, true);
			mob(mc, 3.0, -6.0, false);
		});
		at(a + 222, () -> move(mc, AbilitySlot.SLOT_6));        // C devour
		at(a + 222 + 9, () -> shot(mc, "move_6_devour_seize"));
		at(a + 222 + 17, () -> shot(mc, "move_6_devour_jaws"));
		at(a + 270, () -> move(mc, AbilitySlot.SLOT_1));
		at(a + 290, () -> move(mc, AbilitySlot.SLOT_1));

		// ---- first person
		int f = a + 360;
		at(f, () -> {
			clearMobs(mc);
			camOffset = null;
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.hideGui = false;
			hideHud = true;
			mc.options.fov().set(70);
			face(mc, 180f, 10f);
		});
		at(f + 8, () -> shot(mc, "first_person_idle"));
		at(f + 10, () -> mc.player.swing(InteractionHand.MAIN_HAND));
		at(f + 12, () -> shot(mc, "first_person_swing"));
		at(f + 20, () -> move(mc, AbilitySlot.SLOT_1));
		at(f + 25, () -> shot(mc, "first_person_swipe"));
		at(f + 40, () -> move(mc, AbilitySlot.SLOT_2));
		at(f + 49, () -> shot(mc, "first_person_pound"));
		at(f + 56, () -> {
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.hideGui = true;
			hideHud = false;
		});

		// ---- the melt (Shift+Z in the form)
		int m = f + 64;
		at(m, () -> {
			face(mc, 180f, 8f);
			cam(6.0, 3.2, -8.0, 2.8, 60);
		});
		at(m + 5, () -> frames("melt", 3600));
		at(m + 8, () -> server(mc, p -> {
			p.setShiftKeyDown(true);
			SymbioteAbilityManager.handle(p, AbilitySlot.SLOT_4, true);
			p.setShiftKeyDown(false);
		}));
		at(m + 8 + 6, () -> shot(mc, "melt_1_slump"));
		at(m + 8 + 14, () -> shot(mc, "melt_2_pool"));
		at(m + 8 + 21, () -> shot(mc, "melt_3_rising"));
		at(m + 8 + 34, () -> shot(mc, "melt_4_reformed"));

		// ---- the cocoon again in slow motion (6 tps from just before the charge completes)
		int w = m + 70;
		at(w, () -> cam(6.0, 3.6, -8.5, 3.0, 60));
		at(w + 2, () -> charge(mc));
		at(w + 2 + 57, () -> rate(mc, 6f));
		at(w + 2 + 58, () -> frames("transform_slowmo", 15500));
		at(w + 2 + 58 + 95, () -> rate(mc, 20f));

		// ---- next to the Titan boss (silhouettes)
		int q = w + 2 + 58 + 100;
		at(q, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			var titan = com.projecthero.mod.titan.entity.TitanEntityTypes.TITAN.create(level);
			if (titan != null) {
				titan.moveTo(site.getX() + 14.5, site.getY() + 1, site.getZ() + 0.5, 180f, 0f);
				titan.setNoAi(true);
				titan.setPersistenceRequired();
				level.addFreshEntity(titan);
			}
		}));
		at(q + 10, () -> {
			face(mc, 90f, 0f);
			cam(6.0, 7.0, -30.0, 6.5, 55);
		});
		at(q + 30, () -> shot(mc, "compare_titan_boss_side"));
		at(q + 31, () -> cam(-22.0, 7.0, -22.0, 6.5, 55));
		at(q + 40, () -> shot(mc, "compare_titan_boss_quarter"));
		at(q + 46, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("SYMMON_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17051L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
