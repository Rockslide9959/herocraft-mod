	private static boolean createdWorld;

	public static void init() {
		if (!Files.exists(Path.of("omnitrix_acquire.flag"))) {
			return;
		}
		try {
			only = Files.readString(Path.of("omnitrix_acquire.flag")).trim();
		} catch (java.io.IOException e) {
			only = "";
		}
		ClientTickEvents.END_CLIENT_TICK.register(OmnitrixAcquireHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(900_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "acquire-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static boolean want(String part) {
		return only.isEmpty() || only.contains(part);
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (x, y) -> () -> {
			x.run();
			y.run();
		});
	}

	private static void clearParticles() {
		try {
			var m = net.minecraft.client.particle.ParticleEngine.class.getDeclaredMethod("clearParticles");
			m.setAccessible(true);
			m.invoke(Minecraft.getInstance().particleEngine);
		} catch (ReflectiveOperationException e) {
			System.out.println("[HARNESS] " + e);
		}
	}

	private static void shot(int t, String name, boolean clear) {
		if (clear) {
			at(t - 1, OmnitrixAcquireHarness::clearParticles);
		}
		at(t, () -> {
			Minecraft mc = Minecraft.getInstance();
			Screenshot.grab(mc.gameDirectory, "omnitrix_acquire_" + name + ".png", mc.getMainRenderTarget(), m -> {
			});
			StringBuilder sb = new StringBuilder("[HARNESS] shot " + name);
			if (mc.level != null) {
				for (var pl : mc.level.players()) {
					float c = OmnitrixLatchClient.clock(pl, 0f);
					if (c >= 0f) {
						sb.append(" | ").append(pl.getName().getString()).append(" latch=").append(c);
					}
				}
			}
			System.out.println(sb);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.options.getSoundSourceOptionInstance(net.minecraft.sounds.SoundSource.MASTER).get() > 0.05) {
			mc.options.getSoundSourceOptionInstance(net.minecraft.sounds.SoundSource.MASTER).set(0.05);
		}
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(OmnitrixAcquireHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			mc.options.chatVisibility().set(net.minecraft.world.entity.player.ChatVisiblity.HIDDEN);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(5000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOFIRETICK).set(false, p.server);
				p.server.setDifficulty(Difficulty.PEACEFUL, true);
				p.setGameMode(GameType.CREATIVE);
				site = flatten(level, findFlat(level, p.blockPosition()));
				System.out.println("[HARNESS] site " + site);
				p.teleportTo(level, site.x, site.y, site.z, 0f, 0f);
				ready = true;
			});
		}
		// the script's clock starts once the server has built the site (and had 3 s to settle its light and chunks)
		if (ready && t0 < 0) {
			readyTicks++;
			if (readyTicks > 60) {
				t0 = inWorldTicks;
				AT.put(5, () -> script(mc));
			}
		}
		if (t0 >= 0) {
			server(mc, p -> {
				if (ben != null) {
					ben.tickCount++;
					Omnitrix.tick(ben);
				}
				Omnitrix.tick(p);
				var m = mover;
				if (m != null) {
					m.accept(p);
				}
			});
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	/** Levels a 61 x 61 meadow round {@code at} (grass over dirt over stone, the air above cleared), so the shots are on open ground. */
	private static Vec3 flatten(ServerLevel level, Vec3 at) {
		BlockPos c = BlockPos.containing(at);
		int y0 = c.getY();
		var grass = net.minecraft.world.level.block.Blocks.GRASS_BLOCK.defaultBlockState();
		var dirt = net.minecraft.world.level.block.Blocks.DIRT.defaultBlockState();
		var stone = net.minecraft.world.level.block.Blocks.STONE.defaultBlockState();
		var air = net.minecraft.world.level.block.Blocks.AIR.defaultBlockState();
		for (int dx = -30; dx <= 30; dx++) {
			for (int dz = -30; dz <= 30; dz++) {
				for (int y = y0 - 10; y <= y0 + 30; y++) {
					var s = y < y0 - 4 ? stone : y < y0 - 1 ? dirt : y == y0 - 1 ? grass : air;
					level.setBlock(new BlockPos(c.getX() + dx, y, c.getZ() + dz), s, 2);
				}
			}
		}
		return Vec3.atBottomCenterOf(c);
	}

	/** A flat, dry patch near {@code from}: 5x5 samples 6 apart within 2 blocks of each other. */
	private static Vec3 findFlat(ServerLevel level, BlockPos from) {
		for (int ring = 0; ring < 10; ring++) {
			for (int k = 0; k < Math.max(1, ring * 8); k++) {
				double a = k * Math.PI * 2 / Math.max(1, ring * 8);
				int x = from.getX() + (int) (Math.cos(a) * ring * 24), z = from.getZ() + (int) (Math.sin(a) * ring * 24);
				int min = 999, max = -999;
				boolean wet = false;
				for (int i = -2; i <= 2; i++) {
					for (int j = -2; j <= 2; j++) {
						level.getChunkAt(new BlockPos(x + i * 6, 0, z + j * 6)); // generate it first (an unloaded column reads as the floor)
						BlockPos top = level.getHeightmapPos(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, new BlockPos(x + i * 6, 0, z + j * 6));
						min = Math.min(min, top.getY());
						max = Math.max(max, top.getY());
						wet |= !level.getFluidState(top.below()).isEmpty() || top.getY() < 64;
					}
				}
				if (max - min <= 3 && !wet) {
					BlockPos top = level.getHeightmapPos(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, new BlockPos(x, 0, z));
					return Vec3.atBottomCenterOf(top);
				}
			}
		}
		BlockPos top = level.getHeightmapPos(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, from);
		return Vec3.atBottomCenterOf(top);
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("OA_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.PEACEFUL, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(7171717L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.NORMAL).value()
						.createWorldDimensions(),
				mc.screen);
	}

	private static void camera(ServerPlayer p, Vec3 eye, Vec3 target) {
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		p.teleportTo(p.serverLevel(), eye.x, eye.y - p.getEyeHeight(), eye.z, yaw, pit);
	}

	private static void face(ServerPlayer f, Vec3 target) {
		Vec3 eye = f.getEyePosition();
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		f.teleportTo(f.serverLevel(), f.getX(), f.getY(), f.getZ(), yaw, pit);
		f.setYHeadRot(yaw);
		f.yBodyRot = yaw;
		f.yBodyRotO = yaw;
	}

	private static ServerPlayer fake(ServerPlayer near, String name, Vec3 at, float yaw) {
		ServerLevel level = near.serverLevel();
		GameProfile profile = new GameProfile(java.util.UUID.randomUUID(), name);
		CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
		ServerPlayer f = new ServerPlayer(level.getServer(), level, profile, cookie.clientInformation());
		Connection conn = new Connection(PacketFlow.SERVERBOUND);
		new io.netty.channel.embedded.EmbeddedChannel(conn);
		level.getServer().getPlayerList().placeNewPlayer(conn, f, cookie);
		f.setGameMode(GameType.SURVIVAL);
		f.teleportTo(level, at.x, at.y, at.z, yaw, 0f);
		f.setYHeadRot(yaw);
		f.yBodyRot = yaw;
		f.yBodyRotO = yaw;
		return f;
	}

	private static void gui(Minecraft mc, boolean on) {
		mc.execute(() -> mc.options.hideGui = !on);
	}

	private static Vec3 front(OmnitrixPodEntity e, double d) {
		double yaw = Math.toRadians(e.getYRot());
		return e.position().add(-Math.sin(yaw) * d, 0, Math.cos(yaw) * d);
	}
