/** TEMP-HARNESS (not shipped): v0.17.2 Star Platinum screenshots. Enabled by run/stand.flag. */
// TEMP-HARNESS
public final class StandHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static Vec3 origin;
	private static ServerPlayer jotaro;

	private StandHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("stand.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(StandHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(600_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "stand-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (x, y) -> () -> {
			x.run();
			y.run();
		});
	}

	private static void shot(int t, String name, boolean clear) {
		if (clear) {
			at(t - 1, () -> {
				try {
					var m = net.minecraft.client.particle.ParticleEngine.class.getDeclaredMethod("clearParticles");
					m.setAccessible(true);
					m.invoke(Minecraft.getInstance().particleEngine);
				} catch (ReflectiveOperationException e) {
					System.out.println("[HARNESS] " + e);
				}
			});
		}
		at(t, () -> {
			Minecraft mc = Minecraft.getInstance();
			Screenshot.grab(mc.gameDirectory, "v0172_stand_" + name + ".png", mc.getMainRenderTarget(), m -> {
			});
			System.out.println("[HARNESS] shot " + name);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.options.getSoundSourceOptionInstance(net.minecraft.sounds.SoundSource.MASTER).get() > 0.05) {
			mc.options.getSoundSourceOptionInstance(net.minecraft.sounds.SoundSource.MASTER).set(0.05);
		}
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(StandHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			mc.options.chatVisibility().set(net.minecraft.world.entity.player.ChatVisiblity.HIDDEN);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.EASY, true);
				p.setGameMode(GameType.CREATIVE);
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			AT.put(3, () -> script(mc));
		}
		if (t0 >= 0) {
			server(mc, p -> {
				if (p.tickCount % 5 == 0) {
					for (var e : p.serverLevel().getEntitiesOfClass(net.minecraft.world.entity.monster.Slime.class, p.getBoundingBox().inflate(120))) {
						e.discard();
					}
				}
			});
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("SP_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.EASY, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}

	private static void camera(ServerPlayer p, Vec3 eye, Vec3 target) {
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		p.teleportTo(p.serverLevel(), eye.x, eye.y - p.getEyeHeight(), eye.z, yaw, pit);
	}

	private static ServerPlayer fake(ServerPlayer near, String name, Vec3 at, float yaw) {
		ServerLevel level = near.serverLevel();
		GameProfile profile = new GameProfile(java.util.UUID.randomUUID(), name);
		CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
		ServerPlayer f = new ServerPlayer(level.getServer(), level, profile, cookie.clientInformation());
		Connection conn = new Connection(PacketFlow.SERVERBOUND);
		new io.netty.channel.embedded.EmbeddedChannel(conn);
		level.getServer().getPlayerList().placeNewPlayer(conn, f, cookie);
		f.setGameMode(GameType.SURVIVAL);
		face(f, at, yaw);
		return f;
	}

	private static void face(ServerPlayer f, Vec3 at, float yaw) {
		f.teleportTo(f.serverLevel(), at.x, at.y, at.z, yaw, 0f);
		f.setYHeadRot(yaw);
		f.yBodyRot = yaw;
		f.yBodyRotO = yaw;
	}

	private static void edit(ServerPlayer f, java.util.function.Consumer<StandState> fn) {
		StandState n = Stands.state(f).copy();
		fn.accept(n);
		f.setAttached(ModAttachments.STAND_STATE, n);
	}

	private static void script(Minecraft mc) {
		int t = 10;
		final Vec3[] stage = new Vec3[1];
		at(t, () -> server(mc, p -> {
			ServerLevel l = p.serverLevel();
			stage[0] = origin.add(-20, 0, 0);
			Vec3 s = stage[0];
			net.minecraft.core.BlockPos o = net.minecraft.core.BlockPos.containing(s);
			// a backdrop: a sandstone wall behind, a couple of pillars
			for (int x = -6; x <= 6; x++) {
				for (int y = 0; y <= 4; y++) {
					l.setBlock(o.offset(x, y, -6), net.minecraft.world.level.block.Blocks.SMOOTH_SANDSTONE.defaultBlockState(), 3);
				}
			}
			for (int y = 0; y <= 4; y++) {
				l.setBlock(o.offset(-6, y, 5), net.minecraft.world.level.block.Blocks.CUT_SANDSTONE.defaultBlockState(), 3);
				l.setBlock(o.offset(6, y, 5), net.minecraft.world.level.block.Blocks.CUT_SANDSTONE.defaultBlockState(), 3);
			}
			UniquePowers.forced(() -> Stands.grant(p, StandType.STAR_PLATINUM)); // the camera is a Stand user: it can see Stands
			jotaro = fake(p, "Jotaro", s, 0f);
			UniquePowers.forced(() -> Stands.grant(jotaro, StandType.STAR_PLATINUM));
			Stands.summon(jotaro);
			camera(p, s.add(-2.4, 1.9, 3.6), s.add(-0.2, 1.3, 0));
		}));
		shot(t + 20, "summon", false);
		// at rest, behind the right shoulder
		shot(t + 40, "rest_front", true);
		at(t + 42, () -> server(mc, p -> camera(p, stage[0].add(2.6, 2.2, -3.4), stage[0].add(-0.3, 1.4, 0))));
		shot(t + 48, "rest_back", true);
		at(t + 50, () -> server(mc, p -> camera(p, stage[0].add(-3.6, 1.8, 0.4), stage[0].add(0, 1.4, 0.6))));
		shot(t + 56, "rest_side", true);
		// the ORA Rush (blurred arms, ORA text)
		at(t + 60, () -> server(mc, p -> {
			edit(jotaro, n -> {
				n.rushing = true;
				n.rushStart = p.level().getGameTime();
			});
			camera(p, stage[0].add(-3.2, 1.9, 3.4), stage[0].add(0, 1.3, 0.9));
		}));
		shot(t + 70, "ora_rush", false);
		shot(t + 73, "ora_rush_2", false);
		// Star Finger
		at(t + 80, () -> server(mc, p -> edit(jotaro, n -> {
			n.rushing = false;
			n.animId = StandState.ANIM_FINGER;
			n.animStart = p.level().getGameTime();
		})));
		shot(t + 84, "star_finger", true);
		// Stand Guard
		at(t + 90, () -> server(mc, p -> edit(jotaro, n -> n.guarding = true)));
		shot(t + 96, "guard", true);
		// Star Breaker charging
		at(t + 100, () -> server(mc, p -> edit(jotaro, n -> {
			n.guarding = false;
			n.chargeStart = p.level().getGameTime() - 25;
		})));
		shot(t + 104, "breaker_charge", true);
		// the JoJo pose (the user's body) with the Stand behind
		at(t + 110, () -> server(mc, p -> {
			edit(jotaro, n -> {
				n.chargeStart = 0L;
				n.animId = StandState.ANIM_NONE;
			});
			jotaro.setOnGround(true);
			jotaro.setAttached(ModAttachments.STAND_POSE, new StandPoseState(4, p.level().getGameTime()));
			camera(p, stage[0].add(-1.8, 1.8, 3.4), stage[0].add(-0.2, 1.3, 0));
		}));
		shot(t + 126, "pose_jojo", true);
		at(t + 128, () -> server(mc, p -> jotaro.setAttached(ModAttachments.STAND_POSE, new StandPoseState(2, p.level().getGameTime()))));
		shot(t + 136, "pose_yare_yare", true);
		// the time stop: a zombie knocked into the air and an arrow in flight, frozen; the colour drains
		at(t + 140, () -> server(mc, p -> {
			jotaro.setAttached(ModAttachments.STAND_POSE, StandPoseState.NONE);
			ServerLevel l = p.serverLevel();
			Zombie z = EntityType.ZOMBIE.create(l);
			z.moveTo(stage[0].x + 0.3, stage[0].y, stage[0].z + 3.0, 180f, 0f);
			z.setItemSlot(net.minecraft.world.entity.EquipmentSlot.HEAD, new net.minecraft.world.item.ItemStack(net.minecraft.world.item.Items.CARVED_PUMPKIN));
			l.addFreshEntity(z);
			z.setDeltaMovement(0, 0.9, 0.25);
			var arrow = new net.minecraft.world.entity.projectile.Arrow(EntityType.ARROW, l);
			arrow.setPos(stage[0].x + 4, stage[0].y + 1.6, stage[0].z + 1.5);
			arrow.setDeltaMovement(-1.4, 0.05, 0);
			l.addFreshEntity(arrow);
			camera(p, stage[0].add(-3.0, 2.0, 5.0), stage[0].add(0.3, 1.5, 1.2));
		}));
		at(t + 145, () -> server(mc, p -> {
			Stands.setTimeStopUses(jotaro, 50);
			TimeStop.begin(jotaro);
		}));
		shot(t + 152, "time_stop", false);
		at(t + 154, () -> server(mc, p -> camera(p, stage[0].add(3.4, 1.7, 4.4), stage[0].add(0.2, 1.6, 1.5))));
		shot(t + 162, "time_stop_2", false);
		// the Stand Menu (the camera player's own)
		at(t + 260, () -> {
			mc.options.hideGui = false;
			mc.setScreen(new com.projecthero.mod.client.stand.StandMenuScreen());
		});
		shot(t + 272, "menu", false);
		// first person: your own rush in front of you, the HUD
		at(t + 280, () -> {
			mc.setScreen(null);
			server(mc, p -> {
				p.setGameMode(GameType.SURVIVAL);
				p.getAbilities().flying = false;
				p.onUpdateAbilities();
				Vec3 s = stage[0].add(5, 0, 0);
				p.teleportTo(p.serverLevel(), s.x, s.y, s.z, 0f, 8f);
				ServerLevel l = p.serverLevel();
				Zombie z = EntityType.ZOMBIE.create(l);
				z.moveTo(s.x, s.y, s.z + 2.6, 180f, 0f);
				z.setNoAi(true);
				z.setItemSlot(net.minecraft.world.entity.EquipmentSlot.HEAD, new net.minecraft.world.item.ItemStack(net.minecraft.world.item.Items.CARVED_PUMPKIN));
				l.addFreshEntity(z);
				Stands.summon(p);
				edit(p, n -> n.abilityReadyAt.clear());
			});
		});
		at(t + 300, () -> server(mc, StarPlatinum::startRush));
		shot(t + 308, "first_person_rush_hud", false);
		at(t + 312, () -> server(mc, StarPlatinum::stopRush));
		// third person, the camera player strikes nothing; the non-Stand user's view: the Stand is gone
		at(t + 320, () -> server(mc, p -> {
			Stands.revoke(p);
			for (var z : p.serverLevel().getEntitiesOfClass(Zombie.class, p.getBoundingBox().inflate(64))) {
				z.discard();
			}
			p.setGameMode(GameType.CREATIVE);
			p.getAbilities().flying = true;
			p.onUpdateAbilities();
			mc.execute(() -> mc.options.hideGui = true);
			camera(p, stage[0].add(-2.4, 1.9, 3.6), stage[0].add(-0.2, 1.3, 0));
		}));
		shot(t + 336, "non_stand_user_view", true);
		at(t + 345, () -> {
			System.out.println("[HARNESS] DONE");
			Runtime.getRuntime().halt(0);
		});
	}
}
