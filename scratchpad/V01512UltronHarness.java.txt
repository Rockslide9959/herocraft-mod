	public static void init() {

	private static void watchdog() {
		Thread t = new Thread(() -> {
			while (true) {
				try {
					Thread.sleep(5000);
				} catch (InterruptedException e) {
					return;
				}
				if (System.currentTimeMillis() - lastProgress > 120_000) {
					System.out.println("[HARNESS] WATCHDOG no progress for 120 s");
					Thread.getAllStackTraces().forEach((th, st) -> {
						System.out.println("[HARNESS] WATCHDOG thread " + th.getName());
						for (StackTraceElement e : st) {
							System.out.println("[HARNESS] WATCHDOG    at " + e);
						}
					});
					Runtime.getRuntime().halt(3);
				}
			}
		}, "v01512-harness-watchdog");
		t.setDaemon(true);
		t.start();
	}

	private static ServerPlayer sp(Minecraft mc) {
		MinecraftServer srv = mc.getSingleplayerServer();
		return srv == null || srv.getPlayerList().getPlayers().isEmpty() ? null : srv.getPlayerList().getPlayers().get(0);
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				ServerPlayer p = sp(mc);
				if (p != null) {
					try {
						fn.accept(p);
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		lastProgress = System.currentTimeMillis();
		Screenshot.grab(mc.gameDirectory, "v01512_" + name + ".png", mc.getMainRenderTarget(), c -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	private static void at(int t, Runnable r) {
		Runnable prev = AT.get(t);
		AT.put(t, prev == null ? r : () -> {
			prev.run();
			r.run();
		});
	}

	private static void cam(double x, double y, double z, double ax, double ay, double az) {
		camPos = Vec3.atBottomCenterOf(C).add(x, y, z);
		camAt = Vec3.atBottomCenterOf(C).add(ax, ay, az);
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0) {
				lastProgress = System.currentTimeMillis();
				System.out.println("[HARNESS] waiting " + (mc.screen == null ? "null" : mc.screen.getClass().getName()) + " overlay "
						+ (mc.getOverlay() != null));
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen) {
				createdWorld = true;
				lastProgress = System.currentTimeMillis();
				mc.execute(() -> {
					LevelSettings settings = new LevelSettings("V01512U_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
							Difficulty.HARD, true, new GameRules(), WorldDataConfiguration.DEFAULT);
					mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, WorldOptions.defaultWithRandomSeed(),
							reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
									.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
							mc.screen);
				});
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks % 200 == 0) {
			lastProgress = System.currentTimeMillis();
			System.out.println("[HARNESS] in world " + inWorldTicks);
		}
		if (inWorldTicks == 100) {
			mc.options.hideGui = false;
			mc.options.fov().set(70);
			mc.options.renderDistance().set(10);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.getServer().setDifficulty(Difficulty.HARD, true);
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				C = p.blockPosition();
				p.addEffect(new MobEffectInstance(MobEffects.INVISIBILITY, 999999, 0, false, false));
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				System.out.println("[HARNESS] centre " + C);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				lastProgress = System.currentTimeMillis();
				try {
					r.run();
				} catch (Throwable e) {
					System.out.println("[HARNESS] error " + e);
					e.printStackTrace();
				}
			}
		}
	}

	private static BlockPos ground(ServerLevel level, BlockPos at) {
		return new BlockPos(at.getX(), level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, at.getX(), at.getZ()), at.getZ());
	}

	private static <T extends Mob> T spawn(ServerLevel level, EntityType<T> type, BlockPos at, float yaw, boolean noAi) {
		T m = type.create(level);
		BlockPos g = ground(level, at);
		m.moveTo(g.getX() + 0.5, g.getY(), g.getZ() + 0.5, yaw, 0f);
		m.finalizeSpawn(level, level.getCurrentDifficultyAt(g), MobSpawnType.COMMAND, null);
		m.setNoAi(noAi);
		m.setPersistenceRequired();
		level.addFreshEntity(m);
		m.setYRot(yaw);
		m.setYHeadRot(yaw);
		m.setYBodyRot(yaw);
		return m;
	}

	private static void tp(ServerPlayer p, int dx, int dy, int dz) {
		Vec3 v = Vec3.atBottomCenterOf(C).add(dx, dy, dz);
		p.teleportTo(p.serverLevel(), v.x, v.y, v.z, 0f, 0f);
	}

	private static void time(ServerPlayer p, long t) {
		p.serverLevel().setDayTime(t);
	}
