/**
 * TEMP-HARNESS (not shipped): v0.17.1 Multimouse screenshots. Fake players hold Mullo; the local player is the camera.
 * Shots go to run/screenshots/mm_&lt;name&gt;.png. Enabled by run/mm.flag.
 */
public final class MultimouseHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static Vec3 origin;
	private static ServerPlayer a, b;

	private MultimouseHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("mm.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(MultimouseHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(420_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "mm-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (x, y) -> () -> {
			x.run();
			y.run();
		});
	}

	private static void shot(int t, String name) {
		at(t, () -> {
			Minecraft mc = Minecraft.getInstance();
			Screenshot.grab(mc.gameDirectory, "mm_" + name + ".png", mc.getMainRenderTarget(), m -> {
			});
			System.out.println("[HARNESS] shot " + name);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(MultimouseHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		// the fake players stand still: keep their bodies turned with their heads on this client
		for (var other : mc.level.players()) {
			if (other != mc.player) {
				other.yBodyRot = other.getYHeadRot();
				other.yBodyRotO = other.getYHeadRot();
			}
		}
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.PEACEFUL, true);
				p.setGameMode(GameType.CREATIVE);
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			server(mc, p -> {
				for (ServerPlayer f : new ServerPlayer[] { a, b }) {
					if (f != null) {
						f.tickCount++;
						Miraculous.tick(f);
						f.yBodyRot = f.getYRot();
						f.yBodyRotO = f.getYRot();
						f.setYHeadRot(f.getYRot());
						try {
							java.lang.reflect.Method m = net.minecraft.world.entity.player.Player.class.getDeclaredMethod("updatePlayerPose");
							m.setAccessible(true);
							m.invoke(f);
							f.refreshDimensions();
						} catch (ReflectiveOperationException ex) {
							System.out.println("[HARNESS] pose " + ex);
						}
					}
				}
			});
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("MM_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
				Difficulty.PEACEFUL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	private static java.util.UUID skinUuid(boolean slim) {
		for (int i = 0; i < 1000; i++) {
			java.util.UUID id = java.util.UUID.randomUUID();
			boolean s = net.minecraft.client.resources.DefaultPlayerSkin.get(id).model() == net.minecraft.client.resources.PlayerSkin.Model.SLIM;
			if (s == slim) {
				return id;
			}
		}
		return java.util.UUID.randomUUID();
	}

	private static ServerPlayer fake(ServerPlayer near, String name, Vec3 at, java.util.UUID id) {
		ServerLevel level = near.serverLevel();
		GameProfile profile = new GameProfile(id, name);
		CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
		ServerPlayer f = new ServerPlayer(level.getServer(), level, profile, cookie.clientInformation());
		Connection conn = new Connection(PacketFlow.SERVERBOUND);
		new io.netty.channel.embedded.EmbeddedChannel(conn);
		level.getServer().getPlayerList().placeNewPlayer(conn, f, cookie);
		f.setGameMode(GameType.SURVIVAL);
		f.teleportTo(level, at.x, at.y, at.z, 0f, 0f);
		f.setYHeadRot(0f);
		f.yBodyRot = 0f;
		return f;
	}

	private static void camera(ServerPlayer p, Vec3 eye, Vec3 target) {
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		p.teleportTo(p.serverLevel(), eye.x, eye.y - 1.62, eye.z, yaw, pit);
	}

	private static void settle(ServerPlayer f) {
		MiraculousState n = Miraculous.state(f).copy();
		n.formChangeAt = f.level().getGameTime() - 200L;
		n.animId = MiraculousState.ANIM_NONE;
		f.setAttached(ModAttachments.MIRACULOUS_STATE, n);
	}

	private static void transformNow(ServerPlayer f) {
		settle(f);
		Miraculous.edit(f, n -> n.hungry = false);
		Miraculous.transform(f);
		settle(f);
	}

	private static void ready(ServerPlayer f) {
		Miraculous.edit(f, n -> n.abilityReadyAt.clear());
	}

	private static void face(ServerPlayer f, float yaw, float pitch) {
		f.teleportTo(f.serverLevel(), f.getX(), f.getY(), f.getZ(), yaw, pitch);
		f.setYHeadRot(yaw);
		f.yBodyRot = yaw;
	}

	private static void script(Minecraft mc) {
		int t = 10;
		at(t, () -> server(mc, p -> {
			a = fake(p, "Marinette", origin, skinUuid(false));
			b = fake(p, "Mylene", origin.add(2.4, 0, 0), skinUuid(true));
			Miraculous.grant(a, Kwami.MULLO);
			Miraculous.grant(b, Kwami.MULLO);
			camera(p, origin.add(1.2, 1.9, 5.0), origin.add(1.2, 1.3, 0));
		}));
		shot(t + 20, "mullo_popout_wave");
		t += 80;
		at(t, () -> server(mc, p -> camera(p, origin.add(1.0, 1.8, 2.6), origin.add(0.3, 1.4, 0))));
		shot(t + 6, "mullo_by_the_shoulder");
		at(t + 8, () -> server(mc, p -> {
			KwamiEntity k = KwamiCompanion.find(a);
			if (k != null) {
				Vec3 at = k.position().add(0, 0.33, 0);
				camera(p, at.add(0.4, 0.15, 1.35), at);
			}
		}));
		shot(t + 14, "mullo_closeup");
		at(t + 16, () -> server(mc, p -> {
			KwamiEntity k = KwamiCompanion.find(a);
			if (k != null) {
				camera(p, k.position().add(-1.0, 0.5, -0.9), k.position().add(0, 0.3, 0));
			}
		}));
		shot(t + 22, "mullo_back");
		at(t + 24, () -> server(mc, p -> camera(p, a.position().add(0.0, 1.45, 1.1), a.position().add(0, 1.25, 0))));
		shot(t + 30, "pendant_plain");
		t += 40;
		// the transformation (front)
		at(t, () -> server(mc, p -> {
			Miraculous.edit(a, n -> n.hungry = false);
			settle(a);
			camera(p, origin.add(0.6, 1.5, 3.4), origin.add(0, 1.0, 0));
		}));
		at(t + 4, () -> server(mc, p -> Miraculous.transform(a)));
		shot(t + 24, "transform_reveal");
		shot(t + 50, "transform_done");
		t += 60;
		at(t, () -> server(mc, p -> {
			transformNow(a);
			transformNow(b);
			camera(p, origin.add(1.2, 1.5, 4.2), origin.add(1.2, 1.0, 0));
		}));
		shot(t + 10, "suit_front_wide_slim");
		at(t + 12, () -> server(mc, p -> camera(p, origin.add(1.2, 1.5, -4.2), origin.add(1.2, 1.0, 0))));
		shot(t + 20, "suit_back_tail");
		at(t + 22, () -> server(mc, p -> camera(p, a.position().add(2.3, 1.6, 2.3), a.position().add(0, 1.0, 0))));
		shot(t + 28, "suit_three_quarter");
		at(t + 30, () -> server(mc, p -> camera(p, a.position().add(-2.3, 1.6, -2.3), a.position().add(0, 1.0, 0))));
		shot(t + 36, "suit_three_quarter_back");
		at(t + 38, () -> server(mc, p -> {
			b.teleportTo(p.serverLevel(), origin.x + 2.4, origin.y, origin.z - 7.0, 0f, 0f);
			camera(p, a.position().add(-3.0, 1.4, 0.4), a.position().add(0, 1.0, 0));
		}));
		shot(t + 44, "suit_side");
		at(t + 45, () -> server(mc, p -> b.teleportTo(p.serverLevel(), origin.x + 2.4, origin.y, origin.z, 0f, 0f)));
		at(t + 46, () -> server(mc, p -> camera(p, a.getEyePosition().add(0.0, 0.25, 1.1), a.getEyePosition().add(0, 0.2, 0))));
		shot(t + 52, "face_ears_wide");
		at(t + 54, () -> server(mc, p -> camera(p, b.getEyePosition().add(0.0, 0.25, 1.1), b.getEyePosition().add(0, 0.2, 0))));
		shot(t + 60, "face_ears_slim");
		t += 70;
		// the rope moves
		at(t, () -> server(mc, p -> {
			ready(a);
			camera(p, origin.add(3.2, 1.8, 3.0), origin.add(0, 1.1, 1.5));
			MultimouseMoves.lash(a);
		}));
		shot(t + 4, "move_rope_lash");
		at(t + 20, () -> server(mc, p -> {
			ready(a);
			ServerLevel level = p.serverLevel();
			Vec3 w = origin.add(0, 0, 9);
			for (int x = -2; x <= 2; x++) {
				for (int y = 0; y <= 6; y++) {
					level.setBlockAndUpdate(BlockPos.containing(w.x + x, w.y + y, w.z), Blocks.STONE_BRICKS.defaultBlockState());
				}
			}
			face(a, 0f, -25f);
			MultimouseMoves.ropeSwing(a);
			camera(p, origin.add(4.5, 2.5, 3.0), origin.add(0, 2.0, 4.0));
		}));
		shot(t + 26, "move_rope_swing");
		at(t + 70, () -> server(mc, p -> {
			Miraculous.setVal(a, Multimouse.V_SWING, 0L);
			face(a, 0f, 0f);
			ready(a);
			MultimouseMoves.twirl(a);
			camera(p, origin.add(3.0, 3.0, 3.6), origin.add(0, 1.2, 0));
		}));
		shot(t + 76, "move_rope_twirl");
		at(t + 80, () -> server(mc, p -> {
			Miraculous.setVal(a, Multimouse.V_TWIRL, 0L);
			ready(b);
			Mob pig = EntityType.PIG.create(p.serverLevel());
			pig.moveTo(b.getX(), b.getY(), b.getZ() + 4.0, 180f, 0f);
			pig.setNoAi(true);
			p.serverLevel().addFreshEntity(pig);
			face(b, 0f, 17f);
		}));
		at(t + 82, () -> server(mc, p -> {
			MultimouseMoves.bind(b);
			camera(p, b.position().add(3.4, 1.9, 2.5), b.position().add(0, 0.9, 2.5));
		}));
		shot(t + 92, "move_rope_bind");
		at(t + 96, () -> server(mc, p -> {
			ready(a);
			face(a, 0f, 50f);
			MultimouseMoves.trap(a);
			MultimouseMoves.trap(b);
			face(a, 0f, 0f);
			camera(p, origin.add(1.5, 2.2, 4.5), origin.add(0.8, 0.2, 2.0));
		}));
		shot(t + 102, "move_mouse_trap");
		t += 160;
		// Scurry (Mylene) beside full-size Marinette
		at(t, () -> server(mc, p -> {
			ready(b);
			MultimouseMoves.scurry(b);
			camera(p, origin.add(1.2, 1.3, 3.2), origin.add(1.2, 0.6, 0));
		}));
		shot(t + 10, "scurry");
		at(t + 12, () -> server(mc, p -> camera(p, b.position().add(0.7, 1.7, 1.3), b.position().add(0, 0.2, 0))));
		shot(t + 18, "scurry_close");
		t += 30;
		// Multitude (Marinette): ten copies -- the snares cleared first (a fresh transformation)
		at(t, () -> server(mc, p -> {
			MultimouseMoves.scurry(b);
			Miraculous.detransform(a, false);
			Miraculous.detransform(b, false);
		}));
		at(t + 25, () -> server(mc, p -> {
			transformNow(a);
			transformNow(b);
			b.teleportTo(p.serverLevel(), origin.x + 3.4, origin.y, origin.z - 1.0, 0f, 0f);
			ready(a);
			Multitude.start(a);
			camera(p, origin.add(0.4, 1.9, 2.4), origin.add(0, 0.15, 0));
		}));
		shot(t + 29, "multitude_split");
		shot(t + 60, "multitude");
		at(t + 62, () -> server(mc, p -> camera(p, origin.add(-1.5, 1.8, -1.5), origin.add(0, 0.15, 0))));
		shot(t + 68, "multitude_back");
		at(t + 70, () -> server(mc, p -> camera(p, origin.add(0.3, 1.7, 1.9), origin.add(0, 0.2, 0))));
		shot(t + 76, "multitude_close");
		at(t + 78, () -> server(mc, p -> camera(p, origin.add(0.4, 1.9, 2.4), origin.add(0, 0.15, 0))));
		t += 20;
		at(t + 60, () -> server(mc, p -> Multitude.recombine(a)));
		shot(t + 70, "recombine");
		t += 100;
		// the HUD (the Multitude card) and Mouse Hearing, first person
		at(t, () -> {
			mc.options.hideGui = false;
			mc.options.guiScale().set(2);
			server(mc, p -> {
				p.getInventory().clearContent();
				Miraculous.grant(p, Kwami.MULLO);
				transformNow(p);
				ServerLevel level = p.serverLevel();
				Vec3 w = origin.add(-3, 0, -8);
				for (int x = -3; x <= 3; x++) {
					for (int y = 0; y <= 3; y++) {
						level.setBlockAndUpdate(BlockPos.containing(w.x + x, w.y + y, w.z), Blocks.STONE.defaultBlockState());
					}
				}
				for (EntityType<? extends Mob> type : java.util.List.of(EntityType.PIG, EntityType.COW, EntityType.SHEEP)) {
					Mob m = type.create(level);
					m.moveTo(w.x - 1.5 + level.random.nextInt(3), w.y, w.z - 2.0, 0f, 0f);
					m.setNoAi(true);
					level.addFreshEntity(m);
				}
				camera(p, origin.add(-3, 1.7, -3), origin.add(-3, 1.0, -10));
				MultimouseMoves.hearing(p);
			});
		});
		shot(t + 12, "hearing_hud");
		at(t + 14, () -> server(mc, p -> {
			ready(p);
			Multitude.start(p);
		}));
		shot(t + 50, "multitude_hud");
		at(t + 54, () -> server(mc, p -> {
			Miraculous.revoke(p);
			p.getInventory().setItem(0, new ItemStack(MiraculousItems.MOUSE_PENDANT));
			p.getInventory().setItem(1, new ItemStack(MiraculousItems.CHEESE_WEDGE, 4));
			p.getInventory().selected = 0;
		}));
		shot(t + 64, "items_hotbar");
		at(t + 70, () -> {
			System.out.println("[HARNESS] DONE");
			Runtime.getRuntime().halt(0);
		});
	}
}
