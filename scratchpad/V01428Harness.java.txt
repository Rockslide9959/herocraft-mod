package com.projecthero.mod.client;

import java.nio.file.Files;
import java.nio.file.Path;
import java.util.HashMap;
import java.util.Map;

import com.projecthero.mod.ironman.IronManBlocks;

/** TEMP-HARNESS (not shipped): v0.14.28 suit-up poses / suit-down / send-home visual check. Enabled by run/v01428.flag. */
public final class V01428Harness {
	private static boolean createdWorld;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static Vec3 base;

	private V01428Harness() {
	}

	public static void init() {
		if (Files.exists(Path.of("v01428.flag"))) {
			ClientTickEvents.END_CLIENT_TICK.register(V01428Harness::tick);
		}
	}

	private static ServerPlayer sp(Minecraft mc) {
		MinecraftServer srv = mc.getSingleplayerServer();
		return srv == null || srv.getPlayerList().getPlayers().isEmpty() ? null : srv.getPlayerList().getPlayers().get(0);
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				ServerPlayer p = sp(mc);
				if (p != null) {
					try {
						fn.accept(p);
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		Screenshot.grab(mc.gameDirectory, "v01428_" + name + ".png", mc.getMainRenderTarget(), c -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	private static void at(int t, Runnable r) {
		Runnable prev = AT.get(t);
		AT.put(t, prev == null ? r : () -> {
			prev.run();
			r.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			if (!createdWorld && mc.screen instanceof TitleScreen) {
				createdWorld = true;
				LevelSettings settings = new LevelSettings("V01428" + System.currentTimeMillis() % 100000,
						GameType.SURVIVAL, false, Difficulty.PEACEFUL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
				mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, WorldOptions.defaultWithRandomSeed(),
						reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
								.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
						mc.screen);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.THIRD_PERSON_FRONT);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				base = Vec3.atBottomCenterOf(p.blockPosition());
				TonyStark.grant(p);
			});
			t0 = inWorldTicks;
			buildScript(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	private static void suitUp(Minecraft mc) {
		server(mc, p -> {
			p.teleportTo(p.serverLevel(), base.x, base.y, base.z, 30f, 0f);
			for (EquipmentSlot s : new EquipmentSlot[] { EquipmentSlot.HEAD, EquipmentSlot.CHEST, EquipmentSlot.LEGS, EquipmentSlot.FEET }) {
				p.setItemSlot(s, ItemStack.EMPTY);
			}
			p.getInventory().clearContent();
			for (ArmorItem.Type t : new ArmorItem.Type[] { ArmorItem.Type.HELMET, ArmorItem.Type.CHESTPLATE,
					ArmorItem.Type.LEGGINGS, ArmorItem.Type.BOOTS }) {
				p.getInventory().add(new ItemStack(IronManItems.armor("mark_iii", t)));
			}
			System.out.println("[HARNESS] suit-up started=" + IronManSuitUpManager.beginSuitUp(p, "mark_iii"));
		});
	}

	private static void buildScript(Minecraft mc) {
		int t = 20;
		at(t, () -> suitUp(mc));
		// boots (sequence ticks 1..60), leggings (61..120), chestplate (121..180), helmet (181..240); done at 241
		for (int m : new int[] { 20, 40, 95, 155, 190, 228, 243, 248 }) {
			at(t + 1 + m, () -> shot(mc, "up_t" + String.format("%03d", m)));
		}
		t += 270;
		at(t, () -> server(mc, p -> System.out.println("[HARNESS] suit-down started="
				+ IronManSuitUpManager.beginSuitDown(p, "mark_iii"))));
		for (int m : new int[] { 30, 50, 90, 150, 210, 247 }) {
			at(t + m, () -> shot(mc, "down_t" + String.format("%03d", m)));
		}
		t += 270;
		// send home: the platform 12 blocks ahead and to the side, the camera behind the player
		at(t, () -> {
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			server(mc, p -> {
				p.teleportTo(p.serverLevel(), base.x, base.y, base.z, 0f, 15f);
				BlockPos plat = BlockPos.containing(base).offset(5, 0, 12);
				p.serverLevel().setBlockAndUpdate(plat, IronManBlocks.IRON_MAN_SUIT_PLATFORM.defaultBlockState());
				if (p.serverLevel().getBlockEntity(plat) instanceof IronManSuitPlatformBlockEntity be) {
					be.bindTo(p.getUUID());
				}
				System.out.println("[HARNESS] platform at " + plat);
			});
		});
		at(t + 10, () -> server(mc, p -> {
			System.out.println("[HARNESS] send-home sent=" + IronManSuitCall.sendBack(p, "mark_iii"));
			// turn side-on so the camera (in front of the player) sees the suit standing beside them
			p.teleportTo(p.serverLevel(), base.x, base.y, base.z, 70f, 10f);
		}));
		at(t + 11, () -> mc.options.setCameraType(CameraType.THIRD_PERSON_FRONT));
		for (int m : new int[] { 20, 40, 60, 80, 93, 98, 103, 110, 140 }) {
			at(t + 10 + m, () -> shot(mc, "home_t" + String.format("%03d", m)));
		}
		at(t + 160, () -> server(mc, p -> {
			BlockPos plat = BlockPos.containing(base).offset(5, 0, 12);
			if (p.serverLevel().getBlockEntity(plat) instanceof IronManSuitPlatformBlockEntity be) {
				System.out.println("[HARNESS] docked: helmet=" + be.holds("mark_iii", ArmorItem.Type.HELMET)
						+ " boots=" + be.holds("mark_iii", ArmorItem.Type.BOOTS));
			}
			int armour = 0;
			for (ItemStack s : p.getInventory().items) {
				if (s.getItem() instanceof com.projecthero.mod.ironman.item.IronManArmorItem) {
					armour++;
				}
			}
			System.out.println("[HARNESS] armour left in pack=" + armour);
		}));
		at(t + 180, () -> {
			System.out.println("[HARNESS] done");
			mc.stop();
		});
	}
}
