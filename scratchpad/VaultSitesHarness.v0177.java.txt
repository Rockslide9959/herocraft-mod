/** TEMP-HARNESS (not shipped). Kept as scratchpad/VaultSitesHarness.v0177.java.txt. */
public final class VaultSitesHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks, planTick;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static File out;
	/** The Vault: origin (rotation NONE, so world = origin + local); built over open air so the floors show from outside. */
	static final BlockPos VO = new BlockPos(0, 110, 0);
	static final int F2 = VaultBuilder.F2;
	static final int F3 = VaultBuilder.F3;

	record View(String name, double cx, double cy, double cz, double lx, double ly, double lz, boolean gui) {
	}

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("vaultsites.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(VaultSitesHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(600_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "vaultsites-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		mc.getToasts().clear();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			prev.run();
			r.run();
		});
	}

	static void look(Minecraft mc, Vec3 cam, Vec3 at, boolean gui) {
		mc.options.hideGui = !gui;
		Vec3 d = at.subtract(cam);
		float yaw = (float) Math.toDegrees(Math.atan2(-d.x, d.z));
		float pitch = (float) -Math.toDegrees(Math.atan2(d.y, Math.sqrt(d.x * d.x + d.z * d.z)));
		server(mc, p -> {
			p.getAbilities().mayfly = true;
			p.getAbilities().flying = true;
			p.onUpdateAbilities();
			p.connection.teleport(cam.x, cam.y - 1.62, cam.z, yaw, pitch);
		});
	}

	static Vec3 v(double x, double y, double z) {
		return new Vec3(VO.getX() + x, VO.getY() + y, VO.getZ() + z);
	}

	static void freezeNear(ServerPlayer p, boolean noAi) {
		for (Mob m : p.serverLevel().getEntitiesOfClass(Mob.class, p.getBoundingBox().inflate(90))) {
			m.setNoAi(noAi);
		}
	}

	static BlockPos siteCentre(SiteTheme t) {
		return new BlockPos(400 + t.ordinal() * 64, 100, 0);
	}

	static void prepareSite(ServerLevel level, BlockPos c, SiteTheme t) {
		BlockState top = switch (t.style) {
			case ICE -> Blocks.SNOW_BLOCK.defaultBlockState();
			case TEMPLE -> Blocks.SAND.defaultBlockState();
			default -> Blocks.GRASS_BLOCK.defaultBlockState();
		};
		for (int x = -22; x <= 22; x++) {
			for (int z = -24; z <= 22; z++) {
				for (int y = -4; y <= 16; y++) {
					level.setBlock(c.offset(x, y, z), y < 0 ? Blocks.STONE.defaultBlockState() : y == 0 ? top : Blocks.AIR.defaultBlockState(), 2 | 16);
				}
			}
		}
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(VaultSitesHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(80);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.renderDistance().set(8);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				p.setGameMode(GameType.CREATIVE);
				p.addEffect(new MobEffectInstance(MobEffects.NIGHT_VISION, 20 * 60 * 30, 0, false, false));
			});
			buildPlan(mc);
		}
		if (t < 50) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		} else if (!PLAN.isEmpty()) {
			Runtime.getRuntime().halt(0);
		}
	}

	private static void buildPlan(Minecraft mc) {
		int t = 1;
		// ================================================================ The Vault
		at(t, () -> look(mc, v(-40, 20, -40), v(0, 0, 0), false));
		at(t + 30, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			// a thin crust round the pad only, so the floors hang in the open air (a cutaway)
			for (int x = -14; x <= 14; x++) {
				for (int z = -14; z <= 14; z++) {
					level.setBlock(VO.offset(x, -1, z), Blocks.STONE.defaultBlockState(), 2 | 16);
					level.setBlock(VO.offset(x, 0, z), Blocks.GRASS_BLOCK.defaultBlockState(), 2 | 16);
				}
			}
			VaultPiece piece = new VaultPiece(VO, Rotation.NONE);
			piece.buildAll(level);
			System.out.println("[HARNESS] vault built layout=" + piece.layout() + " box=" + piece.getBoundingBox());
		}));
		int k = t + 90;
		List<View> vault = List.of(
				new View("vault_1_cutaway_side", -48, -14, -40, -6, -30, -30, false),
				new View("vault_2_cutaway_new_floors", 34, -12, -72, -2, -28, -44, false),
				new View("vault_3_maxsec_corridor", -0.5, F2 + 3.2, -13.5, -0.5, F2 + 1.4, -23, true),
				new View("vault_4_laser_grid", 4.5, F2 + 5.2, -25.5, -1, F2 + 0.5, -32, true),
				new View("vault_5_maxsec_hall", 0.5, F2 + 6.6, -35.6, 0.5, F2 + 1, -46, false),
				new View("vault_6_riot_captain", 0.5, F2 + 2.7, -39.5, 0.5, F2 + 2.2, -44, false),
				new View("vault_7_pods", -5.5, F2 + 3, -38, -9, F2 + 1.5, -45, false),
				new View("vault_8_control_room", 6.5, F2 + 3.8, -53.5, -1, F2 + 1, -62, true),
				new View("vault_9_descent", -23, F2 - 4, -55.5, -23, F2 - 12, -44, false));
		for (View vw : vault) {
			int wait = vw.name.contains("cutaway") ? 120 : 50;
			at(k, () -> look(mc, v(vw.cx, vw.cy, vw.cz), v(vw.lx, vw.ly, vw.lz), vw.gui));
			at(k + 20, () -> server(mc, p -> freezeNear(p, true)));
			if (vw.name.contains("riot_captain")) {
				at(k + 24, () -> server(mc, p -> {
					for (RiotCaptainBoss rc : p.serverLevel().getEntitiesOfClass(RiotCaptainBoss.class, p.getBoundingBox().inflate(40))) {
						Vec3 h = v(0.5, F2 + 1, -43.5);
						rc.moveTo(h.x, h.y, h.z, 180f, 0f);
						rc.setYHeadRot(180f);
						rc.setYBodyRot(180f);
						System.out.println("[HARNESS] riot captain placed, scale " + rc.getScale());
					}
				}));
			}
			at(k + wait, () -> shot(mc, vw.name));
			k += wait + 4;
		}
		// the power cut, seen from the pods
		at(k, () -> server(mc, p -> {
			BlockPos b = BlockPos.containing(v(VaultBuilder.BREAKER.getX(), VaultBuilder.BREAKER.getY(), VaultBuilder.BREAKER.getZ()));
			System.out.println("[HARNESS] power down: " + VaultSecurity.shutDown(p.serverLevel(), b, p));
			freezeNear(p, true);
		}));
		at(k + 2, () -> look(mc, v(-5.5, F2 + 3, -38), v(-9, F2 + 1.5, -45), false));
		at(k + 40, () -> shot(mc, "vault_10_power_down_pods_open"));
		k += 46;
		// ================================================================ Kross phase 2: the Breach Slam, filmed; then the gas
		final WardenKrossBoss[] kross = new WardenKrossBoss[1];
		final Husk[] dummy = new Husk[1];
		at(k, () -> look(mc, v(0.5, F3 + 8, 16), v(0.5, F3 + 1, 3), false));
		at(k + 40, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			freezeNear(p, true);
			var found = level.getEntitiesOfClass(WardenKrossBoss.class, p.getBoundingBox().inflate(60));
			System.out.println("[HARNESS] kross found: " + found.size());
			if (found.isEmpty()) {
				return;
			}
			WardenKrossBoss b = found.get(0);
			b.moveTo(v(0, F3 + 1, 10).x, v(0, F3 + 1, 10).y, v(0, F3 + 1, 10).z, 180f, 0f);
			Husk d = EntityType.HUSK.create(level);
			Vec3 dp = v(0.5, F3 + 1, -1.5);
			d.moveTo(dp.x, dp.y, dp.z, 0f, 0f);
			d.setNoAi(true);
			d.setInvulnerable(true);
			level.addFreshEntity(d);
			b.setNoAi(false);
			b.setHealth(b.getMaxHealth() * 0.5f + 2f);
			b.hurt(level.damageSources().playerAttack(p), 4f);
			b.setTarget(d);
			kross[0] = b;
			dummy[0] = d;
		}));
		at(k + 52, () -> server(mc, p -> {
			if (kross[0] != null) {
				kross[0].setTarget(dummy[0]);
				kross[0].startSlam();
			}
		}));
		for (int i = 0; i < 44; i++) {
			final int n = i;
			at(k + 53 + i, () -> shot(mc, String.format("film_kross_breach_slam_%03d", n)));
		}
		at(k + 100, () -> server(mc, p -> {
			if (kross[0] != null) {
				kross[0].setTarget(dummy[0]);
				kross[0].startGas();
			}
		}));
		at(k + 140, () -> shot(mc, "vault_11_kross_gas_volley"));
		at(k + 150, () -> look(mc, v(-6, F3 + 3, 2), v(0, F3 + 2, 8), true));
		at(k + 175, () -> server(mc, p -> {
			if (kross[0] != null) {
				kross[0].setNoAi(true);
			}
		}));
		at(k + 180, () -> shot(mc, "vault_12_kross_phase2_bar"));
		k += 190;
		// ================================================================ the 19 Tier 2 sites
		for (SiteTheme theme : SiteTheme.values()) {
			BlockPos c = siteCentre(theme);
			at(k, () -> look(mc, Vec3.atCenterOf(c).add(-12, 8, 15), Vec3.atCenterOf(c).add(0, 1, -3), false));
			at(k + 30, () -> server(mc, p -> {
				ServerLevel level = p.serverLevel();
				prepareSite(level, c, theme);
				SiteUpgrade.buildAll(level, new SiteAnnexPiece(theme, c.getX(), c.getY() + 1, c.getZ(), 0, 0x51717L * (theme.ordinal() + 1)));
			}));
			at(k + 50, () -> server(mc, p -> freezeNear(p, true)));
			at(k + 90, () -> shot(mc, "site_" + String.format("%02d", theme.ordinal() + 1) + "_" + theme.id));
			k += 96;
		}
		// close-ups of the new rooms
		record Close(SiteTheme theme, String name, double cx, double cy, double cz, double lx, double ly, double lz, boolean openDoor) {
		}
		List<Close> closes = List.of(
				new Close(SiteTheme.GOVERNMENT_SITE, "close_1_hall_captain_panel_door", -4.5, 2.6, 3.5, 3, 1.4, -5, false),
				new Close(SiteTheme.GAMMA_LAB, "close_2_vault_room", 2.5, 2.5, -4, 3, 1, -11, true),
				new Close(SiteTheme.TEMPLE_OF_KHONSHU, "close_3_shrine_back_room", -1.5, 2.8, -6.5, -4, 1.4, -11, false),
				new Close(SiteTheme.RESEARCH_FACILITY, "close_4_specimen_room", -1.5, 2.8, -6.5, -4.5, 1.4, -11, false),
				new Close(SiteTheme.GRAVEYARD, "close_5_ward_stone_tripwire", -3.5, 2.4, -1.5, -3.5, 1, -6, false));
		for (Close cl : closes) {
			BlockPos c = siteCentre(cl.theme);
			Vec3 base = new Vec3(c.getX() + 0.5, c.getY() + 1, c.getZ() + 0.5);
			at(k, () -> server(mc, p -> {
				ServerLevel level = p.serverLevel();
				if (cl.openDoor) {
					BlockPos door = new BlockPos(c.getX() + 2, c.getY() + 1, c.getZ() - 5);
					var list = com.projecthero.mod.alarm.block.BlastDoorBlock.connected(level, door);
					for (BlockPos q : list) {
						com.projecthero.mod.alarm.block.BlastDoorBlock.setOpen(level, q,
								level.getBlockState(q).setValue(com.projecthero.mod.alarm.block.BlastDoorBlock.LOCKED, false), true);
					}
				}
			}));
			at(k + 2, () -> look(mc, base.add(cl.cx, cl.cy, cl.cz), base.add(cl.lx, cl.ly, cl.lz), true));
			at(k + 50, () -> shot(mc, cl.name));
			k += 56;
		}
		at(k + 10, () -> {
		});
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("VAULTSITES_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17077L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
