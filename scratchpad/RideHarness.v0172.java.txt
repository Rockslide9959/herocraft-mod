/** TEMP-HARNESS (not shipped): v0.17.2 ride screenshots. Enabled by run/ride.flag. */
// TEMP-HARNESS
public final class RideHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static Vec3 origin;
	private static final java.util.List<ServerPlayer> FAKES = new java.util.concurrent.CopyOnWriteArrayList<>();
	/** entity id -> the body yaw the client should draw (the fakes are never ticked by a connection). */
	private static final Map<Integer, Float> YAW = new ConcurrentHashMap<>();
	private static volatile java.util.function.Consumer<ServerPlayer> mover;

	private RideHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("ride.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(RideHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(500_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "ride-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (x, y) -> () -> {
			x.run();
			y.run();
		});
	}

	private static void shot(int t, String name) {
		at(t - 1, () -> {
			try {
				var m = net.minecraft.client.particle.ParticleEngine.class.getDeclaredMethod("clearParticles");
				m.setAccessible(true);
				m.invoke(Minecraft.getInstance().particleEngine);
			} catch (ReflectiveOperationException e) {
				System.out.println("[HARNESS] " + e);
			}
		});
		at(t, () -> {
			Minecraft mc = Minecraft.getInstance();
			Screenshot.grab(mc.gameDirectory, "v0172_" + name + ".png", mc.getMainRenderTarget(), m -> {
			});
			System.out.println("[HARNESS] cam " + mc.player.position() + " yaw " + mc.player.getYRot());
			for (Entity e : mc.level.entitiesForRendering()) {
				if (e instanceof net.minecraft.world.entity.player.Player pl && pl != mc.player) {
					System.out.println("[HARNESS]   " + pl.getName().getString() + " at " + pl.position() + " body " + pl.yBodyRot + " head "
							+ pl.getYHeadRot() + " vehicle " + (pl.getVehicle() == null ? "-" : pl.getVehicle().getName().getString()));
				}
			}
			System.out.println("[HARNESS] shot " + name);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.options.getSoundSourceOptionInstance(net.minecraft.sounds.SoundSource.MASTER).get() > 0.05) {
			mc.options.getSoundSourceOptionInstance(net.minecraft.sounds.SoundSource.MASTER).set(0.05);
		}
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(RideHarness::createWorld);
			}
			return;
		}
		// the client draws each fake's body at the yaw we asked for
		for (Map.Entry<Integer, Float> e : YAW.entrySet()) {
			Entity en = mc.level.getEntity(e.getKey());
			if (en instanceof LivingEntity le && !le.isPassenger()) {
				le.yBodyRot = e.getValue();
				le.yBodyRotO = e.getValue();
				le.setYHeadRot(e.getValue());
				le.yHeadRotO = e.getValue();
				le.setYRot(e.getValue());
				le.yRotO = e.getValue();
			}
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			mc.options.chatVisibility().set(net.minecraft.world.entity.player.ChatVisiblity.HIDDEN);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.PEACEFUL, true);
				p.setGameMode(GameType.CREATIVE);
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			AT.put(3, () -> script(mc));
		}
		if (t0 >= 0) {
			server(mc, p -> {
				for (ServerPlayer f : FAKES) {
					f.tickCount++;
					Omnitrix.tick(f);
				}
				var m = mover;
				if (m != null) {
					m.accept(p);
				}
			});
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("RIDE_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.PEACEFUL, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}

	private static void camera(ServerPlayer p, Vec3 eye, Vec3 target) {
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		p.teleportTo(p.serverLevel(), eye.x, eye.y - p.getEyeHeight(), eye.z, yaw, pit);
	}

	private static ServerPlayer fake(ServerPlayer near, String name, Vec3 at, float yaw) {
		ServerLevel level = near.serverLevel();
		GameProfile profile = new GameProfile(java.util.UUID.randomUUID(), name);
		CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
		ServerPlayer f = new ServerPlayer(level.getServer(), level, profile, cookie.clientInformation());
		Connection conn = new Connection(PacketFlow.SERVERBOUND);
		new io.netty.channel.embedded.EmbeddedChannel(conn);
		level.getServer().getPlayerList().placeNewPlayer(conn, f, cookie);
		f.setGameMode(GameType.SURVIVAL);
		face(f, at, yaw);
		FAKES.add(f);
		return f;
	}

	private static void face(ServerPlayer f, Vec3 at, float yaw) {
		f.teleportTo(f.serverLevel(), at.x, at.y, at.z, yaw, 0f);
		f.setYHeadRot(yaw);
		f.yBodyRot = yaw;
		f.yBodyRotO = yaw;
		YAW.put(f.getId(), yaw);
	}

	private static void become(ServerPlayer f, String alien) {
		var n = Omnitrix.state(f).copy();
		n.rechargeUntil = 0;
		n.selected = alien;
		Omnitrix.save(f, n);
		Omnitrix.transform(f);
		f.setOnGround(true);
	}

	private static void script(Minecraft mc) {
		int t = 10;
		Vec3 wStage = origin.add(-24, 0, 0);
		Vec3 gStage = origin.add(-24, 0, 12);
		ServerPlayer[] wild = new ServerPlayer[1], rider = new ServerPlayer[1], host = new ServerPlayer[1], gm = new ServerPlayer[1];
		at(t, () -> server(mc, p -> {
			p.setGameMode(GameType.SPECTATOR);
			wild[0] = fake(p, "Ben", wStage, 180f);
			UniquePowers.forced(() -> Omnitrix.grant(wild[0]));
			rider[0] = fake(p, "Gwen", wStage.add(1.5, 0, 0), 180f);
			host[0] = fake(p, "Kevin", gStage, 180f);
			gm[0] = fake(p, "Galvan", gStage.add(1.5, 0, 0), 180f);
			UniquePowers.forced(() -> Omnitrix.grant(gm[0]));
		}));
		at(t + 10, () -> server(mc, p -> {
			become(wild[0], "wildmutt");
			face(wild[0], wStage, 180f);
			become(gm[0], "grey_matter");
			face(gm[0], gStage.add(1.5, 0, 0), 180f);
		}));
		at(t + 20, () -> server(mc, p -> {
			System.out.println("[HARNESS] mount wildmutt " + OmnitrixRiding.mountWildmutt(rider[0], wild[0]));
			System.out.println("[HARNESS] shoulder " + OmnitrixRiding.shoulder(gm[0], host[0], false));
		}));
		t += 40;
		// Wildmutt faces 180 (north, -Z): his front is -Z
		at(t, () -> server(mc, p -> camera(p, wStage.add(-2.4, 2.0, -3.0), wStage.add(0, 1.3, -0.2))));
		shot(t + 12, "ride_wildmutt_three_quarter");
		at(t + 13, () -> server(mc, p -> camera(p, wStage.add(-3.6, 1.6, -0.2), wStage.add(0, 1.4, -0.2))));
		shot(t + 18, "ride_wildmutt_side");
		at(t + 19, () -> server(mc, p -> camera(p, wStage.add(2.2, 2.4, 2.8), wStage.add(0, 1.5, 0))));
		shot(t + 24, "ride_wildmutt_back");
		at(t + 25, () -> server(mc, p -> camera(p, wStage.add(-1.3, 2.3, -1.0), wStage.add(0, 1.9, 0.1))));
		shot(t + 30, "ride_wildmutt_close");
		// galloping north: the bob
		at(t + 31, () -> {
			mover = p -> {
				ServerPlayer w = wild[0];
				Vec3 np = w.position().add(0, 0, -0.32);
				face(w, np, 180f);
				camera(p, np.add(-3.6, 1.6, -0.2), np.add(0, 1.4, -0.2));
			};
		});
		shot(t + 47, "ride_wildmutt_gallop_a");
		shot(t + 50, "ride_wildmutt_gallop_b");
		at(t + 51, () -> mover = null);
		t += 60;
		// Grey Matter on Kevin's right shoulder (Kevin faces north, -Z: his right is -X)
		// (facing north his RIGHT is east, +X)
		at(t, () -> server(mc, p -> camera(p, gStage.add(1.0, 1.8, -1.5), gStage.add(0.25, 1.4, 0))));
		shot(t + 10, "ride_greymatter_front");
		at(t + 11, () -> server(mc, p -> camera(p, gStage.add(1.5, 1.6, 0.0), gStage.add(0.37, 1.5, 0))));
		shot(t + 18, "ride_greymatter_side");
		at(t + 19, () -> server(mc, p -> camera(p, gStage.add(0.9, 2.0, 1.4), gStage.add(0.3, 1.5, 0))));
		shot(t + 26, "ride_greymatter_back");
		at(t + 27, () -> server(mc, p -> camera(p, gStage.add(0.75, 1.65, -0.75), gStage.add(0.37, 1.55, 0))));
		shot(t + 34, "ride_greymatter_close");
		t += 6;
		// Kevin turns to face east (yaw -90, +X): Grey Matter stays on his right shoulder (now on the south side)
		at(t + 29, () -> server(mc, p -> {
			face(host[0], gStage, -90f);
			// his own client turns his view with the carrier (OmnitrixRiding.carryRider); the fake has none -- do it here
			gm[0].setYRot(-90f);
			gm[0].setYHeadRot(-90f);
			camera(p, gStage.add(1.8, 1.9, 1.6), gStage.add(0, 1.5, 0.3));
		}));
		shot(t + 40, "ride_greymatter_turned");
		at(t + 50, () -> {
			System.out.println("[HARNESS] DONE");
			Runtime.getRuntime().halt(0);
		});
	}
}
