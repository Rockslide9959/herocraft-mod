/**
 * TEMP-HARNESS (not shipped, v0.17.5 Sentry & the Void): builds a little stage at spawn, photographs the Sentry suit and the Void
 * Form front and back, the HUD with both meters, each big move, films A Million Suns, Erase and a takeover (the body attacking while
 * the ring eats the ground), the Shame Rooms screen, the golden flash and kneel, the eaten ground and the ground grown back. Enabled
 * by run/sentry.flag = "<out dir>". Kept as scratchpad/SentryHarness.v0175.java.txt.

public final class SentryHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static File out;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static int planTick;
	private static BlockPos site;
	private static ArmorStand camera;
	private static Vec3 camAt, camLook;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("sentry.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(SentryHarness::tick);
		ClientTickEvents.START_CLIENT_TICK.register(mc -> placeCamera(mc));
		// the level never draws the local player unless it is the camera: draw it ourselves for the fixed camera
		net.fabricmc.fabric.api.client.rendering.v1.WorldRenderEvents.AFTER_ENTITIES.register(ctx -> {
			Minecraft mc = Minecraft.getInstance();
			if (camAt == null || mc.player == null || ctx.consumers() == null) {
				return;
			}
			float pt = ctx.tickCounter().getGameTimeDeltaPartialTick(false);
			var p = mc.player;
			Vec3 cam = ctx.camera().getPosition();
			double x = net.minecraft.util.Mth.lerp(pt, p.xo, p.getX()) - cam.x, y = net.minecraft.util.Mth.lerp(pt, p.yo, p.getY()) - cam.y,
					z = net.minecraft.util.Mth.lerp(pt, p.zo, p.getZ()) - cam.z;
			var disp = mc.getEntityRenderDispatcher();
			disp.render(p, x, y, z, net.minecraft.util.Mth.lerp(pt, p.yRotO, p.getYRot()), pt, ctx.matrixStack(), ctx.consumers(), disp.getPackedLightCoords(p, pt));
		});
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(720_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "sentry-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		mc.getToasts().clear();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		if (!name.startsWith("film_")) {
			System.out.println("[HARNESS] shot " + name);
		}
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			prev.run();
			r.run();
		});
	}

	private static Vec3 s(double x, double y, double z) {
		return new Vec3(site.getX() + 0.5 + x, site.getY() + 1 + y, site.getZ() + 0.5 + z);
	}

	/** A fixed camera at site-relative {@code at}, looking at {@code look}. */
	private static void fixed(Minecraft mc, Vec3 at, Vec3 look) {
		if (camera == null || camera.level() != mc.level) {
			camera = new ArmorStand(EntityType.ARMOR_STAND, mc.level);
			camera.setInvisible(true);
		}
		camAt = at;
		camLook = look;
		mc.options.setCameraType(CameraType.FIRST_PERSON);
		mc.setCameraEntity(camera);
		placeCamera(mc);
	}

	private static void playerView(Minecraft mc, CameraType type) {
		camAt = null;
		mc.setCameraEntity(mc.player);
		mc.options.setCameraType(type);
	}

	private static void placeCamera(Minecraft mc) {
		if (camAt == null || camera == null) {
			return;
		}
		Vec3 to = camLook.subtract(camAt);
		float yaw = (float) Math.toDegrees(Math.atan2(-to.x, to.z));
		float pitch = (float) -Math.toDegrees(Math.atan2(to.y, Math.sqrt(to.x * to.x + to.z * to.z)));
		double eye = camera.getEyeHeight();
		camera.moveTo(camAt.x, camAt.y - eye, camAt.z, yaw, pitch);
		camera.xo = camera.getX();
		camera.yo = camera.getY();
		camera.zo = camera.getZ();
		camera.xOld = camera.getX();
		camera.yOld = camera.getY();
		camera.zOld = camera.getZ();
		camera.yRotO = yaw;
		camera.xRotO = pitch;
		camera.setYHeadRot(yaw);
	}

	private static void edit(ServerPlayer p, Consumer<SentryState> c) {
		SentryState n = Sentry.state(p).copy();
		c.accept(n);
		p.setAttached(SentryAttachments.STATE, n);
	}

	private static void press(Minecraft mc, AbilitySlot slot, boolean shift, boolean down) {
		server(mc, p -> {
			p.setShiftKeyDown(shift);
			SentryAbilityManager.handle(p, slot, down);
			p.setShiftKeyDown(false);
		});
	}

	/** Puts the player at site (x, z), facing yaw, on the ground. */
	private static void stand(Minecraft mc, double x, double z, float yaw) {
		server(mc, p -> {
			p.getAbilities().flying = false;
			p.onUpdateAbilities();
			p.connection.teleport(site.getX() + 0.5 + x, site.getY() + 1, site.getZ() + 0.5 + z, yaw, 0f);
		});
	}

	private static Mob dummy(ServerLevel level, EntityType<? extends Mob> type, double x, double z, boolean ai) {
		Mob m = type.create(level);
		m.moveTo(site.getX() + 0.5 + x, site.getY() + 1, site.getZ() + 0.5 + z, 180f, 0f);
		m.setNoAi(!ai);
		m.setPersistenceRequired();
		m.setItemSlot(net.minecraft.world.entity.EquipmentSlot.HEAD, new net.minecraft.world.item.ItemStack(net.minecraft.world.item.Items.LEATHER_HELMET));
		if (m.getAttribute(net.minecraft.world.entity.ai.attributes.Attributes.MAX_HEALTH) != null) {
			m.getAttribute(net.minecraft.world.entity.ai.attributes.Attributes.MAX_HEALTH).setBaseValue(400);
			m.setHealth(400);
		}
		level.addFreshEntity(m);
		return m;
	}

	private static void clearMobs(ServerLevel level) {
		for (Mob m : level.getEntitiesOfClass(Mob.class, new AABB(site).inflate(48))) {
			m.discard();
		}
	}

	static void prepare(ServerLevel level) {
		for (int x = -24; x <= 24; x++) {
			for (int z = -24; z <= 24; z++) {
				for (int y = -4; y <= 24; y++) {
					BlockState st = y < -1 ? Blocks.STONE.defaultBlockState() : y < 0 ? Blocks.DIRT.defaultBlockState()
							: y == 0 ? Blocks.GRASS_BLOCK.defaultBlockState() : Blocks.AIR.defaultBlockState();
					level.setBlock(site.offset(x, y, z), st, 2 | 16);
				}
			}
		}
		// a small stone-brick hut with a chest (the Void skips containers), a tree, a path
		for (int x = 6; x <= 11; x++) {
			for (int z = -11; z <= -6; z++) {
				for (int y = 1; y <= 4; y++) {
					boolean wall = x == 6 || x == 11 || z == -11 || z == -6;
					boolean door = z == -6 && x == 8 && y <= 2;
					if (y == 4) {
						level.setBlock(site.offset(x, y, z), Blocks.OAK_PLANKS.defaultBlockState(), 2 | 16);
					} else if (wall && !door) {
						level.setBlock(site.offset(x, y, z), Blocks.STONE_BRICKS.defaultBlockState(), 2 | 16);
					}
				}
			}
		}
		level.setBlock(site.offset(9, 1, -9), Blocks.CHEST.defaultBlockState(), 3);
		for (int y = 1; y <= 5; y++) {
			level.setBlock(site.offset(-8, y, -8), Blocks.OAK_LOG.defaultBlockState(), 2 | 16);
		}
		for (int x = -10; x <= -6; x++) {
			for (int z = -10; z <= -6; z++) {
				for (int y = 4; y <= 6; y++) {
					if (level.getBlockState(site.offset(x, y, z)).isAir() && Math.abs(x + 8) + Math.abs(z + 8) + (y - 4) < 5) {
						level.setBlock(site.offset(x, y, z), Blocks.OAK_LEAVES.defaultBlockState().setValue(net.minecraft.world.level.block.LeavesBlock.PERSISTENT, true), 2 | 16);
					}
				}
			}
		}
		for (int x = -20; x <= 20; x++) {
			level.setBlock(site.offset(x, 0, 3), Blocks.DIRT_PATH.defaultBlockState(), 2 | 16);
		}
		for (int i = 0; i < 18; i++) {
			int x = -18 + (i * 7) % 36, z = 8 + (i * 5) % 12;
			level.setBlock(site.offset(x, 1, z), (i % 2 == 0 ? Blocks.POPPY : Blocks.DANDELION).defaultBlockState(), 2 | 16);
		}
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(SentryHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(75);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.renderDistance().set(5);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(com.projecthero.mod.loot.finds.SupplyDrops.RULE).set(false, p.server);
				p.setGameMode(GameType.CREATIVE);
				BlockPos sp = p.blockPosition();
				int y = level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, sp.getX(), sp.getZ());
				site = new BlockPos(sp.getX(), y - 1, sp.getZ());
				prepare(level);
				com.projecthero.mod.hero.PowerGrants.grantHero(p, "sentry");
				p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 20 * 60 * 30, 4, false, false));
				System.out.println("[HARNESS] site " + site + " sentry=" + Sentry.hasPower(p));
			});
		}
		if (t == 45 && site == null) {
			return;
		}
		if (t == 46) {
			buildPlan(mc);
		}
		if (t < 50 || site == null) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		}
	}

				s.nightfallUntil = 0L;
			});
			p.connection.teleport(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5, 180f, 0f);
			for (int i = 0; i < 5; i++) {
				double a = i * Math.PI * 2 / 5.0;
				Mob m = dummy(level, i % 2 == 0 ? EntityType.ZOMBIE : EntityType.PIG, Math.cos(a) * 9, Math.sin(a) * 9, true);
				m.setHealth(400f);
			}
		}));
		at(t + 5, () -> {
			playerView(mc, CameraType.THIRD_PERSON_BACK);
			mc.options.hideGui = false;
		});
		at(t + 10, () -> server(mc, p -> {
			// (night vision for the camera only: midnight around the Void is otherwise nearly black on film)
			p.addEffect(new MobEffectInstance(MobEffects.NIGHT_VISION, 20 * 60 * 5, 0, false, false));
			Sentry.setVoid(p, 100f);
		}));
		for (int f = 0; f < 220; f++) {
			final int n = f;
			at(t + 12 + f, () -> shot(mc, String.format("film_takeover_%03d", n)));
		}
		at(t + 120, () -> shot(mc, "20_takeover_view"));
		// later: the eaten ground from above (a wide fixed camera, the HUD's takeover view is the owner's own)
		at(t + 400, () -> {
			fixed(mc, s(-16, 14, 16), s(0, 0, 0));
			mc.options.hideGui = true;
		});
		at(t + 430, () -> shot(mc, "21_takeover_eating_terrain"));
		// the Shame Rooms: open the mind early (the harness skips the 30 s wait)
		at(t + 440, () -> {
			playerView(mc, CameraType.THIRD_PERSON_BACK);
			mc.options.hideGui = false;
			server(mc, p -> edit(p, s -> {
				long now = p.level().getGameTime();
				s.mindLockedUntil = now;
				s.roomStartedAt = now;
			}));
		});
		at(t + 470, () -> shot(mc, "22_shame_rooms"));
		at(t + 520, () -> shot(mc, "23_shame_rooms_later"));
		// a friend reaching in is not filmable with one client: throw the Void out by command, the golden flash and the kneel
		at(t + 530, () -> {
			mc.setScreen(null);
			server(mc, p -> {
				VoidTakeover.end(p, VoidTakeover.End.COMMAND);
				p.connection.teleport(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5, 180f, 0f);
			});
		});
		at(t + 534, () -> fixed(mc, s(-3, 2.0, -3.5), s(0, 0.8, 0)));
		at(t + 537, () -> shot(mc, "24_thrown_out_flash"));
		at(t + 560, () -> shot(mc, "25_kneel"));
		at(t + 575, () -> {
			fixed(mc, s(-16, 14, 16), s(0, 0, 0));
			mc.options.hideGui = true;
			server(mc, p -> clearMobs(p.serverLevel()));
		});
		at(t + 600, () -> shot(mc, "26_eaten_ground_after"));
		// the world grows back about a minute after control returns
		at(t + 530 + 1200 + 140, () -> shot(mc, "27_ground_growing_back"));
		at(t + 530 + 1200 + 420, () -> shot(mc, "28_ground_restored"));
		at(t + 530 + 1200 + 430, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("SENTRY_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17051L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
