// v0.17.7 Iron Man I-page harness: the full suit page (IronManSuitInfoScreen) for several suits at several scroll points, the
// unsuited Tony Stark page (PowerInfoScreen) and the Guidebook's Iron Man chapter. To use: copy to
// src/client/java/com/projecthero/mod/client/IronManInfoHarness.java, add "com.projecthero.mod.client.IronManInfoHarness" to
// fabric.mod.json client entrypoints, write run/iminfo.flag = <out dir>. Remove both from src before committing.
package com.projecthero.mod.client;

import java.io.File;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;
import java.util.function.Consumer;

import com.projecthero.mod.client.gui.IronManSuitInfoScreen;
import com.projecthero.mod.ironman.IronManEnergy;
import com.projecthero.mod.ironman.TonyStark;
import com.projecthero.mod.ironman.item.IronManItems;

import net.fabricmc.api.ClientModInitializer;
import net.fabricmc.fabric.api.client.event.lifecycle.v1.ClientTickEvents;
import net.minecraft.client.Minecraft;
import net.minecraft.client.Screenshot;
import net.minecraft.client.gui.screens.TitleScreen;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.Difficulty;
import net.minecraft.world.entity.EquipmentSlot;
import net.minecraft.world.item.ArmorItem;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.GameRules;
import net.minecraft.world.level.GameType;
import net.minecraft.world.level.LevelSettings;
import net.minecraft.world.level.WorldDataConfiguration;
import net.minecraft.world.level.levelgen.WorldOptions;
import net.minecraft.world.level.levelgen.presets.WorldPresets;

/** TEMP-HARNESS (not shipped). Kept as scratchpad/IronManInfoHarness.v0177.java.txt. */
public final class IronManInfoHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks, planTick;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static File out;
	private static final EquipmentSlot[] SLOTS = { EquipmentSlot.HEAD, EquipmentSlot.CHEST, EquipmentSlot.LEGS, EquipmentSlot.FEET };
	private static final ArmorItem.Type[] TYPES = { ArmorItem.Type.HELMET, ArmorItem.Type.CHESTPLATE, ArmorItem.Type.LEGGINGS,
			ArmorItem.Type.BOOTS };

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("iminfo.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(IronManInfoHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(300_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "iminfo-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		mc.getToasts().clear();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			try {
				prev.run();
				r.run();
			} catch (Throwable e) {
				System.out.println("[HARNESS] step failed: " + e);
				e.printStackTrace();
			}
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(IronManInfoHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.guiScale().set(2);
			mc.resizeDisplay();
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
		}
		if (t == 46) {
			buildPlan(mc);
		}
		if (t < 50) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		} else if (PLAN.size() > 0) {
			Runtime.getRuntime().halt(0);
		}
	}

	private static void wear(ServerPlayer p, String suit) {
		for (int i = 0; i < 4; i++) {
			p.setItemSlot(SLOTS[i], suit == null ? ItemStack.EMPTY : new ItemStack(IronManItems.armor(suit, TYPES[i])));
		}
		if (suit != null) {
			TonyStark.setActiveSuit(p, suit);
			IronManEnergy.setIntegrity(p, suit, IronManEnergy.maxIntegrity(suit));
		}
	}

	private static void page(Minecraft mc, int t, String suit, float[] fractions) {
		at(t, () -> {
			mc.setScreen(null);
			server(mc, p -> wear(p, suit));
		});
		at(t + 12, () -> mc.setScreen(new IronManSuitInfoScreen()));
		for (int i = 0; i < fractions.length; i++) {
			float f = fractions[i];
			int n = i;
			at(t + 16 + i * 4, () -> {
				if (mc.screen instanceof IronManSuitInfoScreen s) {
					s.scrollFraction(f);
				}
			});
			at(t + 18 + i * 4, () -> shot(mc, suit + "_" + n));
		}
	}

	private static void buildPlan(Minecraft mc) {
		int t = 10;
		at(t, () -> server(mc, p -> {
			TonyStark.grant(p);
			System.out.println("[HARNESS] tony=" + TonyStark.hasPower(p));
		}));
		// unsuited Tony Stark: the I page
		at(t + 20, () -> mc.setScreen(com.projecthero.mod.client.gui.PowerInfoScreen.create()));
		at(t + 26, () -> shot(mc, "unsuited_0"));
		at(t + 28, () -> {
			if (mc.screen instanceof com.projecthero.mod.client.gui.PowerInfoScreen s) {
				s.scrollFraction(0.04f);
			}
		});
		at(t + 32, () -> shot(mc, "unsuited_1"));
		t += 40;
		page(mc, t, "mark_1", new float[] { 0f, 0.2f, 0.4f, 0.6f, 0.8f, 1f });
		t += 50;
		for (String s : new String[] { "mark_iii", "mark_2", "mark_6", "mark_9", "mark_8" }) {
			page(mc, t, s, new float[] { 0f, 0.35f, 0.7f });
			t += 36;
		}
		int done = t;
		at(done, () -> {
			mc.setScreen(null);
			System.out.println("[HARNESS] done");
		});
		at(done + 4, () -> {
		});
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("IMINFO_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.PEACEFUL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17051L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
