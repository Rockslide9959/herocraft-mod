/**
 * TEMP-HARNESS (not shipped, v0.17.5 gear tiers): the three armour sets on armour stands (front, back, night glow,
 * close-ups), item icons + tooltips, a Grapple Line film and a Celestial Warhammer slam film (frames per tick).
 * Enabled by run/gear.flag = "<out dir>". Install: copy to src/client/java/com/projecthero/mod/client/GearHarness.java
 * and add it to the "client" entrypoints in fabric.mod.json. Remove both before committing.
 */
public final class GearHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static BlockPos base;
	private static File out;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("gear.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(GearHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(480_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "gear-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	private static void at(Minecraft mc, double dx, double dy, double dz, float yaw, float pitch) {
		server(mc, p -> p.connection.teleport(base.getX() + 0.5 + dx, base.getY() + dy, base.getZ() + 0.5 + dz, yaw, pitch));
	}

	/** Teleports the player and aims at a point. */
	private static void aimAt(Mc m, double px, double pz, Vec3 target) {
		Minecraft mc = m.mc;
		Vec3 eye = new Vec3(base.getX() + 0.5 + px, base.getY() + 1.62, base.getZ() + 0.5 + pz);
		Vec3 d = target.subtract(eye);
		float yaw = (float) (Math.toDegrees(Math.atan2(d.z, d.x)) - 90.0);
		float pitch = (float) -Math.toDegrees(Math.atan2(d.y, Math.hypot(d.x, d.z)));
		at(mc, px, 0, pz, yaw, pitch);
	}

	private record Mc(Minecraft mc) {
	}

	private static final Item[][] SETS = {
			{ GearItems.HERO_ALLOY_HELMET, GearItems.HERO_ALLOY_CHESTPLATE, GearItems.HERO_ALLOY_LEGGINGS, GearItems.HERO_ALLOY_BOOTS, GearItems.HERO_ALLOY_SWORD, GearItems.ALLOY_BATON },
			{ GearItems.VIBRANIUM_HELMET, GearItems.VIBRANIUM_CHESTPLATE, GearItems.VIBRANIUM_LEGGINGS, GearItems.VIBRANIUM_BOOTS, GearItems.VIBRANIUM_BLADE, GearItems.VIBRANIUM_GRAPPLE },
			{ GearItems.CELESTIAL_HELMET, GearItems.CELESTIAL_CHESTPLATE, GearItems.CELESTIAL_LEGGINGS, GearItems.CELESTIAL_BOOTS, GearItems.CELESTIAL_WARHAMMER, null } };

	private static void stage(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		level.setDayTime(6000L);
		level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
		p.server.setDifficulty(Difficulty.EASY, true);
		p.setGameMode(GameType.SURVIVAL);
		base = p.blockPosition();
		for (int x = -10; x <= 10; x++) {
			for (int z = -6; z <= 22; z++) {
				level.setBlock(base.offset(x, -1, z), Blocks.SMOOTH_STONE.defaultBlockState(), 3);
				for (int y = 0; y < 14; y++) {
					level.setBlock(base.offset(x, y, z), Blocks.AIR.defaultBlockState(), 3);
				}
			}
		}
		// the grapple target: a tall pillar with a ledge, 16 blocks south-east of the stage
		for (int y = 0; y < 9; y++) {
			for (int x = 6; x <= 8; x++) {
				for (int z = 18; z <= 20; z++) {
					level.setBlock(base.offset(x, y, z), Blocks.POLISHED_DEEPSLATE.defaultBlockState(), 3);
				}
			}
		}
		for (int i = 0; i < 3; i++) {
			ArmorStand s = new ArmorStand(EntityType.ARMOR_STAND, level);
			s.moveTo(base.getX() + 0.5 + (i - 1) * 2.5, base.getY(), base.getZ() + 4.5, 180f, 0f);
			CompoundTag tag = new CompoundTag();
			s.saveWithoutId(tag);
			tag.putBoolean("ShowArms", true);
			tag.putBoolean("NoBasePlate", true);
			s.load(tag);
			s.setYRot(180f);
			s.setYBodyRot(180f);
			s.setYHeadRot(180f);
			Item[] set = SETS[i];
			s.setItemSlot(EquipmentSlot.HEAD, new ItemStack(set[0]));
			s.setItemSlot(EquipmentSlot.CHEST, new ItemStack(set[1]));
			s.setItemSlot(EquipmentSlot.LEGS, new ItemStack(set[2]));
			s.setItemSlot(EquipmentSlot.FEET, new ItemStack(set[3]));
			s.setItemSlot(EquipmentSlot.MAINHAND, new ItemStack(set[4]));
			if (set[5] != null) {
				s.setItemSlot(EquipmentSlot.OFFHAND, new ItemStack(set[5]));
			}
			level.addFreshEntity(s);
		}
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(GearHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 40) {
			mc.options.fov().set(70);
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, GearHarness::stage);
		}
		if (t < 90 || base == null) {
			return;
		}
		int s = t - 90;
		Mc m = new Mc(mc);
		// ---- the three sets: front, back, night, close-ups
		if (s == 0) {
			mc.options.hideGui = true;
			at(mc, 0, 0, 0.3, 0f, 12f);
		}
		if (s == 25) {
			shot(mc, "01_sets_front");
			at(mc, 0, 0, 8.7, 180f, 12f);
		}
		if (s == 50) {
			shot(mc, "02_sets_back");
		}
		String[] names = { "alloy", "vibranium", "celestial" };
		for (int i = 0; i < 3; i++) {
			int b = 55 + i * 40;
			double x = (i - 1) * 2.5;
			if (s == b) {
				at(mc, x, 0, 2.4, 0f, 15f);
			}
			if (s == b + 15) {
				shot(mc, "03_" + names[i] + "_front_close");
				at(mc, x, 0, 6.6, 180f, 15f);
			}
			if (s == b + 32) {
				shot(mc, "04_" + names[i] + "_back_close");
			}
		}
		if (s == 180) {
			server(mc, p -> p.serverLevel().setDayTime(18000L));
			at(mc, 0, 0, 0.6, 0f, 12f);
		}
		if (s == 215) {
			shot(mc, "05_sets_front_night_celestial_glow");
			at(mc, 2.5, 0, 2.6, 0f, 12f);
		}
		if (s == 235) {
			shot(mc, "06_celestial_night_close");
			server(mc, p -> p.serverLevel().setDayTime(6000L));
		}
		// ---- icons + tooltips
		if (s == 240) {
			mc.options.hideGui = false;
			mc.setScreen(new ItemsScreen(0));
		}
		for (int i = 0; i < ItemsScreen.PICK.length; i++) {
			if (s == 246 + i * 6) {
				mc.setScreen(new ItemsScreen(i));
			}
			if (s == 250 + i * 6) {
				shot(mc, String.format("07_items_tooltip_%02d", i));
			}
		}
		int g0 = 250 + ItemsScreen.PICK.length * 6 + 6;
		// ---- grapple film (third person, real right-click through the use key)
		Vec3 ledge = new Vec3(base.getX() + 7.5, base.getY() + 9.0, base.getZ() + 18.4);
		if (s == g0) {
			mc.setScreen(null);
			mc.options.hideGui = true;
			server(mc, p -> {
				p.getInventory().clearContent();
				p.getInventory().setItem(0, new ItemStack(GearItems.GRAPPLE_LINE));
				p.getInventory().setItem(1, new ItemStack(GearItems.CELESTIAL_WARHAMMER));
				p.getInventory().selected = 0;
			});
			mc.player.getInventory().selected = 0;
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			aimAt(m, 0, 2, ledge);
		}
		if (s >= g0 + 30 && s < g0 + 30 + 70) {
			int f = s - (g0 + 30);
			if (f == 6) {
				KeyMapping.click(InputConstants.Type.MOUSE.getOrCreate(1));
			}
			if (f == 8) { // look across the line of travel so the cable reads (the pull keeps going)
				mc.player.setYRot(mc.player.getYRot() + 72f); // client-side look only: the client owns its rotation, no teleport
				mc.player.setXRot(12f);
			}
			shot(mc, String.format("grapple/f%03d", f));
		}
		if (s == g0 + 30 + 75) {
			System.out.println("[HARNESS] after grapple player at " + mc.player.position() + " ledge " + ledge);
		}
		// ---- celestial slam film
		int c0 = g0 + 110;
		if (s == c0) {
			server(mc, p -> {
				p.setItemSlot(EquipmentSlot.HEAD, new ItemStack(GearItems.CELESTIAL_HELMET));
				p.setItemSlot(EquipmentSlot.CHEST, new ItemStack(GearItems.CELESTIAL_CHESTPLATE));
				p.setItemSlot(EquipmentSlot.LEGS, new ItemStack(GearItems.CELESTIAL_LEGGINGS));
				p.setItemSlot(EquipmentSlot.FEET, new ItemStack(GearItems.CELESTIAL_BOOTS));
				p.getInventory().selected = 1;
				p.getCooldowns().removeCooldown(GearItems.CELESTIAL_WARHAMMER);
				ServerLevel level = p.serverLevel();
				for (Husk old : level.getEntitiesOfClass(Husk.class, p.getBoundingBox().inflate(40))) {
					old.discard();
				}
				double[][] spots = { { -2.5, 10 }, { 2.5, 10 }, { 0, 12.5 }, { -3.5, 13 }, { 3.5, 13 }, { 0, 17 } };
				for (double[] sp : spots) {
					Husk h = new Husk(EntityType.HUSK, level);
					h.moveTo(base.getX() + 0.5 + sp[0], base.getY(), base.getZ() + 0.5 + sp[1], 180f, 0f);
					h.setNoAi(true);
					h.setPersistenceRequired();
					level.addFreshEntity(h);
				}
			});
			mc.player.getInventory().selected = 1;
			at(mc, 0, 0, 8.5, 0f, 25f);
		}
		if (s >= c0 + 30 && s < c0 + 30 + 80) {
			int f = s - (c0 + 30);
			if (f == 5) {
				KeyMapping.click(InputConstants.Type.MOUSE.getOrCreate(1));
				mc.options.keyUse.setDown(true);
			}
			if (f == 42) {
				mc.options.keyUse.setDown(false);
			}
			if (f == 46) {
				server(mc, p -> {
					for (Husk h : p.serverLevel().getEntitiesOfClass(Husk.class, p.getBoundingBox().inflate(20))) {
						System.out.println("[HARNESS] husk " + String.format("%.1f", h.getHealth()) + " hp at " + h.position());
					}
				});
			}
			shot(mc, String.format("slam/f%03d", f));
		}
		if (s == c0 + 30 + 85) {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		}
	}

	/** Icons + names on the right, one tooltip (page {@code page}) on the left. */
	static final class ItemsScreen extends Screen {
		static final Item[] PICK = { GearItems.GRAPPLE_LINE, GearItems.SMOKE_BOMB, GearItems.FLASHBANG, GearItems.HERO_ALLOY_CHESTPLATE,
				GearItems.ALLOY_BATON, GearItems.UTILITY_BELT, GearItems.VIBRANIUM_HELMET, GearItems.VIBRANIUM_GRAPPLE,
				GearItems.CELESTIAL_CHESTPLATE, GearItems.CELESTIAL_WARHAMMER, GearItems.CELESTIAL_UPGRADE_TEMPLATE };
		private final int page;

		ItemsScreen(int page) {
			super(Component.literal("gear"));
			this.page = page;
		}

		@Override
		public void render(GuiGraphics g, int mouseX, int mouseY, float partial) {
			g.fill(0, 0, width, height, 0xFF1B1E24);
			List<Item> l = new ArrayList<>(GearItems.all());
			for (int i = 0; i < l.size(); i++) {
				int col = i / 12, row = i % 12;
				int x = width - 252 + col * 126, y = 6 + row * 18;
				ItemStack st = new ItemStack(l.get(i));
				g.renderItem(st, x, y);
				g.drawString(font, st.getHoverName(), x + 19, y + 4, 0xFFFFFF, true);
			}
			ItemStack tip = new ItemStack(PICK[page % PICK.length]);
			g.renderTooltip(font, Screen.getTooltipFromItem(minecraft, tip), tip.getTooltipImage(), -6, 22);
		}

		@Override
		public boolean isPauseScreen() {
			return false;
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("GEAR_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.EASY, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	static {
		LootItems.class.getName();
	}
}
