/** TEMP-HARNESS (not shipped). Kept as scratchpad/SymbioteBugsHarness.v0176.java.txt. */
public final class SymbioteMonsterHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks, planTick;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static BlockPos site;
	private static File out;
	public static volatile net.minecraft.world.phys.Vec3 camOffset;
	public static volatile double camLookY = 1.2;

	private static void cam(double x, double y, double z, double lookY, int fov) {
		camOffset = new net.minecraft.world.phys.Vec3(x, y, z);
		camLookY = lookY;
		Minecraft.getInstance().options.fov().set(fov);
	}

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("symmon.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(SymbioteMonsterHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(480_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "symbugs-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		mc.getToasts().clear();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			prev.run();
			r.run();
		});
	}

	private static float faceYaw = Float.NaN, facePitch;

	private static void face(Minecraft mc, float yaw, float pitch) {
		server(mc, p -> p.connection.teleport(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5, yaw, pitch));
		faceYaw = yaw;
		facePitch = pitch;
	}

	private static void openInfo(Minecraft mc, String name) {
		// exactly what the I key does (ProjectHeroModClient.handlePowerInfo)
		if (com.projecthero.mod.client.gui.IronManSuitInfoScreen.wornSuit(mc.player) != null) {
			mc.setScreen(new com.projecthero.mod.client.gui.IronManSuitInfoScreen());
		} else {
			mc.setScreen(com.projecthero.mod.client.gui.PowerInfoScreen.create());
		}
		StringBuilder sb = new StringBuilder();
		if (mc.screen != null) {
			for (var c : mc.screen.children()) {
				if (c instanceof AbstractWidget w) {
					sb.append('[').append(w.getMessage().getString()).append("] ");
				}
			}
		}
		System.out.println("[HARNESS] " + name + " screen=" + (mc.screen == null ? null : mc.screen.getClass().getSimpleName()) + " widgets=" + sb
				+ " normalHost=" + Symbiote.isNormalHost(mc.player) + " owns=" + Symbiote.ownsKeysOverPrimary(mc.player));
	}

	private static void charge(Minecraft mc) {
		server(mc, p -> {
			SymbioteVitalsManager.restoreBiomass(p);
			SymbioteState c = Symbiote.state(p).copy();
			c.abilityCooldowns.set(SymbioteMonster.CDI_MONSTER, 0L);
			p.setAttached(ModAttachments.SYMBIOTE_STATE, c);
			SymbioteMonster.handleCharge(p, true, p.level().getGameTime());
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(SymbioteMonsterHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(70);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(com.projecthero.mod.loot.finds.SupplyDrops.RULE).set(false, p.server);
				BlockPos sp = p.blockPosition();
				int y = level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, sp.getX(), sp.getZ());
				site = new BlockPos(sp.getX(), y - 1, sp.getZ());
				for (int x = -18; x <= 30; x++) {
					for (int z = -22; z <= 14; z++) {
						for (int yy = -2; yy <= 22; yy++) {
							BlockState st = yy < 0 ? Blocks.DIRT.defaultBlockState() : yy == 0 ? Blocks.GRASS_BLOCK.defaultBlockState() : Blocks.AIR.defaultBlockState();
							level.setBlock(site.offset(x, yy, z), st, 2 | 16);
						}
					}
				}
				p.setGameMode(GameType.SURVIVAL);
				p.connection.teleport(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5, 180f, 8f);
				p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 20 * 60 * 30, 4, false, false));
				p.addEffect(new MobEffectInstance(MobEffects.SATURATION, 20 * 60 * 30, 4, false, false));
			});
		}
		if (t == 46) {
			buildPlan(mc);
		}
		if (t < 50 || site == null) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		} else if (PLAN.size() > 0) {
			Runtime.getRuntime().halt(0);
		}
		if (!Float.isNaN(faceYaw)) {
			mc.player.setYRot(faceYaw);
			mc.player.setXRot(facePitch);
			mc.player.setYHeadRot(faceYaw);
			mc.player.yHeadRotO = faceYaw;
			mc.player.yBodyRot = faceYaw;
			mc.player.yBodyRotO = faceYaw;
		}
	}

	private static void buildPlan(Minecraft mc) {
		String mode = readMode();
		int t = 10;
		at(t, () -> mc.options.setCameraType(CameraType.THIRD_PERSON_BACK));
		if (!mode.equals("tongue")) {
			at(t, () -> server(mc, p -> System.out.println("[HARNESS] kryptonian=" + com.projecthero.mod.hero.PowerGrants.grantHero(p, "kryptonian").message().getString())));
			at(t + 10, () -> server(mc, p -> System.out.println("[HARNESS] symbiote=" + Symbiote.grant(p, false))));
			at(t + 30, () -> openInfo(mc, "i1_kryptonian_suit_off"));
			at(t + 36, () -> shot(mc, "i1_kryptonian_suit_off"));
			at(t + 38, () -> mc.setScreen(null));
			at(t + 40, () -> server(mc, Symbiote::toggle));
			at(t + 110, () -> openInfo(mc, "i2_kryptonian_suit_on"));
			at(t + 116, () -> shot(mc, "i2_kryptonian_suit_on"));
			at(t + 118, () -> mc.setScreen(null));
			at(t + 120, () -> server(mc, Symbiote::toggle));
			at(t + 180, () -> server(mc, p -> {
				com.projecthero.mod.hero.PowerGrants.revokeHeroTier(p, "kryptonian");
				System.out.println("[HARNESS] omnitrix=" + com.projecthero.mod.hero.PowerGrants.grantHero(p, "omnitrix").message().getString());
			}));
			at(t + 210, () -> openInfo(mc, "i3_omnitrix_suit_off"));
			at(t + 216, () -> shot(mc, "i3_omnitrix_suit_off"));
			at(t + 218, () -> mc.setScreen(null));
			at(t + 220, () -> server(mc, Symbiote::toggle));
			at(t + 290, () -> openInfo(mc, "i4_omnitrix_suit_on"));
			at(t + 296, () -> shot(mc, "i4_omnitrix_suit_on"));
			at(t + 298, () -> mc.setScreen(null));
			at(t + 300, () -> server(mc, p -> {
				Symbiote.toggle(p);
				com.projecthero.mod.hero.PowerGrants.revokeHeroTier(p, "omnitrix");
			}));
			t += 370;
		} else {
			at(t, () -> server(mc, p -> System.out.println("[HARNESS] symbiote=" + Symbiote.grant(p, false))));
			t += 20;
		}
		// ---- Monster face close-ups
		at(t, () -> server(mc, p -> SymbioteSuitColours.setPreset(p, SymbioteColourPreset.CLASSIC)));
		at(t + 2, () -> server(mc, p -> {
			if (!Symbiote.isActive(p)) {
				Symbiote.toggle(p);
			}
		}));
		at(t + 2, () -> mc.options.hideGui = true);
		at(t + 75, () -> charge(mc));
		int burst = t + 75 + 61 + 50;
		int s = burst + 60;
		at(s, () -> {
			face(mc, 180f, 8f);
			cam(0, 5.0, -6.0, 4.6, 40);
		});
		at(s + 6, () -> shot(mc, "t1_face_front"));
		at(s + 7, () -> cam(2.2, 5.2, -5.6, 4.6, 40));
		at(s + 13, () -> shot(mc, "t2_face_three_quarter"));
		at(s + 14, () -> cam(6.2, 4.6, -1.0, 4.2, 45));
		at(s + 20, () -> shot(mc, "t3_face_side"));
		at(s + 21, () -> cam(1.0, 3.2, -6.5, 4.2, 40));
		at(s + 27, () -> shot(mc, "t4_face_low"));
		at(s + 30, () -> {
		});
	}

	private static String readMode() {
		try {
			return Files.readString(Path.of("symbugs.mode")).trim();
		} catch (Exception e) {
			return "all";
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("SYMBUGS_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17051L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
