/**
 * TEMP-HARNESS (not shipped): Supervillain boss model screenshots. Enabled by run/svm.flag (its text = the shot prefix,
 * "old" or "new"). Needs SvCamMixin in projecthero.client.mixins.json and SvModelHarness.init() in onInitializeClient.
 */
public final class SvModelHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	public static Vec3 camPos;
	public static Vec3 camAt;
	private static BlockPos C;
	private static String prefix = "old";
	private static volatile long lastProgress = System.currentTimeMillis();
	/** NEW mode: plays a named clip on the boss (wired to the new animation API). */
	public static java.util.function.BiConsumer<EmpoweredZombie, String> PLAY = (b, n) -> {
	};
	/** NEW mode: the clips shown per boss, with the server tick to shoot at. */
	public static Object[][] MOVES = {};

	private SvModelHarness() {
	}

	public static void init() {
		Path flag = Path.of("svm.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			prefix = Files.readString(flag).trim();
		} catch (Exception e) {
			prefix = "old";
		}
		if (prefix.startsWith("new")) {
			PLAY = (b, n) -> b.playSupervillainAnim(com.projecthero.mod.event.entity.SupervillainAnims.byName(n));
			MOVES = new Object[][] { { "melee", 7 }, { "melee_alt", 7 }, { "shoot", 7 }, { "slam", 7 }, { "repel", 7 }, { "cast_2", 14 }, { "cast_2", 25 }, { "cast_4", 30 }, { "cast_4", 45 }, { "burst", 7 } };
		}
		ClientTickEvents.END_CLIENT_TICK.register(SvModelHarness::tick);
		Thread wd = new Thread(() -> {
			while (true) {
				try {
					Thread.sleep(5000);
				} catch (InterruptedException e) {
					return;
				}
				if (System.currentTimeMillis() - lastProgress > 120_000L) {
					System.out.println("[HARNESS] WATCHDOG: no progress for 120 s");
					Runtime.getRuntime().halt(3);
				}
			}
		}, "svm-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static ServerPlayer sp(Minecraft mc) {
		MinecraftServer srv = mc.getSingleplayerServer();
		return srv == null || srv.getPlayerList().getPlayers().isEmpty() ? null : srv.getPlayerList().getPlayers().get(0);
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				ServerPlayer p = sp(mc);
				if (p != null) {
					try {
						fn.accept(p);
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		lastProgress = System.currentTimeMillis();
		Screenshot.grab(mc.gameDirectory, "svm_" + prefix + "_" + name + ".png", mc.getMainRenderTarget(), c -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	private static void at(int t, Runnable r) {
		Runnable prev = AT.get(t);
		AT.put(t, prev == null ? r : () -> {
			prev.run();
			r.run();
		});
	}

	private static void cam(double x, double y, double z, double ax, double ay, double az) {
		camPos = Vec3.atBottomCenterOf(C).add(x, y, z);
		camAt = Vec3.atBottomCenterOf(C).add(ax, ay, az);
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0) {
				lastProgress = System.currentTimeMillis();
				if (!createdWorld && mc.getOverlay() == null && mc.screen != null && !(mc.screen instanceof TitleScreen)) {
					mc.options.onboardAccessibility = false;
					mc.setScreen(new TitleScreen());
				}
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				lastProgress = System.currentTimeMillis();
				mc.execute(() -> {
					LevelSettings settings = new LevelSettings("SVM_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
							Difficulty.PEACEFUL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
					mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, WorldOptions.defaultWithRandomSeed(),
							reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
									.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
							mc.screen);
				});
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks % 20 == 0) {
			lastProgress = System.currentTimeMillis();
		}
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(60);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.getServer().setDifficulty(Difficulty.NORMAL, true);
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				C = p.blockPosition();
				p.addEffect(new MobEffectInstance(MobEffects.INVISIBILITY, 999999, 0, false, false));
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				lastProgress = System.currentTimeMillis();
				try {
					r.run();
				} catch (Throwable e) {
					System.out.println("[HARNESS] error " + e);
					e.printStackTrace();
				}
			}
		}
	}

	private static BlockPos ground(ServerLevel level, BlockPos at) {
		return new BlockPos(at.getX(), level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, at.getX(), at.getZ()), at.getZ());
	}

	private static EmpoweredZombie spawn(ServerLevel level, SupervillainVariant v, BlockPos at, float yaw) {
		EmpoweredZombie m = RaidEntityTypes.EMPOWERED_ZOMBIE.create(level);
		BlockPos g = ground(level, at);
		m.moveTo(g.getX() + 0.5, g.getY(), g.getZ() + 0.5, yaw, 0f);
		m.finalizeSpawn(level, level.getCurrentDifficultyAt(g), MobSpawnType.COMMAND, null);
		m.configureAsSupervillain(v, "power_05_geokinesis", 1);
		m.setNoAi(true);
		m.setPersistenceRequired();
		level.addFreshEntity(m);
		m.setYRot(yaw);
		m.setYHeadRot(yaw);
		m.setYBodyRot(yaw);
		return m;
	}

	private static void clear(ServerPlayer p) {
		for (Entity e : p.serverLevel().getEntities((Entity) null, new AABB(C).inflate(80), e -> e instanceof EmpoweredZombie)) {
			e.discard();
		}
	}

	private static void weather(ServerPlayer p, long time, boolean rain) {
		ServerLevel l = p.serverLevel();
		l.setDayTime(time);
		if (rain) {
			l.setWeatherParameters(0, 6000, true, false);
		} else {
			l.setWeatherParameters(6000, 0, false, false);
		}
	}

	private static void script(Minecraft mc) {
		int t = 20;
		at(t, () -> server(mc, p -> {
			ServerLevel l = p.serverLevel();
			spawn(l, SupervillainVariant.CHIMERA, C.offset(-6, 0, 0), 0f);
			spawn(l, SupervillainVariant.ARSENAL, C, 0f);
			spawn(l, SupervillainVariant.OMEGA_MAGE, C.offset(6, 0, 0), 0f);
			p.teleportTo(l, C.getX(), C.getY() + 4, C.getZ() + 16, 0f, 0f);
		}));
		at(t + 4, () -> cam(0, 3.2, 15, 0, 2.2, 0));
		at(t + 60, () -> shot(mc, "00_lineup_day"));
		at(t + 62, () -> server(mc, p -> weather(p, 13400, false)));
		at(t + 100, () -> shot(mc, "00_lineup_dusk"));
		at(t + 102, () -> server(mc, p -> weather(p, 18000, false)));
		at(t + 130, () -> shot(mc, "00_lineup_night"));
		at(t + 132, () -> server(mc, p -> {
			weather(p, 6000, false);
			clear(p);
		}));
		t += 140;
		int i = 0;
		for (SupervillainVariant v : SupervillainVariant.values()) {
			final String name = String.format("%02d_%s", ++i, v.id());
			final EmpoweredZombie[] m = new EmpoweredZombie[1];
			final double h = v.scale() * 1.95;
			final double d = h * 1.25;
			at(t, () -> server(mc, p -> {
				m[0] = spawn(p.serverLevel(), v, C, 0f);
				p.teleportTo(p.serverLevel(), C.getX(), C.getY() + h, C.getZ() + d, 0f, 0f);
			}));
			at(t + 4, () -> cam(0, h * 0.55, d * 1.05, 0, h * 0.5, 0));
			at(t + 28, () -> shot(mc, name + "_0_intro"));
			at(t + 70, () -> shot(mc, name + "_a_front"));
			at(t + 72, () -> cam(d * 0.78, h * 0.62, d * 0.78, 0, h * 0.5, 0));
			at(t + 78, () -> shot(mc, name + "_b_threequarter"));
			at(t + 80, () -> cam(d * 1.05, h * 0.55, 0, 0, h * 0.5, 0));
			at(t + 86, () -> shot(mc, name + "_c_side"));
			at(t + 88, () -> cam(d * 0.78, h * 0.62, d * 0.78, 0, h * 0.5, 0));
			int tt = t + 90;
			for (Object[] mv : MOVES) {
				final String clip = (String) mv[0];
				final int fire = (Integer) mv[1];
				at(tt, () -> server(mc, p -> PLAY.accept(m[0], clip)));
				at(tt + fire, () -> shot(mc, name + "_d_" + clip + "_" + fire));
				com.projecthero.mod.event.entity.SupervillainAnims.Clip cl = com.projecthero.mod.event.entity.SupervillainAnims.byName(clip);
				tt += Math.max(fire + 6, cl == null ? 30 : cl.length() + 6);
			}
			final int dusk = tt;
			at(dusk, () -> server(mc, p -> weather(p, 13400, false)));
			at(dusk + 30, () -> shot(mc, name + "_e_dusk"));
			at(dusk + 31, () -> server(mc, p -> m[0].hurt(p.serverLevel().damageSources().genericKill(), 99999f)));
			at(dusk + 31 + 22, () -> shot(mc, name + "_f_death"));
			at(dusk + 60, () -> server(mc, p -> {
				weather(p, 6000, false);
				clear(p);
			}));
			t = dusk + 70;
		}
		final int end = t + 10;
		at(end, () -> {
			System.out.println("[HARNESS] DONE");
			mc.stop();
		});
	}
}
