/**
 * TEMP-HARNESS (not shipped), v0.17.3 Beast Boy HQ round (T-Rex, pterodactyl, snake, fly, hawk, bat): stills, a turntable, a
 * walk / flight sequence (take-off, climb, glide, bank, landing) and every move of each, plus the T-Rex and snake quick morphs, through
 * BeastFormArt. Variant of BeastFixHarness. Enabled by run/beasthq.flag (its text filters shots by substring, comma-separated).
 * Frames go to $BEASTART_OUT/screenshots as hq_<shot>_<frame>.png.
 * Hook: node scratchpad/beastboy/hq_harness_hook.js <worktree> [off].
 */
public final class BeastHQHarness {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks, t0 = -1;
	private static Vec3 origin;
	private static File outDir;
	private static String filter = "";

	record Shot(String name, String kind, BeastForm form, float yaw, int frames) {
	}

	private static final List<Shot> SHOTS = new ArrayList<>();
	private static int shotIdx = -1, shotTick;
	private static volatile Shot cur;
	private static volatile int curFrame;

	private static final BeastForm[] FORMS = { BeastForm.TREX, BeastForm.PTERODACTYL, BeastForm.HAWK, BeastForm.BAT, BeastForm.SNAKE, BeastForm.FLY };

	private BeastHQHarness() {
	}

	static float size(BeastForm f) {
		return switch (f) {
			case TREX -> 4.6f;
			case PTERODACTYL -> 2.8f;
			case HAWK -> 1.2f;
			case BAT -> 1.2f;
			case SNAKE -> 0.75f;
			case FLY -> 0.2f;
			default -> 1f;
		};
	}

	static boolean flyer(BeastForm f) {
		return f == BeastForm.PTERODACTYL || f == BeastForm.HAWK || f == BeastForm.BAT || f == BeastForm.FLY;
	}

	public static void init() {
		if (!Files.exists(Path.of("beasthq.flag"))) {
			return;
		}
		try {
			filter = Files.readString(Path.of("beasthq.flag")).trim();
		} catch (Exception e) {
			filter = "";
		}
		String out = System.getenv("BEASTART_OUT");
		outDir = out != null ? new File(out) : Minecraft.getInstance().gameDirectory;
		buildShots();
		ClientTickEvents.END_CLIENT_TICK.register(BeastHQHarness::tick);
		WorldRenderEvents.AFTER_ENTITIES.register(BeastHQHarness::render);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(1_200_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "beasthq-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void add(String name, String kind, BeastForm f, float yaw, int frames) {
		boolean ok = filter.isEmpty();
		for (String part : filter.split(",")) {
			if (!part.isBlank() && name.contains(part.trim())) {
				ok = true;
			}
		}
		if (ok) {
			SHOTS.add(new Shot(name, kind, f, yaw, frames));
		}
	}

	private static void buildShots() {
		for (BeastForm f : FORMS) {
			add("f34_" + f, "still", f, 35f, 1);
			add("side_" + f, "still", f, 90f, 1);
			if (flyer(f) && f != BeastForm.FLY) {
				add("f34air_" + f, "airstill", f, 35f, 1);
				add("topair_" + f, "airtop", f, 20f, 1);
				add("frontair_" + f, "airstill", f, 0f, 1);
			}
			add("turn_" + f, "turn", f, 0f, 72);
			add("seq_" + f, "seq", f, 60f, 150);
		}
		for (String m : new String[] { "bite", "roar", "swipe", "stomp" }) {
			add("move_TREX_" + m, "move:" + m, BeastForm.TREX, 55f, m.equals("roar") || m.equals("stomp") ? 30 : 22);
		}
		add("move_PTERODACTYL_dive", "airmove:dive", BeastForm.PTERODACTYL, 60f, 26);
		add("move_PTERODACTYL_screech", "move:screech", BeastForm.PTERODACTYL, 50f, 22);
		add("move_PTERODACTYL_grab", "airmove:grab", BeastForm.PTERODACTYL, 60f, 22);
		add("move_HAWK_strike", "airmove:strike", BeastForm.HAWK, 60f, 22);
		add("move_HAWK_screech", "move:screech", BeastForm.HAWK, 45f, 22);
		add("move_BAT_screech", "airmove:screech", BeastForm.BAT, 45f, 22);
		add("move_SNAKE_strike", "move:strike", BeastForm.SNAKE, 70f, 24);
		add("quick_TREX", "quick", BeastForm.TREX, 30f, 28);
		add("quick_SNAKE", "quick", BeastForm.SNAKE, 60f, 28);
		System.out.println("[HARNESS] shots=" + SHOTS.size());
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(BeastHQHarness::createWorld);
			}
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 60) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.setGameMode(GameType.CREATIVE);
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				BlockPos b = p.blockPosition();
				for (int x = -30; x <= 30; x++) {
					for (int z = -40; z <= 6; z++) {
						level.setBlockAndUpdate(b.offset(x, -1, z), (x + z & 1) == 0 ? Blocks.GRASS_BLOCK.defaultBlockState() : Blocks.MOSS_BLOCK.defaultBlockState());
					}
				}
				origin = new Vec3(b.getX() + 0.5, b.getY(), b.getZ() + 0.5);
			});
		}
		if (inWorldTicks == 100) {
			t0 = inWorldTicks;
		}
		if (t0 < 0 || origin == null) {
			return;
		}
		if (shotIdx < 0 || shotTick >= cur.frames() + 6) {
			shotIdx++;
			if (shotIdx >= SHOTS.size()) {
				System.out.println("[HARNESS] done");
				mc.stop();
				return;
			}
			cur = SHOTS.get(shotIdx);
			shotTick = 0;
			placeCamera(mc, cur);
		}
		mc.player.getAbilities().flying = true;
		curFrame = Math.max(0, shotTick - 6);
		if (shotTick >= 6) {
			String name = String.format("hq_%s_%03d.png", cur.name(), curFrame);
			Screenshot.grab(outDir, name, mc.getMainRenderTarget(), c -> {
			});
		}
		shotTick++;
	}

	private static Vec3 stage() {
		return origin.add(0, 0, -6);
	}

	private static void placeCamera(Minecraft mc, Shot s) {
		Vec3 st = stage();
		BeastForm f = s.form();
		float sz = size(f);
		float dist = sz * 0.95f + 0.4f;
		float mid = Math.max(f.height, sz * 0.25f) * 0.45f;
		float eye = mid + dist * 0.3f;
		if (f == BeastForm.FLY) {
			dist = 0.36f;
			mid = 0.1f;
			eye = 0.2f;
		}
		if (s.kind().equals("seq") && flyer(f)) {
			dist *= 1.45f;
			mid += sz * 0.35f;
			eye = mid + dist * 0.12f;
		} else if (s.kind().startsWith("air")) {
			mid += sz * 0.3f;
			eye = mid + dist * (s.kind().equals("airtop") ? 0.9f : 0.25f);
		} else if (s.kind().equals("quick")) {
			dist = f == BeastForm.SNAKE ? 3.6f : 3.0f;
			mid = 1.4f;
			eye = 2.0f;
		}
		float pitch = (float) Math.toDegrees(Math.atan2(eye - mid, dist));
		Vec3 cam = new Vec3(st.x, st.y + eye - 1.62, st.z + dist);
		server(mc, p -> {
			p.teleportTo(p.serverLevel(), cam.x, cam.y, cam.z, 180f, pitch);
			p.getAbilities().flying = true;
			p.onUpdateAbilities();
		});
	}

	/** Flight sequence keyframes: {airborne 0/1, vy, walkAmt, tailLag, altitude (blocks), extra yaw}. */
	private static float[] seqState(BeastForm f, int fr) {
		float sz = size(f);
		float maxAlt = f == BeastForm.FLY ? 0.2f : sz * 0.6f;
		if (fr < 22) {
			return new float[] { 0, 0, fr > 12 ? 0.4f : 0f, 0, 0, 0 };
		}
		if (fr < 34) {
			float k = (fr - 22) / 12f;
			return new float[] { 1, 0.3f, 0.3f, 0, maxAlt * 0.35f * k, 0 };
		}
		if (fr < 64) {
			float k = (fr - 34) / 30f;
			return new float[] { 1, 0.14f, 0.55f, 0, maxAlt * (0.35f + 0.5f * k), 0 };
		}
		if (fr < 92) {
			float k = (fr - 64) / 28f;
			return new float[] { 1, -0.12f, 0.95f, 0, maxAlt * (0.85f - 0.2f * k), 0 };
		}
		if (fr < 114) {
			float k = (fr - 92) / 22f;
			return new float[] { 1, 0f, 0.8f, 0.32f * Mth.sin(k * Mth.PI), maxAlt * 0.65f, 90f * k };
		}
		if (fr < 132) {
			float k = (fr - 114) / 18f;
			return new float[] { 1, -0.2f, 0.3f * (1 - k), 0, maxAlt * 0.65f * (1 - k), 90f };
		}
		return new float[] { 0, 0, 0, 0, 0, 90f };
	}

	private static BeastPose poseFor(Shot s, float time, int frame, float[] extra) {
		BeastPose b = new BeastPose();
		b.age = time;
		b.entityId = 5000 + shotIdx;
		BeastForm f = s.form();
		String k = s.kind();
		if (k.equals("airstill") || k.equals("airtop") || k.startsWith("airmove:")) {
			b.airborne = true;
			b.walkAmt = 0.6f;
			b.walkPos = time * 0.9f;
			b.vy = k.equals("airtop") ? -0.15f : 0.05f;
			extra[0] = size(f) * 0.3f;
		}
		if (k.startsWith("move:") || k.startsWith("airmove:")) {
			b.move = k.substring(k.indexOf(':') + 1);
			b.prog = Math.min(1f, frame / (float) (s.frames() - 4));
		}
		if (k.equals("seq")) {
			if (flyer(f)) {
				float[] q = seqState(f, frame);
				b.airborne = q[0] > 0.5f;
				b.vy = q[1];
				b.walkAmt = q[2];
				b.walkPos = time * 0.9f;
				b.tailLag = q[3];
				extra[0] = q[4];
				extra[1] = q[5];
			} else if (f == BeastForm.SNAKE) {
				float w = frame < 40 ? 0f : frame < 110 ? Math.min(1f, (frame - 40) / 15f) : Math.max(0f, 1f - (frame - 110) / 12f);
				b.walkAmt = w * 0.7f;
				b.walkPos = Math.max(0, frame - 40) * 0.55f;
				b.headYaw = Mth.sin(time * 0.05f) * 0.4f;
			} else {
				float w = frame < 70 ? 1f : Math.max(0f, 1f - (frame - 70) / 15f);
				b.walkAmt = w;
				b.walkPos = Math.min(frame, 77) * 0.9f;
				b.tailLag = frame < 60 ? 0.12f * Mth.sin(frame * 0.08f) : 0f;
				if (frame > 90) {
					b.headYaw = Mth.sin((frame - 90) * 0.09f) * 0.7f;
					b.headPitch = Mth.sin((frame - 90) * 0.05f) * 0.3f;
				}
			}
		}
		if (k.equals("turn")) {
			extra[1] = frame * 5f;
		}
		return b;
	}

	private static void render(WorldRenderContext ctx) {
		Minecraft mc = Minecraft.getInstance();
		Shot s = cur;
		if (s == null || origin == null || mc.level == null || ctx.consumers() == null || ctx.matrixStack() == null) {
			return;
		}
		PoseStack pose = ctx.matrixStack();
		MultiBufferSource buffers = ctx.consumers();
		Vec3 cam = ctx.camera().getPosition();
		float pt = ctx.tickCounter().getGameTimeDeltaPartialTick(false);
		float time = shotTick + pt;
		Vec3 st = stage();
		int light = LevelRenderer.getLightColor(mc.level, BlockPos.containing(st.add(0, 1, 2)));
		BeastForm f = s.form();
		if (s.kind().equals("quick")) {
			pose.pushPose();
			pose.translate(st.x - cam.x, st.y - cam.y, st.z - cam.z);
			pose.mulPose(Axis.YP.rotationDegrees(s.yaw()));
			BeastFormArt.renderQuickMorphAtFeet(f, Math.min(1f, curFrame / (float) (s.frames() - 2)), time, pose, buffers, light, 0xFFFFFFFF);
			pose.popPose();
			return;
		}
		float[] extra = new float[2];
		BeastPose b = poseFor(s, time, curFrame, extra);
		pose.pushPose();
		pose.translate(st.x - cam.x, st.y - cam.y + extra[0], st.z - cam.z);
		pose.mulPose(Axis.YP.rotationDegrees(s.yaw() + extra[1]));
		BeastFormArt.renderPose(f, b, pose, buffers, light, 0xFFFFFFFF);
		pose.popPose();
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("HQ_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.PEACEFUL, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}
}
