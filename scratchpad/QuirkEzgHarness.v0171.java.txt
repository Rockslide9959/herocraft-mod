	public static void init() {

			return;

			only = Files.readString(Path.of("quirkezg.flag")).trim();
		} catch (java.io.IOException e) {
			only = "";
		}
		ClientTickEvents.END_CLIENT_TICK.register(QuirkEzgHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(560_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "quirkezg-harness-watchdog");

		wd.start();

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				ServerPlayer local = srv.getPlayerList().getPlayers().stream().filter(sp -> sp != fake).findFirst().orElse(null);
				if (local != null) {
					try {
						fn.accept(local);
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (x, y) -> () -> {
			x.run();
			y.run();
		});
	}

	private static void shot(int t, String name) {
		at(t, () -> {
			Minecraft mc = Minecraft.getInstance();
			mc.gui.setOverlayMessage(net.minecraft.network.chat.Component.empty(), false);
			Screenshot.grab(mc.gameDirectory, "qezg_" + name + ".png", mc.getMainRenderTarget(), m -> {
			});
			System.out.println("[HARNESS] shot " + name);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(QuirkEzgHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			mc.options.chatVisibility().set(net.minecraft.world.entity.player.ChatVisiblity.HIDDEN);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(1000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.NORMAL, true);
				p.setGameMode(GameType.CREATIVE);
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
				for (int x = -14; x <= 14; x++) {
					for (int z = -14; z <= 14; z++) {
						for (int y = 1; y <= 3; y++) {
							level.setBlock(BlockPos.containing(origin.x + x, origin.y - y, origin.z + z),
									y == 1 ? Blocks.GRASS_BLOCK.defaultBlockState() : Blocks.DIRT.defaultBlockState(), 3);
						}
						for (int y = 0; y <= 6; y++) {
							level.setBlock(BlockPos.containing(origin.x + x, origin.y + y, origin.z + z), Blocks.AIR.defaultBlockState(), 3);
						}
					}
				}
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("QEZG_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}

	private static void camera(ServerPlayer p, Vec3 eye, Vec3 target) {
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		p.teleportTo(p.serverLevel(), eye.x, eye.y - 1.62, eye.z, yaw, pit);
	}

	private static ServerPlayer fake(ServerPlayer near, String name, Vec3 at) {
		ServerLevel level = near.serverLevel();
		GameProfile profile = new GameProfile(java.util.UUID.randomUUID(), name);
		CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
		ServerPlayer f = new ServerPlayer(level.getServer(), level, profile, cookie.clientInformation());
		Connection conn = new Connection(PacketFlow.SERVERBOUND);
		new io.netty.channel.embedded.EmbeddedChannel(conn);
		level.getServer().getPlayerList().placeNewPlayer(conn, f, cookie);
		f.setGameMode(GameType.SURVIVAL);
		f.teleportTo(level, at.x, at.y, at.z, 0f, 0f);
		f.setYHeadRot(0f);
		f.yBodyRot = 0f;
		return f;
	}

	private static void face(ServerPlayer f, Vec3 target) {
		f.lookAt(EntityAnchorArgument.Anchor.EYES, target);
		f.setYHeadRot(f.getYRot());
		f.yBodyRot = f.getYRot();
		f.yBodyRotO = f.getYRot();
		f.connection.teleport(f.getX(), f.getY(), f.getZ(), f.getYRot(), f.getXRot());
	}

	private static void suit(ServerPlayer f, Quirk q) {
		for (ArmorItem.Type t : QuirkSuits.Suit.of(q).pieces()) {
			f.setItemSlot(t.getSlot(), new ItemStack(QuirkSuits.piece(q, t)));
		}
	}

	private static Mob mob(ServerLevel level, EntityType<? extends Mob> type, Vec3 at, float yaw) {
		Mob m = type.create(level);
		m.moveTo(at.x, at.y, at.z, yaw, 0f);
		m.setYHeadRot(yaw);
		m.yBodyRot = yaw;
		m.setNoAi(false);
		try {
			for (String n : new String[] { "goalSelector", "targetSelector" }) {
				java.lang.reflect.Field fl = Mob.class.getDeclaredField(n);
				fl.setAccessible(true);
				((net.minecraft.world.entity.ai.goal.GoalSelector) fl.get(m)).removeAllGoals(g -> true);
			}
		} catch (ReflectiveOperationException e) {
			m.setNoAi(true);
		}
		m.setPersistenceRequired();
		m.getAttribute(Attributes.MAX_HEALTH).setBaseValue(120);
		m.setHealth(120);
		level.addFreshEntity(m);
		MOBS.add(m);
		return m;
	}

	private static void flatten(ServerLevel level) {
		for (int x = -14; x <= 14; x++) {
			for (int z = -14; z <= 14; z++) {
				level.setBlock(BlockPos.containing(origin.x + x, origin.y - 1, origin.z + z), Blocks.GRASS_BLOCK.defaultBlockState(), 3);
				for (int y = 0; y <= 4; y++) {
					level.setBlock(BlockPos.containing(origin.x + x, origin.y + y, origin.z + z), Blocks.AIR.defaultBlockState(), 3);
				}
			}
		}
	}

	private static void clearMobs() {
		for (Mob m : MOBS) {
			m.discard();
		}
		MOBS.clear();
	}

	private static void gui(Minecraft mc, boolean on) {

	private static void third(Minecraft mc) {
		mc.execute(() -> mc.options.setCameraType(CameraType.FIRST_PERSON));

	private static boolean want(String scene) {

	private static void reset(Minecraft mc, Quirk q, String name) {
		at(t, () -> server(mc, p -> {
			clearMobs();
			if (fake != null) {
				Quirks.revoke(fake);
				fake.teleportTo(origin.x, origin.y, origin.z);
			} else {
				fake = fake(p, name, origin);
			}
			com.projecthero.mod.hero.UniquePowers.forced(() -> Quirks.grant(fake, q));
			suit(fake, q);
			fake.teleportTo(fake.serverLevel(), origin.x, origin.y, origin.z, 0f, 0f);
			Quirks.edit(fake, n -> n.abilityReadyAt.clear());
		}));
		t += 4;
	}

		at(t + 98, () -> server(mc, p -> ErasureMoves.reelStart(p)));
		shot(t + 112, "er_fp_reel");
		at(t + 116, () -> server(mc, p -> {
			ErasureMoves.reelStop(p);
			Quirks.edit(p, n -> n.abilityReadyAt.clear());
			PlusUltra.set(p, 100f);
			com.projecthero.mod.quirk.QuirkKits.of(Quirk.ERASURE).ultimate(p);
		}));
		shot(t + 120, "er_fp_takedown_a");
		shot(t + 132, "er_fp_takedown_b");
		t += 170;
		at(t, () -> {
			gui(mc, false);
			server(mc, p -> {
				com.projecthero.mod.quirk.QuirkKits.of(Quirk.ERASURE).stopAll(p);
				Quirks.revoke(p);
				for (ArmorItem.Type ty : QuirkSuits.TYPES) {
					p.setItemSlot(ty.getSlot(), ItemStack.EMPTY);
				}
				p.setGameMode(GameType.CREATIVE);
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				clearMobs();
			});
		});
		t += 6;
	}
