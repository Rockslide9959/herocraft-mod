/**
 * TEMP-HARNESS (not shipped, v0.17.5 Symbiote suit): front / back / side / face / emblem shots of the suit, every preset, a custom
 * colour, the colour screen, first-person arms, and films of the suit flowing on, the sheen + veins, and the tendril / blade /
 * spikes in a non-default colour. Enabled by run/symsuit.flag = "<out dir>" (second line "before" = only the comparison shots).
 * Kept as scratchpad/SymbioteSuitHarness.v0175.java.txt.
 */
public final class SymbioteSuitHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static boolean beforeOnly;
	private static boolean filmsOnly;
	private static int waitTicks, inWorldTicks;
	private static File out;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static int planTick;
	private static BlockPos site;
	private static ArmorStand camera;
	private static Vec3 camAt, camLook;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("symsuit.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			List<String> lines = Files.readAllLines(flag);
			out = new File(lines.get(0).trim());
			beforeOnly = lines.size() > 1 && lines.get(1).trim().equals("before");
			filmsOnly = lines.size() > 1 && lines.get(1).trim().equals("films");
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(SymbioteSuitHarness::tick);
		ClientTickEvents.START_CLIENT_TICK.register(SymbioteSuitHarness::placeCamera);
		net.fabricmc.fabric.api.client.rendering.v1.WorldRenderEvents.AFTER_ENTITIES.register(ctx -> {
			Minecraft mc = Minecraft.getInstance();
			if (camAt == null || mc.player == null || ctx.consumers() == null) {
				return;
			}
			float pt = ctx.tickCounter().getGameTimeDeltaPartialTick(false);
			var p = mc.player;
			Vec3 cam = ctx.camera().getPosition();
			double x = net.minecraft.util.Mth.lerp(pt, p.xo, p.getX()) - cam.x, y = net.minecraft.util.Mth.lerp(pt, p.yo, p.getY()) - cam.y,
					z = net.minecraft.util.Mth.lerp(pt, p.zo, p.getZ()) - cam.z;
			var disp = mc.getEntityRenderDispatcher();
			disp.render(p, x, y, z, net.minecraft.util.Mth.lerp(pt, p.yRotO, p.getYRot()), pt, ctx.matrixStack(), ctx.consumers(),
					disp.getPackedLightCoords(p, pt));
		});
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(480_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "symsuit-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		mc.getToasts().clear();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		if (!name.startsWith("film_")) {
			System.out.println("[HARNESS] shot " + name);
		}
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			prev.run();
			r.run();
		});
	}

	private static Vec3 s(double x, double y, double z) {
		return new Vec3(site.getX() + 0.5 + x, site.getY() + 1 + y, site.getZ() + 0.5 + z);
	}

	private static void fixed(Minecraft mc, Vec3 at, Vec3 look) {
		if (camera == null || camera.level() != mc.level) {
			camera = new ArmorStand(EntityType.ARMOR_STAND, mc.level);
			camera.setInvisible(true);
		}
		followOff = null;
		camAt = at;
		camLook = look;
		mc.options.setCameraType(CameraType.FIRST_PERSON);
		mc.setCameraEntity(camera);
		placeCamera(mc);
	}

	private static Vec3 followOff, followLook;

	private static void follow(Minecraft mc, Vec3 off, Vec3 look) {
		fixed(mc, mc.player.position().add(off), mc.player.position().add(look));
		followOff = off;
		followLook = look;
	}

	private static void playerView(Minecraft mc, CameraType type) {
		followOff = null;
		camAt = null;
		mc.setCameraEntity(mc.player);
		mc.options.setCameraType(type);
	}

	private static void placeCamera(Minecraft mc) {
		if (mc.player != null && site != null) {
			mc.player.yBodyRot = mc.player.getYRot();
			mc.player.yBodyRotO = mc.player.getYRot();
		}
		if (followOff != null && mc.player != null) {
			camAt = mc.player.position().add(followOff);
			camLook = mc.player.position().add(followLook);
		}
		if (camAt == null || camera == null) {
			return;
		}
		Vec3 to = camLook.subtract(camAt);
		float yaw = (float) Math.toDegrees(Math.atan2(-to.x, to.z));
		float pitch = (float) -Math.toDegrees(Math.atan2(to.y, Math.sqrt(to.x * to.x + to.z * to.z)));
		double eye = camera.getEyeHeight();
		camera.moveTo(camAt.x, camAt.y - eye, camAt.z, yaw, pitch);
		camera.xo = camera.getX();
		camera.yo = camera.getY();
		camera.zo = camera.getZ();
		camera.xOld = camera.getX();
		camera.yOld = camera.getY();
		camera.zOld = camera.getZ();
		camera.yRotO = yaw;
		camera.xRotO = pitch;
		camera.setYHeadRot(yaw);
	}

	private static void stand(Minecraft mc, double x, double z, float yaw) {
		server(mc, p -> {
			p.getAbilities().flying = false;
			p.onUpdateAbilities();
			p.connection.teleport(site.getX() + 0.5 + x, site.getY() + 1, site.getZ() + 0.5 + z, yaw, 0f);
			p.setYHeadRot(yaw);
		});
	}

	private static void suitUp(ServerPlayer p, boolean instant) {
		Symbiote.state(p).toggleReadyAt = 0L;
		Symbiote.toggle(p);
		if (instant) {
			Symbiote.state(p).transformStartTick -= 10_000L;
			Symbiote.tick(p);
		}
	}

	static void prepare(ServerLevel level) {
		for (int x = -14; x <= 14; x++) {
			for (int z = -14; z <= 14; z++) {
				for (int y = -2; y <= 16; y++) {
					BlockState st = y < 0 ? Blocks.DIRT.defaultBlockState() : y == 0 ? Blocks.GRASS_BLOCK.defaultBlockState() : Blocks.AIR.defaultBlockState();
					level.setBlock(site.offset(x, y, z), st, 2 | 16);
				}
			}
		}
		for (int y = 1; y <= 5; y++) {
			level.setBlock(site.offset(-6, y, 8), Blocks.OAK_LOG.defaultBlockState(), 2 | 16);
		}
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(SymbioteSuitHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(70);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(com.projecthero.mod.loot.finds.SupplyDrops.RULE).set(false, p.server);
				BlockPos sp = p.blockPosition();
				int y = level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, sp.getX(), sp.getZ());
				site = new BlockPos(sp.getX(), y - 1, sp.getZ());
				prepare(level);
				p.setGameMode(GameType.SURVIVAL);
				Symbiote.grant(p);
				p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 20 * 60 * 30, 4, false, false));
				p.addEffect(new MobEffectInstance(MobEffects.SATURATION, 20 * 60 * 30, 4, false, false));
				System.out.println("[HARNESS] site " + site + " bonded=" + Symbiote.hasSymbiote(p) + " normal=" + Symbiote.isNormalHost(p));
			});
		}
		if (t == 45 && site == null) {
			return;
		}
		if (t == 46) {
			buildPlan(mc);
		}
		if (t < 50 || site == null) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		}
	}

	/** The player is ~2.7 blocks tall suited: aim the cameras at the chest (y 1.9) and the face (y 2.4). */

	private static void buildPlan(Minecraft mc) {
		int t = 10;
		at(t, () -> {
			stand(mc, 0, 0, 180f);
			mc.options.hideGui = true;
			server(mc, p -> suitUp(p, true));
		});
		t += 40;
		if (!filmsOnly) {
			views(mc, t, beforeOnly ? "before_" : "after_");
		}
		t += 120;
		if (beforeOnly) {
			at(t, () -> {
				System.out.println("[HARNESS] done");
				Runtime.getRuntime().halt(0);
			});
			return;
		}
		if (!filmsOnly) { // ---- every preset, front + back
		for (SymbioteColourPreset preset : SymbioteColourPreset.values()) {
			final SymbioteColourPreset pr = preset;
			at(t, () -> server(mc, p -> SymbioteSuitColours.setPreset(p, pr)));
			at(t + 1, () -> fixed(mc, s(0, 1.75, -3.1), s(0, 1.3, 0)));
			at(t + 10, () -> shot(mc, "preset_" + pr.id() + "_front"));
			at(t + 11, () -> fixed(mc, s(0, 1.75, 3.1), s(0, 1.3, 0)));
			at(t + 20, () -> shot(mc, "preset_" + pr.id() + "_back"));
			t += 22;
		}
		// ---- a custom colour: teal body, amber emblem, cyan eyes
		at(t, () -> server(mc, p -> SymbioteSuitColours.set(p, new SymbioteColours(0x0E6B60, 0xFFB000, 0x22E5FF))));
		at(t + 1, () -> fixed(mc, s(-1.6, 1.8, -2.7), s(0, 1.3, 0)));
		at(t + 12, () -> shot(mc, "custom_teal_amber_cyan_three_quarter"));
		at(t + 13, () -> fixed(mc, s(0, 2.25, -1.15), s(0, 2.22, 0)));
		at(t + 24, () -> shot(mc, "custom_face"));
		// night: the eye glow
		at(t + 26, () -> server(mc, p -> p.serverLevel().setDayTime(18000L)));
		at(t + 27, () -> fixed(mc, s(0, 2.1, -2.0), s(0, 1.9, 0)));
		at(t + 40, () -> shot(mc, "custom_night_eye_glow"));
		at(t + 42, () -> server(mc, p -> {
			p.serverLevel().setDayTime(18000L);
			SymbioteSuitColours.setPreset(p, SymbioteColourPreset.CLASSIC);
		}));
		at(t + 52, () -> shot(mc, "classic_night_eye_glow"));
		at(t + 54, () -> server(mc, p -> p.serverLevel().setDayTime(6000L)));
		t += 60;
		// ---- the colour screen (suit worn: the live 3D preview) and the power page's button
		at(t, () -> {
			playerView(mc, CameraType.FIRST_PERSON);
			mc.options.hideGui = false;
			server(mc, p -> SymbioteSuitColours.set(p, new SymbioteColours(0x0E6B60, 0xFFB000, 0x22E5FF)));
		});
		at(t + 6, () -> mc.setScreen(com.projecthero.mod.client.gui.PowerInfoScreen.create()));
		at(t + 16, () -> shot(mc, "screen_power_info_suit_colours_button"));
		at(t + 18, () -> mc.setScreen(new com.projecthero.mod.client.gui.SymbioteColourScreen(null)));
		at(t + 30, () -> shot(mc, "screen_colour_3d_preview"));
		t += 34;
		// ---- first person: the sleeve in a custom colour, then with the blade out
		at(t, () -> {
			mc.setScreen(null);
			playerView(mc, CameraType.FIRST_PERSON);
			mc.options.hideGui = false;
		});
		at(t + 10, () -> shot(mc, "first_person_arm_custom"));
		at(t + 12, () -> server(mc, p -> SymbioteSuitColours.setPreset(p, SymbioteColourPreset.CLASSIC)));
		at(t + 22, () -> shot(mc, "first_person_arm_classic"));
		t += 26;
		} // !filmsOnly
		// ---- FILM: the suit flowing off and on (Classic), front camera
		at(t, () -> {
			mc.options.hideGui = true;
			stand(mc, 0, 0, 180f);
			follow(mc, new Vec3(-1.3, 1.8, -2.8), new Vec3(0, 1.3, 0));
			server(mc, p -> suitUp(p, false)); // retract
		});
		for (int f = 0; f < 34; f++) {
			final int n = f;
			at(t + 1 + f, () -> shot(mc, String.format("film_flow_%03d", n)));
		}
		at(t + 40, () -> server(mc, p -> suitUp(p, false))); // on again
		for (int f = 0; f < 52; f++) {
			final int n = 34 + f;
			at(t + 40 + f, () -> shot(mc, String.format("film_flow_%03d", n)));
		}
		t += 100;
		// ---- FILM: sheen + veins, a close look at the chest and arm over one full sweep
		at(t, () -> { stand(mc, 0, 0, 180f); follow(mc, new Vec3(-0.95, 1.95, -1.45), new Vec3(0, 1.65, 0)); });
		for (int f = 0; f < 120; f++) {
			final int n = f;
			at(t + 6 + f, () -> shot(mc, String.format("film_sheen_%03d", n)));
		}
		t += 130;
		// ---- FILM: tendril strike, spike shot and the blade in Toxin colours (not the default)
		at(t, () -> server(mc, p -> {
			SymbioteSuitColours.setPreset(p, SymbioteColourPreset.TOXIN);
			p.connection.teleport(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5, 180f, 0f);
			Zombie z = new Zombie(EntityType.ZOMBIE, p.serverLevel());
			z.moveTo(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5 - 6, 0f, 0f);
			z.setNoAi(true);
			z.setPersistenceRequired();
			z.setInvulnerable(true);
			p.serverLevel().addFreshEntity(z);
		}));
		at(t + 4, () -> follow(mc, new Vec3(-3.0, 2.0, -4.2), new Vec3(0, 1.4, -1.6)));
		for (int f = 0; f < 90; f++) {
			final int n = f;
			at(t + 6 + f, () -> shot(mc, String.format("film_moves_%03d", n)));
		}
		at(t + 10, () -> server(mc, p -> {
			p.connection.teleport(p.getX(), p.getY(), p.getZ(), 180f, 0f);
			SymbioteAbilityManager.handle(p, AbilitySlot.SLOT_1, true);
		}));
		at(t + 11, () -> server(mc, p -> SymbioteAbilityManager.handle(p, AbilitySlot.SLOT_1, false)));
		at(t + 18, () -> shot(mc, "moves_tendril_toxin"));
		at(t + 36, () -> server(mc, p -> {
			SymbioteAbilityManager.handle(p, AbilitySlot.SLOT_2, true);
		}));
		at(t + 37, () -> server(mc, p -> SymbioteAbilityManager.handle(p, AbilitySlot.SLOT_2, false)));
		at(t + 44, () -> shot(mc, "moves_spikes_toxin"));
		at(t + 60, () -> server(mc, p -> {
			p.connection.teleport(p.getX(), p.getY(), p.getZ(), 180f, 0f);
			SymbioteAbilityManager.handle(p, AbilitySlot.SLOT_5, true);
			System.out.println("[HARNESS] blade=" + p.getAttachedOrElse(ModAttachments.SYMBIOTE_VITALS, null).bladeActive);
		}));
		at(t + 61, () -> server(mc, p -> SymbioteAbilityManager.handle(p, AbilitySlot.SLOT_5, false)));
		at(t + 80, () -> shot(mc, "moves_blade_toxin"));
		t += 100;
		at(t, () -> {
			playerView(mc, CameraType.FIRST_PERSON);
			mc.options.hideGui = false;
		});
		at(t + 8, () -> shot(mc, "first_person_blade_toxin"));
		// ---- the colour screen with the suit off: the paper dolls
		at(t + 10, () -> server(mc, p -> {
			SymbioteAbilityManager.handle(p, AbilitySlot.SLOT_5, true);
			suitUp(p, true); // retract, instantly
		}));
		at(t + 11, () -> server(mc, p -> SymbioteAbilityManager.handle(p, AbilitySlot.SLOT_5, false)));
		at(t + 20, () -> mc.setScreen(new com.projecthero.mod.client.gui.SymbioteColourScreen(null)));
		at(t + 30, () -> shot(mc, "screen_colour_dolls"));
		at(t + 40, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("SYMSUIT_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17051L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
