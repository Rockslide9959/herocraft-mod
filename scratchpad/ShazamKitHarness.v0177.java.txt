/** TEMP-HARNESS (not shipped): films of the v0.17.7 Shazam kit. Enabled by run/shazamkit.flag = "<out dir>|<mode>". */
public final class ShazamKitHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static BlockPos base;
	private static File out;
	private static String mode = "all";
	private static final java.util.Map<Integer, List<Runnable>> AT = new java.util.HashMap<>();
	private static int film0 = -1, filmEnd = -1;
	private static String filmDir = "film";
	private static float camAngle, camDist, camH, camLookAhead, camLookH = 1.1f;
	private static boolean camOn;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("shazamkit.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			String[] parts = Files.readString(flag).trim().split("\\|");
			out = new File(parts[0]);
			if (parts.length > 1) {
				mode = parts[1];
			}
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(ShazamKitHarness::tick);
		ClientTickEvents.START_CLIENT_TICK.register(mc -> placeCam(mc));
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(900_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "shazamkit-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		new File(out, "screenshots/" + name + ".png").getParentFile().mkdirs();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
	}

	private static void at(int t, Runnable r) {
		AT.computeIfAbsent(t, k -> new ArrayList<>()).add(r);
	}

	private static void cam3(float angle, float dist, float h, float lookAhead, float lookH) {
		camAngle = angle;
		camDist = dist;
		camH = h;
		camLookAhead = lookAhead;
		camLookH = lookH;
		camOn = true;
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = true;
		placeCam(mc);
	}

	private static void fp() {
		camOn = false;
		camPose = null;
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = false;
		mc.options.setCameraType(CameraType.FIRST_PERSON);
		if (mc.player != null) {
			mc.setCameraEntity(mc.player);
		}
	}

	public static volatile double[] camPose;

	private static void placeCam(Minecraft mc) {
		if (!camOn || mc.player == null || mc.level == null || base == null) {
			camPose = null;
			return;
		}
		Vec3 c = new Vec3(base.getX() + 0.5, base.getY(), base.getZ() + 0.5);
		double r = Math.toRadians(camAngle);
		Vec3 eye = c.add(Math.sin(r) * camDist, camH, Math.cos(r) * camDist);
		Vec3 look = c.add(0, camLookH, camLookAhead);
		Vec3 d = look.subtract(eye);
		float yaw = (float) (Math.toDegrees(Math.atan2(-d.x, d.z)));
		float pitch = (float) (-Math.toDegrees(Math.atan2(d.y, d.horizontalDistance())));
		camPose = new double[] { eye.x, eye.y, eye.z, yaw, pitch };
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
	}

	private static float pitch = 8f;

	private static void face(Minecraft mc) {
		var p = mc.player;
		p.setYRot(0f);
		p.yRotO = 0f;
		p.setXRot(pitch);
		p.xRotO = pitch;
		p.yHeadRot = 0f;
		p.yHeadRotO = 0f;
	}

	private static void stage(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		level.setDayTime(6000L);
		level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
		level.setWeatherParameters(6000, 0, false, false);
		p.server.setDifficulty(Difficulty.EASY, true);
		p.setGameMode(GameType.SURVIVAL);
		base = p.blockPosition();
		for (int x = -20; x <= 20; x++) {
			for (int z = -12; z <= 34; z++) {
				level.setBlock(base.offset(x, -1, z), (x + z) % 2 == 0 ? Blocks.SMOOTH_STONE.defaultBlockState() : Blocks.POLISHED_ANDESITE.defaultBlockState(), 3);
			}
		}
		p.getInventory().clearContent();
		System.out.println("[HARNESS] grant: " + Shazam.grant(p));
		System.out.println("[HARNESS] word: " + Shazam.speak(p));
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, pitch);
	}

	/** Back to the start: position, health, no cooldowns, a full bar, on the ground, every old mob gone. */
	private static void reset(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		for (Mob m : level.getEntitiesOfClass(Mob.class, new AABB(base).inflate(60))) {
			m.discard();
		}
		com.projecthero.mod.shazam.ShazamFlight.stop(p);
		p.removeAllEffects();
		p.setHealth(p.getMaxHealth());
		Shazam.clearCooldowns(p);
		Shazam.setLightning(p, 100f);
		com.projecthero.mod.shazam.data.ShazamState n = Shazam.state(p).copy(); // no buff left over from the scene before
		n.wisdomUntil = 0L;
		n.standUntil = 0L;
		n.thunderUntil = 0L;
		n.boostUntil = 0L;
		p.setAttached(com.projecthero.mod.attachment.ModAttachments.SHAZAM_STATE, n);
		Shazam.reconcile(p);
		p.setDeltaMovement(Vec3.ZERO);
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, pitch);
	}

	private static Husk husk(ServerPlayer p, double x, double z) {
		ServerLevel level = p.serverLevel();
		Husk h = new Husk(EntityType.HUSK, level);
		h.setNoAi(true);
		h.getAttribute(Attributes.MAX_HEALTH).setBaseValue(250); // under the 300 boss line, so they fly
		h.setHealth(250);
		h.setPersistenceRequired();
		h.moveTo(base.getX() + 0.5 + x, base.getY(), base.getZ() + 0.5 + z, 180f, 0f);
		h.setYHeadRot(180f);
		level.addFreshEntity(h);
		return h;
	}

	private static void crowd(ServerPlayer p) {
		husk(p, 0.0, 3.2);
		husk(p, -2.2, 5.0);
		husk(p, 2.4, 5.5);
		husk(p, -3.5, 8.0);
		husk(p, 3.2, 9.0);
		husk(p, 0.5, 11.0);
	}

	private static void press(ServerPlayer p, int slot, boolean shift) {
		p.setShiftKeyDown(shift);
		com.projecthero.mod.hero.AbilityRouter.handleInput(p, slot, true);
		p.setShiftKeyDown(false);
	}

	private static void release(ServerPlayer p, int slot) {
		com.projecthero.mod.hero.AbilityRouter.handleInput(p, slot, false);
	}

	private static void film(int from, int to, String dir) {
		at(from, () -> {
			film0 = from;
			filmEnd = to;
			filmDir = dir;
		});
	}

	/** One move: reset, crowd, press {slot, shift}, release after {@code holdTicks} (a tap: one tick), with a still. */
	private static int move(Minecraft mc, int t, int slot, boolean shift, int holdTicks, int len, String still, int stillAt) {
		at(t, com.projecthero.mod.client.shazam.ShazamBoltFx::reset);
		at(t, () -> server(mc, p -> {
			reset(p);
			crowd(p);
		}));
		at(t + 6, () -> server(mc, p -> press(p, slot, shift)));
		at(t + 6 + Math.max(1, holdTicks), () -> server(mc, p -> release(p, slot)));
		if (still != null) {
			at(t + 6 + stillAt, () -> shot(mc, still));
		}
		return t + len;
	}

	private static void plan() {
		Minecraft mc = Minecraft.getInstance();
		int t = 0;
		final int R = 1, G = 2, X = 3, Z = 4, V = 5, C = 6;

		// ================= R: the Strength of Hercules -- the combo, the uppercut
		at(t, () -> cam3(65f, 4.8f, 1.9f, 1.4f, 1.1f));
		int r0 = t;
		at(t, () -> server(mc, p -> {
			reset(p);
			ShazamKitHarness.husk(p, 0.0, 2.2);
		}));
		for (int k = 0; k < 3; k++) {
			final int kk = k;
			at(t + 8 + k * 9, () -> server(mc, p -> press(p, R, false)));
			at(t + 9 + k * 9, () -> server(mc, p -> release(p, R)));
			at(t + 11 + k * 9, () -> shot(mc, "r_combo_" + (kk + 1)));
		}
		t += 8 + 27 + 22;
		t = move(mc, t, R, true, 0, 40, "r_uppercut", 7);
		film(r0 + 4, t - 2, "film_r_hercules");

		// ================= G: the Power of Zeus -- the bolt and its chain, the slam

		int g0 = t;

		t = move(mc, t, G, true, 0, 44, "g_thunderhead_slam", 9);
		film(g0 + 4, t - 2, "film_g_zeus");

		at(t, () -> cam3(70f, 9.5f, 3.0f, 5.0f, 1.2f));
		int c0 = t;
		t = move(mc, t, C, false, 0, 34, "c_mercury_blitz", 3);
		t = move(mc, t, C, true, 0, 36, "c_thunderclap", 7);
		film(c0 + 4, t - 2, "film_c_mercury");

		// ================= Z: the Stamina of Atlas -- the quake
		at(t, () -> cam3(62f, 8.0f, 2.8f, 3.5f, 1.2f));
		int z0 = t;
		t = move(mc, t, Z, false, 0, 40, "z_atlas_quake_knee", 3);
		at(z0 + 6 + 7, () -> shot(mc, "z_atlas_quake_stomp"));
		film(z0 + 4, t - 2, "film_z_atlas");

		at(t, () -> cam3(60f, 10.0f, 3.5f, 4.0f, 2.0f));
		int x0 = t;
		t = move(mc, t, X, false, 0, 28, "x_mercury_dash", 2);
		t = move(mc, t, X, true, 0, 26, "x_sonic_leap", 4);
		at(t - 6, () -> server(mc, p -> press(p, X, false))); // Mercury Burst in the air
		at(t - 5, () -> server(mc, p -> release(p, X)));
		at(t - 3, () -> shot(mc, "x_mercury_burst"));
		t += 16;
		t = move(mc, t, V, false, 0, 36, "v_wisdom", 10);
		t = move(mc, t, V, true, 0, 40, "v_courage", 12);
		film(x0 + 4, t - 2, "film_xv_mercury_wisdom_courage");

		// ================= the ultimate
		at(t, () -> cam3(55f, 12.0f, 3.8f, 4.0f, 2.2f));
		int u0 = t;
		t = move(mc, t, Z, true, 0, 70, "ult_shout", 6);
		at(u0 + 6 + 17, () -> shot(mc, "ult_bolt"));
		at(u0 + 6 + 25, () -> shot(mc, "ult_release"));
		at(u0 + 6 + 40, () -> shot(mc, "ult_charged"));
		film(u0 + 4, t - 2, "film_ult_thunderstrike");

		at(t, () -> cam3(62f, 7.5f, 2.6f, 3.0f, 1.3f));

		t = move(mc, t, R, false, 40, 62, "hold_r_haymaker_charge", 30);
		at(h0 + 6 + 42, () -> shot(mc, "hold_r_haymaker_release"));
		int h1 = t;
		t = move(mc, t, G, false, 44, 56, "hold_g_zeus_wrath", 26);
		int h2 = t;
		t = move(mc, t, C, false, 40, 48, "hold_c_mercury_rush", 22);

			com.projecthero.mod.client.shazam.ShazamHud.forceShift = false;

		at(end, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(ShazamKitHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 20) {
			GLFW.glfwSetWindowSize(mc.getWindow().getWindow(), 854, 480);
		}
		if (t == 40) {
			mc.options.fov().set(70);
			mc.options.fovEffectScale().set(0.0);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			mc.getToasts().clear();
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, ShazamKitHarness::stage);
		}
		if (t < 140 || base == null) {
			return;
		}
		face(mc);
		int s = t - 140;
		if (s == 0) {
			plan();
		}
		if (s % 10 == 0) {
			mc.gui.getChat().clearMessages(false);
		}
		List<Runnable> todo = AT.remove(s);
		if (todo != null) {
			todo.forEach(Runnable::run);
		}
		if (s >= film0 && s < filmEnd) {
			shot(mc, String.format(filmDir + "/f%03d", s - film0));
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(GameRules.RULE_DOMOBSPAWNING).set(false, null);
		LevelSettings settings = new LevelSettings("SHAZAMKIT_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.EASY, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}

package com.projecthero.mod.client.mixin;

import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.Shadow;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

import net.minecraft.client.Camera;
import net.minecraft.world.entity.Entity;
import net.minecraft.world.level.BlockGetter;

/** TEMP-HARNESS (not shipped): the ShazamKitHarness's fixed camera. */
@Mixin(Camera.class)
public abstract class ShazamKitHarnessCameraMixin {
	@Shadow
	private boolean detached;

	@Shadow
	protected abstract void setRotation(float yaw, float pitch);

	@Shadow
	protected abstract void setPosition(double x, double y, double z);

	@Inject(method = "setup", at = @At("TAIL"))
	private void projecthero$harnessCam(BlockGetter level, Entity entity, boolean detached, boolean mirrored, float partial, CallbackInfo ci) {
		double[] c = com.projecthero.mod.client.ShazamKitHarness.camPose;
		if (c != null) {
			this.detached = true;
			setPosition(c[0], c[1], c[2]);
			setRotation((float) c[3], (float) c[4]);
		}
	}
}
