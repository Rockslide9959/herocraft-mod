package com.projecthero.mod.client;

import java.nio.file.Files;
import java.nio.file.Path;

import com.projecthero.mod.omnitrix.Omnitrix;
import com.projecthero.mod.omnitrix.alien.Aliens;
import com.projecthero.mod.omnitrix.alien.cannonbolt.CannonboltKit;

import net.fabricmc.fabric.api.client.event.lifecycle.v1.ClientTickEvents;
import net.minecraft.client.CameraType;
import net.minecraft.client.Minecraft;
import net.minecraft.client.Screenshot;
import net.minecraft.client.gui.screens.TitleScreen;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.Difficulty;
import net.minecraft.world.level.GameRules;
import net.minecraft.world.level.GameType;
import net.minecraft.world.level.LevelSettings;
import net.minecraft.world.level.WorldDataConfiguration;
import net.minecraft.world.level.levelgen.WorldOptions;
import net.minecraft.world.level.levelgen.presets.WorldPresets;

/** TEMP-HARNESS (not shipped): Cannonbolt standing, the dial icon, curling up and rolling. Enabled by cannonbolt.flag in the run dir. */
public final class CannonboltHarness {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;

	private CannonboltHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("cannonbolt.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(CannonboltHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(300_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "cb-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		Screenshot.grab(mc.gameDirectory, "cb_" + name + ".png", mc.getMainRenderTarget(), m -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	private static CannonboltKit kit() {
		return (CannonboltKit) Aliens.kit(CannonboltKit.ID);
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(CannonboltHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 80) {
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			server(mc, p -> {
				p.serverLevel().setDayTime(6000L);
				p.serverLevel().getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				p.serverLevel().getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.setGameMode(GameType.SURVIVAL);
				Omnitrix.grant(p);
				com.projecthero.mod.omnitrix.alien.Aliens.discover(p, CannonboltKit.ID, true);
				Omnitrix.noGlitch(p);
				p.connection.teleport(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5, 180f, 30f);
			});
		}
		if (t == 100) {
			server(mc, Omnitrix::activate);
		}
		if (t == 120) {
			server(mc, p -> Omnitrix.select(p, CannonboltKit.ID));
		}
		if (t == 150) {
			shot(mc, "dial");
		}
		if (t == 160) {
			server(mc, p -> {
				Omnitrix.select(p, CannonboltKit.ID);
				Omnitrix.transform(p);
				// a husk for scale
				var h = net.minecraft.world.entity.EntityType.HUSK.create(p.serverLevel());
				h.moveTo(p.getX() + 2.2, p.getY(), p.getZ() - 1.0, 180f, 0f);
				h.setNoAi(true);
				p.serverLevel().addFreshEntity(h);
			});
		}
		if (t == 200) {
			server(mc, p -> kit().setBall(p, true));
		}
		if (t == 210) {
			mc.options.hideGui = true;
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			server(mc, p -> p.connection.teleport(p.getX(), p.getY(), p.getZ(), 0f, 12f));
		}
		if (t == 215) {
			mc.options.keyUp.setDown(true);
			mc.options.keySprint.setDown(true);
		}
		if (t > 215 && t <= 300) {
			double sp = Math.hypot(mc.player.getX() - mc.player.xo, mc.player.getZ() - mc.player.zo);
			if (t % 5 == 0) {
				System.out.println("[HARNESS] t=" + t + " speed=" + String.format("%.3f", sp) + " b/t (" + String.format("%.1f", sp * 20) + " b/s) sprint="
						+ mc.player.isSprinting() + " ground=" + mc.player.onGround());
			}
		}
		if (t == 300) {
			mc.options.keyJump.setDown(true);
		}
		if (t == 302) {
			mc.options.keyJump.setDown(false);
		}
		if (t >= 296 && t < 344 && t % 2 == 0) {
			shot(mc, String.format("launch_%03d", (t - 296) / 2));
		}
		if (t > 300 && t < 344) {
			System.out.println("[HARNESS] t=" + t + " y=" + String.format("%.2f", mc.player.getY()) + " v=" + String.format("%.2f", Math.hypot(mc.player.getX() - mc.player.xo, mc.player.getZ() - mc.player.zo)));
		}
		if (t == 344) {
			mc.options.keyUp.setDown(false);
			mc.options.keySprint.setDown(false);
		}
		if (t == 352) {
			server(mc, p -> p.connection.teleport(p.getX(), p.getY() + 30, p.getZ(), 0f, 35f));
		}
		if (t >= 352 && t < 452 && t % 2 == 0) {
			shot(mc, String.format("fall_%03d", (t - 352) / 2));
			final int tt = t;
			server(mc, p -> System.out.println("[HARNESS] t=" + tt + " y=" + String.format("%.2f", p.getY()) + " bounces=" + CannonboltKit.bouncesLeft(p)
					+ " hp=" + p.getHealth()));
		}
		if (t == 460) {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		}
		if (t == 420) {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		}
	}

	private static int camId = -1;

	private static void cam(Minecraft mc, double dist, double h, String tag) {
		server(mc, p -> {
			var s = net.minecraft.world.entity.EntityType.ARMOR_STAND.create(p.serverLevel());
			net.minecraft.world.phys.Vec3 f = net.minecraft.world.phys.Vec3.directionFromRotation(0, p.getYRot());
			net.minecraft.world.phys.Vec3 eye = p.position().add(f.scale(dist)).add(0, h, 0);
			net.minecraft.world.phys.Vec3 look = p.position().add(0, Math.min(h, 1.4), 0);
			net.minecraft.world.phys.Vec3 d = look.subtract(eye);
			float yaw = (float) Math.toDegrees(Math.atan2(-d.x, d.z)), pitch = (float) -Math.toDegrees(Math.atan2(d.y, Math.sqrt(d.x * d.x + d.z * d.z)));
			net.minecraft.world.phys.Vec3 at = eye.add(d.normalize().scale(4.0)); // third-person back pulls the camera 4 blocks behind its eyes
			s.moveTo(at.x, at.y - s.getEyeHeight(), at.z, yaw, pitch);
			s.setYHeadRot(yaw);
			s.setInvisible(true);
			s.setNoGravity(true);
			s.setInvulnerable(true);
			p.serverLevel().addFreshEntity(s);
			camId = s.getId();
		});
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
		new Thread(() -> {
			try {
				Thread.sleep(400);
			} catch (InterruptedException e) {
				return;
			}
			mc.execute(() -> {
				var e = mc.level.getEntity(camId);
				if (e != null) {
					mc.setCameraEntity(e);
				}
				System.out.println("[HARNESS] cam " + tag + " " + (e != null));
			});
		}).start();
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("CB_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.EASY, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}
}
