 * mindscape, before and after. Enabled by run/fxharness.flag; its text filters scenes (empty = all) and a word "tag=xyz" names
 * the frames fx_xyz_<scene>_NNN.png. New-code calls go through reflection so the same file runs on the old code (the before shots).

public final class FateShazamFxHarness {
	public static volatile Vec3 camPos;
	public static volatile Vec3 camLook;
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorld;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static String recording;
	private static int recFrame;
	private static int recUntil;
	private static Vec3 origin;
	private static String only = "";
	private static String tag = "x";

	private FateShazamFxHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("fxharness.flag"))) {
			return;
		}
		try {
			only = Files.readString(Path.of("fxharness.flag")).trim();
		} catch (Exception e) {
			only = "";
		}
		for (String w : only.split("\\s+")) {
			if (w.startsWith("tag=")) {
				tag = w.substring(4);
			}
		}
		only = only.replaceAll("tag=\\S+", "").trim();
		ClientTickEvents.END_CLIENT_TICK.register(FateShazamFxHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(900_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "fx-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] server step failed: " + e);
						e.printStackTrace(System.out);
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(FateShazamFxHarness::createWorld);
			}
			return;
		}
		inWorld++;
		if (inWorld == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.bobView().set(false);
			mc.options.chatVisibility().set(net.minecraft.world.entity.player.ChatVisiblity.HIDDEN);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			server(mc, p -> {
				ServerLevel l = p.serverLevel();
				l.setDayTime(13500L);
				l.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				l.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				l.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.EASY, true);
				origin = new Vec3(Math.floor(p.getX()) + 0.5, Math.floor(p.getY()), Math.floor(p.getZ()) + 0.5);
				BlockPos o = BlockPos.containing(origin);
				for (int x = -16; x <= 16; x++) {
					for (int z = -10; z <= 18; z++) {
						l.setBlock(o.offset(x, -1, z), (x + z) % 7 == 0 ? Blocks.CHISELED_SANDSTONE.defaultBlockState() : Blocks.SMOOTH_SANDSTONE.defaultBlockState(), 3);
					}
				}
				p.setGameMode(GameType.SURVIVAL);
				p.getInventory().clearContent();
			});
			t0 = inWorld + 20;
			script(mc);
		}
		if (t0 < 0 || inWorld < t0) {
			return;
		}
		Runnable r = AT.remove(inWorld - t0);
		if (r != null) {
			r.run();
		}
		if (recording != null) {
			Screenshot.grab(mc.gameDirectory, String.format("fx_%s_%s_%03d.png", tag, recording, recFrame++), mc.getMainRenderTarget(), c -> {
			});
			if (inWorld - t0 >= recUntil) {
				System.out.println("[HARNESS] recorded " + recording + " frames=" + recFrame);
				recording = null;
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("FX_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.EASY, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}

	private static void record(int start, String name, int ticks) {
		at(start, () -> {
			recording = name;
			recFrame = 0;
			recUntil = start + ticks;
		});
	}

	private static boolean want(String name) {
		return only.isEmpty() || only.contains(name);
	}

	private static void cam(int t, double x, double y, double z, double lx, double ly, double lz) {
		at(t, () -> {
			camPos = origin.add(x, y, z);
			camLook = origin.add(lx, ly, lz);
		});
	}

	private static void face(ServerPlayer p, Vec3 target) {
		Vec3 eye = p.getEyePosition();
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		p.teleportTo(p.serverLevel(), p.getX(), p.getY(), p.getZ(), yaw, pit);
	}

	private static void clear(ServerPlayer p) {
		ServerLevel l = p.serverLevel();
		for (var e : l.getEntitiesOfClass(net.minecraft.world.entity.Mob.class, new AABB(origin, origin).inflate(64))) {
			e.discard();
		}
	}

	private static void dummy(Minecraft mc, int t, double dx, double dz, float hp) {
		at(t, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			Husk h = EntityType.HUSK.create(level);
			h.moveTo(origin.x + dx, origin.y, origin.z + dz, 180f, 0f);
			h.setYHeadRot(180f);
			h.setPersistenceRequired();
			h.getAttribute(Attributes.MAX_HEALTH).setBaseValue(hp);
			h.setHealth(hp);
			h.setSilent(true);
			h.setNoAi(true);
			level.addFreshEntity(h);
		}));
	}

	/** Fate: back to the spot, helmet on, Dominion 10, cooldowns clear. */
	private static void fate(Minecraft mc, int t) {
		at(t, () -> server(mc, p -> {
			clear(p);
			if (Shazam.transformed(p)) {
				Shazam.forceUntransform(p);
			}
			if (!DoctorFate.hasPower(p)) {
				DoctorFate.grant(p);
			}
			DoctorFate.clearTransient(p);
			DoctorFate.setHelmet(p, true);
			DoctorFate.setDominion(p, 10f);
			DoctorFate.clearCooldowns(p);
			p.setDeltaMovement(Vec3.ZERO);
			p.teleportTo(p.serverLevel(), origin.x, origin.y, origin.z, 0f, 0f);
			p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 100000, 10, false, false));
			p.setHealth(p.getMaxHealth());
		}));
	}

	private static void cast(Minecraft mc, int t, double tx, double ty, double tz, String id) {
		at(t, () -> server(mc, p -> {
			DoctorFate.clearCooldowns(p);
			face(p, origin.add(tx, ty, tz));
			DoctorFateSpells.testOverride = false;
			DoctorFateSpells.use(p, id);
		}));
	}

	/** Shazam: back to the spot, transformed, full bar. */
	private static void shazam(Minecraft mc, int t, boolean transformed) {
		at(t, () -> server(mc, p -> {
			clear(p);
			if (DoctorFate.hasPower(p)) {
				DoctorFate.setDominion(p, 0f);
				DoctorFate.revoke(p);
			}
			if (!Shazam.hasPower(p)) {
				Shazam.grant(p);
			}
			p.setDeltaMovement(Vec3.ZERO);
			p.teleportTo(p.serverLevel(), origin.x, origin.y, origin.z, 0f, 0f);
			Shazam.clearCooldowns(p);
			if (Shazam.transformed(p) != transformed) {
				if (transformed) {
					Shazam.speak(p);
				} else {
					Shazam.forceUntransform(p);
				}
				Shazam.clearCooldowns(p);
			}
			Shazam.clearTransient(p);
			Shazam.setLightning(p, 100f);
			p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 100000, 10, false, false));
			p.setHealth(p.getMaxHealth());
		}));
	}

	private static void move(Minecraft mc, int t, double tx, double ty, double tz, String id) {
		at(t, () -> server(mc, p -> {
			face(p, origin.add(tx, ty, tz));
			ShazamMoves.use(p, id);
		}));
	}

	/** A hero landing in the named style (new code; skipped on the old). */
	private static void land(Minecraft mc, int t, String style) {
		at(t, () -> server(mc, p -> {
			try {
				var s = com.projecthero.mod.flight.landing.HeroLandingStyle.valueOf(style);
				com.projecthero.mod.flight.landing.HeroLanding.trigger(p, s, 0.9f);
			} catch (IllegalArgumentException e) {
				System.out.println("[HARNESS] no landing style " + style);
			}
		}));
	}

			record(t + 6, "s_stand", 40);
			move(mc, t + 8, 0, 1.6, 6, ShazamAbilityManager.ACHILLES_STAND);
			t += 50;
		}
		if (want("s_landing")) {
			shazam(mc, t, true);
			cam(t + 2, 3.2, 2.0, 4.2, 0, 0.6, 0);
			record(t + 6, "s_landing", 40);
			land(mc, t + 10, "SHAZAM");
			t += 50;
		}
		at(t + 10, () -> {
			System.out.println("[HARNESS] DONE");
			mc.stop();
		});
	}
