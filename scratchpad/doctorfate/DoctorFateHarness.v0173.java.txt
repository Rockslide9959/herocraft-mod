	private static boolean readLate() {
		try {
			return Files.readString(Path.of("doctorfate.flag")).contains("late");
		} catch (Exception e) {
			return false;
		}
	}

	private DoctorFateHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("doctorfate.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(DoctorFateHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(540_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "doctorfate-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] server step failed: " + e);
						e.printStackTrace(System.out);
					}
				}
			});
		}
	}

	private static void shot(String name) {
		Minecraft mc = Minecraft.getInstance();
		Screenshot.grab(mc.gameDirectory, "df_" + name + ".png", mc.getMainRenderTarget(), m -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	private static void cam(Vec3 rel, Vec3 lookRel) {
		camPos = origin.add(rel);
		camLook = origin.add(lookRel);
	}

	private static void face(ServerPlayer p, Vec3 target) {
		Vec3 eye = p.getEyePosition();
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		p.teleportTo(p.serverLevel(), p.getX(), p.getY(), p.getZ(), yaw, pit);
	}

	private static Husk husk(ServerLevel l, Vec3 at) {
		Husk h = EntityType.HUSK.create(l);
		h.moveTo(at.x, at.y, at.z, 180f, 0f);
		h.setNoAi(true);
		h.setPersistenceRequired();
		h.getAttribute(Attributes.MAX_HEALTH).setBaseValue(100.0);
		h.setHealth(100f);
		l.addFreshEntity(h);
		return h;
	}

	private static void husks(ServerPlayer p) {
		ServerLevel l = p.serverLevel();
		l.getEntitiesOfClass(Husk.class, p.getBoundingBox().inflate(40)).forEach(h -> h.discard());
		husk(l, origin.add(0, 0, 7));
		husk(l, origin.add(-2.5, 0, 9));
		husk(l, origin.add(2.5, 0, 9));
	}

	private static Husk nearest(ServerPlayer p, Vec3 rel) {
		Vec3 at = origin.add(rel);
		return p.serverLevel().getEntitiesOfClass(Husk.class, p.getBoundingBox().inflate(40)).stream()
				.min(java.util.Comparator.comparingDouble(h -> h.position().distanceToSqr(at))).orElse(null);
	}

	private static void cast(Minecraft mc, Vec3 targetRel, String id) {
		server(mc, p -> {
			DoctorFate.clearCooldowns(p);
			if (targetRel != null) {
				face(p, origin.add(targetRel).add(0, 1.0, 0));
			}
			DoctorFateSpells.testOverride = false;
			DoctorFateSpells.use(p, id);
		});
	}

	private static void startClip(String name, int step) {
		clip = name;
		frame = 0;
		every = step;
	}

	private static void stopClip() {
		clip = null;
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("DF_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.EASY, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}
