// TEMP-HARNESS
public final class WildmuttHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static Vec3 origin;
	private static ServerPlayer ben;
	private static Zombie dummy;
	/** Runs on the server every tick (a walking fake, a camera that follows). */
	private static volatile java.util.function.Consumer<ServerPlayer> mover;
	private static String only = System.getProperty("wm.only", "");

	private WildmuttHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("wildmutt.flag"))) {
			return;
		}
		try {
			only = Files.readString(Path.of("wildmutt.flag")).trim();
		} catch (java.io.IOException e) {
			only = "";
		}
		ClientTickEvents.END_CLIENT_TICK.register(WildmuttHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(600_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "wildmutt-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (x, y) -> () -> {
			x.run();
			y.run();
		});
	}

	private static void shot(int t, String name) {
		at(t - 1, () -> {
			if (!name.startsWith("quills") && !name.startsWith("slam") && !name.startsWith("frenzy")) {
				try {
					var m = net.minecraft.client.particle.ParticleEngine.class.getDeclaredMethod("clearParticles");
					m.setAccessible(true);
					m.invoke(Minecraft.getInstance().particleEngine);
				} catch (ReflectiveOperationException e) {
					System.out.println("[HARNESS] " + e);
				}
			}
		});
		at(t, () -> {
			Minecraft mc = Minecraft.getInstance();
			Screenshot.grab(mc.gameDirectory, "omnitrix_wildmutt2_" + name + ".png", mc.getMainRenderTarget(), m -> {
			});
			System.out.println("[HARNESS] shot " + name);
			if (mc.level != null) {
				for (var pl : mc.level.players()) {
					if (Omnitrix.hasPower(pl) && Omnitrix.state(pl).transformed()) {
						var s = Omnitrix.state(pl);
						var mgr = com.projecthero.mod.client.omnitrix.alien.wildmutt.WildmuttAnimatable.INSTANCE.getAnimatableInstanceCache()
								.getManagerForId(pl.getId());
						var ctl = mgr == null ? null : mgr.getAnimationControllers().get("main");
						System.out.println("[HARNESS]   " + pl.getName().getString() + " anim=" + s.animId + " since=" + (pl.level().getGameTime() - s.animStart)
								+ " flags=" + s.kitFlags + " main=" + (ctl == null || ctl.getCurrentAnimation() == null ? "-" : ctl.getCurrentAnimation().animation().name())
								+ " onGround=" + pl.onGround());
					}
				}
			}
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.options.getSoundSourceOptionInstance(net.minecraft.sounds.SoundSource.MASTER).get() > 0.05) {
			mc.options.getSoundSourceOptionInstance(net.minecraft.sounds.SoundSource.MASTER).set(0.05);
		}
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(WildmuttHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			mc.options.chatVisibility().set(net.minecraft.world.entity.player.ChatVisiblity.HIDDEN);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.EASY, true);
				p.setGameMode(GameType.CREATIVE);
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			AT.put(3, () -> script(mc));
		}
		if (t0 >= 0) {
			server(mc, p -> {
				if (ben != null) {
					ben.tickCount++;
					Omnitrix.tick(ben);
				}
				var m = mover;
				if (m != null) {
					m.accept(p);
				}
				if (p.tickCount % 5 == 0) {
					for (var e : p.serverLevel().getEntitiesOfClass(net.minecraft.world.entity.monster.Slime.class, p.getBoundingBox().inflate(120))) {
						e.discard();
					}
				}
			});
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("WM_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.EASY, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}

	private static void camera(ServerPlayer p, Vec3 eye, Vec3 target) {
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		p.teleportTo(p.serverLevel(), eye.x, eye.y - p.getEyeHeight(), eye.z, yaw, pit);
	}

	private static void face(ServerPlayer f, Vec3 target) {
		Vec3 eye = f.getEyePosition();
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		f.teleportTo(f.serverLevel(), f.getX(), f.getY(), f.getZ(), yaw, pit);
		f.setYHeadRot(yaw);
		f.yBodyRot = yaw;
		f.yBodyRotO = yaw;
		f.swing(net.minecraft.world.InteractionHand.MAIN_HAND, true);
	}

	private static ServerPlayer fake(ServerPlayer near, String name, Vec3 at, float yaw) {
		ServerLevel level = near.serverLevel();
		GameProfile profile = new GameProfile(java.util.UUID.randomUUID(), name);
		CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
		ServerPlayer f = new ServerPlayer(level.getServer(), level, profile, cookie.clientInformation());
		Connection conn = new Connection(PacketFlow.SERVERBOUND);
		new io.netty.channel.embedded.EmbeddedChannel(conn);
		level.getServer().getPlayerList().placeNewPlayer(conn, f, cookie);
		f.setGameMode(GameType.SURVIVAL);
		f.teleportTo(level, at.x, at.y, at.z, yaw, 0f);
		f.setYHeadRot(yaw);
		f.yBodyRot = yaw;
		f.yBodyRotO = yaw;
		return f;
	}

	private static void gui(Minecraft mc, boolean on) {
		mc.execute(() -> mc.options.hideGui = !on);
	}

	private static void become(ServerPlayer f, String alien) {
		if (Omnitrix.state(f).transformed()) {
			Omnitrix.revert(f, Omnitrix.Reason.REVOKE);
		}
		var n = Omnitrix.state(f).copy();
		n.rechargeUntil = 0;
		n.selected = alien;
		Omnitrix.save(f, n);
		Omnitrix.transform(f);
		f.setOnGround(true);
	}

	private static WildmuttKit kit() {
		return (WildmuttKit) Aliens.kit(WildmuttKit.ID);
	}

	private static Zombie dummy(ServerPlayer near, Vec3 at) {
		if (dummy != null) {
			dummy.discard();
		}
		ServerLevel l = near.serverLevel();
		Zombie z = EntityType.ZOMBIE.create(l);
		z.moveTo(at.x, at.y, at.z, 0f, 0f);
		z.setNoAi(true);
		z.setPersistenceRequired();
		z.getAttribute(net.minecraft.world.entity.ai.attributes.Attributes.MAX_HEALTH).setBaseValue(500);
		z.setHealth(500);
		z.setItemSlot(net.minecraft.world.entity.EquipmentSlot.HEAD, new net.minecraft.world.item.ItemStack(net.minecraft.world.item.Items.LEATHER_HELMET));
		l.addFreshEntity(z);
		dummy = z;
		return z;
	}

	private static void info(Minecraft mc, String scrollTo) {
		PowerInfoScreen s = new PowerInfoScreen();
		mc.setScreen(s);
		if (scrollTo != null && !s.scrollTo(scrollTo)) {
			System.out.println("[HARNESS] scrollTo miss: " + scrollTo);
		}
	}

	private static boolean want(String part) {
		return only.isEmpty() || only.contains(part);
	}

	private static void script(Minecraft mc) {
		int t = 10;
		at(t, () -> {
			gui(mc, true);
			server(mc, p -> {
				p.teleportTo(p.serverLevel(), origin.x - 12, origin.y, origin.z - 12, 200f, 10f);
				UniquePowers.forced(() -> Omnitrix.grant(p));
			});
		});
		t += 110;
		if (want("fp")) {
			// ---------------------------------------------------------------- first person: the paws (a sword in the slot: hidden),
			// a turn, the swipe, a walk, the maul
			at(t, () -> {
				mc.options.bobView().set(true);
				server(mc, p -> {
					p.setGameMode(GameType.SURVIVAL);
					p.getAbilities().mayfly = false;
					p.getAbilities().flying = false;
					p.onUpdateAbilities();
					p.teleportTo(p.serverLevel(), origin.x, origin.y, origin.z, 200f, 15f);
					become(p, WildmuttKit.ID);
					p.setItemInHand(net.minecraft.world.InteractionHand.MAIN_HAND, new net.minecraft.world.item.ItemStack(net.minecraft.world.item.Items.DIAMOND_SWORD));
					p.setItemInHand(net.minecraft.world.InteractionHand.OFF_HAND, new net.minecraft.world.item.ItemStack(net.minecraft.world.item.Items.TORCH));
				});
			});
			at(t + 1, () -> gui(mc, true)); // F1 hides the hands too
			shot(t + 40, "fp_idle");
			// turning: the client's view yaw runs ahead of the hands' (yBob) -- the paws trail
			for (int i = 0; i < 6; i++) {
				at(t + 42 + i, () -> {
					mc.player.setYRot(mc.player.getYRot() + 14f);
					mc.player.setXRot(mc.player.getXRot() + 2f);
				});
			}
			shot(t + 46, "fp_turn");
			at(t + 56, () -> mc.player.swing(net.minecraft.world.InteractionHand.MAIN_HAND));
			shot(t + 59, "fp_swipe_a");
			shot(t + 60, "fp_swipe_b");
			shot(t + 61, "fp_swipe_c");
			shot(t + 62, "fp_swipe_d");
			// walking forward
			at(t + 66, () -> mc.options.keyUp.setDown(true));
			shot(t + 80, "fp_walk_a");
			shot(t + 83, "fp_walk_b");
			at(t + 86, () -> mc.options.keyUp.setDown(false));
			at(t + 96, () -> server(mc, p -> kit().maul(p)));
			shot(t + 99, "fp_maul");
			t += 120;
		}
		if ("fp".equals(only)) {
			at(t, () -> {
				System.out.println("[HARNESS] DONE");
				Runtime.getRuntime().halt(0);
			});
			return;
		}
		// ---------------------------------------------------------------- third person: Ben as Wildmutt
		Vec3 stage = origin.add(-24, 0, 0);
		at(t, () -> {
			gui(mc, false);
			server(mc, p -> {
				if (Omnitrix.state(p).transformed()) {
					Omnitrix.revoke(p);
				}
				p.setGameMode(GameType.SPECTATOR);
				ben = fake(p, "Ben", stage, 180f);
				UniquePowers.forced(() -> Omnitrix.grant(ben));
				become(ben, WildmuttKit.ID);
				camera(p, stage.add(-1.9, 1.5, -2.7), stage.add(0, 0.85, 0));
			});
		});
		at(t + 20, () -> server(mc, p -> face(ben, stage.add(-0.8, 0.9, -5))));
		shot(t + 40, "idle_three_quarter");
		at(t + 41, () -> server(mc, p -> camera(p, stage.add(-3.5, 1.1, -0.2), stage.add(0, 0.8, -0.2))));
		shot(t + 45, "idle_side");
		at(t + 46, () -> server(mc, p -> camera(p, stage.add(2.0, 1.9, 2.8), stage.add(0, 0.9, 0))));
		shot(t + 50, "idle_back");
		at(t + 51, () -> server(mc, p -> camera(p, stage.add(-0.3, 1.0, -2.7), stage.add(0, 0.85, 0))));
		shot(t + 55, "idle_front");
		t += 60;
		// walk and gallop: Ben moves toward -Z; the camera rides alongside
		final int walkStart = t;
		at(t, () -> server(mc, p -> {
			ben.teleportTo(p.serverLevel(), stage.x, stage.y, stage.z + 14, 180f, 0f);
			ben.setYHeadRot(180f);
			mover = q -> {
				double step = ben.isSprinting() ? 0.47 : 0.29;
				Vec3 at = ben.position().add(0, 0, -step);
				ben.teleportTo(q.serverLevel(), at.x, at.y, at.z, 180f, 0f);
				ben.setYHeadRot(180f);
				camera(q, at.add(-3.4, 1.2, -0.6), at.add(0, 0.8, -0.4));
			};
		}));
		shot(walkStart + 24, "walk");
		shot(walkStart + 26, "walk_b");
		at(walkStart + 30, () -> server(mc, p -> ben.setSprinting(true)));
		shot(walkStart + 50, "gallop");
		shot(walkStart + 52, "gallop_b");
		shot(walkStart + 54, "gallop_c");
		at(walkStart + 56, () -> server(mc, p -> {
			mover = null;
			ben.setSprinting(false);
			ben.teleportTo(p.serverLevel(), stage.x, stage.y, stage.z, 180f, 0f);
		}));
		t = walkStart + 70;
		// the moves, each against a zombie
		at(t, () -> server(mc, p -> {
			Zombie z = dummy(p, stage.add(0.3, 0, -3.2));
			face(ben, z.position().add(0, 1.0, 0));
			camera(p, stage.add(-3.6, 1.6, -0.8), stage.add(0, 0.9, -1.4));
		}));
		at(t + 15, () -> server(mc, p -> {
			kit().pounce(ben);
			ben.teleportTo(p.serverLevel(), ben.getX(), ben.getY() + 0.5, ben.getZ() - 0.6, ben.getYRot(), ben.getXRot());
		}));
		shot(t + 20, "pounce");
		at(t + 23, () -> server(mc, p -> ben.teleportTo(p.serverLevel(), stage.x, stage.y, stage.z - 0.6, ben.getYRot(), ben.getXRot())));
		shot(t + 24, "pounce_bite");
		at(t + 40, () -> server(mc, p -> {
			ben.teleportTo(p.serverLevel(), stage.x, stage.y, stage.z, ben.getYRot(), ben.getXRot());
			dummy.moveTo(stage.x + 0.3, stage.y, stage.z - 4.5, 0f, 0f);
			face(ben, dummy.position().add(0, 1.6, 0));
			camera(p, stage.add(-3.8, 1.5, 0.8), stage.add(0, 1.1, -0.8));
			kit().roar(ben);
		}));
		shot(t + 52, "roar");
		at(t + 70, () -> server(mc, p -> {
			dummy.moveTo(stage.x + 0.2, stage.y, stage.z - 2.4, 0f, 0f);
			face(ben, dummy.position().add(0, 1.0, 0));
			camera(p, stage.add(-3.6, 1.7, -0.2), stage.add(0, 1.2, -1.0));
		}));
		at(t + 80, () -> server(mc, p -> kit().maul(ben)));
		shot(t + 82, "maul");
		shot(t + 87, "maul_b");
		shot(t + 94, "maul_c");
		at(t + 110, () -> server(mc, p -> kit().slam(ben)));
		shot(t + 117, "slam_up");
		shot(t + 120, "slam");
		at(t + 135, () -> server(mc, p -> {
			face(ben, stage.add(0, 1.0, -6));
			kit().quills(ben);
		}));
		shot(t + 143, "quills");
		shot(t + 146, "quills_fire");
		at(t + 160, () -> server(mc, p -> {
			if (dummy != null) {
				dummy.discard();
			}
			kit().roll(ben);
		}));
		shot(t + 166, "roll");
		at(t + 196, () -> server(mc, p -> kit().frenzy(ben)));
		shot(t + 204, "frenzy");
		t += 230;
		// climbing a wall: a stone wall ahead, Wall Climb on, Ben going up it
		Vec3 wallAt = stage.add(8, 0, 0);
		at(t, () -> server(mc, p -> {
			ServerLevel l = p.serverLevel();
			BlockPos w = BlockPos.containing(wallAt.x, wallAt.y, wallAt.z - 1.0);
			for (int x = -3; x <= 3; x++) {
				for (int y = 0; y < 8; y++) {
					l.setBlock(w.offset(x, y, 0), Blocks.STONE_BRICKS.defaultBlockState(), 2);
				}
			}
			kit().setFlag(ben, WildmuttKit.F_CLIMB, true);
			ben.teleportTo(l, wallAt.x, wallAt.y + 1.0, wallAt.z - 0.45, 180f, 0f);
			ben.setYHeadRot(180f);
			ben.setOnGround(false);
			mover = q -> {
				Vec3 at = ben.position().add(0, 0.12, 0);
				if (at.y > wallAt.y + 2.6) {
					at = new Vec3(at.x, wallAt.y + 0.9, at.z);
				}
				ben.teleportTo(q.serverLevel(), at.x, at.y, at.z, 180f, 0f);
				ben.setYHeadRot(180f);
				ben.setOnGround(false);
			};
			camera(p, wallAt.add(-3.2, 2.2, 2.6), wallAt.add(0, 1.9, -0.6));
		}));
		shot(t + 22, "climb");
		at(t + 23, () -> server(mc, p -> camera(p, wallAt.add(4.4, 2.6, 1.6), wallAt.add(0, 2.8, -0.6))));
		shot(t + 27, "climb_b");
		at(t + 30, () -> {
			mover = null;
		});
		t += 40;
		at(t, () -> {
			System.out.println("[HARNESS] DONE");
			Runtime.getRuntime().halt(0);
		});
	}
