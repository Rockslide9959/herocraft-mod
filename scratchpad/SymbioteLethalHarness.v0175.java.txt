/**
 * TEMP-HARNESS (not shipped, v0.17.5 Symbiote Lethal Protector): views of the suit, face close-ups (closed / open / snarl / tongue
 * out), Classic side by side, the style screen, first person, and films of the idle mouth, a snarl + tongue lash on attack and the
 * suit flowing on. Enabled by run/symlethal.flag = "<out dir>" (second line "quick" = only the views + face shots). Kept as
 * scratchpad/SymbioteLethalHarness.v0175.java.txt.
 */
public final class SymbioteLethalHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static boolean quick;
	private static int waitTicks, inWorldTicks;
	private static File out;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static final List<BooleanSupplier> GATES = new ArrayList<>();
	private static int planTick;
	private static BlockPos site;
	private static ArmorStand camera;
	private static Vec3 camAt, camLook;
	private static Zombie dummy;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("symlethal.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			List<String> lines = Files.readAllLines(flag);
			out = new File(lines.get(0).trim());
			quick = lines.size() > 1 && lines.get(1).trim().equals("quick");
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(SymbioteLethalHarness::tick);
		ClientTickEvents.START_CLIENT_TICK.register(SymbioteLethalHarness::placeCamera);
		net.fabricmc.fabric.api.client.rendering.v1.WorldRenderEvents.AFTER_ENTITIES.register(ctx -> {
			Minecraft mc = Minecraft.getInstance();
			if (camAt == null || mc.player == null || ctx.consumers() == null) {
				return;
			}
			float pt = ctx.tickCounter().getGameTimeDeltaPartialTick(false);
			var p = mc.player;
			Vec3 cam = ctx.camera().getPosition();
			double x = net.minecraft.util.Mth.lerp(pt, p.xo, p.getX()) - cam.x, y = net.minecraft.util.Mth.lerp(pt, p.yo, p.getY()) - cam.y,
					z = net.minecraft.util.Mth.lerp(pt, p.zo, p.getZ()) - cam.z;
			var disp = mc.getEntityRenderDispatcher();
			disp.render(p, x, y, z, net.minecraft.util.Mth.lerp(pt, p.yRotO, p.getYRot()), pt, ctx.matrixStack(), ctx.consumers(),
					disp.getPackedLightCoords(p, pt));
		});
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(540_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "symlethal-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		mc.getToasts().clear();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		if (!name.startsWith("film_")) {
			var p = mc.player;
			SymbioteMouth.Pose pose = p == null ? null : SymbioteMouth.compute(p, 0f, new SymbioteMouth.Pose());
			System.out.println("[HARNESS] shot " + name + (pose == null ? "" : String.format(" jaw=%.1f ext=%.1f loll=%.2f snarl=%.2f lash=%.2f",
					pose.jaw, pose.ext, pose.loll, pose.snarl, pose.lash)));
		}
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
			GATES.add(null);
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			prev.run();
			r.run();
		});
	}

	/** The plan waits at {@code ticks} until {@code until} holds (max 400 ticks). */
	static void gate(int ticks, BooleanSupplier until) {
		at(ticks, () -> {
		});
		GATES.set(ticks - 1, until);
	}

	private static int gateWait;

	private static Vec3 s(double x, double y, double z) {
		return new Vec3(site.getX() + 0.5 + x, site.getY() + 1 + y, site.getZ() + 0.5 + z);
	}

	private static void fixed(Minecraft mc, Vec3 at, Vec3 look) {
		if (camera == null || camera.level() != mc.level) {
			camera = new ArmorStand(EntityType.ARMOR_STAND, mc.level);
			camera.setInvisible(true);
		}
		camAt = at;
		camLook = look;
		mc.options.setCameraType(CameraType.FIRST_PERSON);
		mc.setCameraEntity(camera);
		placeCamera(mc);
	}

	private static void playerView(Minecraft mc, CameraType type) {
		camAt = null;
		mc.setCameraEntity(mc.player);
		mc.options.setCameraType(type);
	}

	private static void placeCamera(Minecraft mc) {
		if (mc.player != null && site != null) {
			mc.player.yBodyRot = mc.player.getYRot();
			mc.player.yBodyRotO = mc.player.getYRot();
		}
		if (camAt == null || camera == null) {
			return;
		}
		Vec3 to = camLook.subtract(camAt);
		float yaw = (float) Math.toDegrees(Math.atan2(-to.x, to.z));
		float pitch = (float) -Math.toDegrees(Math.atan2(to.y, Math.sqrt(to.x * to.x + to.z * to.z)));
		double eye = camera.getEyeHeight();
		camera.moveTo(camAt.x, camAt.y - eye, camAt.z, yaw, pitch);
		camera.xo = camera.getX();
		camera.yo = camera.getY();
		camera.zo = camera.getZ();
		camera.xOld = camera.getX();
		camera.yOld = camera.getY();
		camera.zOld = camera.getZ();
		camera.yRotO = yaw;
		camera.xRotO = pitch;
		camera.setYHeadRot(yaw);
	}

	private static void stand(Minecraft mc, double x, double z, float yaw) {
		server(mc, p -> {
			p.getAbilities().flying = false;
			p.onUpdateAbilities();
			p.connection.teleport(site.getX() + 0.5 + x, site.getY() + 1, site.getZ() + 0.5 + z, yaw, 0f);
			p.setYHeadRot(yaw);
		});
	}

	private static void suitUp(ServerPlayer p, boolean instant) {
		Symbiote.state(p).toggleReadyAt = 0L;
		Symbiote.toggle(p);
		if (instant) {
			Symbiote.state(p).transformStartTick -= 10_000L;
			Symbiote.tick(p);
		}
	}

	static void prepare(ServerLevel level) {
		for (int x = -14; x <= 14; x++) {
			for (int z = -14; z <= 14; z++) {
				for (int y = -2; y <= 16; y++) {
					BlockState st = y < 0 ? Blocks.DIRT.defaultBlockState() : y == 0 ? Blocks.GRASS_BLOCK.defaultBlockState() : Blocks.AIR.defaultBlockState();
					level.setBlock(site.offset(x, y, z), st, 2 | 16);
				}
			}
		}
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(SymbioteLethalHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(70);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(com.projecthero.mod.loot.finds.SupplyDrops.RULE).set(false, p.server);
				BlockPos sp = p.blockPosition();
				int y = level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, sp.getX(), sp.getZ());
				site = new BlockPos(sp.getX(), y - 1, sp.getZ());
				prepare(level);
				p.setGameMode(GameType.SURVIVAL);
				Symbiote.grant(p);
				p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 20 * 60 * 30, 4, false, false));
				p.addEffect(new MobEffectInstance(MobEffects.SATURATION, 20 * 60 * 30, 4, false, false));
				SymbioteSuitColours.setStyle(p, SymbioteSuitStyle.LETHAL_PROTECTOR);
				System.out.println("[HARNESS] site " + site + " normal=" + Symbiote.isNormalHost(p));
			});
		}
		if (t == 45 && site == null) {
			return;
		}
		if (t == 46) {
			buildPlan(mc);
			System.out.println("[HARNESS] lethal available=" + com.projecthero.mod.client.symbiote.SymbioteSuitTint.lethalAvailable());
		}
		if (t < 50 || site == null) {
			return;
		}
		if (planTick < PLAN.size()) {
			BooleanSupplier g = GATES.get(planTick);
			if (g != null && !g.getAsBoolean() && gateWait++ < 400) {
				return;
			}
			gateWait = 0;
			PLAN.get(planTick++).run();
		}
	}

	private static SymbioteMouth.Pose pose(Minecraft mc) {
		return SymbioteMouth.compute(mc.player, 0f, new SymbioteMouth.Pose());
	}

	private static void spawnDummy(Minecraft mc, double dz) {
		server(mc, p -> {
			if (dummy != null) {
				dummy.discard();
			}
			Zombie z = new net.minecraft.world.entity.monster.Husk(EntityType.HUSK, p.serverLevel());
			z.moveTo(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5 + dz, 0f, 0f);
			z.setNoAi(true);
			z.setPersistenceRequired();
			z.getAttribute(Attributes.MAX_HEALTH).setBaseValue(1000.0);
			z.getAttribute(Attributes.KNOCKBACK_RESISTANCE).setBaseValue(1.0);
			z.setHealth(1000f);
			p.serverLevel().addFreshEntity(z);
			dummy = z;
		});
	}

	private static void hit(Minecraft mc) {
		server(mc, p -> {
			p.swing(InteractionHand.MAIN_HAND, true);
			if (dummy != null && dummy.isAlive()) {
				p.attack(dummy);
				dummy.setHealth(1000f);
			}
		});
	}

	private static final Vec3 FACE_FRONT = new Vec3(0, 2.34, -1.3), FACE_AT = new Vec3(0, 2.26, 0);

	private static void side(Minecraft mc) {
		fixed(mc, s(-2.0, 2.62, -0.45), s(0, 2.3, -0.15));
	}

	private static void face(Minecraft mc, double x, double z) {
		fixed(mc, s(x, FACE_FRONT.y, z), s(0, FACE_AT.y, 0));
	}

	private static void buildPlan(Minecraft mc) {
		int t = 10;
		at(t, () -> {
			stand(mc, 0, 0, 180f);
			mc.options.hideGui = true;
			server(mc, p -> suitUp(p, true));
		});
		t += 40;
		// ---- views
		String[][] views = { { "01_front", "0", "1.75", "-3.1" }, { "02_back", "0", "1.75", "3.1" }, { "03_side", "3.1", "1.75", "0" },
				{ "04_three_quarter", "-1.6", "1.8", "-2.7" } };
		for (String[] v : views) {
			final String[] vv = v;
			at(t, () -> fixed(mc, s(Double.parseDouble(vv[1]), Double.parseDouble(vv[2]), Double.parseDouble(vv[3])), s(0, 1.3, 0)));
			at(t + 12, () -> shot(mc, "lethal_" + vv[0]));
			t += 14;
		}
		// ---- the face: closed (idle, between lolls), open (the loll), snarl, tongue lash
		at(t, () -> face(mc, 0, FACE_FRONT.z));
		gate(t + 2, () -> { var ps = pose(mc); return ps.loll == 0f && ps.jaw < 1.6f; });
		at(t + 3, () -> shot(mc, "face_01_closed"));
		at(t + 4, () -> face(mc, -0.95, -1.0));
		at(t + 6, () -> shot(mc, "face_02_closed_three_quarter"));
		at(t + 7, () -> side(mc));
		at(t + 9, () -> shot(mc, "face_03_closed_side"));
		at(t + 10, () -> face(mc, 0, FACE_FRONT.z));
		gate(t + 12, () -> pose(mc).loll > 0.97f);
		at(t + 13, () -> shot(mc, "face_04_open_tongue_lolling"));
		at(t + 14, () -> face(mc, -0.95, -1.0));
		at(t + 16, () -> shot(mc, "face_05_lolling_three_quarter"));
		at(t + 17, () -> side(mc));
		at(t + 19, () -> shot(mc, "face_06_lolling_side"));
		t += 22;
		at(t, () -> { face(mc, 0, FACE_FRONT.z); spawnDummy(mc, -1.6); });
		gate(t + 20, () -> pose(mc).loll == 0f);
		at(t + 21, () -> server(mc, p -> p.swing(InteractionHand.MAIN_HAND, true))); // a swing at nothing: snarl
		at(t + 28, () -> shot(mc, "face_07_snarl"));
		at(t + 29, () -> face(mc, -0.95, -1.0));
		at(t + 31, () -> shot(mc, "face_08_snarl_three_quarter"));
		at(t + 60, () -> face(mc, 0, FACE_FRONT.z));
		at(t + 62, () -> hit(mc)); // a hit: the tongue lashes
		at(t + 68, () -> shot(mc, "face_09_tongue_lash"));
		at(t + 90, () -> side(mc));
		at(t + 92, () -> hit(mc));
		at(t + 98, () -> shot(mc, "face_10_tongue_lash_side"));
		at(t + 120, () -> face(mc, -0.95, -1.0));
		at(t + 122, () -> hit(mc));
		at(t + 128, () -> shot(mc, "face_11_tongue_lash_three_quarter"));
		t += 150;
		at(t, () -> server(mc, p -> { if (dummy != null) dummy.discard(); }));
		// ---- Classic beside it (same views), then back
		at(t + 1, () -> server(mc, p -> SymbioteSuitColours.setStyle(p, SymbioteSuitStyle.CLASSIC)));
		at(t + 2, () -> fixed(mc, s(0, 1.75, -3.1), s(0, 1.3, 0)));
		at(t + 12, () -> shot(mc, "classic_01_front"));
		at(t + 13, () -> face(mc, 0, FACE_FRONT.z));
		at(t + 22, () -> shot(mc, "classic_face"));
		at(t + 23, () -> face(mc, -0.95, -1.0));
		at(t + 32, () -> shot(mc, "classic_face_three_quarter"));
		at(t + 34, () -> server(mc, p -> SymbioteSuitColours.setStyle(p, SymbioteSuitStyle.LETHAL_PROTECTOR)));
		t += 40;
		// ---- recoloured: Carnage and a custom colour on the Lethal Protector
		at(t, () -> server(mc, p -> SymbioteSuitColours.setPreset(p, SymbioteColourPreset.CARNAGE)));
		at(t + 1, () -> fixed(mc, s(-1.6, 1.8, -2.7), s(0, 1.3, 0)));
		at(t + 12, () -> shot(mc, "lethal_carnage_three_quarter"));
		at(t + 13, () -> server(mc, p -> SymbioteSuitColours.setPreset(p, SymbioteColourPreset.ANTI_VENOM)));
		at(t + 24, () -> shot(mc, "lethal_anti_venom_three_quarter"));
		at(t + 25, () -> server(mc, p -> SymbioteSuitColours.setColours(p, new SymbioteColours(0x0E6B60, 0xFFB000, 0x22E5FF))));
		at(t + 36, () -> shot(mc, "lethal_custom_teal_three_quarter"));
		at(t + 37, () -> face(mc, 0, FACE_FRONT.z));
		at(t + 48, () -> shot(mc, "lethal_custom_teal_face"));
		at(t + 50, () -> server(mc, p -> SymbioteSuitColours.setPreset(p, SymbioteColourPreset.CLASSIC)));
		t += 54;
		if (quick) {
			at(t, () -> {
				System.out.println("[HARNESS] done");
				Runtime.getRuntime().halt(0);
			});
			return;
		}
		// ---- the style screen (suit worn: live 3D preview) and first person
		at(t, () -> { playerView(mc, CameraType.FIRST_PERSON); mc.options.hideGui = false; });
		at(t + 6, () -> mc.setScreen(new com.projecthero.mod.client.gui.SymbioteColourScreen(null)));
		at(t + 20, () -> shot(mc, "screen_style_lethal"));
		at(t + 22, () -> mc.setScreen(null));
		at(t + 30, () -> shot(mc, "first_person_lethal"));
		at(t + 32, () -> mc.options.hideGui = true);
		t += 36;
		// ---- FILM: the idle mouth, close-up, through a loll (breathing, tongue out and swaying, drool)
		at(t, () -> face(mc, -0.7, -1.05));
		gate(t + 2, () -> { double l = pose(mc).loll; return l > 0f && l < 0.05f; });
		at(t + 2, () -> { });
		for (int f = 0; f < 150; f++) {
			final int n = f;
			at(t + 3 + f, () -> shot(mc, String.format("film_idle_%03d", n)));
		}
		t += 160;
		// ---- FILM: snarl + tongue lash on attack, and a tendril strike
		at(t, () -> { stand(mc, 0, 0, 180f); spawnDummy(mc, -2.2); fixed(mc, s(2.3, 2.45, -1.5), s(-0.1, 2.05, -0.7)); });
		gate(t + 4, () -> pose(mc).loll == 0f && pose(mc).ext < 0.1f);
		for (int f = 0; f < 120; f++) {
			final int n = f;
			at(t + 5 + f, () -> shot(mc, String.format("film_attack_%03d", n)));
		}
		at(t + 10, () -> server(mc, p -> p.swing(InteractionHand.MAIN_HAND, true)));
		at(t + 40, () -> hit(mc));
		at(t + 70, () -> hit(mc));
		at(t + 95, () -> server(mc, p -> {
			p.connection.teleport(p.getX(), p.getY(), p.getZ(), 180f, 0f);
			SymbioteAbilityManager.handle(p, AbilitySlot.SLOT_1, true);
		}));
		at(t + 96, () -> server(mc, p -> SymbioteAbilityManager.handle(p, AbilitySlot.SLOT_1, false)));
		t += 130;
		at(t, () -> server(mc, p -> { if (dummy != null) dummy.discard(); }));
		// ---- FILM: the suit flowing off and on again, three-quarter on the head
		at(t + 1, () -> { stand(mc, 0, 0, 180f); fixed(mc, s(-1.3, 2.1, -2.0), s(0, 1.7, 0)); server(mc, p -> suitUp(p, false)); });
		for (int f = 0; f < 30; f++) {
			final int n = f;
			at(t + 2 + f, () -> shot(mc, String.format("film_flow_%03d", n)));
		}
		at(t + 40, () -> server(mc, p -> suitUp(p, false)));
		for (int f = 0; f < 60; f++) {
			final int n = 30 + f;
			at(t + 40 + f, () -> shot(mc, String.format("film_flow_%03d", n)));
		}
		t += 110;
		at(t, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("SYMLETHAL_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17051L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
