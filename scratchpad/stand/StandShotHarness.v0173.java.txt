/** TEMP-HARNESS (not shipped): films Star Platinum. Enabled by run/standshot.flag (its text = the scene list, empty = all). */
public final class StandShotHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static String recording;
	private static int recFrame;
	private static int recUntil;
	private static Vec3 origin;
	private static Vec3 base;
	private static volatile String logTag;
	private static String scenes = "";

	/** A fixed camera (world), or one that follows the player (offsets from the player's feet). */
	public static volatile Vec3 camPos;
	public static volatile Vec3 camLook;
	private static volatile Vec3 followOff;
	private static volatile Vec3 followLook;

	private StandShotHarness() {
	}

	public static void init() {
		Path flag = Path.of("standshot.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			scenes = Files.readString(flag).trim();
		} catch (java.io.IOException e) {
			scenes = "";
		}
		ClientTickEvents.END_CLIENT_TICK.register(StandShotHarness::tick);
		ServerTickEvents.END_SERVER_TICK.register(StandShotHarness::serverLog);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(720_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "standshot-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	/** Asked by the camera mixin every frame. */
	public static void updateCamera(Entity entity, float partial) {
		Vec3 off = followOff;
		if (off != null && entity != null) {
			Vec3 at = entity.getPosition(partial);
			camPos = at.add(off);
			camLook = at.add(followLook);
		}
	}

	private static void serverLog(MinecraftServer srv) {
		String tag = logTag;
		if (tag == null || srv.getPlayerList().getPlayers().isEmpty()) {
			return;
		}
		ServerPlayer p = srv.getPlayerList().getPlayers().get(0);
		StandState s = Stands.state(p);
		long now = p.level().getGameTime();
		StringBuilder b = new StringBuilder(String.format("[SHOT] %s S t=%d anim=%d age=%d st=%.1f rush=%b chase=%d/%d dash=%b grab=%d stop=%b pos=(%.2f,%.2f,%.2f) v=%.2f",
				tag, now, s.animId, now - s.animStart, s.stamina, s.rushing, s.chaseStart == 0 ? 0 : now - s.chaseStart, s.chaseGrabAt == 0 ? 0 : now - s.chaseGrabAt,
				s.dashUntil > now, s.grabbedEntity, TimeStop.active(), p.getX(), p.getY(), p.getZ(), p.getDeltaMovement().length()));
		for (Zombie z : p.serverLevel().getEntitiesOfClass(Zombie.class, p.getBoundingBox().inflate(64))) {
			b.append(String.format(" | z hp=%.1f (%.1f,%.1f,%.1f) t%d", z.getHealth(), z.getX() - p.getX(), z.getY() - p.getY(), z.getZ() - p.getZ(), z.tickCount));
		}
		System.out.println(b);
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(StandShotHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = false;
			mc.options.fov().set(70);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.EASY, true);
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			if (recording != null) {
				Screenshot.grab(mc.gameDirectory, String.format("sp_%s_%03d.png", recording, recFrame++), mc.getMainRenderTarget(), c -> {
				});
				if (inWorldTicks - t0 >= recUntil) {
					System.out.println("[HARNESS] recorded " + recording + " frames=" + recFrame);
					recording = null;
					logTag = null;
				}
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("SP_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false,
				Difficulty.EASY, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	private static void record(int start, String name, int ticks) {
		at(start, () -> {
			recording = name;
			recFrame = 0;
			recUntil = start + ticks;
			logTag = name;
			System.out.println("[HARNESS] recording " + name);
		});
	}

	private static Zombie zombie(ServerPlayer p, Vec3 at, double hp, boolean ai) {
		Zombie z = EntityType.ZOMBIE.create(p.serverLevel());
		z.moveTo(at.x, at.y, at.z, 180f, 0f);
		z.setNoAi(false);
		if (!ai) {
			// a target that stays put but still flies when hit (a no-AI mob never moves at all)
			z.addEffect(new net.minecraft.world.effect.MobEffectInstance(net.minecraft.world.effect.MobEffects.MOVEMENT_SLOWDOWN, 100000, 10, false, false));
		}
		z.setPersistenceRequired();
		z.getAttribute(Attributes.MAX_HEALTH).setBaseValue(hp);
		z.setHealth((float) hp);
		z.setItemSlot(net.minecraft.world.entity.EquipmentSlot.HEAD, new ItemStack(net.minecraft.world.item.Items.LEATHER_HELMET));
		z.setYHeadRot(180f);
		z.setYBodyRot(180f);
		p.serverLevel().addFreshEntity(z);
		return z;
	}

	/** Back to the start of a scene: everything cleared, full Stamina, the Stand out (unless {@code summoned} is false). */
	private static void reset(ServerPlayer p, boolean summoned) {
		p.serverLevel().getEntitiesOfClass(Entity.class, p.getBoundingBox().inflate(80),
				e -> e instanceof Zombie || e instanceof net.minecraft.world.entity.item.ItemEntity
						|| e instanceof com.projecthero.mod.stand.entity.BallBearingEntity).forEach(Entity::discard);
		if (Stands.summoned(p) != summoned) {
			if (summoned) {
				Stands.summon(p);
			} else {
				Stands.dismiss(p);
			}
		}
		StandState n = Stands.state(p).copy();
		n.abilityReadyAt.clear();
		n.stamina = 100f;
		n.regenSlowUntil = 0L;
		n.animId = 0;
		p.setAttached(ModAttachments.STAND_STATE, n);
		p.setItemInHand(InteractionHand.MAIN_HAND, ItemStack.EMPTY);
		p.setDeltaMovement(Vec3.ZERO);
		p.teleportTo(p.serverLevel(), base.x, base.y, base.z, 0f, 0f);
		p.setYHeadRot(0f);
		p.yBodyRot = 0f;
		p.setHealth(p.getMaxHealth());
	}

	interface Setup {
		void run(ServerPlayer p);
	}

	/** A scene: a fresh spot, the camera (null = the player's own first-person view), the setup, then the view straight ahead. */
	private static void scene(Minecraft mc, int t, int index, CameraType type, Vec3 cam, Vec3 look, boolean follow, boolean summoned, float pitch, Setup setup) {
		at(t, () -> {
			mc.gui.clear();
			mc.gui.getChat().clearMessages(false);
			base = origin.add(index * 48, 0, 0);
			mc.options.setCameraType(type);
			if (follow) {
				followOff = cam;
				followLook = look;
			} else {
				followOff = null;
				camPos = cam == null ? null : base.add(cam);
				camLook = look == null ? null : base.add(look);
			}
			server(mc, p -> {
				reset(p, summoned);
				if (setup != null) {
					setup.run(p);
				}
			});
		});
		at(t + 6, () -> {
			if (mc.player != null) {
				mc.player.setYRot(0f);
				mc.player.setXRot(pitch);
				mc.player.yRotO = 0f;
				mc.player.xRotO = pitch;
			}
		});
	}

	private static void click(int t) {
		at(t, () -> KeyMapping.click(InputConstants.Type.MOUSE.getOrCreate(0)));
	}

	private static void hold(KeyMapping key, int t, int ticks) {
		at(t, () -> key.setDown(true));
		at(t + ticks, () -> key.setDown(false));
	}

	/** Shift held around a key press (the server reads Shift off the sneak state). */
	private static void shiftPress(KeyMapping key, int t, int ticks) {
		hold(Minecraft.getInstance().options.keyShift, t - 3, ticks + 6);
		hold(key, t, ticks);
	}

	/** Mining as a held left button does it (the harness window has no grabbed mouse, so the button itself is not read). */
	private static void mine(Minecraft mc, int t, int ticks) {
		for (int i = 0; i < ticks; i++) {
			at(t + i, () -> {
				if (mc.player != null && mc.gameMode != null && mc.hitResult instanceof net.minecraft.world.phys.BlockHitResult bh
						&& bh.getType() == net.minecraft.world.phys.HitResult.Type.BLOCK) {
					mc.gameMode.continueDestroyBlock(bh.getBlockPos(), bh.getDirection());
					mc.player.swing(InteractionHand.MAIN_HAND);
				}
			});
		}
		at(t + ticks, () -> {
			if (mc.gameMode != null) {
				mc.gameMode.stopDestroyBlock();
			}
		});
	}

	private static void turn(Minecraft mc, int t, float yaw, float pitch) {
		at(t, () -> {
			if (mc.player != null) {
				mc.player.setYRot(yaw);
				mc.player.setXRot(pitch);
			}
		});
	}

	private static boolean want(String name) {
		return scenes.isEmpty() || ("," + scenes + ",").contains("," + name + ",");
	}

	private static void script(Minecraft mc) {
		int t = 10;
		at(t, () -> server(mc, p -> {
			base = origin;
			p.setGameMode(GameType.SURVIVAL);
			Stands.grant(p, StandType.STAR_PLATINUM);
		}));
		t += 40;
		int idx = 1;
		Vec3 side = new Vec3(-5.5, 2.0, 1.8);
		Vec3 sideLook = new Vec3(0, 1.2, 1.8);
		KeyMapping r = ModKeyBindings.ABILITY_1, g = ModKeyBindings.ABILITY_2, x = ModKeyBindings.ABILITY_3, z = ModKeyBindings.ABILITY_4,
				v = ModKeyBindings.ABILITY_5, c = ModKeyBindings.ABILITY_6;
		if (want("m1")) {
			// left clicks: two at the air, then a 3-hit chain on a zombie
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, side, sideLook, false, true, 8f, null);
			record(t + 14, "m1", 80);
			click(t + 20);
			click(t + 32);
			at(t + 40, () -> server(mc, p -> zombie(p, base.add(0, 0, 2.6), 200, false)));
			click(t + 48);
			click(t + 60);
			click(t + 72);
			t += 104;
		}
		if (want("m1fp")) {
			scene(mc, t, idx++, CameraType.FIRST_PERSON, null, null, false, true, 8f, p -> zombie(p, base.add(0, 0, 2.6), 200, false));
			record(t + 14, "m1fp", 60);
			click(t + 20);
			click(t + 34);
			click(t + 48);
			t += 84;
		}
		if (want("float")) {
			// walking: it floats after you, no walk cycle
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-6.0, 1.8, 0.5), new Vec3(0, 1.2, 0.5), true, true, 0f, null);
			record(t + 12, "float", 80);
			hold(mc.options.keyUp, t + 16, 60);
			t += 100;
		}
		if (want("summon")) {
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-4.5, 1.8, -0.5), new Vec3(0, 1.1, -0.3), false, false, 0f, null);
			record(t + 12, "summon", 70);
			at(t + 16, StandClient::pressH);
			at(t + 46, StandClient::pressH);
			at(t + 62, StandClient::pressH);
			t += 90;
		}
		if (want("mining")) {
			// a stone pillar and a log in front; the attack held on each
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-4.5, 2.4, 0.6), new Vec3(0, 1.2, 1.6), false, true, 20f, p -> {
				BlockPos b = BlockPos.containing(base.add(0, 0, 2));
				p.serverLevel().setBlockAndUpdate(b, Blocks.STONE.defaultBlockState());
				p.serverLevel().setBlockAndUpdate(b.above(), Blocks.OAK_LOG.defaultBlockState());
			});
			turn(mc, t + 8, 0f, 30f);
			record(t + 12, "mining", 90);
			mine(mc, t + 16, 30);
			at(t + 48, () -> server(mc, p -> p.getInventory().clearContent())); // (the cobblestone it picked up would fill the hand)
			turn(mc, t + 50, 0f, 4f);
			mine(mc, t + 54, 36);
			t += 110;
		}
		if (want("smash")) {
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, side, sideLook, false, true, 8f, p -> zombie(p, base.add(0, 0, 2.8), 200, false));
			record(t + 14, "smash", 70);
			for (int i = 0; i < 4; i++) {
				hold(r, t + 20 + i * 9, 2);
			}
			t += 92;
		}
		if (want("barrage")) {
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, side, sideLook, false, true, 8f, p -> zombie(p, base.add(0, 0, 2.8), 200, false));
			record(t + 14, "barrage", 80);
			hold(r, t + 20, 40);
			t += 100;
		}
		if (want("finger")) {
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-7, 3.0, 6), new Vec3(0, 1.4, 8), false, true, 2f, p -> {
				zombie(p, base.add(0, 0, 6), 200, false);
				zombie(p, base.add(0, 0, 16), 200, false);
			});
			record(t + 14, "finger", 60);
			shiftPress(r, t + 20, 2);
			t += 80;
		}
		if (want("chase")) {
			// sent straight ahead, then steered to a zombie off to the left
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-4, 6.0, -3), new Vec3(1.5, 1.0, 5), false, true, 0f,
					p -> zombie(p, base.add(2.5, 0, 6.5), 200, false));
			record(t + 14, "chase", 150);
			hold(g, t + 20, 2);
			for (int i = 0; i < 12; i++) {
				turn(mc, t + 26 + i, -i * 2.0f, 0f); // the mouse drifts left: Star Platinum follows it
			}
			t += 170;
		}
		if (want("flick")) {
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-5, 2.2, 4), new Vec3(0, 1.3, 5), false, true, 4f,
					p -> zombie(p, base.add(0, 0, 10), 200, false));
			record(t + 14, "flick", 40);
			shiftPress(g, t + 20, 2);
			t += 60;
		}
		if (want("breaker")) {
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, side, sideLook, false, true, 8f, p -> zombie(p, base.add(0, 0, 2.8), 200, false));
			record(t + 14, "breaker", 60);
			hold(z, t + 18, 26);
			t += 80;
		}
		if (want("jump")) {
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-12, 5, 8), new Vec3(0, 3, 8), false, true, -15f, null);
			record(t + 12, "jump", 60);
			hold(x, t + 16, 2);
			t += 80;
		}
		if (want("orarush")) {
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-7, 3, 5), new Vec3(0, 1, 5), false, true, 0f, p -> {
				zombie(p, base.add(0, 0, 4), 200, false);
				zombie(p, base.add(0.6, 0, 7), 200, false);
			});
			record(t + 12, "orarush", 50);
			shiftPress(x, t + 16, 2);
			t += 70;
		}
		if (want("inhale")) {
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-6, 2.5, 3), new Vec3(0, 1.2, 3), false, true, 0f, p -> {
				zombie(p, base.add(0, 0, 7), 200, false);
				zombie(p, base.add(-1.5, 0, 6), 200, false);
			});
			record(t + 12, "inhale", 60);
			hold(c, t + 16, 2);
			t += 75;
		}
		if (want("uppercut")) {
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, side, sideLook, false, true, 8f, p -> zombie(p, base.add(0, 0, 2.6), 200, false));
			record(t + 14, "uppercut", 50);
			shiftPress(c, t + 20, 2);
			t += 70;
		}
		if (want("guard")) {
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, side, sideLook, false, true, 0f, p -> zombie(p, base.add(0, 0, 2.2), 200, true));
			record(t + 12, "guard", 60);
			hold(v, t + 16, 40);
			t += 76;
		}
		if (want("grab")) {
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, side, sideLook, false, true, 8f, p -> zombie(p, base.add(0, 0, 2.4), 200, false));
			record(t + 14, "grab", 170);
			shiftPress(v, t + 18, 2); // grab
			for (int i = 0; i < 5; i++) {
				hold(r, t + 34 + i * 7, 2); // the 5-hit combo, the fifth knocks it free
			}
			at(t + 80, () -> server(mc, p -> {
				StandState n = Stands.state(p).copy();
				n.abilityReadyAt.clear();
				p.setAttached(ModAttachments.STAND_STATE, n);
				p.serverLevel().getEntitiesOfClass(Zombie.class, p.getBoundingBox().inflate(30)).forEach(zz -> zz.teleportTo(base.x, base.y, base.z + 2.4));
			}));
			shiftPress(v, t + 88, 2); // grab again
			shiftPress(v, t + 112, 2); // Shift+V: set it down gently
			at(t + 124, () -> server(mc, p -> {
				StandState n = Stands.state(p).copy();
				n.abilityReadyAt.clear();
				p.setAttached(ModAttachments.STAND_STATE, n);
			}));
			shiftPress(v, t + 130, 2); // and again
			hold(v, t + 156, 2); // V throws
			t += 190;
		}
		if (want("world")) {
			// The World: zombies walking in (the far one 20 blocks out) freeze mid-step, skeletons' arrows and a falling sand block hang in
			// the air, the colour drains, the caster flies about; Shift+Z again and everything moves on

					zombie(p, base.add(-5 + i * 2.5, 0, 9 + (i % 3) * 5.5), 200, true);
				}
				for (int i = 0; i < 2; i++) {
					var sk = EntityType.SKELETON.create(p.serverLevel());
					sk.moveTo(base.x - 4 + i * 8, base.y, base.z + 14, 180f, 0f);
					sk.setItemSlot(net.minecraft.world.entity.EquipmentSlot.MAINHAND, new ItemStack(net.minecraft.world.item.Items.BOW));
					sk.setItemSlot(net.minecraft.world.entity.EquipmentSlot.HEAD, new ItemStack(net.minecraft.world.item.Items.LEATHER_HELMET));
					sk.setPersistenceRequired();
					p.serverLevel().addFreshEntity(sk);
				}
				p.setGameMode(GameType.SURVIVAL);
			});
			at(t + 50, () -> server(mc, p -> net.minecraft.world.entity.item.FallingBlockEntity.fall(p.serverLevel(),
					BlockPos.containing(base.add(2, 9, 6)), Blocks.SAND.defaultBlockState())));
			record(t + 14, "world", 190);
			shiftPress(z, t + 20, 2);
			// flying during the stop (time stops at t+59): take off, rise, glide forward among the frozen mobs
			// a double-tap of jump takes off (as a player does); then up, then forward at 10 b/s
			hold(mc.options.keyJump, t + 64, 2);
			hold(mc.options.keyJump, t + 68, 2);
			hold(mc.options.keyJump, t + 78, 12);
			hold(mc.options.keyUp, t + 92, 30);
			shiftPress(z, t + 150, 2); // Shift+Z again: time resumes
			t += 220;
		}
		at(t, () -> {
			System.out.println("[HARNESS] DONE");
			Minecraft.getInstance().stop();
		});
	}
