// v0.17.7 after-image check: films the plain Super Speed trail and the Speed Force Speedster trail from the side (no armour, then
// diamond armour). To use: copy to src/client/java/com/projecthero/mod/client/TrailHarness.java and
// scratchpad/TrailHarnessCameraMixin.v0177.java.txt to src/client/java/com/projecthero/mod/client/mixin/TrailHarnessCameraMixin.java,
// add the entrypoint to fabric.mod.json and the mixin to projecthero.client.mixins.json, write run/trail.flag = <out dir>.
// Granting the Speed Force Speedster already turns Speed Mode on (do not toggle it). For same-skin before/after runs use
// ./gradlew runClient --args="--username Player713".
package com.projecthero.mod.client;

import java.io.File;
import java.nio.file.Files;
import java.nio.file.Path;

import net.fabricmc.api.ClientModInitializer;
import net.fabricmc.fabric.api.client.event.lifecycle.v1.ClientTickEvents;
import net.minecraft.client.CameraType;
import net.minecraft.client.KeyMapping;
import net.minecraft.client.Minecraft;
import net.minecraft.client.Screenshot;
import net.minecraft.client.gui.screens.TitleScreen;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.Difficulty;
import net.minecraft.world.entity.EquipmentSlot;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.item.Items;
import net.minecraft.world.level.GameRules;
import net.minecraft.world.level.GameType;
import net.minecraft.world.level.LevelSettings;
import net.minecraft.world.level.WorldDataConfiguration;
import net.minecraft.world.level.levelgen.WorldOptions;
import net.minecraft.world.level.levelgen.presets.WorldPresets;
import net.minecraft.world.phys.Vec3;

/** TEMP-HARNESS (not shipped). */
public final class TrailHarness implements ClientModInitializer {
	public static Vec3 camOffset;
	public static Vec3 camLook = new Vec3(0, 1.0, -3.0);
	private static boolean created;
	private static int wait, t, frame;
	private static String filmDir;
	private static File out;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("trail.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(TrailHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(300_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		});
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[TRAIL] error " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void grab(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		new File(out, "screenshots/" + name).getParentFile().mkdirs();
		Screenshot.grab(out, name, mc.getMainRenderTarget(), m -> {
		});
	}

	private static void armour(ServerPlayer p, boolean on) {
		p.setItemSlot(EquipmentSlot.HEAD, on ? new ItemStack(Items.DIAMOND_HELMET) : ItemStack.EMPTY);
		p.setItemSlot(EquipmentSlot.CHEST, on ? new ItemStack(Items.DIAMOND_CHESTPLATE) : ItemStack.EMPTY);
		p.setItemSlot(EquipmentSlot.LEGS, on ? new ItemStack(Items.DIAMOND_LEGGINGS) : ItemStack.EMPTY);
		p.setItemSlot(EquipmentSlot.FEET, on ? new ItemStack(Items.DIAMOND_BOOTS) : ItemStack.EMPTY);
	}

	private static void superSpeedMode(ServerPlayer p) {
		com.projecthero.mod.hero.Power power = com.projecthero.mod.hero.Powers.byKey(com.projecthero.mod.hero.power.p04.SuperSpeedHandlers.KEY);
		com.projecthero.mod.hero.ExperimentalPowers.grant(p, power);
		com.projecthero.mod.hero.ExperimentalPowers.setActive(p, power);
		for (int s = 1; s <= 6; s++) {
			var a = power.ability(com.projecthero.mod.hero.AbilitySlot.byNumber(s));
			if (a != null && a.id().equals("speed_mode")) {
				com.projecthero.mod.hero.AbilityRouter.handleInput(p, s, true);
				com.projecthero.mod.hero.AbilityRouter.handleInput(p, s, false);
			}
		}
		System.out.println("[TRAIL] super speed mode=" + com.projecthero.mod.hero.power.p04.SuperSpeedHandlers.speedMode(p));
	}

	/** One run: setup at {@code at}, sprint forward 30 ticks while filming into {@code dir}. */
	private static int run(Minecraft mc, int at, String dir, java.util.function.Consumer<ServerPlayer> setup) {
		if (t == at) {
			server(mc, p -> {
				p.connection.teleport(0.5, p.getY(), 0.5, 0f, 0f);
				setup.accept(p);
			});
		}
		if (t == at + 20) {
			System.out.println("[TRAIL] film " + dir + " clientSpeedForceActive=" + com.projecthero.mod.speedforce.SpeedForceSpeedster.speedActive(mc.player) + " superTrail=" + com.projecthero.mod.hero.visual.MutationVisuals.hasFlag(mc.player, com.projecthero.mod.hero.revamp.v0145.SuperSpeedV0145.TRAIL));
			filmDir = dir;
			frame = 0;
		}
		if (t >= at + 20 && t < at + 55) {
			mc.player.setYRot(0f);
			mc.player.setXRot(0f);
			KeyMapping.set(mc.options.keyUp.getDefaultKey(), true);
			mc.player.setSprinting(true);
		}
		if (t == at + 55) {
			KeyMapping.set(mc.options.keyUp.getDefaultKey(), false);
			filmDir = null;
		}
		return at + 70;
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			wait++;
			if (!created && mc.getOverlay() == null && mc.screen instanceof TitleScreen && wait > 40) {
				created = true;
				mc.execute(TrailHarness::createWorld);
			}
			return;
		}
		t++;
		if (t == 30) {
			mc.options.hideGui = true;
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.fovEffectScale().set(0.0);
			camOffset = new Vec3(4.5, 1.6, 1.5);
			server(mc, p -> {
				p.serverLevel().setDayTime(6000L);
				p.serverLevel().getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				p.setGameMode(GameType.SURVIVAL);
			});
		}
		int next = 40;
		next = run(mc, next, "super_bare", p -> {
			superSpeedMode(p);
			armour(p, false);
		});
		next = run(mc, next, "super_armour", p -> armour(p, true));
		next = run(mc, next, "speedforce_bare", p -> {
			com.projecthero.mod.hero.HeroTiers.wipeAll(p);
			armour(p, false);
			System.out.println("[TRAIL] grant " + com.projecthero.mod.hero.PowerGrants.grantHero(p, "speed_force_speedster", true).message().getString());
		});
		next = run(mc, next, "speedforce_armour", p -> armour(p, true));
		if (filmDir != null) {
			grab(mc, filmDir + "/f" + String.format("%03d", frame++) + ".png");
		}
		if (t == next + 5) {
			Runtime.getRuntime().halt(0);
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(GameRules.RULE_DOMOBSPAWNING).set(false, null);
		LevelSettings s = new LevelSettings("TRAIL_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.PEACEFUL, true, rules,
				WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(s.levelName(), s, new WorldOptions(7L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
