/**
 * TEMP-HARNESS (not shipped, v0.17.5 Power Armour round 2: proportions + side-aware opening). run/pa.flag = "&lt;out dir&gt;|before"
 * or "&lt;out dir&gt;|after". before: the suit next to the player (front / side / back / three-quarter) only. after: the same, then the
 * seven tiers, close-ups, the sealed wearer next to a frame and a normal player, a first-person look, and films of entering from
 * the front and the back and stepping out each way. Install: copy to src/client/java/com/projecthero/mod/client/PowerArmourHarness.java,
 * add it to the "client" entrypoints in fabric.mod.json and scratchpad/PowerArmourHarnessCameraMixin.v0175.java.txt to the client
 * mixins. Remove all three before committing.
 */
public final class PowerArmourHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static BlockPos base;
	private static File out;
	private static boolean after;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("pa.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			String[] parts = Files.readString(flag).trim().split("\\|");
			out = new File(parts[0]);
			after = parts.length > 1 && parts[1].equals("after");
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		for (String d : new String[] { "enter_front", "enter_back", "exit_front", "exit_back" }) {
			new File(out, "screenshots/" + d).mkdirs();
		}
		ClientTickEvents.END_CLIENT_TICK.register(PowerArmourHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(540_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "pa-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
	}

	private static Vec3 at(double dx, double dy, double dz) {
		return new Vec3(base.getX() + 0.5 + dx, base.getY() + dy, base.getZ() + 0.5 + dz);
	}

	public static volatile Vec3 camPos, camLook;

	private static void cam(Minecraft mc, Vec3 eye, Vec3 target) {
		camPos = eye;
		camLook = target;
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
	}

	private static void playerCam(Minecraft mc) {
		camPos = null;
		camLook = null;
		mc.options.setCameraType(CameraType.FIRST_PERSON);
	}

	private static SuitData suit(ArmourTier t, ModuleType... mods) {
		SuitData d = SuitData.fresh(t);
		d.core = t.ordinal() >= ArmourTier.VIBRANIUM.ordinal() ? CoreType.STAR_CORE : t.ordinal() >= ArmourTier.DIAMOND.ordinal() ? CoreType.ARC_CORE
				: CoreType.FUSION_CELL;
		d.coreCharge = d.core.capacity;
		for (ModuleType m : mods) {
			for (int i = 0; i < d.modules.length; i++) {
				if (d.modules[i] == null && t.slotPart(i) == m.part) {
					d.modules[i] = m;
					break;
				}
			}
		}
		return d;
	}

	private static PowerArmourFrameEntity frame(ServerLevel level, Vec3 pos, float yaw, SuitData suit) {
		PowerArmourFrameEntity f = PowerArmourMod.FRAME.create(level);
		f.moveTo(pos.x, pos.y, pos.z, yaw, 0f);
		f.setSuit(suit);
		level.addFreshEntity(f);
		return f;
	}

	private static List<PowerArmourFrameEntity> frames(ServerPlayer p) {
		return new ArrayList<>(p.serverLevel().getEntitiesOfClass(PowerArmourFrameEntity.class, p.getBoundingBox().inflate(60)));
	}

	private static void put(ServerPlayer p, Vec3 v, float yaw) {
		p.connection.teleport(v.x, v.y, v.z, yaw, 0f);
		p.setYHeadRot(yaw);
		p.setYBodyRot(yaw);
	}

	private static void stage(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		level.setDayTime(6000L);
		level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
		p.server.setDifficulty(Difficulty.PEACEFUL, true);
		p.setGameMode(GameType.SURVIVAL);
		base = p.blockPosition();
		for (int x = -14; x <= 14; x++) {
			for (int z = -10; z <= 16; z++) {
				level.setBlock(base.offset(x, -1, z), (x + z) % 2 == 0 ? Blocks.POLISHED_DEEPSLATE.defaultBlockState()
						: Blocks.DEEPSLATE_TILES.defaultBlockState(), 3);
				for (int y = 0; y < 12; y++) {
					level.setBlock(base.offset(x, y, z), Blocks.AIR.defaultBlockState(), 3);
				}
			}
		}
		for (int x = -14; x <= 14; x++) {
			for (int y = 0; y < 6; y++) {
				level.setBlock(base.offset(x, y, 15), Blocks.LIGHT_GRAY_CONCRETE.defaultBlockState(), 3);
				level.setBlock(base.offset(x, y, -9), Blocks.LIGHT_GRAY_CONCRETE.defaultBlockState(), 3);
			}
		}
		for (int z = -9; z <= 15; z++) {
			for (int y = 0; y < 6; y++) {
				level.setBlock(base.offset(14, y, z), Blocks.LIGHT_GRAY_CONCRETE.defaultBlockState(), 3);
			}
		}
		frame(level, at(0, 0, 4), 180f, suit(ArmourTier.IRON));
		p.getInventory().clearContent();
		put(p, at(1.05, 0, 4), 180f);
	}

	private static void press(int slot, boolean down, boolean shift) {
		ClientPlayNetworking.send(new AbilityInputPayload(slot, down, shift, 0));
	}

	/** One film: the frame at (0, 0, 4) facing north (yaw 180), the player walks in from {@code from}. */
	private static void film(Minecraft mc, int s, int f0, String dir, Vec3 from, float fromYaw, Vec3 eye, Vec3 look) {
		if (s == f0 - 12) {
			server(mc, p -> {
				frames(p).forEach(Entity::discard);
				frame(p.serverLevel(), at(0, 0, 4), 180f, suit(ArmourTier.HERO_ALLOY, ModuleType.MISSILE_POD, ModuleType.GATLING));
				put(p, from, fromYaw);
			});
			cam(mc, eye, look);
		}
		if (s == f0) {
			server(mc, p -> {
				PowerArmour.tryEnter(p, frames(p).get(0));
				var st = PowerArmour.state(p);
				System.out.println("[HARNESS] " + dir + ": side " + (st == null ? "?" : st.side));
			});
		}
		if (s >= f0 && s < f0 + 100) {
			shot(mc, String.format("%s/f%03d", dir, s - f0));
		}
	}

	private static void exitFilm(Minecraft mc, int s, int x0, String dir, Vec3 eye, Vec3 look) {
		if (s == x0 - 6) {
			cam(mc, eye, look);
			server(mc, p -> {
				var st = PowerArmour.state(p);
				if (st != null) {
					st.readyAt.clear();
					PowerArmour.sync(p);
				}
			});
		}
		if (s == x0 + 4) {
			press(5, true, true);
		}
		if (s == x0 + 4 + 22) {
			press(5, false, false);
		}
		if (s >= x0 && s < x0 + 80) {
			shot(mc, String.format("%s/f%03d", dir, s - x0));
		}
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(PowerArmourHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 40) {
			mc.options.fov().set(60);
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			server(mc, PowerArmourHarness::stage);
		}
		if (t < 100 || base == null) {
			return;
		}
		int s = t - 100;
		if (s % 10 == 0) {
			mc.gui.getChat().clearMessages(false);
		}
		String pre = after ? "after_" : "before_";
		// ---------------- 1. the suit next to the player: front, side, back, three-quarter
		if (s == 0) {
			mc.options.hideGui = true;
			cam(mc, at(0.52, 1.15, -0.6), at(0.52, 1.05, 4));
		}
		if (s == 16) {
			shot(mc, pre + "compare_front");
			server(mc, p -> put(p, at(0, 0, 5.05), 90f));
			cam(mc, at(5.4, 1.15, 4.52), at(0, 1.05, 4.52));
		}
		if (s == 30) {
			shot(mc, pre + "compare_side");
			server(mc, p -> put(p, at(1.05, 0, 4), 180f));
			cam(mc, at(0.52, 1.15, 8.6), at(0.52, 1.05, 4));
		}
		if (s == 44) {
			shot(mc, pre + "compare_back");
			cam(mc, at(3.2, 1.6, 0.9), at(0.4, 1.0, 4));
		}
		if (s == 58) {
			shot(mc, pre + "compare_three_quarter");
		}
		if (s == 61) {
			if (!after) {
				System.out.println("[HARNESS] done (before)");
				try { Thread.sleep(1500L); } catch (InterruptedException e) { }
				Runtime.getRuntime().halt(0);
			}
		}
		// ---------------- 2. the seven tiers, front / back / side
		if (s == 62) {
			server(mc, p -> {
				frames(p).forEach(Entity::discard);
				ArmourTier[] tiers = ArmourTier.values();
				for (int i = 0; i < tiers.length; i++) {
					frame(p.serverLevel(), at(6 - i * 2, 0, 4), 180f, suit(tiers[i]));
				}
				put(p, at(0, 0, -6), 0f);
			});
			cam(mc, at(0, 1.5, -5.4), at(0, 1.0, 4));
		}
		if (s == 80) {
			shot(mc, "tiers_front");
			cam(mc, at(0, 1.5, 12.6), at(0, 1.0, 4));
		}
		if (s == 94) {
			shot(mc, "tiers_back");
			cam(mc, at(8.6, 1.4, 1.2), at(1.5, 1.0, 4.6));
		}
		if (s == 108) {
			shot(mc, "tiers_angle");
			cam(mc, at(0.0, 1.62, 2.0), at(0, 1.45, 4));
		}
		// ---------------- 3. close-ups (Hero Alloy at x 0, Diamond at -2, Celestial at -6, Netherite at -2... by index)
		if (s == 122) {
			shot(mc, "closeup_diamond_chest");
			cam(mc, at(-1.4, 1.95, 2.6), at(-2, 1.7, 4));
		}
		if (s == 136) {
			shot(mc, "closeup_netherite_helmet");
			cam(mc, at(-5.0, 1.3, 2.4), at(-6, 1.1, 4));
		}
		if (s == 150) {
			shot(mc, "closeup_celestial");
			cam(mc, at(-4.0, 1.4, 6.0), at(-4, 1.2, 4));
		}
		if (s == 164) {
			shot(mc, "closeup_vibranium_back");
			cam(mc, at(-2.6, 0.7, 2.2), at(-2, 0.6, 4));
		}
		if (s == 178) {
			shot(mc, "closeup_netherite_legs");
			cam(mc, at(2.9, 1.5, 2.5), at(2, 1.2, 4));
		}
		if (s == 184) {
			shot(mc, "closeup_hero_alloy");
		}
		// ---------------- 4. the wearer next to a frame of the same tier and a normal player's height reference
		if (s == 186) {
			server(mc, p -> {
				frames(p).forEach(Entity::discard);
				frame(p.serverLevel(), at(-0.6, 0, 4), 180f, suit(ArmourTier.IRON));
				PowerArmourFrameEntity f = frame(p.serverLevel(), at(0.6, 0, 4), 180f, suit(ArmourTier.IRON));
				put(p, at(0.6, 0, 4), 180f);
				PowerArmour.enterInstant(p, f);
				put(p, at(0.6, 0, 4), 180f);
			});
			cam(mc, at(0, 1.2, -0.8), at(0, 1.1, 4));
		}
		if (s == 204) {
			shot(mc, "wearer_next_to_frame");
			server(mc, p -> System.out.println("[HARNESS] sealed bbox h=" + p.getBbHeight() + " w=" + p.getBbWidth() + " eye=" + p.getEyeHeight()));
			cam(mc, at(4.4, 1.2, 4.0), at(0, 1.1, 4));
		}
		if (s == 216) {
			shot(mc, "wearer_next_to_frame_side");
			playerCam(mc);
			mc.options.hideGui = false;
			server(mc, p -> put(p, at(0.6, 0, 4), 0f));
		}
		if (s == 232) {
			shot(mc, "first_person_sealed");
			mc.options.hideGui = true;
			server(mc, p -> {
				PowerArmour.exitInstant(p);
				frames(p).forEach(Entity::discard);
			});
		}
		// ---------------- 5. films: enter front, exit front, enter back, exit back
		int e0 = 260;
		film(mc, s, e0, "enter_front", at(-0.7, 0, 1.3), 20f, at(3.0, 1.9, 1.2), at(0.2, 1.0, 3.6));
		int x0 = e0 + 110;
		exitFilm(mc, s, x0, "exit_front", at(3.0, 1.9, 1.2), at(0.2, 1.0, 3.6));
		int b0 = x0 + 96;
		film(mc, s, b0, "enter_back", at(-0.7, 0, 6.7), 160f, at(3.0, 1.9, 6.8), at(0.2, 1.0, 4.4));
		int y0 = b0 + 110;
		exitFilm(mc, s, y0, "exit_back", at(3.0, 1.9, 6.8), at(0.2, 1.0, 4.4));
		// open-side stills (frames opened on one side)
		int o0 = y0 + 90;
		if (s == o0) {
			server(mc, p -> {
				frames(p).forEach(Entity::discard);
				PowerArmourFrameEntity a = frame(p.serverLevel(), at(-1.0, 0, 4), 180f, suit(ArmourTier.DIAMOND));
				a.setSide(0);
				a.setAnim(PowerArmourFrameEntity.OPEN);
				PowerArmourFrameEntity b = frame(p.serverLevel(), at(1.0, 0, 4), 180f, suit(ArmourTier.DIAMOND));
				b.setSide(1);
				b.setAnim(PowerArmourFrameEntity.OPEN);
				put(p, at(0, 0, -6), 0f);
			});
			cam(mc, at(2.4, 1.7, 1.2), at(0, 1.0, 4));
		}
		if (s == o0 + 14) {
			shot(mc, "open_front_vs_back_front_view");
			cam(mc, at(2.4, 1.7, 6.8), at(0, 1.0, 4));
		}
		if (s == o0 + 28) {
			shot(mc, "open_front_vs_back_back_view");
			cam(mc, at(4.4, 1.5, 4.0), at(0, 1.0, 4));
		}
		if (s == o0 + 42) {
			shot(mc, "open_front_vs_back_side_view");
		}
		if (s == o0 + 48) {
			System.out.println("[HARNESS] done");
			try { Thread.sleep(1500L); } catch (InterruptedException e) { }
			Runtime.getRuntime().halt(0);
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("PA2_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.PEACEFUL, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
