/** TEMP-HARNESS (not shipped): films the v0.17.3 Undead kit (Z, Shift+Z, Shift+V, the katana, the shoulder sword). Enabled by run/undeadkit.flag. */
public final class UndeadKitHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static String recording;
	private static int recFrame;
	private static int recUntil;
	private static Vec3 origin;
	private static Vec3 base;
	private static final List<Mob> dummies = new ArrayList<>();
	private static String only;
	/** An orbit for the camera: {centre x, y, z, radius, height, start tick, degrees per tick, start degrees}; null for a fixed camera. */
	private static double[] orbit;

	/** The third-person camera (UndeadKitHarnessCameraMixin): a fixed spot looking at a fixed spot. */
	public static volatile Vec3 camPos;
	public static volatile Vec3 camLook;

	private UndeadKitHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("undeadkit.flag"))) {
			return;
		}
		try {
			only = Files.readString(Path.of("undeadkit.flag")).trim();
		} catch (Exception e) {
			only = "";
		}
		ClientTickEvents.END_CLIENT_TICK.register(UndeadKitHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(1_200_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "undeadkit-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(UndeadKitHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.NORMAL, true);
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			if (orbit != null) {
				double a = Math.toRadians(orbit[7] + (inWorldTicks - t0 - orbit[5]) * orbit[6]);
				camPos = new Vec3(orbit[0] + Math.sin(a) * orbit[3], orbit[1] + orbit[4], orbit[2] + Math.cos(a) * orbit[3]);
				camLook = new Vec3(orbit[0], orbit[1], orbit[2]);
			}
			if (recording != null) {
				String name = String.format("uk_%s_%03d.png", recording, recFrame++);
				Screenshot.grab(mc.gameDirectory, name, mc.getMainRenderTarget(), c -> {
				});
				if (inWorldTicks - t0 >= recUntil) {
					System.out.println("[HARNESS] recorded " + recording + " frames=" + recFrame);
					recording = null;
				}
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("UK_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.NORMAL, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(), mc.screen);
	}

	private static void clearDummies() {
		for (Mob m : dummies) {
			m.discard();
		}
		dummies.clear();
	}

	private static Mob dummy(ServerPlayer p, Vec3 at, float yaw) {
		Mob m = EntityType.ZOMBIE.create(p.serverLevel());
		m.setNoAi(true);
		m.setPersistenceRequired();
		m.getAttribute(Attributes.MAX_HEALTH).setBaseValue(400);
		m.setHealth(400);
		m.moveTo(at.x, at.y, at.z, yaw, 0f);
		m.setYHeadRot(yaw);
		m.setItemSlot(net.minecraft.world.entity.EquipmentSlot.HEAD, new ItemStack(Items.CARVED_PUMPKIN));
		p.serverLevel().addFreshEntity(m);
		dummies.add(m);
		return m;
	}

	private static void record(int start, String name, int ticks) {
		at(start, () -> {
			recording = name;
			recFrame = 0;
			recUntil = start + ticks;
		});
	}

	private static void reset(ServerPlayer p, float pitch) {
		if (Undead.state(p).boneSword()) {
			UndeadBoneSword.putAway(p, false, false);
		}
		if (UndeadShoulderSword.has(p)) {
			p.removeAttached(UndeadShoulderSword.ATTACHMENT);
		}
		p.serverLevel().getEntitiesOfClass(net.minecraft.world.entity.Entity.class, p.getBoundingBox().inflate(128),
				e -> e instanceof net.minecraft.world.entity.monster.Slime || e instanceof Arrow || e instanceof net.minecraft.world.entity.item.ItemEntity || e instanceof com.projecthero.mod.undead.entity.UndeadPartEntity)
				.forEach(e -> e.discard());
		p.clearFire();
		p.removeAllEffects();
		Undead.restore(p);
		UndeadState n = Undead.state(p).copy();
		n.abilityReadyAt.clear();
		n.animId = 0;
		p.setAttached(ModAttachments.UNDEAD_STATE, n);
		p.setItemInHand(InteractionHand.MAIN_HAND, ItemStack.EMPTY);
		p.setShiftKeyDown(false);
		p.setDeltaMovement(Vec3.ZERO);
		p.teleportTo(p.serverLevel(), base.x, base.y, base.z, 0f, pitch);
		p.setYHeadRot(0f);
		p.yBodyRot = 0f;
		p.getFoodData().setFoodLevel(20);
		p.getFoodData().setSaturation(20f);
		p.setHealth(p.getMaxHealth());
	}

	private static void clearCd(ServerPlayer p) {
		UndeadState n = Undead.state(p).copy();
		n.abilityReadyAt.clear();
		p.setAttached(ModAttachments.UNDEAD_STATE, n);
	}

	/** A fresh spot for each scene. */
	private static void scene(Minecraft mc, int t, int index, CameraType type, Vec3 cam, Vec3 look, float pitch, java.util.function.Consumer<ServerPlayer> setup) {
		at(t, () -> {
			base = origin.add(index * 60, 0, 0);
			orbit = null;
			mc.options.setCameraType(type);
			mc.gui.clear();
			mc.gui.getChat().clearMessages(false);
			mc.options.hideGui = type != CameraType.FIRST_PERSON;
			camPos = cam == null ? null : base.add(cam);
			camLook = look == null ? null : base.add(look);
			server(mc, p -> {
				clearDummies();
				reset(p, pitch);
				setup.accept(p);
			});
		});
	}

	private static void key(Minecraft mc, int t, AbilitySlot slot, boolean shift, boolean down) {
		at(t, () -> server(mc, p -> {
			if (down) {
				clearCd(p);
			}
			p.setShiftKeyDown(shift);
			UndeadAbilityManager.handle(p, slot, down);
			if (!down) {
				p.setShiftKeyDown(false);
			}
		}));
	}

	private static void tap(Minecraft mc, int t, AbilitySlot slot, boolean shift) {
		key(mc, t, slot, shift, true);
		key(mc, t, slot, shift, false);
	}

	private static boolean want(String name) {
		return only == null || only.isEmpty() || only.contains(name);
	}

	private static void script(Minecraft mc) {
		int t = 10;
		at(t, () -> server(mc, p -> {
			base = origin;
			p.setGameMode(GameType.SURVIVAL);
			Undead.grant(p);
			reset(p, 0f);
		}));
		t += 30;
		int idx = 1;
		if (want("vortex")) {
			// Z: a line of zombies, seen from the side; the twist, the drill, the burst at its end
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-5.0, 2.6, -3.0), new Vec3(0.5, 1.2, 9.0), 0f, p -> {
				for (int i = 0; i < 6; i++) {
					dummy(p, base.add(0.1, 0, 4 + i * 3.5), 180f);
				}
			});
			record(t + 25, "vortex_tp", 70);
			tap(mc, t + 30, AbilitySlot.SLOT_4, false);
			t += 110;
			scene(mc, t, idx++, CameraType.FIRST_PERSON, null, null, 2f, p -> {
				for (int i = 0; i < 6; i++) {
					dummy(p, base.add(0.1, 0, 4 + i * 3.5), 180f);
				}
			});
			record(t + 25, "vortex_fp", 50);
			tap(mc, t + 30, AbilitySlot.SLOT_4, false);
			t += 90;
		}
		if (want("meteor")) {
			// Shift + hold Z: the 5 s charge and the impact on a crowd 22 blocks out
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-16.0, 6.0, -8.0), new Vec3(0.0, 8.0, 14.0), 4.6f, p -> {
				for (int i = 0; i < 9; i++) {
					double a = i * Math.PI * 2 / 9;
					dummy(p, base.add(Math.cos(a) * 4.5, 0, 20 + Math.sin(a) * 4.5), 180f);
				}
			});
			record(t + 20, "meteor_tp", 175);
			key(mc, t + 25, AbilitySlot.SLOT_4, true, true);
			key(mc, t + 160, AbilitySlot.SLOT_4, true, false);
			t += 220;
			scene(mc, t, idx++, CameraType.FIRST_PERSON, null, null, 4.6f, p -> {
				for (int i = 0; i < 6; i++) {
					double a = i * Math.PI * 2 / 6;
					dummy(p, base.add(Math.cos(a) * 3.5, 0, 20 + Math.sin(a) * 3.5), 180f);
				}
			});
			record(t + 20, "meteor_fp", 165);
			key(mc, t + 25, AbilitySlot.SLOT_4, true, true);
			key(mc, t + 150, AbilitySlot.SLOT_4, true, false);
			t += 210;
		}
		if (want("rib")) {
			// Shift+V held: arrows from in front stop on the ribs
			for (int pass = 0; pass < 2; pass++) {
				boolean fp = pass == 1;
				scene(mc, t, idx++, fp ? CameraType.FIRST_PERSON : CameraType.THIRD_PERSON_BACK, fp ? null : new Vec3(-2.6, 1.9, 2.4),
						fp ? null : new Vec3(0.0, 1.1, 0.9), 0f, p -> {
						});
				String n = fp ? "rib_fp" : "rib_tp";
				record(t + 20, n, 95);
				key(mc, t + 25, AbilitySlot.SLOT_5, true, true);
				for (int k = 0; k < 8; k++) {
					int kk = k;
					at(t + 32 + k * 7, () -> server(mc, p -> {
						Vec3 from = base.add((kk % 3 - 1) * 0.6, 1.3 + (kk % 2) * 0.35, 9.0);
						Vec3 to = base.add(0, 1.25, 0);
						Arrow a = new Arrow(p.serverLevel(), from.x, from.y, from.z, new ItemStack(Items.ARROW), null);
						Vec3 v = to.subtract(from).normalize().scale(1.6);
						a.setDeltaMovement(v);
						p.serverLevel().addFreshEntity(a);
					}));
				}
				key(mc, t + 100, AbilitySlot.SLOT_5, true, false);
				t += 130;
			}
		}
		if (want("katana")) {
			// the katana: drawn, then a close turntable round it, and a still in first person
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-2.0, 1.6, 2.0), new Vec3(0, 1.0, 0.3), 0f, UndeadBoneSword::toggle);
			at(t + 30, () -> orbit = new double[] { base.x - 0.4, base.y + 0.75, base.z + 1.1, 1.9, 0.45, inWorldTicks - t0, 4.0, -60 });
			record(t + 30, "katana_turntable", 90);
			t += 130;
			scene(mc, t, idx++, CameraType.FIRST_PERSON, null, null, 12f, UndeadBoneSword::toggle);
			record(t + 30, "katana_fp_hold", 40);
			t += 80;
			// each sword skill
			String[] names = { "k_crescent", "k_great", "k_flash", "k_whirl" };
			AbilitySlot[] slots = { AbilitySlot.SLOT_1, AbilitySlot.SLOT_2, AbilitySlot.SLOT_4, AbilitySlot.SLOT_6 };
			int[] lens = { 40, 55, 40, 35 };
			for (int i = 0; i < names.length; i++) {
				int ii = i;
				boolean whirl = i == 3;
				scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, whirl ? new Vec3(-4.5, 3.0, -3.5) : new Vec3(-5.5, 2.8, -2.5),
						whirl ? new Vec3(0, 1.0, 0.3) : new Vec3(0.3, 1.2, 6.0), 3f, p -> {
							UndeadBoneSword.toggle(p);
							if (ii == 3) {
								for (int k = 0; k < 6; k++) {
									double a = k * Math.PI * 2 / 6;
									dummy(p, base.add(Math.cos(a) * 3.0, 0, Math.sin(a) * 3.0), 0f);
								}
							} else {
								dummy(p, base.add(0.3, 0, ii == 2 ? 4.0 : 6.0), 180f);
								dummy(p, base.add(-1.2, 0, ii == 2 ? 6.0 : 9.0), 180f);
								dummy(p, base.add(1.4, 0, 12.0), 180f);
							}
						});
				record(t + 35, names[i], lens[i]);
				tap(mc, t + 40, slots[i], false);
				t += 45 + lens[i];
			}
			// the kit in first person
			scene(mc, t, idx++, CameraType.FIRST_PERSON, null, null, 3f, p -> {
				UndeadBoneSword.toggle(p);
				dummy(p, base.add(0.3, 0, 5.0), 180f);
				dummy(p, base.add(-1.2, 0, 8.0), 180f);
			});
			record(t + 35, "k_fp", 150);
			tap(mc, t + 40, AbilitySlot.SLOT_1, false);
			tap(mc, t + 65, AbilitySlot.SLOT_2, false);
			tap(mc, t + 110, AbilitySlot.SLOT_6, false);
			tap(mc, t + 140, AbilitySlot.SLOT_4, false);
			t += 200;
		}
		if (want("shoulder")) {
			// the shoulder sword: stabbed in, a turntable round it, then bare-handed hits that cut like the sword
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(1.6, 2.0, 1.8), new Vec3(0.1, 1.4, 0), 0f, p -> {
				ItemStack s = new ItemStack(Items.DIAMOND_SWORD);
				s.enchant(p.serverLevel().registryAccess().lookupOrThrow(Registries.ENCHANTMENT).getOrThrow(Enchantments.FIRE_ASPECT), 2);
				p.setItemInHand(InteractionHand.MAIN_HAND, s);
			});
			record(t + 20, "shoulder_in", 40);
			at(t + 25, () -> server(mc, p -> {
				p.setShiftKeyDown(true);
				UndeadShoulderSword.interact(p);
				p.setShiftKeyDown(false);
			}));
			at(t + 60, () -> orbit = new double[] { base.x, base.y + 1.5, base.z, 1.7, 0.25, inWorldTicks - t0, 4.5, 20 });
			record(t + 60, "shoulder_turntable", 80);
			t += 150;
			scene(mc, t, idx++, CameraType.THIRD_PERSON_BACK, new Vec3(-3.2, 2.0, 1.2), new Vec3(0, 1.2, 1.2), 0f, p -> {
				ItemStack s = new ItemStack(Items.DIAMOND_SWORD);
				s.enchant(p.serverLevel().registryAccess().lookupOrThrow(Registries.ENCHANTMENT).getOrThrow(Enchantments.FIRE_ASPECT), 2);
				p.setItemInHand(InteractionHand.MAIN_HAND, s);
				p.setShiftKeyDown(true);
				UndeadShoulderSword.interact(p);
				p.setShiftKeyDown(false);
				dummy(p, base.add(0, 0, 2.0), 180f);
			});
			record(t + 25, "shoulder_hit_tp", 70);
			for (int k = 0; k < 3; k++) {
				at(t + 32 + k * 18, () -> server(mc, p -> {
					if (!dummies.isEmpty()) {
						p.resetAttackStrengthTicker();
						try {
							java.lang.reflect.Field f = net.minecraft.world.entity.LivingEntity.class.getDeclaredField("attackStrengthTicker");
							f.setAccessible(true);
							f.setInt(p, 100);
						} catch (ReflectiveOperationException e) {
							// (dev names: unmapped in a mapped run would not happen)
						}
						p.swing(InteractionHand.MAIN_HAND, true);
						p.attack(dummies.get(0));
					}
				}));
			}
			t += 110;
			scene(mc, t, idx++, CameraType.FIRST_PERSON, null, null, 10f, p -> {
				ItemStack s = new ItemStack(Items.DIAMOND_SWORD);
				s.enchant(p.serverLevel().registryAccess().lookupOrThrow(Registries.ENCHANTMENT).getOrThrow(Enchantments.FIRE_ASPECT), 2);
				p.setItemInHand(InteractionHand.MAIN_HAND, s);
				p.setShiftKeyDown(true);
				UndeadShoulderSword.interact(p);
				p.setShiftKeyDown(false);
				dummy(p, base.add(0, 0, 2.0), 180f);
			});
			record(t + 25, "shoulder_hit_fp", 60);
			for (int k = 0; k < 3; k++) {
				at(t + 32 + k * 16, () -> {
					Minecraft.getInstance().player.swing(InteractionHand.MAIN_HAND);
					server(mc, p -> {
						if (!dummies.isEmpty()) {
							try {
								java.lang.reflect.Field f = net.minecraft.world.entity.LivingEntity.class.getDeclaredField("attackStrengthTicker");
								f.setAccessible(true);
								f.setInt(p, 100);
							} catch (ReflectiveOperationException e) {
								// ignore
							}
							p.attack(dummies.get(0));
						}
					});
				});
			}
			t += 100;
		}
		at(t, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}
}
