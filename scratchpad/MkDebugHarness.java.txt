package com.projecthero.mod.client;

import java.util.HashMap;
import java.util.Map;

import com.projecthero.mod.moonknight.MoonKnight;
import com.projecthero.mod.moonknight.MoonKnightAlter;
import com.projecthero.mod.moonknight.data.MoonKnightState;

import net.fabricmc.fabric.api.client.event.lifecycle.v1.ClientTickEvents;

import net.minecraft.client.CameraType;
import net.minecraft.client.Minecraft;
import net.minecraft.client.Screenshot;
import net.minecraft.client.gui.screens.TitleScreen;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.Difficulty;
import net.minecraft.world.level.GameRules;
import net.minecraft.world.level.GameType;
import net.minecraft.world.level.LevelSettings;
import net.minecraft.world.level.WorldDataConfiguration;
import net.minecraft.world.level.levelgen.WorldOptions;
import net.minecraft.world.level.levelgen.presets.WorldPresets;
import net.minecraft.world.phys.Vec3;

/**
 * TEMP-HARNESS (not shipped): v0.13.21 Moon Knight alter suits / swap / glide / block screenshot tour.
 * To use: copy to src/client/java/com/projecthero/mod/client/MkDebugHarness.java, copy HarnessCameraMixin.java.txt
 * into client/mixin with V01315DebugHarness -> MkDebugHarness, add "HarnessCameraMixin" to
 * projecthero.client.mixins.json, call MkDebugHarness.init() in ProjectHeroModClient, ./gradlew runClient -PmodpackModsDir=
 * (shots land in run/screenshots/mk21_*.png, the client quits itself). Remove all four again before committing.
 * The glide is driven for real: survival, 30 blocks up, keyShift.setDown(true) -> the server starts the glide.
 */
public final class MkDebugHarness {
	public static Vec3 camOffset;
	public static double camLookY = 1.1;
	private static boolean createdWorld;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();

	private MkDebugHarness() {
	}

	public static void init() {
		ClientTickEvents.END_CLIENT_TICK.register(MkDebugHarness::tick);
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					fn.accept(srv.getPlayerList().getPlayers().get(0));
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		Screenshot.grab(mc.gameDirectory, "mk21_" + name + ".png", mc.getMainRenderTarget(), x -> {
		});
	}

	private static void at(int t, Runnable r) {
		AT.put(t, r);
	}

	private static void alter(ServerPlayer p, MoonKnightAlter a) {
		MoonKnightState c = MoonKnight.state(p).copy();
		c.abilityReadyAt.clear();
		MoonKnight.saveState(p, c);
		com.projecthero.mod.moonknight.ability.MoonKnightAlters.select(p, a.ordinal());
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			if (!createdWorld && mc.screen instanceof TitleScreen) {
				createdWorld = true;
				LevelSettings settings = new LevelSettings("MkDebug" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
						Difficulty.PEACEFUL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
				mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, WorldOptions.defaultWithRandomSeed(),
						reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
								.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
						mc.screen);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.hideGui = true;
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				MoonKnight.grant(p, true);
				p.setYRot(0);
				p.setYHeadRot(0);
				p.setYBodyRot(0);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	private static void script(Minecraft mc) {
		Vec3 front = new Vec3(1.2, 1.3, 3.2);
		Vec3 back = new Vec3(-1.2, 1.5, -3.2);
		at(5, () -> server(mc, p -> MoonKnight.setTransformedForTesting(p, true)));
		at(8, () -> camOffset = front);
		at(20, () -> shot(mc, "a01_marc_front"));
		at(22, () -> camOffset = back);
		at(26, () -> shot(mc, "a02_marc_back"));
		at(28, () -> camOffset = front);
		at(30, () -> server(mc, p -> alter(p, MoonKnightAlter.STEVEN)));
		at(38, () -> shot(mc, "a03_swap_marc_to_steven_25"));
		at(45, () -> shot(mc, "a04_swap_marc_to_steven_50"));
		at(65, () -> shot(mc, "a05_steven_front"));
		at(67, () -> camOffset = back);
		at(71, () -> shot(mc, "a06_steven_back"));
		at(73, () -> camOffset = front);
		at(75, () -> server(mc, p -> alter(p, MoonKnightAlter.JAKE)));
		at(90, () -> shot(mc, "a07_swap_steven_to_jake_50"));
		at(110, () -> shot(mc, "a08_jake_front"));
		at(112, () -> camOffset = back);
		at(116, () -> shot(mc, "a09_jake_back"));
		at(118, () -> server(mc, p -> alter(p, MoonKnightAlter.MARC)));
		// the glide: survival, up in the air, then hold Sneak for real (the client sends it, the server starts the glide)
		at(150, () -> server(mc, p -> {
			p.setGameMode(GameType.SURVIVAL);
			p.teleportTo(p.getX(), p.getY() + 30, p.getZ());
			p.setYRot(0);
			p.setXRot(10);
		}));
		at(156, () -> mc.options.keyShift.setDown(true));
		at(158, () -> camOffset = new Vec3(4.0, 0.6, 0.0));
		at(168, () -> shot(mc, "b01_glide_side"));
		at(170, () -> camOffset = new Vec3(-1.5, 3.2, -3.0));
		at(174, () -> shot(mc, "b02_glide_above_behind"));
		at(176, () -> camOffset = new Vec3(1.0, -0.8, 4.0));
		at(180, () -> shot(mc, "b03_glide_front_below"));
		at(182, () -> camOffset = new Vec3(0.3, 4.5, -0.3));
		at(186, () -> shot(mc, "b04_glide_top"));
		at(188, () -> server(mc, p -> alter(p, MoonKnightAlter.JAKE)));
		at(220, () -> camOffset = new Vec3(3.0, 1.5, -2.5));
		at(224, () -> shot(mc, "b05_glide_jake"));
		at(226, () -> mc.options.keyShift.setDown(false));
		at(232, () -> shot(mc, "b06_glide_released"));
		// the cape block on the ground
		at(240, () -> server(mc, p -> {
			p.setGameMode(GameType.CREATIVE);
			p.teleportTo(p.getX(), -60, p.getZ());
			p.setXRot(0);
		}));
		at(250, () -> mc.options.keyUse.setDown(true));
		at(256, () -> camOffset = front);
		at(262, () -> shot(mc, "c01_cape_block_front"));
		at(264, () -> camOffset = new Vec3(-2.5, 1.5, -2.0));
		at(268, () -> shot(mc, "c02_cape_block_back"));
		at(270, () -> mc.options.keyUse.setDown(false));
		at(275, () -> camOffset = new Vec3(3.5, 1.4, 0.5));
		at(278, () -> server(mc, p -> com.projecthero.mod.moonknight.ability.MoonKnightDash.dash(p)));
		at(280, () -> shot(mc, "c03_dash"));
		at(290, () -> mc.stop());
	}
}
