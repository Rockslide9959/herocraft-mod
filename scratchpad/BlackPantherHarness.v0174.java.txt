/** TEMP-HARNESS (not shipped): films Black Panther v0.17.4 -- both movesets, the necklace, the claws, the recipe. Enabled by run/bp.flag. */
public final class BlackPantherHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static String recording;
	private static int recFrame;
	private static int recUntil;
	private static Vec3 origin;
	private static Vec3 base;
	private static final List<Mob> dummies = new ArrayList<>();
	private static String only;

	public static volatile Vec3 camPos;
	public static volatile Vec3 camLook;

	private BlackPantherHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("bp.flag"))) {
			return;
		}
		try {
			only = Files.readString(Path.of("bp.flag")).trim();
		} catch (Exception e) {
			only = "";
		}
		ClientTickEvents.END_CLIENT_TICK.register(BlackPantherHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(900_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "bp-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static boolean want(String n) {
		return only == null || only.isEmpty() || only.contains(n);
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(BlackPantherHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.NORMAL, true);
				p.setGameMode(GameType.SURVIVAL);
				BlackPanther.grant(p);
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			if (recording != null) {
				String name = String.format("bp_%s_%03d.png", recording, recFrame++);
				Screenshot.grab(mc.gameDirectory, name, mc.getMainRenderTarget(), c -> {
				});
				if (inWorldTicks - t0 >= recUntil) {
					System.out.println("[HARNESS] recorded " + recording + " frames=" + recFrame);
					recording = null;
				}
			}
		}
	}

	private static void shot(Minecraft mc, int t, String name) {
		at(t, () -> Screenshot.grab(mc.gameDirectory, "bp_still_" + name + ".png", mc.getMainRenderTarget(), c -> {
		}));
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("BP_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.NORMAL, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}

	private static void clearDummies() {
		for (Mob m : dummies) {
			m.discard();
		}
		dummies.clear();
	}

	private static Mob dummy(ServerPlayer p, EntityType<? extends Mob> type, Vec3 at, float yaw) {
		Mob m = type.create(p.serverLevel());
		m.setNoAi(true);
		m.setPersistenceRequired();
		m.getAttribute(Attributes.MAX_HEALTH).setBaseValue(1000);
		m.setHealth(1000);
		m.addEffect(new MobEffectInstance(MobEffects.FIRE_RESISTANCE, 100000, 0, false, false));
		m.setItemSlot(EquipmentSlot.HEAD, new ItemStack(Items.LEATHER_HELMET));
		m.moveTo(at.x, at.y, at.z, yaw, 0f);
		m.setYHeadRot(yaw);
		m.setYBodyRot(yaw);
		p.serverLevel().addFreshEntity(m);
		dummies.add(m);
		return m;
	}

	private static void record(int start, String name, int ticks) {
		at(start, () -> {
			recording = name;
			recFrame = 0;
			recUntil = start + ticks;
		});
	}

	private static void reset(ServerPlayer p, float yaw, boolean suited) {
		Minecraft mc = Minecraft.getInstance();
		mc.options.keyShift.setDown(false);
		p.clearFire();
		p.getInventory().clearContent();
		p.removeAllEffects();
		p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 100000, 4, false, false));
		p.setDeltaMovement(Vec3.ZERO);
		p.teleportTo(p.serverLevel(), base.x, base.y, base.z, yaw, 0f);
		p.setYHeadRot(yaw);
		p.yBodyRot = yaw;
		p.setHealth(p.getMaxHealth());
		var s = BlackPanther.state(p).copy();
		s.abilityReadyAt.clear();
		s.animId = 0;
		s.habit = true;
		s.necklaceOff = false;
		s.clawsOut = false;
		s.absorbing = false;
		s.stalking = false;
		s.senses = false;
		if (suited) {
			s.suited = true;
			s.suitChangeAt = p.level().getGameTime() - 100;
		} else {
			s.suited = false;
			s.suitChangeAt = p.level().getGameTime() - 100;
		}
		p.setAttached(com.projecthero.mod.attachment.ModAttachments.BLACK_PANTHER_STATE, s);
		BlackPanther.reconcile(p);
	}

	private static void scene(Minecraft mc, int t, int index, boolean suited, CameraType type, Vec3 cam, Vec3 look, float yaw,
			Consumer<ServerPlayer> setup) {
		at(t, () -> {
			base = origin.add(index * 30, 0, 0);
			mc.options.setCameraType(type);
			mc.options.hideGui = type != CameraType.FIRST_PERSON;
			camPos = cam == null ? null : base.add(cam);
			camLook = look == null ? null : base.add(look);
			server(mc, p -> {
				clearDummies();
				reset(p, yaw, suited);
				setup.accept(p);
			});
		});
	}

	private static void look(ServerPlayer p, Mob m) {
		p.lookAt(net.minecraft.commands.arguments.EntityAnchorArgument.Anchor.EYES, m.getEyePosition().add(0, -0.3, 0));
		p.connection.teleport(p.getX(), p.getY(), p.getZ(), p.getYRot(), p.getXRot());
	}

	/** One filmed move: a scene, a beat, the move, and the recording. */
	private static int move(Minecraft mc, int t, int[] idx, String name, boolean suited, Vec3 cam, Vec3 look, int len, Consumer<ServerPlayer> setup,
			Consumer<ServerPlayer> act) {
		if (!want(name)) {
			return t;
		}
		scene(mc, t, idx[0]++, suited, CameraType.THIRD_PERSON_BACK, cam, look, 0f, setup);
		at(t + 12, () -> server(mc, act));
		record(t + 8, name, len);
		return t + len + 20;
	}

	private static int fp(Minecraft mc, int t, int[] idx, String name, boolean suited, int len, Consumer<ServerPlayer> setup,
			Consumer<ServerPlayer> act) {
		if (!want(name)) {
			return t;
		}
		scene(mc, t, idx[0]++, suited, CameraType.FIRST_PERSON, null, null, 0f, setup);
		at(t + 12, () -> server(mc, act));
		record(t + 8, name, len);
		return t + len + 20;
	}

	private static Consumer<ServerPlayer> zombieAhead(double dist) {
		return p -> dummy(p, EntityType.ZOMBIE, base.add(0, 0, dist), 180f);
	}

		}, p -> {
			look(p, dummies.get(0));
			BlackPantherAbilities.dagger(p);
		});
		t = move(mc, t, idx, "s_flash", true, WIDE, WIDE_LOOK, 30, zombieAhead(6), p -> {
			look(p, dummies.get(0));
			BlackPantherAbilities.kimoyo(p);
		});
		t = move(mc, t, idx, "s_absorb", true, SIDE, SIDE_LOOK, 40, zombieAhead(1.8), BlackPantherAbilities::raiseStance);
		at(t - 40, () -> server(mc, p -> {
			if (!dummies.isEmpty()) {
				dummies.get(0).swing(net.minecraft.world.InteractionHand.MAIN_HAND);
				p.hurt(p.damageSources().mobAttack(dummies.get(0)), 4f);
			}
		}));
		at(t - 22, () -> server(mc, BlackPantherAbilities::lowerStance));
		t = move(mc, t, idx, "s_return", true, WIDE, WIDE_LOOK, 26, p -> {
			Mob sk = dummy(p, EntityType.SKELETON, base.add(0, 0, 8), 180f);
			Arrow a = new Arrow(EntityType.ARROW, p.serverLevel());
			a.setOwner(sk);
			p.hurt(p.damageSources().arrow(a, sk), 6f);
		}, BlackPantherAbilities::kineticReturn);

		// ---------------------------------------------------------------- first person
		t = fp(mc, t, idx, "fp_u_combo", false, 28, zombieAhead(2.2), BlackPantherUnsuited::combo);
		t = fp(mc, t, idx, "fp_u_takedown", false, 26, zombieAhead(2.0), BlackPantherUnsuited::takedown);
		t = fp(mc, t, idx, "fp_u_parry", false, 30, zombieAhead(1.8), p -> {
			BlackPantherUnsuited.raiseParry(p);
			p.hurt(p.damageSources().mobAttack(dummies.get(0)), 4f);
		});
		at(t - 4, () -> server(mc, BlackPantherUnsuited::lowerParry));
		t = fp(mc, t, idx, "fp_u_resolve", false, 32, p -> {
		}, BlackPantherUnsuited::resolve);
		t = fp(mc, t, idx, "fp_s_claws", true, 32, zombieAhead(2.2), BlackPantherAbilities::claws);
		t = fp(mc, t, idx, "fp_s_pulse", true, 24, p -> BlackPanther.setKinetic(p, 60f), BlackPantherAbilities::pulse);
		t = fp(mc, t, idx, "fp_s_suit_up", false, 52, p -> {
		}, BlackPanther::suitUp);

		at(t, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
