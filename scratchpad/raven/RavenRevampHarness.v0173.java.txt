/**
 * TEMP-HARNESS (not shipped): the v0.17.3 Raven revamp proof -- the wheel, the HUD, every main move and every spell. Enabled by
 * run/ravenrevamp.flag (its text filters scenes). Frames: run/screenshots/rr_*. Hook in/out with
 * scratchpad/raven/revamp_hook.js; films with scratchpad/raven/revamp_films.sh.
 */
public final class RavenRevampHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static String recording;
	private static int recFrame;
	private static int recUntil;
	private static Vec3 origin;
	private static String only;

	public static volatile Float orbitYaw;
	public static volatile float orbitHeight = 0.15f;
	public static volatile float orbitRadius = 2.0f;
	public static volatile float orbitSpin = 0f;
	public static volatile float lookHeight = 1.55f;

	private RavenRevampHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("ravenrevamp.flag"))) {
			return;
		}
		try {
			only = Files.readString(Path.of("ravenrevamp.flag")).trim();
		} catch (Exception e) {
			only = "";
		}
		ClientTickEvents.END_CLIENT_TICK.register(RavenRevampHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(900_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "ravenrevamp-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(RavenRevampHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.EASY, true);
				p.setGameMode(GameType.SURVIVAL);
				p.removeAllEffects();
				p.getAbilities().invulnerable = true;
				p.onUpdateAbilities();
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			if (orbitYaw != null && orbitSpin != 0f) {
				orbitYaw = orbitYaw + orbitSpin;
			}
			if (recording != null) {
				String name = String.format("rr_%s_%03d.png", recording, recFrame++);
				Screenshot.grab(mc.gameDirectory, name, mc.getMainRenderTarget(), c -> {
				});
				if (inWorldTicks - t0 >= recUntil) {
					System.out.println("[HARNESS] recorded " + recording + " frames=" + recFrame);
					recording = null;
				}
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("RR_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false,
				Difficulty.EASY, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	private static void record(int start, String name, int ticks) {
		at(start, () -> {
			recording = name;
			recFrame = 0;
			recUntil = start + ticks;
		});
	}

	private static boolean want(String name) {
		return only == null || only.isEmpty() || only.contains(name);
	}

	/** A fresh Raven at the origin facing +z, full and calm, nothing left over from the last scene. */
	private static void raven(Minecraft mc, int t, float pitch, float composure) {
		at(t, com.projecthero.mod.client.raven.RavenFx::reset); // nothing left over from the last scene's effects
		at(t, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			AABB box = new AABB(origin, origin).inflate(48);
			for (Entity e : level.getEntitiesOfClass(Entity.class, box, e -> e instanceof Husk || e instanceof ItemEntity || e instanceof Arrow)) {
				e.discard();
			}
			RavenRestore.restoreAll();
			if (!Raven.hasPower(p)) {
				Raven.grant(p);
			}
			Raven.clearTransient(p);
			p.removeAllEffects();
			p.setDeltaMovement(Vec3.ZERO);
			p.getAbilities().invulnerable = true;
			p.onUpdateAbilities();
			p.teleportTo(level, origin.x, origin.y, origin.z, 0f, pitch);
			p.setYHeadRot(0f);
			p.yBodyRot = 0f;
			p.setHealth(p.getMaxHealth());
			Raven.clearCooldowns(p);
			Raven.setEnergy(p, 100f);
			Raven.setComposure(p, composure);
		}));
	}

	private static void husk(Minecraft mc, int t, double dx, double dz) {
		at(t, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			Husk h = EntityType.HUSK.create(level);
			if (h == null) {
				return;
			}
			h.moveTo(origin.x + dx, origin.y, origin.z + dz, 180f, 0f);
			h.setYHeadRot(180f);
			h.setPersistenceRequired();
			h.getAttribute(Attributes.MAX_HEALTH).setBaseValue(300);
			h.setHealth(300);
			h.getAttribute(Attributes.MOVEMENT_SPEED).setBaseValue(0);
			h.setSilent(true);
			h.setNoAi(true);
			h.setBaby(false);
			level.addFreshEntity(h);
		}));
	}

	private static void cam(int t, Float yaw, float height, float radius, float spin, float look) {
		at(t, () -> {
			orbitYaw = yaw;
			orbitHeight = height;
			orbitRadius = radius;
			orbitSpin = spin;
			lookHeight = look;
		});
	}

	private static void act(Minecraft mc, int t, Consumer<ServerPlayer> fn) {
		at(t, () -> server(mc, fn));
	}

	private static void use(Minecraft mc, int t, String id) {
		act(mc, t, p -> {
			Raven.clearCooldowns(p);
			Raven.setEnergy(p, 100f);
			boolean ok = RavenMoves.use(p, id);
			System.out.println("[HARNESS] use " + id + " -> " + ok);
		});
	}

	private static void spell(Minecraft mc, int t, RavenSpell s) {
		act(mc, t, p -> {
			Raven.clearCooldowns(p);
			Raven.setEnergy(p, 100f);
			RavenSpells.select(p, s.ordinal());
			boolean ok = RavenSpells.cast(p, s, false);
			System.out.println("[HARNESS] spell " + s.id + " -> " + ok);
		});
	}

	/** The client player's look (the client owns it in single player), turned smoothly from one to the other. */
	private static void turn(Minecraft mc, int t, int ticks, float yaw0, float yaw1, float pitch0, float pitch1) {
		for (int i = 0; i <= ticks; i++) {
			float k = i / (float) ticks;
			float yaw = yaw0 + (yaw1 - yaw0) * k, pitch = pitch0 + (pitch1 - pitch0) * k;
			at(t + i, () -> {
				if (mc.player != null) {
					mc.player.setYRot(yaw);
					mc.player.setXRot(pitch);
					mc.player.setYHeadRot(yaw);
					mc.player.yBodyRot = yaw;
				}
			});
		}
	}

	/** The client player looks at a spot (offset from the origin). */
	private static void lookAt(Minecraft mc, int t, double dx, double dy, double dz) {
		at(t, () -> {
			if (mc.player != null) {
				mc.player.lookAt(net.minecraft.commands.arguments.EntityAnchorArgument.Anchor.EYES, origin.add(dx, dy, dz));
				mc.player.yBodyRot = mc.player.getYRot();
			}
		});
	}

	private static void arrowAt(Minecraft mc, int t, double fromDx, double fromDz) {
		act(mc, t, p -> {
			ServerLevel level = p.serverLevel();
			Arrow a = new Arrow(EntityType.ARROW, level);
			Husk owner = level.getEntitiesOfClass(Husk.class, new AABB(origin, origin).inflate(20)).stream().findFirst().orElse(null);
			a.setOwner(owner);
			Vec3 from = origin.add(fromDx, 1.4, fromDz);
			a.setPos(from.x, from.y, from.z);
			Vec3 to = p.position().add(0, 1.0, 0).subtract(from).normalize().scale(1.2);
			a.setDeltaMovement(to);
			level.addFreshEntity(a);
		});
	}

	private static void view(Minecraft mc, int t, boolean gui) {
		at(t, () -> {
			mc.options.hideGui = !gui;
			clean(mc);
		});
	}

	/** No chat, title, toast or tutorial over the HUD shots. */
	private static void clean(Minecraft mc) {
		mc.gui.getChat().clearMessages(false);
		mc.gui.clear();
		mc.getToasts().clear();
		mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
	}

	private static int scene(Minecraft mc, int t, String name, int len, java.util.function.IntConsumer body) {
		if (!want(name)) {
			return t;
		}
		body.accept(t);
		return t + len;
	}
