	/** Holds movement keys down for {@code ticks} ticks from {@code t} (re-pressed every tick, released after). */
	private static void keys(Minecraft mc, int t, int ticks, boolean up, boolean down, boolean sprint) {
		for (int i = 0; i < ticks; i++) {
			at(t + i, () -> {
				mc.options.keyUp.setDown(up);
				mc.options.keyDown.setDown(down);
				mc.options.keySprint.setDown(sprint);
				if (mc.player != null) {
					mc.player.setSprinting(sprint);
				}
			});
		}
		at(t + ticks, () -> {
			mc.options.keyUp.setDown(false);
			mc.options.keyDown.setDown(false);
			mc.options.keySprint.setDown(false);
			if (mc.player != null) {
				mc.player.setSprinting(false);
			}
		});
	}

	/** Up in the air and levitating. */
	private static void fly(Minecraft mc, int t, double up) {
		act(mc, t, p -> {
			p.teleportTo(p.serverLevel(), origin.x, origin.y + up, origin.z, 0f, 0f);
			com.projecthero.mod.raven.RavenFlight.toggle(p);
			System.out.println("[HARNESS] flying " + Raven.isFlying(p));
		});
	}

	private static void camera(Minecraft mc, int t, CameraType type) {
		at(t, () -> mc.options.setCameraType(type));
	}
