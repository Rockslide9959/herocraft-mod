/**
 * TEMP-HARNESS (not shipped): v0.16.6 Ladybug stills. A fake player is Ladybug; the local player is the camera (and,
 * at the end, Ladybug herself for the first-person and Ladybug Vision shots). Stills go to run/screenshots/lb_*.png.
 * Enabled by run/lb.flag.
 */
public final class LadybugHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static ServerPlayer lb;
	private static Vec3 origin;
	private static final List<Mob> dummies = new ArrayList<>();
	private static boolean move;

	private LadybugHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("lb.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(LadybugHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(600_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "lb-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void snap(Minecraft mc, int t, String name) {
		at(t, () -> Screenshot.grab(mc.gameDirectory, "lb_" + name + ".png", mc.getMainRenderTarget(), c -> {
		}));
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(LadybugHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.NORMAL, true);
				p.setGameMode(GameType.CREATIVE);
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			server(mc, p -> physics());
		}
	}

	private static void physics() {
		if (lb == null || lb.isRemoved()) {
			return;
		}
		lb.tickCount++;
		Miraculous.tick(lb);
		if (move) {
			Vec3 v = lb.getDeltaMovement();
			lb.move(MoverType.SELF, v);
			double vy = lb.onGround() ? 0 : v.y - 0.08;
			lb.setDeltaMovement(v.x * 0.91, vy * 0.98, v.z * 0.91);
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("LB_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
				Difficulty.NORMAL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	/** A uuid whose default skin is {@code skin} (e.g. "wide/steve"), so the fake wears it offline. */
	private static java.util.UUID uuidFor(String skin) {
		for (int i = 0; i < 100000; i++) {
			java.util.UUID u = java.util.UUID.randomUUID();
			if (net.minecraft.client.resources.DefaultPlayerSkin.get(u).texture().getPath().endsWith(skin + ".png")) {
				return u;
			}
		}
		return java.util.UUID.randomUUID();
	}

	private static ServerPlayer fake(ServerPlayer near, String name, Vec3 at, String skin) {
		ServerLevel level = near.serverLevel();
		GameProfile profile = new GameProfile(uuidFor(skin), name);
		CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
		ServerPlayer f = new ServerPlayer(level.getServer(), level, profile, cookie.clientInformation());
		Connection conn = new Connection(PacketFlow.SERVERBOUND);
		new io.netty.channel.embedded.EmbeddedChannel(conn);
		level.getServer().getPlayerList().placeNewPlayer(conn, f, cookie);
		f.setGameMode(GameType.SURVIVAL);
		f.teleportTo(level, at.x, at.y, at.z, 0f, 0f);
		f.setYHeadRot(0f);
		f.yBodyRot = 0f;
		return f;
	}

	private static void camera(ServerPlayer p, Vec3 eye, Vec3 target) {
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		p.teleportTo(p.serverLevel(), eye.x, eye.y - 1.62, eye.z, yaw, pit);
	}

	private static void face(ServerPlayer r, float yaw, float pitch) {
		r.teleportTo(r.serverLevel(), r.getX(), r.getY(), r.getZ(), yaw, pitch);
		r.setYHeadRot(yaw);
		r.yBodyRot = yaw;
		r.yBodyRotO = yaw;
	}

	private static void home(ServerPlayer r) {
		r.setDeltaMovement(Vec3.ZERO);
		r.teleportTo(r.serverLevel(), origin.x, origin.y, origin.z, 0f, 0f);
		face(r, 0f, 0f);
	}

	private static void ready(ServerPlayer r) {
		Miraculous.edit(r, n -> {
			n.abilityReadyAt.clear();
			n.formChangeAt = r.level().getGameTime() - 200L;
			n.animId = MiraculousState.ANIM_NONE;
		});
	}

	private static void clearDummies() {
		for (Mob m : dummies) {
			m.discard();
		}
		dummies.clear();
	}

	private static <T extends Mob> T dummy(ServerPlayer p, EntityType<T> type, Vec3 at, float yaw) {
		T m = type.create(p.serverLevel());
		m.setNoAi(true);
		m.setPersistenceRequired();
		m.getAttribute(Attributes.MAX_HEALTH).setBaseValue(100);
		m.setHealth(100);
		m.moveTo(at.x, at.y, at.z, yaw, 0f);
		m.setYHeadRot(yaw);
		m.setItemSlot(net.minecraft.world.entity.EquipmentSlot.HEAD, new ItemStack(net.minecraft.world.item.Items.CARVED_PUMPKIN));
		p.serverLevel().addFreshEntity(m);
		dummies.add(m);
		return m;
	}

	private static void block(ServerPlayer p, Vec3 rel, net.minecraft.world.level.block.Block b) {
		p.serverLevel().setBlock(BlockPos.containing(origin.add(rel)), b.defaultBlockState(), 3);
	}

	private static Vec3 side(double along, double dist, double h) {
		return origin.add(-dist, h, along);
	}
