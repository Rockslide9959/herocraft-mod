/**
 * TEMP-HARNESS (not shipped): films every Gyro Zeppeli move by calling GyroMoves directly. Enabled by run/gyro.flag (its text
 * filters scenes). Frames: run/screenshots/gy_*. Hook in/out with scratchpad/gyro/harness_hook.js; films with
 * scratchpad/gyro/films.sh.
 */
public final class GyroHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static String recording;
	private static int recFrame;
	private static int recUntil;
	private static Vec3 origin;
	private static String only;

	public static volatile Float orbitYaw;
	public static volatile float orbitHeight = 0.15f;
	public static volatile float orbitRadius = 2.0f;
	public static volatile float orbitSpin = 0f;
	public static volatile float lookHeight = 1.55f;

	private GyroHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("gyro.flag"))) {
			return;
		}
		try {
			only = Files.readString(Path.of("gyro.flag")).trim();
		} catch (Exception e) {
			only = "";
		}
		ClientTickEvents.END_CLIENT_TICK.register(GyroHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(900_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "gyro-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(GyroHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.EASY, true);
				p.setGameMode(GameType.SURVIVAL);
				p.removeAllEffects();
				Gyro.grant(p);
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			if (orbitYaw != null && orbitSpin != 0f) {
				orbitYaw = orbitYaw + orbitSpin;
			}
			if (recording != null) {
				String name = String.format("gy_%s_%03d.png", recording, recFrame++);
				Screenshot.grab(mc.gameDirectory, name, mc.getMainRenderTarget(), c -> {
				});
				if (inWorldTicks - t0 >= recUntil) {
					System.out.println("[HARNESS] recorded " + recording + " frames=" + recFrame);
					recording = null;
				}
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("GY_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false,
				Difficulty.EASY, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	private static void record(int start, String name, int ticks) {
		if (!want(name)) {
			return;
		}
		at(start, () -> {
			recording = name;
			recFrame = 0;
			recUntil = start + ticks;
		});
	}

	private static boolean want(String name) {
		return only == null || only.isEmpty() || only.contains(name);
	}

	/** Back to the spot facing south; dummies gone; full Spin, both balls home, no cooldowns. */
	private static void reset(Minecraft mc, int t, float pitch) {
		at(t, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			for (Husk h : level.getEntitiesOfClass(Husk.class, new AABB(origin, origin).inflate(48))) {
				h.discard();
			}
			if (p.getVehicle() != null) {
				p.stopRiding();
			}
			p.removeAllEffects();
			p.setDeltaMovement(Vec3.ZERO);
			p.teleportTo(level, origin.x, origin.y, origin.z, 0f, pitch);
			p.setYHeadRot(0f);
			p.yBodyRot = 0f;
			p.setXRot(pitch);
			p.setHealth(p.getMaxHealth());
			Gyro.clearTransient(p);
			Gyro.clearCooldowns(p);
			Gyro.setSpin(p, 100f);
		}));
	}

	private static void dummy(Minecraft mc, int t, double dx, double dz) {
		at(t, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			Husk h = EntityType.HUSK.create(level);
			if (h == null) {
				return;
			}
			h.moveTo(origin.x + dx, origin.y, origin.z + dz, 180f, 0f);
			h.setYHeadRot(180f);
			h.setPersistenceRequired();
			h.getAttribute(Attributes.MAX_HEALTH).setBaseValue(280);
			h.setHealth(280);
			h.getAttribute(Attributes.MOVEMENT_SPEED).setBaseValue(0);
			h.setSilent(true);
			h.setNoAi(true);
			level.addFreshEntity(h);
		}));
	}

	private static void use(Minecraft mc, int t, String id) {
		at(t, () -> server(mc, p -> GyroMoves.use(p, id)));
	}

	private static void charge(Minecraft mc, int t, String id, int hold) {
		at(t, () -> server(mc, p -> GyroMoves.startCharge(p, id)));
		at(t + hold, () -> server(mc, p -> GyroMoves.releaseCharge(p, id)));
	}

	private static void cam(int t, Float yaw, float height, float radius, float spin, float look) {
		at(t, () -> {
			orbitYaw = yaw;
			orbitHeight = height;
			orbitRadius = radius;
			orbitSpin = spin;
			lookHeight = look;
		});
	}

	private static void firstPerson(Minecraft mc, int t, boolean on) {
		at(t, () -> {
			if (on) {
				orbitYaw = null;
				mc.options.setCameraType(CameraType.FIRST_PERSON);
				mc.options.hideGui = false;
			} else {
				mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
				mc.options.hideGui = true;
			}
		});
	}

	private static void script(Minecraft mc) {
		int t = 20;
		if (want("look")) {
			reset(mc, t, 0f);
			cam(t + 1, 0f, 0.1f, 2.4f, 6f, 1.1f);
			record(t + 4, "look", 60);
			t += 68;
		}
		if (want("hud")) {
			reset(mc, t, 0f);
			firstPerson(mc, t + 1, true);
			record(t + 4, "hud", 30);
			use(mc, t + 10, GyroAbilityManager.STEEL_BALL);
			firstPerson(mc, t + 36, false);
			t += 40;
		}
		if (want("steel_ball")) {
			reset(mc, t, 2f);
			dummy(mc, t + 1, 0, 9);
			cam(t + 1, 110f, 0.8f, 5.5f, 0f, 1.3f);
			record(t + 4, "steel_ball", 50);
			use(mc, t + 8, GyroAbilityManager.STEEL_BALL);
			t += 58;
		}
		if (want("twin_balls")) {
			reset(mc, t, 2f);
			dummy(mc, t + 1, 0, 9);
			cam(t + 1, 160f, 3.5f, 6.0f, 0f, 1.0f);
			record(t + 4, "twin_balls", 50);
			use(mc, t + 8, GyroAbilityManager.TWIN_BALLS);
			t += 58;
		}
		if (want("spin_touch")) {
			reset(mc, t, 8f);
			dummy(mc, t + 1, 0, 2.2);
			cam(t + 1, 75f, 0.5f, 4.0f, 0f, 1.2f);
			record(t + 4, "spin_touch", 50);
			use(mc, t + 8, GyroAbilityManager.SPIN_TOUCH);
			t += 58;
		}
		if (want("golden_fp")) {
			reset(mc, t, 2f);
			dummy(mc, t + 1, 0, 7);
			dummy(mc, t + 1, 0.2, 11);
			firstPerson(mc, t + 1, true);
			record(t + 4, "golden_fp", 56);
			charge(mc, t + 8, GyroAbilityManager.GOLDEN_SPIN, 30);
			firstPerson(mc, t + 62, false);
			t += 66;
		}
		if (want("golden_tp")) {
			reset(mc, t, 2f);
			dummy(mc, t + 1, 0, 7);
			dummy(mc, t + 1, 0.2, 11);
			cam(t + 1, 100f, 1.5f, 7.0f, 0f, 1.3f);
			record(t + 4, "golden_tp", 56);
			charge(mc, t + 8, GyroAbilityManager.GOLDEN_SPIN, 30);
			t += 64;
		}
		if (want("scan")) {
			reset(mc, t, 30f);
			dummy(mc, t + 1, 6, 14);
			dummy(mc, t + 1, -8, 20);
			// a wall between him and the dummies: the outlines show through it
			at(t + 1, () -> server(mc, p -> {
				for (int x = -10; x <= 10; x++) {
					for (int y = 0; y < 4; y++) {
						p.serverLevel().setBlockAndUpdate(BlockPos.containing(origin.x + x, origin.y + y, origin.z + 5), Blocks.STONE_BRICKS.defaultBlockState());
					}
				}
			}));
			firstPerson(mc, t + 2, true);
			record(t + 4, "scan", 50);
			use(mc, t + 8, GyroAbilityManager.SCAN);
			at(t + 30, () -> server(mc, p -> p.setXRot(0f)));
			at(t + 56, () -> server(mc, p -> {
				for (int x = -10; x <= 10; x++) {
					for (int y = 0; y < 4; y++) {
						p.serverLevel().setBlockAndUpdate(BlockPos.containing(origin.x + x, origin.y + y, origin.z + 5), Blocks.AIR.defaultBlockState());
					}
				}
			}));
			firstPerson(mc, t + 57, false);
			t += 60;
		}
		if (want("ball_breaker")) {
			reset(mc, t, 2f);
			dummy(mc, t + 1, 0, 7);
			cam(t + 1, 70f, 1.2f, 7.0f, 0f, 1.4f);
			record(t + 4, "ball_breaker", 75);
			charge(mc, t + 8, GyroAbilityManager.BALL_BREAKER, 30);
			t += 82;
		}
		if (want("breaker_fp")) {
			reset(mc, t, 2f);
			dummy(mc, t + 1, 0, 7);
			firstPerson(mc, t + 1, true);
			record(t + 4, "breaker_fp", 70);
			charge(mc, t + 8, GyroAbilityManager.BALL_BREAKER, 30);
			firstPerson(mc, t + 76, false);
			t += 80;
		}
		if (want("spin_leap")) {
			reset(mc, t, 0f);
			cam(t + 1, 95f, 2.0f, 11.0f, 0f, 2.5f);
			record(t + 4, "spin_leap", 50);
			use(mc, t + 8, GyroAbilityManager.SPIN_LEAP);
			t += 58;
		}
		if (want("valkyrie")) {
			reset(mc, t, 0f);
			cam(t + 1, 60f, 2.0f, 9.0f, 0f, 1.2f);
			record(t + 4, "valkyrie", 80);
			use(mc, t + 8, GyroAbilityManager.VALKYRIE);
			t += 88;
		}
		if (want("spin_quake")) {
			reset(mc, t, 0f);
			for (int i = 0; i < 5; i++) {
				double a = i * Math.PI * 2 / 5 + 0.3;
				dummy(mc, t + 1, Math.sin(a) * 3.5, Math.cos(a) * 3.5);
			}
			cam(t + 1, 30f, 3.5f, 9.0f, 0f, 0.8f);
			record(t + 4, "spin_quake", 40);
			use(mc, t + 10, GyroAbilityManager.SPIN_QUAKE);
			t += 48;
		}
		if (want("spin_shield")) {
			reset(mc, t, 0f);
			cam(t + 1, 70f, 0.8f, 5.0f, 0f, 1.2f);
			record(t + 4, "spin_shield", 80);
			use(mc, t + 8, GyroAbilityManager.SPIN_SHIELD);
			for (int k = 0; k < 4; k++) {
				int kk = k;
				at(t + 20 + k * 12, () -> server(mc, p -> {
					Arrow arrow = new Arrow(EntityType.ARROW, p.serverLevel());
					double a = kk * 1.4;
					arrow.moveTo(p.getX() + Math.sin(a) * 9, p.getY() + 1.2, p.getZ() + Math.cos(a) * 9);
					Vec3 v = p.position().add(0, 1.0, 0).subtract(arrow.position()).normalize().scale(1.6);
					arrow.setDeltaMovement(v);
					p.serverLevel().addFreshEntity(arrow);
				}));
			}
			t += 88;
		}
		if (want("healing_spin")) {
			reset(mc, t, 0f);
			at(t + 2, () -> server(mc, p -> p.setHealth(10f)));
			firstPerson(mc, t + 2, true);
			record(t + 4, "healing_spin", 70);
			use(mc, t + 8, GyroAbilityManager.HEALING_SPIN);
			firstPerson(mc, t + 76, false);
			t += 80;
		}
		if (want("face_spin")) {
			reset(mc, t, 0f);
			cam(t + 1, 0f, 0.1f, 2.2f, 0f, 1.5f);
			record(t + 4, "face_spin", 40);
			use(mc, t + 10, GyroAbilityManager.FACE_SPIN);
			t += 48;
		}
		if (want("riding")) {
			reset(mc, t, 2f);
			dummy(mc, t + 1, 0, 12);
			use(mc, t + 3, GyroAbilityManager.VALKYRIE);
			cam(t + 5, 110f, 1.6f, 7.5f, 0f, 1.6f);
			record(t + 60, "riding", 50);
			at(t + 62, () -> server(mc, p -> GyroMoves.startCharge(p, GyroAbilityManager.GOLDEN_SPIN)));
			at(t + 62 + 15, () -> server(mc, p -> GyroMoves.releaseCharge(p, GyroAbilityManager.GOLDEN_SPIN)));
			t += 118;
		}
		at(t, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}
}
