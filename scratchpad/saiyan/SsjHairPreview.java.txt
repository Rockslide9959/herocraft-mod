import java.awt.image.BufferedImage;
import java.io.File;
import java.util.List;

import javax.imageio.ImageIO;

import com.projecthero.mod.client.saiyan.hair.SsjHairModel;

/**
 * Offline preview of SsjHairModel (no Minecraft): z-buffered orthographic renders of the mesh on a Steve-ish head, for fast
 * shape iteration. Build: copy to SsjHairPreview.java next to a copy of SsjHairModel.java and
 *   javac -cp joml.jar -d out SsjHairModel.java SsjHairPreview.java && java -cp out;joml.jar SsjHairPreview out.png
 * Views: front, 3/4, side, back, top (yaw about the vertical axis, then a little pitch).
 */
public class SsjHairPreview {
	static final int W = 360, H = 360;

	public static void main(String[] a) throws Exception {
		List<SsjHairModel.Quad> mesh = SsjHairModel.mesh();
		if (a.length > 1) {
			mesh = mesh.subList(0, Integer.parseInt(a[1]));
		}
		System.out.println("quads " + mesh.size());
		float[][] views = { { 0, 0, 13, 0.82f }, { 40, 10, 13, 0.82f }, { 90, 0, 13, 0.82f }, { 180, 8, 13, 0.82f }, { 30, 55, 13, 0.82f }, { 0, 0, 34, 1.25f } };
		BufferedImage out = new BufferedImage(W * views.length, H, BufferedImage.TYPE_INT_RGB);
		for (int v = 0; v < views.length; v++) {
			SCALE = views[v][2];
			BASE = views[v][3];
			render(out, v * W, mesh, views[v][0], views[v][1]);
		}
		ImageIO.write(out, "png", new File(a.length > 0 ? a[0] : "preview.png"));
	}

	static float[] zb = new float[W * H];
	static float SCALE = 13f, BASE = 0.82f;

	static void render(BufferedImage img, int ox, List<SsjHairModel.Quad> mesh, float yawDeg, float pitchDeg) {
		java.util.Arrays.fill(zb, -1e9f);
		for (int y = 0; y < H; y++) {
			for (int x = 0; x < W; x++) {
				img.setRGB(ox + x, y, 0xAACCF0);
			}
		}
		// head box: model space x -4..4, y -8..0, z -4..4 (face at -z)
		float[][] c = new float[8][];
		for (int i = 0; i < 8; i++) {
			c[i] = new float[] { (i & 1) == 0 ? -4 : 4, (i & 2) == 0 ? -8 : 0, (i & 4) == 0 ? -4 : 4 };
		}
		int[][] faces = { { 0, 1, 3, 2 }, { 4, 5, 7, 6 }, { 0, 2, 6, 4 }, { 1, 3, 7, 5 }, { 0, 1, 5, 4 }, { 2, 3, 7, 6 } };
		int[] fcol = { 0xC69680, 0x5A3A22, 0xA87A62, 0xA87A62, 0x4A2E1A, 0xB08068 };
		for (int f = 0; f < 6; f++) {
			int[] fi = faces[f];
			float[] xyz = new float[12];
			for (int k = 0; k < 4; k++) {
				System.arraycopy(c[fi[k]], 0, xyz, k * 3, 3);
			}
			int col = fcol[f];
			quad(img, ox, xyz, new int[] { col, col, col, col }, yawDeg, pitchDeg);
		}
		// eyes on the face (z = -4.01): model y -4..-3 rows
		eye(img, ox, -3, -1, yawDeg, pitchDeg, 0xFFFFFF);
		eye(img, ox, 1, 3, yawDeg, pitchDeg, 0xFFFFFF);
		eye(img, ox, -2, -1, yawDeg, pitchDeg, 0x3C46A0);
		eye(img, ox, 1, 2, yawDeg, pitchDeg, 0x3C46A0);
		for (SsjHairModel.Quad q : mesh) {
			quad(img, ox, q.xyz(), q.rgb(), yawDeg, pitchDeg);
		}
		if (OUTLINE) {
			for (SsjHairModel.Quad q : SsjHairModel.outline()) {
				cull = CULL_SIGN;
				quad(img, ox, q.xyz(), q.rgb(), yawDeg, pitchDeg);
				cull = 0;
			}
		}
	}

	static boolean OUTLINE = !"0".equals(System.getenv("SSJ_OUTLINE"));
	static int CULL_SIGN = Integer.parseInt(System.getenv().getOrDefault("SSJ_CULL", "1"));
	static int cull = 0;

	static void eye(BufferedImage img, int ox, float x0, float x1, float yaw, float pitch, int col) {
		float[] xyz = { x0, -4, -4.02f, x1, -4, -4.02f, x1, -3, -4.02f, x0, -3, -4.02f };
		quad(img, ox, xyz, new int[] { col, col, col, col }, yaw, pitch);
	}

	/** Model space -> screen: the camera looks at the face from yaw 0 (in front, at -z). */
	static float[] proj(float x, float y, float z, float yaw, float pitch) {
		double a = Math.toRadians(yaw), p = Math.toRadians(pitch);
		// view from the front: screen x = -model x (the wearer's left is the viewer's right), screen up = -model y
		double X = x, Y = -y, Z = -z; // Z toward the camera
		double rx = X * Math.cos(a) + Z * Math.sin(a), rz = -X * Math.sin(a) + Z * Math.cos(a);
		double ry = Y * Math.cos(p) - rz * Math.sin(p), rz2 = Y * Math.sin(p) + rz * Math.cos(p);
		float s = SCALE;
		return new float[] { (float) (W / 2 + rx * s), (float) (H * BASE - (ry - 0) * s), (float) rz2 };
	}

	static void quad(BufferedImage img, int ox, float[] xyz, int[] rgb, float yaw, float pitch) {
		float[][] p = new float[4][];
		for (int k = 0; k < 4; k++) {
			p[k] = proj(xyz[k * 3], xyz[k * 3 + 1], xyz[k * 3 + 2], yaw, pitch);
		}
		if (cull != 0) {
			float ar = (p[1][0] - p[0][0]) * (p[2][1] - p[0][1]) - (p[2][0] - p[0][0]) * (p[1][1] - p[0][1]);
			if (Math.signum(ar) == cull) {
				return;
			}
		}
		tri(img, ox, p[0], p[1], p[2], rgb[0], rgb[1], rgb[2]);
		tri(img, ox, p[0], p[2], p[3], rgb[0], rgb[2], rgb[3]);
	}

	static void tri(BufferedImage img, int ox, float[] a, float[] b, float[] c, int ca, int cb, int cc) {
		int x0 = (int) Math.max(0, Math.floor(Math.min(a[0], Math.min(b[0], c[0]))));
		int x1 = (int) Math.min(W - 1, Math.ceil(Math.max(a[0], Math.max(b[0], c[0]))));
		int y0 = (int) Math.max(0, Math.floor(Math.min(a[1], Math.min(b[1], c[1]))));
		int y1 = (int) Math.min(H - 1, Math.ceil(Math.max(a[1], Math.max(b[1], c[1]))));
		float d = (b[0] - a[0]) * (c[1] - a[1]) - (c[0] - a[0]) * (b[1] - a[1]);
		if (Math.abs(d) < 1e-6f) {
			return;
		}
		for (int y = y0; y <= y1; y++) {
			for (int x = x0; x <= x1; x++) {
				float px = x + 0.5f, py = y + 0.5f;
				float w1 = ((px - a[0]) * (c[1] - a[1]) - (c[0] - a[0]) * (py - a[1])) / d;
				float w2 = ((b[0] - a[0]) * (py - a[1]) - (px - a[0]) * (b[1] - a[1])) / d;
				float w0 = 1 - w1 - w2;
				if (w0 < 0 || w1 < 0 || w2 < 0) {
					continue;
				}
				float z = w0 * a[2] + w1 * b[2] + w2 * c[2];
				if (z <= zb[y * W + x]) {
					continue;
				}
				zb[y * W + x] = z;
				int r = (int) (w0 * ((ca >> 16) & 255) + w1 * ((cb >> 16) & 255) + w2 * ((cc >> 16) & 255));
				int g = (int) (w0 * ((ca >> 8) & 255) + w1 * ((cb >> 8) & 255) + w2 * ((cc >> 8) & 255));
				int bl = (int) (w0 * (ca & 255) + w1 * (cb & 255) + w2 * (cc & 255));
				img.setRGB(ox + x, y, (r << 16) | (g << 8) | bl);
			}
		}
	}
}
