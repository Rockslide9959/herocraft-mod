/**
 * TEMP-HARNESS (not shipped): films the v0.17.4 Saiyan -- the 3D flame aura orbited (front, side, back, top), each colour, the
 * aura in first person, the reworked first-person arm clips (real presses), and logs every sound the client plays while
 * recording (frame, file, volume, pitch, loop stop) so scratchpad/saiyan/v0174_films.js can lay the real clips under the MP4.
 * Enabled by run/saiyanv4.flag (its text filters scenes). Frames: run/screenshots/sv4_*.
 */
public final class SaiyanV4Harness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static String recording;
	private static int recFrame;
	private static int recUntil;
	private static Vec3 origin;
	private static String only;
	private static final List<String> SOUNDS = new ArrayList<>();
	private static final Map<SoundInstance, Integer> LOOPS = new HashMap<>();

	public static volatile Float orbitYaw;
	public static volatile float orbitHeight = 0.15f;
	public static volatile float orbitRadius = 2.0f;
	public static volatile float orbitSpin = 0f;
	public static volatile float lookHeight = 1.0f;
	public static volatile float orbitBob = 0f;
	private static float orbitBase;

	private SaiyanV4Harness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("saiyanv4.flag"))) {
			return;
		}
		try {
			only = Files.readString(Path.of("saiyanv4.flag")).trim();
		} catch (Exception e) {
			only = "";
		}
		ClientTickEvents.END_CLIENT_TICK.register(SaiyanV4Harness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(900_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "saiyanv4-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void cmd(Minecraft mc, int t, String c) {
		at(t, () -> server(mc, p -> p.server.getCommands().performPrefixedCommand(p.createCommandSourceStack().withPermission(4), c)));
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(SaiyanV4Harness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			System.out.println("[HARNESS] dbz sounds active=" + SaiyanDbzSounds.active());
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.bobView().set(false);
			mc.getSoundManager().addListener((sound, acc, range) -> {
				if (recording != null && sound.getLocation().getNamespace().equals("projecthero")) {
					SOUNDS.add(recFrame + " " + sound.getSound().getLocation() + " " + sound.getVolume() + " " + sound.getPitch() + " "
							+ (sound.isLooping() ? "loop" : "once") + " " + sound.getLocation());
					if (sound.isLooping()) {
						LOOPS.put(sound, SOUNDS.size() - 1);
					}
				}
			});
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.NORMAL, true);
				p.setGameMode(GameType.CREATIVE);
				p.removeAllEffects();
				Saiyan.grant(p);
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			if (orbitYaw != null && orbitSpin != 0f) {
				orbitYaw = orbitYaw + orbitSpin;
				if (orbitBob != 0f) {
					orbitHeight = orbitBase + orbitBob * (float) Math.sin(Math.toRadians(orbitYaw * 2f));
				}
			}
			if (recording != null) {
				// loops that stopped this tick
				for (var it = LOOPS.entrySet().iterator(); it.hasNext();) {
					var e = it.next();
					if (!mc.getSoundManager().isActive(e.getKey())) {
						SOUNDS.set(e.getValue(), SOUNDS.get(e.getValue()) + " stop=" + recFrame);
						it.remove();
					}
				}
				String name = String.format("sv4_%s_%03d.png", recording, recFrame++);
				Screenshot.grab(mc.gameDirectory, name, mc.getMainRenderTarget(), c -> {
				});
				if (inWorldTicks - t0 >= recUntil) {
					System.out.println("[HARNESS] recorded " + recording + " frames=" + recFrame);
					try {
						Files.write(mc.gameDirectory.toPath().resolve("screenshots").resolve("sv4_" + recording + "_sounds.txt"), SOUNDS);
					} catch (Exception e) {
						System.out.println("[HARNESS] sound log failed " + e);
					}
					SOUNDS.clear();
					LOOPS.clear();
					recording = null;
				}
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("S4_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
				Difficulty.NORMAL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	private static void record(int start, String name, int ticks) {
		at(start, () -> {
			recording = name;
			recFrame = 0;
			recUntil = start + ticks;
			SOUNDS.clear();
			LOOPS.clear();
		});
	}

	private static void still(int t, String name) {
		at(t, () -> {
			Screenshot.grab(Minecraft.getInstance().gameDirectory, "sv4_still_" + name + ".png", Minecraft.getInstance().getMainRenderTarget(), c -> {
			});
			System.out.println("[HARNESS] still " + name);
		});
	}

	private static boolean want(String name) {
		return only == null || only.isEmpty() || only.contains(name);
	}

	/** Back to the spot facing south, dummies gone, cooldowns cleared, full Ki. */
	private static void reset(Minecraft mc, int t, float pitch) {
		at(t, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			for (Husk h : level.getEntitiesOfClass(Husk.class, new AABB(origin, origin).inflate(48))) {
				h.discard();
			}
			p.removeAllEffects();
			p.getAbilities().flying = false;
			p.onUpdateAbilities();
			p.setDeltaMovement(Vec3.ZERO);
			p.teleportTo(level, origin.x, origin.y, origin.z, 0f, pitch);
			p.setYHeadRot(0f);
			p.yBodyRot = 0f;
			p.setXRot(pitch);
			SaiyanKit.resetForTest(p);
		}));
		cmd(mc, t + 1, "projecthero power tune saiyan ki 100");
	}

	private static void dummy(Minecraft mc, int t, double dx, double dz) {
		at(t, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			Husk h = EntityType.HUSK.create(level);
			if (h == null) {
				return;
			}
			h.moveTo(origin.x + dx, origin.y, origin.z + dz, 180f, 0f);
			h.setYHeadRot(180f);
			h.setPersistenceRequired();
			h.getAttribute(Attributes.MAX_HEALTH).setBaseValue(400);
			h.setHealth(400);
			h.getAttribute(Attributes.MOVEMENT_SPEED).setBaseValue(0);
			h.setSilent(true);
			level.addFreshEntity(h);
		}));
	}

	private static void press(Minecraft mc, int t, String id, boolean down) {
		at(t, () -> server(mc, p -> SaiyanKit.press(p, id, down)));
	}

	private static void cam(int t, Float yaw, float height, float radius, float spin, float look) {
		at(t, () -> {
			orbitYaw = yaw;
			orbitHeight = height;
			orbitRadius = radius;
			orbitSpin = spin;
			lookHeight = look;
			orbitBase = height;
			orbitBob = 0f;
		});
	}

	private static void firstPerson(Minecraft mc, int t, boolean on) {
		at(t, () -> {
			if (on) {
				orbitYaw = null;
				mc.options.setCameraType(CameraType.FIRST_PERSON);
				mc.options.hideGui = false;
			} else {
				mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
				mc.options.hideGui = true;
			}
		});
	}

	private static void pitch(Minecraft mc, int t, float x) {
		at(t, () -> server(mc, p -> {
			p.setXRot(x);
			p.connection.teleport(p.getX(), p.getY(), p.getZ(), 0f, x);
		}));
	}

	private static void dbg(Minecraft mc, int t, SaiyanForm f, boolean charging, boolean kaioken) {
		at(t, () -> {
			SaiyanFx.debugForm(mc.player, f, false);
			SaiyanFx.debugCharging(mc.player, charging);
			SaiyanFx.debugKaioken(mc.player, kaioken);
		});
	}

	private static void dbgClear(Minecraft mc, int t) {
		at(t, () -> SaiyanFx.debugClear(mc.player));
	}

	private static void script(Minecraft mc) {
		int t = 20;
		reset(mc, t, 0f);
		t += 10;
		// ---- the 3D aura, real Super Saiyan (the transformation's sound and burst), then the orbit and top-down
		if (want("orbit")) {
			cam(t, 20f, 0.6f, 6.0f, 0f, 1.0f);
			record(t + 2, "transform", 50);
			cmd(mc, t + 6, "projecthero power tune saiyan ssj");
			t += 60;
			cam(t, 0f, 0.2f, 3.6f, 3f, 1.0f);
			at(t + 1, () -> orbitBob = 0.9f);
			record(t + 2, "ssj_orbit", 120);
			t += 125;
			cam(t, 0f, 3.6f, 0.3f, 1.5f, 1.0f);
			record(t + 2, "ssj_top", 50);
			still(t + 30, "ssj_top");
			t += 55;
			cam(t, 0f, 0.1f, 3.4f, 0f, 1.0f);
			still(t + 15, "ssj_front");
			cam(t + 16, 90f, 0.1f, 3.4f, 0f, 1.0f);
			still(t + 30, "ssj_side");
			cam(t + 31, 180f, 0.1f, 3.4f, 0f, 1.0f);
			still(t + 45, "ssj_back");
			cam(t + 46, 35f, 1.6f, 3.6f, 0f, 1.0f);
			still(t + 60, "ssj_high");
			t += 65;
			// charging (V, real: the charge sounds), slow orbit
			cam(t, 30f, 0.4f, 4.2f, 1.2f, 1.1f);
			record(t + 2, "ssj_charge", 110);
			at(t + 10, () -> server(mc, p -> Saiyan.setCharging(p, true)));
			at(t + 90, () -> server(mc, p -> Saiyan.setCharging(p, false)));
			t += 120;
			cmd(mc, t, "projecthero power tune saiyan end_ssj");
			t += 30;
		}
		// ---- every colour (client overrides): Kaioken, Ultra Instinct, a Base charge, Super Saiyan charging
		if (want("colours")) {
			String[] names = { "kaioken", "ultra", "base_charge", "ssj_charging", "ssj_kaioken" };
			SaiyanForm[] forms = { SaiyanForm.BASE, SaiyanForm.ULTRA, SaiyanForm.BASE, SaiyanForm.SSJ, SaiyanForm.SSJ };
			boolean[] ch = { false, false, true, true, false };
			boolean[] kk = { true, false, false, false, true };
			for (int i = 0; i < names.length; i++) {
				dbg(mc, t, forms[i], ch[i], kk[i]);
				cam(t + 1, 0f, 0.1f, 3.6f, 0f, 1.0f);
				still(t + 25, names[i] + "_front");
				cam(t + 26, 40f, 1.4f, 3.8f, 0f, 1.0f);
				still(t + 40, names[i] + "_high");
				t += 42;
			}
			// Kaioken orbit
			dbg(mc, t, SaiyanForm.BASE, false, true);
			cam(t + 1, 0f, 0.3f, 3.6f, 4f, 1.0f);
			record(t + 15, "kaioken_orbit", 90);
			t += 110;
			dbg(mc, t, SaiyanForm.ULTRA, false, false);
			cam(t + 1, 0f, 0.3f, 3.6f, 4f, 1.0f);
			record(t + 15, "ultra_orbit", 90);
			t += 110;
			dbgClear(mc, t);
			t += 5;
		}
		// ---- the aura in first person: looking down at the flame round the feet, ahead, then charging
		if (want("fp_aura")) {
			reset(mc, t, 55f);
			firstPerson(mc, t + 1, true);
			dbg(mc, t + 2, SaiyanForm.SSJ, false, false);
			record(t + 10, "fp_aura", 120);
			pitch(mc, t + 40, 15f);
			at(t + 60, () -> SaiyanFx.debugCharging(mc.player, true));
			pitch(mc, t + 90, 70f);
			at(t + 125, () -> SaiyanFx.debugCharging(mc.player, false));
			dbgClear(mc, t + 131);
			firstPerson(mc, t + 132, false);
			t += 135;
		}
		// ---- the first-person arm clips, real presses, in Super Saiyan (sounds logged)
		if (want("fp_moves")) {
			reset(mc, t, 0f);
			dummy(mc, t + 2, 0, 2.3);
			firstPerson(mc, t + 3, true);
			cmd(mc, t + 4, "projecthero power tune saiyan ssj");
			t += 50;
			record(t, "fp_combo", 48);
			for (int i = 0; i < 5; i++) {
				press(mc, t + 4 + i * 7, SaiyanKit.COMBO, true);
			}
			t += 50;
			reset(mc, t, 0f);
			dummy(mc, t + 2, 0, 6);
			record(t + 4, "fp_blasts", 40);
			press(mc, t + 6, SaiyanKit.KI_BLAST, true);
			press(mc, t + 34, SaiyanKit.KI_BLAST, false);
			t += 46;
			reset(mc, t, 0f);
			dummy(mc, t + 2, 0, 6);
			record(t + 4, "fp_kamehameha", 100);
			press(mc, t + 6, SaiyanKit.BEAM, true);
			press(mc, t + 56, SaiyanKit.BEAM, false);
			t += 106;
			reset(mc, t, 0f);
			dummy(mc, t + 2, 0, 2.3);
			record(t + 4, "fp_rush", 40);
			press(mc, t + 6, SaiyanKit.RUSH, true);
			t += 46;
			reset(mc, t, 0f);
			dummy(mc, t + 2, 0, 4);
			record(t + 4, "fp_smash", 40);
			press(mc, t + 6, SaiyanKit.STRONG, true);
			t += 46;
			reset(mc, t, 0f);
			dummy(mc, t + 2, 0, 3);
			record(t + 4, "fp_wave", 30);
			press(mc, t + 6, SaiyanKit.AOE, true);
			t += 36;
			reset(mc, t, 0f);
			record(t + 4, "fp_flare", 26);
			press(mc, t + 6, SaiyanKit.UTILITY2, true);
			t += 32;
			reset(mc, t, 0f);
			record(t + 4, "fp_powerup", 70);
			at(t + 8, () -> server(mc, p -> Saiyan.setCharging(p, true)));
			at(t + 58, () -> server(mc, p -> Saiyan.setCharging(p, false)));
			t += 76;
			// the transformation in first person: back to Base, then Super Saiyan again
			cmd(mc, t, "projecthero power tune saiyan end_ssj");
			t += 30;
			reset(mc, t, 0f);
			record(t + 4, "fp_transform", 40);
			cmd(mc, t + 8, "projecthero power tune saiyan ssj");
			t += 46;
			reset(mc, t, -10f);
			record(t + 4, "fp_spirit_bomb", 110);
			at(t + 6, () -> server(mc, p -> SaiyanSpiritBomb.press(p, true)));
			at(t + 80, () -> server(mc, p -> SaiyanSpiritBomb.press(p, false)));
			t += 120;
			firstPerson(mc, t, false);
		}
		at(t, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}
}
