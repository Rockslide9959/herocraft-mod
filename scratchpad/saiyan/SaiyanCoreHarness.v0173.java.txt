/**
 * TEMP-HARNESS (not shipped): films the v0.17.3 Saiyan revamp core with the REAL server state -- the HUD, Rage filling in a
 * fight, the Super Saiyan transform + timer + its end, Kaioken (Shift + hold V wind-up, red aura, health drain), V charge (tap
 * and hold) and flight. Enabled by run/saiyanfx.flag (its text filters scenes). Frames: run/screenshots/sc_*.
 * Keeps the class name SaiyanFxHarness so the camera mixin (SaiyanAuraHarnessCamMixin) works unchanged.
 */
public final class SaiyanFxHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static String recording;
	private static int recFrame;
	private static int recUntil;
	private static Vec3 origin;
	private static String only;
	private static Zombie dummy;

	public static volatile Float orbitYaw;
	public static volatile float orbitHeight = 0.15f;
	public static volatile float orbitRadius = 2.0f;
	public static volatile float orbitSpin = 0f;
	public static volatile float lookHeight = 1.55f;
	public static volatile float orbitBob = 0f;
	public static volatile int variant = 0;

	private SaiyanFxHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("saiyanfx.flag"))) {
			return;
		}
		try {
			only = Files.readString(Path.of("saiyanfx.flag")).trim();
		} catch (Exception e) {
			only = "";
		}
		ClientTickEvents.END_CLIENT_TICK.register(SaiyanFxHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(900_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "saiyanfx-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(SaiyanFxHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = false;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.EASY, true);
				p.setGameMode(GameType.SURVIVAL);
				p.removeAllEffects();
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
				if (!Saiyan.hasPower(p)) {
					Saiyan.grant(p);
				}
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			if (orbitYaw != null && orbitSpin != 0f) {
				orbitYaw = orbitYaw + orbitSpin;
			}
			if (recording != null) {
				String name = String.format("sc_%s_%03d.png", recording, recFrame++);
				Screenshot.grab(mc.gameDirectory, name, mc.getMainRenderTarget(), c -> {
				});
				if (inWorldTicks - t0 >= recUntil) {
					System.out.println("[HARNESS] recorded " + recording + " frames=" + recFrame);
					recording = null;
				}
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("SC_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false,
				Difficulty.EASY, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	private static void record(int start, String name, int ticks) {
		at(start, () -> {
			recording = name;
			recFrame = 0;
			recUntil = start + ticks;
		});
	}

	private static void still(int t, String name) {
		at(t, () -> {
			Screenshot.grab(Minecraft.getInstance().gameDirectory, "sc_still_" + name + ".png", Minecraft.getInstance().getMainRenderTarget(), c -> {
			});
			System.out.println("[HARNESS] still " + name);
		});
	}

	private static boolean want(String name) {
		return only == null || only.isEmpty() || only.contains(name);
	}

	private static void reset(Minecraft mc, int t) {
		at(t, () -> server(mc, p -> {
			p.removeAllEffects();
			p.setDeltaMovement(Vec3.ZERO);
			p.teleportTo(p.serverLevel(), origin.x, origin.y, origin.z, 0f, 0f);
			p.setYHeadRot(0f);
			p.yBodyRot = 0f;
			p.setHealth(p.getMaxHealth());
			p.getFoodData().setFoodLevel(20);
		}));
	}

	private static void edit(ServerPlayer p, java.util.function.Consumer<SaiyanState> change) {
		SaiyanState n = Saiyan.state(p).copy();
		change.accept(n);
		p.setAttached(ModAttachments.SAIYAN_STATE, n);
		Saiyan.reconcile(p);
	}

	private static void cam(int t, Float yaw, float height, float radius, float spin, float look) {
		at(t, () -> {
			orbitYaw = yaw;
			orbitHeight = height;
			orbitRadius = radius;
			orbitSpin = spin;
			lookHeight = look;
		});
	}

	private static void key(Minecraft mc, int t, AbilitySlot slot, boolean shift, boolean down) {
		at(t, () -> server(mc, p -> {
			p.setShiftKeyDown(shift);
			SaiyanAbilityManager.handle(p, slot, down);
		}));
	}

	private static void script(Minecraft mc) {
		int t = 20;
		reset(mc, t);
		t += 20;
		// the HUD: Base, part-filled bars; then Shift (the Shift moves) and Left-Alt (the move list)
		if (want("hud")) {
			at(t, () -> server(mc, p -> {
				Saiyan.setKi(p, 64f);
				Saiyan.setRage(p, 35f);
			}));
			cam(t + 1, null, 0f, 0f, 0f, 0f);
			still(t + 10, "hud_base");
			at(t + 12, () -> SaiyanHud.forceShift = true);
			still(t + 16, "hud_shift");
			at(t + 18, () -> {
				SaiyanHud.forceShift = false;
				MoveListOverlay.forceHeld = true;
			});
			still(t + 22, "hud_alt");
			at(t + 24, () -> MoveListOverlay.forceHeld = false);
			t += 30;
		}
		// V tap: charges on its own until Ki is full
		if (want("vtap")) {
			reset(mc, t);
			at(t + 1, () -> server(mc, p -> Saiyan.setKi(p, 30f)));
			cam(t + 1, 20f, 0.3f, 3.6f, 0f, 1.5f);
			key(mc, t + 10, AbilitySlot.SLOT_5, false, true);
			key(mc, t + 12, AbilitySlot.SLOT_5, false, false);
			record(t + 4, "vcharge_tap", 140);
			t += 150;
		}
		// V hold: charges while held, stops on release
		if (want("vhold")) {
			reset(mc, t);
			at(t + 1, () -> server(mc, p -> Saiyan.setKi(p, 20f)));
			cam(t + 1, -20f, 0.3f, 3.6f, 0f, 1.5f);
			key(mc, t + 10, AbilitySlot.SLOT_5, false, true);
			key(mc, t + 70, AbilitySlot.SLOT_5, false, false);
			record(t + 4, "vcharge_hold", 100);
			t += 110;
		}
		// flight: take off in the air, fly forward, hover, land
		if (want("flight")) {
			reset(mc, t);
			cam(t + 1, null, 0f, 0f, 0f, 0f);
			at(t + 2, () -> server(mc, p -> {
				Saiyan.setKi(p, 100f);
				p.teleportTo(p.serverLevel(), origin.x, origin.y + 2.5, origin.z, 0f, 0f);
			}));
			at(t + 4, () -> server(mc, p -> com.projecthero.mod.saiyan.SaiyanFlight.toggle(p)));
			at(t + 12, () -> mc.options.keyUp.setDown(true));
			at(t + 14, () -> mc.options.keySprint.setDown(true));
			at(t + 50, () -> mc.options.keySprint.setDown(false));
			at(t + 70, () -> mc.options.keyUp.setDown(false));
			at(t + 100, () -> server(mc, p -> com.projecthero.mod.saiyan.SaiyanFlight.toggle(p)));
			record(t + 2, "flight", 120);
			t += 130;
		}
		// Rage: trading blows with a big zombie (500 HP = a stronger foe); the bar fills to 100 and pulses "H"
		if (want("rage") || want("ssj")) {
			reset(mc, t);
			at(t + 1, () -> server(mc, p -> {
				Saiyan.setRage(p, 30f);
				Saiyan.setKi(p, 80f);
				dummy = EntityType.ZOMBIE.create(p.serverLevel());
				dummy.setNoAi(true);
				dummy.getAttribute(Attributes.MAX_HEALTH).setBaseValue(500);
				dummy.setHealth(500f);
				dummy.setItemSlot(net.minecraft.world.entity.EquipmentSlot.HEAD, new net.minecraft.world.item.ItemStack(net.minecraft.world.item.Items.LEATHER_HELMET));
				dummy.moveTo(origin.x, origin.y, origin.z + 1.6, 180f, 0f);
				dummy.setYHeadRot(180f);
				p.serverLevel().addFreshEntity(dummy);
			}));
			cam(t + 2, 60f, 0.4f, 4.2f, 0f, 1.3f);
			for (int k = 0; k < 24; k++) {
				final int kk = k;
				at(t + 10 + k * 6, () -> server(mc, p -> {
					p.swing(InteractionHand.MAIN_HAND, true);
					dummy.invulnerableTime = 0;
					dummy.hurt(p.damageSources().playerAttack(p), 3f);
					if (kk % 3 == 1) {
						p.invulnerableTime = 0;
						p.hurt(p.damageSources().mobAttack(dummy), 4f);
					}
				}));
			}
			record(t + 4, "rage", 160);
			t += 170;
		}
		// Super Saiyan: H at full Rage -- the burst, the hair and aura, the 3:00 timer
		if (want("ssj")) {
			at(t, () -> server(mc, p -> {
				if (dummy != null) {
					dummy.discard();
				}
				p.setHealth(p.getMaxHealth());
				Saiyan.setRage(p, 100f);
			}));
			cam(t + 1, 15f, 0.6f, 5.5f, 0.6f, 1.4f);
			at(t + 15, () -> server(mc, Saiyan::transform));
			record(t + 5, "ssj_transform", 150);
			still(t + 120, "ssj_hud");
			t += 160;
			// fast-forward to the last 2 s: the end, the exhaustion lockout on the HUD
			at(t, () -> server(mc, p -> edit(p, s -> s.ssjUntil = p.level().getGameTime() + 40)));
			cam(t + 1, -25f, 0.4f, 4.5f, 0f, 1.4f);
			record(t + 2, "ssj_end", 90);
			t += 100;
		}
		// Kaioken: Shift + hold V 3 s (the wind-up row on the HUD), the red aura, the health drain; then a tap ends it
		if (want("kaioken")) {
			reset(mc, t);
			cam(t + 1, 20f, 0.4f, 4.0f, 0.5f, 1.4f);
			key(mc, t + 10, AbilitySlot.SLOT_5, true, true);
			key(mc, t + 75, AbilitySlot.SLOT_5, true, false);
			at(t + 76, () -> server(mc, p -> p.setShiftKeyDown(false)));
			record(t + 4, "kaioken", 200);
			still(t + 150, "kaioken_hud");
			at(t + 205, () -> server(mc, p -> Saiyan.stopKaioken(p, true)));
			t += 215;
		}
		at(t, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}
}
