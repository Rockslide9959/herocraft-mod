// v0.17.7 Black Panther Ancestral Plane harness: films the WHOLE dream in one continuous run, one frame per client tick.
// To use: copy to src/client/java/com/projecthero/mod/client/BpAncestralHarness.java, add
// "com.projecthero.mod.client.BpAncestralHarness" to the client entrypoints in src/main/resources/fabric.mod.json, write
// run/bp_ancestral.flag = <out dir>, run node scratchpad/low_graphics_options.js run/options.txt, then ./gradlew runClient.
// Remove both from src before committing.
// The run: a fresh world at dusk, a bed, Bast's Blessing, lie down -> asleep into the plane -> the arrival -> walk (sprint) to the
// ancestors -> the greeting -> the procession -> walk into the shrine, touch the altar -> Bast wakes -> the Trial of the Shades
// (an autopilot that steps in and strikes the nearest shade) -> the spirit path forming -> the course (scripted hops, with one
// staged fall caught at checkpoint 3) -> the finale -> waking in bed as the Black Panther.
// Frames: film_NNNN.png (+ marks.txt "name frame") -> node scratchpad/v0177/bp_ancestral_film.js <out dir>
package com.projecthero.mod.client;

/** TEMP-HARNESS (not shipped). Kept as scratchpad/BpAncestralHarness.v0177.java.txt. */
public final class BpAncestralHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static File out;
	private static FileWriter marks;
	private static int frame;
	private static boolean filming;
	private static String phase = "setup";
	private static int phaseTicks;
	private static int endTicks = -1;
	private static volatile int bx, by, bz;
	private static int hop, hopTick;
	private static boolean fallDone, falling;
	private static int lastAttack;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("bp_ancestral.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim().split("\\R")[0].trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		try {
			marks = new FileWriter(new File(out, "marks.txt"));
		} catch (Exception e) {
			marks = null;
		}
		ClientTickEvents.END_CLIENT_TICK.register(BpAncestralHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(1_500_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] watchdog");
			Runtime.getRuntime().halt(3);
		}, "bp-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static ServerPlayer sp(Minecraft mc) {
		MinecraftServer srv = mc.getSingleplayerServer();
		return srv == null || srv.getPlayerList().getPlayers().isEmpty() ? null : srv.getPlayerList().getPlayers().get(0);
	}

	private static AncestralPlane.Stage stage(Minecraft mc) {
		ServerPlayer p = sp(mc);
		return p == null ? null : AncestralPlane.stage(p);
	}

	private static void mark(String name) {
		System.out.println("[HARNESS] mark " + name + " " + frame);
		try {
			if (marks != null) {
				marks.write(name + " " + frame + "\n");
				marks.flush();
			}
		} catch (Exception ignored) {
		}
	}

	private static void shot(Minecraft mc) {
		mc.getToasts().clear();
		Screenshot.grab(out, String.format("film_%04d.png", frame), mc.getMainRenderTarget(), m -> {
		});
		frame++;
	}

	private static void go(String p) {
		phase = p;
		phaseTicks = 0;
		mark(p);
	}

	private static float yawTo(Vec3 from, Vec3 to) {
		return (float) Math.toDegrees(Math.atan2(-(to.x - from.x), to.z - from.z));
	}

	private static void walk(Minecraft mc, Vec3 target, boolean sprint) {
		float yaw = yawTo(mc.player.position(), target);
		mc.player.setYRot(yaw);
		mc.player.setYHeadRot(yaw);
		mc.player.setXRot(6f);
		mc.options.keyUp.setDown(true);
		mc.options.keySprint.setDown(sprint);
		mc.player.setSprinting(sprint);
	}

	private static void stop(Minecraft mc) {
		mc.options.keyUp.setDown(false);
		mc.options.keySprint.setDown(false);
		if (mc.player != null) {
			mc.player.setSprinting(false);
		}
	}

	private static BlockPos origin(Minecraft mc) {
		ServerPlayer p = sp(mc);
		BlockPos o = p == null ? null : AncestralPlane.origin(p);
		return o == null ? new BlockPos(0, PlaneRealm.BASE_Y, 0) : o;
	}

	/** Standing spot on platform {@code i} of the spirit path. */
	private static Vec3 platformTop(BlockPos o, int i) {
		ParkourCourse.Platform pl = ParkourCourse.platforms().get(i);
		return PlaneGeometry.at(o, PlaneGeometry.PATH_X + pl.centreX(), PlaneGeometry.PATH_BASE + pl.y() + 1, pl.centreZ());
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(960, 540);
				mc.execute(BpAncestralHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		phaseTicks++;
		if (t == 30) {
			mc.options.fov().set(75);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.renderDistance().set(8);
			mc.options.framerateLimit().set(60);
			mc.options.entityDistanceScaling().set(2.0);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			if (mc.getSingleplayerServer() != null) {
				mc.getSingleplayerServer().getPlayerList().setViewDistance(8);
			}
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(13200L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.setWeatherParameters(24000, 0, false, false);
				p.setGameMode(GameType.SURVIVAL);
				p.getInventory().clearContent();
				// warm the plane's slot 0 up front, so nothing in the film waits on a build
				ServerLevel plane = PlaneRealm.level(p.server);
				PlaneRealm.buildNow(plane, 0);
			});
		}
		if (t == 120) {
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				int x = p.blockPosition().getX(), z = p.blockPosition().getZ();
				int y = level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, x, z);
				// a little stone floor with a bed on it
				for (int dx = -3; dx <= 3; dx++) {
					for (int dz = -3; dz <= 3; dz++) {
						level.setBlock(new BlockPos(x + dx, y - 1, z + dz), Blocks.SPRUCE_PLANKS.defaultBlockState(), 2);
						for (int h = 0; h < 4; h++) {
							level.setBlock(new BlockPos(x + dx, y + h, z + dz), Blocks.AIR.defaultBlockState(), 2);
						}
					}
				}
				BlockState foot = Blocks.PURPLE_BED.defaultBlockState().setValue(BedBlock.FACING, Direction.NORTH).setValue(BedBlock.PART, BedPart.FOOT);
				level.setBlock(new BlockPos(x, y, z), foot, 3);
				level.setBlock(new BlockPos(x, y, z - 1), foot.setValue(BedBlock.PART, BedPart.HEAD), 3);
				bx = x;
				by = y;
				bz = z - 1;
				p.connection.teleport(x + 0.5, y, z + 1.6, 180f, 30f);
				BastBlessing.give(p);
			});
		}
		if (t == 170) {
			filming = true;
			go("bed");
			server(mc, p -> {
				var r = p.startSleepInBed(new BlockPos(bx, by, bz));
				System.out.println("[HARNESS] sleep " + r);
			});
		}
		switch (phase) {
			case "bed" -> {
				if (stage(mc) == AncestralPlane.Stage.ARRIVAL) {
					go("arrival");
				}
			}
			case "arrival" -> {
				if (stage(mc) == AncestralPlane.Stage.WANDER) {
					go("walk");
				}
			}
			case "walk" -> {
				BlockPos o = origin(mc);
				walk(mc, PlaneGeometry.ancestor(o, 0), true);
				if (phaseTicks == 70) {
					server(mc, p -> p.connection.teleport(o.getX() + 46.5, o.getY(), o.getZ() + 0.5, -90f, 6f));
				}
				if (stage(mc) == AncestralPlane.Stage.GREETING) {
					stop(mc);
					go("greeting");
				}
				if (phaseTicks > 600) {
					stop(mc);
					server(mc, p -> p.connection.teleport(o.getX() + 62.5, o.getY(), o.getZ() + 0.5, -90f, 6f));
				}
			}
			case "greeting" -> {
				if (stage(mc) == AncestralPlane.Stage.PROCESSION) {
					go("procession");
				}
			}
			case "procession" -> {
				if (stage(mc) == AncestralPlane.Stage.SHRINE) {
					go("shrine");
				}
			}
			case "shrine" -> {
				BlockPos o = origin(mc);
				Vec3 altar = PlaneGeometry.at(o, PlaneGeometry.SHRINE_X + PlaneGeometry.ALTAR_DX - 1.5, PlaneGeometry.SHRINE_FLOOR, 0.5);
				if (phaseTicks < 90 && mc.player.position().distanceTo(altar) > 1.5) {
					walk(mc, altar, false);
				} else {
					stop(mc);
					if (phaseTicks == 95 || phaseTicks == 140) {
						server(mc, AncestralPlane::touchShrineForTest);
					}
				}
				if (stage(mc) == AncestralPlane.Stage.TRIAL_INTRO) {
					stop(mc);
					go("intro");
				}
			}
			case "intro" -> {
				if (stage(mc) == AncestralPlane.Stage.TRIAL) {
					go("trial");
				}
			}
			case "trial" -> {
				fight(mc);
				if (phaseTicks == 2600) {
					server(mc, p -> AncestralPlane.setTrialRound(p, 3));
				}
				if (stage(mc) == AncestralPlane.Stage.PATH) {
					stop(mc);
					go("path");
				}
			}
			case "path" -> {
				if (stage(mc) == AncestralPlane.Stage.PARKOUR) {
					hop = 0;
					hopTick = 0;
					go("parkour");
				}
			}
			case "parkour" -> parkour(mc);
			case "finale" -> {
				if (stage(mc) == null) {
					go("wake");
				}
			}
			case "wake" -> {
				if (phaseTicks == 1) {
					mc.player.setXRot(-10f);
				}
				if (phaseTicks >= 150) {
					go("done");
				}
			}
			case "done" -> {
				filming = false;
				if (endTicks < 0) {
					endTicks = 0;
					System.out.println("[HARNESS] done frames=" + frame);
				}
				if (++endTicks > 40) {
					Runtime.getRuntime().halt(0);
				}
			}
			default -> {
			}
		}
		if (filming) {
			shot(mc);
		}
	}

	/** The trial's autopilot: face the nearest shade, step in, strike when in reach. */
	private static void fight(Minecraft mc) {
		List<PantherShadeEntity> shades = mc.level.getEntitiesOfClass(PantherShadeEntity.class, mc.player.getBoundingBox().inflate(20), e -> !e.spirit());
		PantherShadeEntity near = null;
		double best = 1e9;
		for (PantherShadeEntity e : shades) {
			double d = e.distanceTo(mc.player);
			if (d < best) {
				best = d;
				near = e;
			}
		}
		if (near == null) {
			stop(mc);
			return;
		}
		Vec3 eye = mc.player.getEyePosition();
		Vec3 at = near.position().add(0, 0.5, 0);
		float yaw = yawTo(eye, at);
		double h = Math.hypot(at.x - eye.x, at.z - eye.z);
		float pitch = (float) -Math.toDegrees(Math.atan2(at.y - eye.y, h));
		mc.player.setYRot(yaw);
		mc.player.setYHeadRot(yaw);
		mc.player.setXRot(pitch);
		// wait for it to come in (crouching or stalking close), then step in and strike
		boolean close = best < 4.6 || near.phase() == PantherShadeEntity.WINDUP || near.phase() == PantherShadeEntity.POUNCE;
		mc.options.keyUp.setDown(close && best > 2.4);
		if (best < 3.0 && inWorldTicks - lastAttack > 8 && near.tickCount > 8) {
			lastAttack = inWorldTicks;
			mc.gameMode.attack(mc.player, near);
			mc.player.swing(InteractionHand.MAIN_HAND);
		}
	}

	/** The course: scripted hops from stone to stone (server teleports along an arc), one staged fall at checkpoint 3. */
	private static void parkour(Minecraft mc) {
		BlockPos o = origin(mc);
		List<ParkourCourse.Platform> all = ParkourCourse.platforms();
		if (stage(mc) == AncestralPlane.Stage.FINALE) {
			go("finale");
			return;
		}
		if (falling) {
			// let the plane catch the fall, then go on from checkpoint 3
			if (phaseTicks > 8 && mc.player.getY() > o.getY() + PlaneGeometry.PATH_BASE + 2 && mc.player.getDeltaMovement().y > -0.2) {
				falling = false;
				hopTick = -20;
				mark("caught");
			}
			return;
		}
		if (hop >= all.size() - 1) {
			return;
		}
		int dur = 12;
		hopTick++;
		if (hopTick <= 0) {
			return; // a pause on the stone
		}
		Vec3 a = platformTop(o, hop), b = platformTop(o, hop + 1);
		double k = Math.min(1.0, hopTick / (double) dur);
		Vec3 pos = a.lerp(b, k).add(0, 1.15 * Math.sin(Math.PI * k), 0);
		float yaw = yawTo(a, b);
		server(mc, p -> p.connection.teleport(pos.x, pos.y, pos.z, yaw, 18f));
		if (k >= 1.0) {
			hop++;
			hopTick = all.get(hop).kind() == ParkourCourse.Kind.CHECKPOINT ? -14 : -3;
			if (hop == 11 && !fallDone) {
				// checkpoint 3: step off the edge and fall
				fallDone = true;
				falling = true;
				phaseTicks = 0;
				mark("fall");
				Vec3 off = b.add(0, 0, -3.4);
				server(mc, p -> p.connection.teleport(off.x, off.y + 0.2, off.z, 180f, 40f));
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		LevelSettings settings = new LevelSettings("BPPLANE_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.PEACEFUL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(7177L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.NORMAL).value().createWorldDimensions(),
				mc.screen);
	}
}
