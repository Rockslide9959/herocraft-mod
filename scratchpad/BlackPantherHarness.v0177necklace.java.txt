package com.projecthero.mod.client;

import java.nio.file.Files;
import java.nio.file.Path;
import java.util.HashMap;
import java.util.Map;
import java.util.function.Consumer;

import com.projecthero.mod.blackpanther.BlackPanther;
import com.projecthero.mod.blackpanther.item.BlackPantherItems;
import com.projecthero.mod.client.blackpanther.BlackPantherClient;

import net.fabricmc.fabric.api.client.event.lifecycle.v1.ClientTickEvents;

import net.minecraft.client.CameraType;
import net.minecraft.client.Minecraft;
import net.minecraft.client.Screenshot;
import net.minecraft.client.gui.screens.TitleScreen;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.Difficulty;
import net.minecraft.world.effect.MobEffectInstance;
import net.minecraft.world.effect.MobEffects;
import net.minecraft.world.entity.EquipmentSlot;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.GameRules;
import net.minecraft.world.level.GameType;
import net.minecraft.world.level.LevelSettings;
import net.minecraft.world.level.WorldDataConfiguration;
import net.minecraft.world.level.levelgen.WorldOptions;
import net.minecraft.world.level.levelgen.presets.WorldPresets;
import net.minecraft.world.phys.Vec3;

/** TEMP-HARNESS (not shipped): v0.17.7 Black Panther necklace -- the model on the suit and the H suit-up. Enabled by run/bp.flag. */
public final class BlackPantherHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static String recording;
	private static int recFrame;
	private static int recUntil;
	private static Vec3 origin;
	private static Vec3 base;

	public static volatile Vec3 camPos;
	public static volatile Vec3 camLook;

	private BlackPantherHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("bp.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(BlackPantherHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(420_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "bp-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(BlackPantherHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.setGameMode(GameType.SURVIVAL);
				BlackPanther.grant(p);
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
				base = origin;
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			if (recording != null) {
				String name = String.format("bpn_%s_%03d.png", recording, recFrame++);
				Screenshot.grab(mc.gameDirectory, name, mc.getMainRenderTarget(), c -> {
				});
				if (inWorldTicks - t0 >= recUntil) {
					System.out.println("[HARNESS] recorded " + recording + " frames=" + recFrame);
					recording = null;
				}
			}
		}
	}

	private static void shot(Minecraft mc, int t, String name) {
		at(t, () -> Screenshot.grab(mc.gameDirectory, "bpn_still_" + name + ".png", mc.getMainRenderTarget(), c -> {
		}));
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("BPN_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.PEACEFUL, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}

	private static void record(int start, String name, int ticks) {
		at(start, () -> {
			recording = name;
			recFrame = 0;
			recUntil = start + ticks;
		});
	}

	/** Stand still at the base, the necklace worn or not, the suit on (finished forming) or off. */
	private static void reset(ServerPlayer p, boolean necklace, boolean suited) {
		p.getInventory().clearContent();
		p.removeAllEffects();
		p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 100000, 4, false, false));
		p.setDeltaMovement(Vec3.ZERO);
		p.teleportTo(p.serverLevel(), base.x, base.y, base.z, 0f, 0f);
		p.setYHeadRot(0f);
		p.yBodyRot = 0f;
		if (necklace) {
			p.setItemSlot(EquipmentSlot.CHEST, new ItemStack(BlackPantherItems.NECKLACE));
		}
		var s = BlackPanther.state(p).copy();
		s.suited = suited;
		s.animId = 0;
		s.suitChangeAt = p.level().getGameTime() - 100;
		p.setAttached(com.projecthero.mod.attachment.ModAttachments.BLACK_PANTHER_STATE, s);
		BlackPanther.reconcile(p);
	}

	private static void scene(Minecraft mc, int t, int index, boolean necklace, boolean suited, CameraType type, Vec3 cam, Vec3 look) {
		at(t, () -> {
			base = origin.add(index * 20, 0, 0);
			mc.options.setCameraType(type);
			camPos = cam == null ? null : base.add(cam);
			camLook = look == null ? null : base.add(look);
			server(mc, p -> reset(p, necklace, suited));
		});
	}

	private static void cam(int t, Vec3 pos, Vec3 look) {
		at(t, () -> {
			camPos = base.add(pos);
			camLook = base.add(look);
		});
	}

	/** H on the client, through the real path: ownsH, then the payload. */
	private static void pressH(Minecraft mc) {
		boolean owns = mc.player != null && BlackPantherClient.ownsH(mc.player);
		System.out.println("[HARNESS] H owned by the necklace: " + owns);
		if (owns) {
			BlackPantherClient.pressH();
		}
	}

	private static void script(Minecraft mc) {
		int t = 40;
		int idx = 1;
		Vec3 chest = new Vec3(0, 1.3, 0);
		// suited with the necklace: front close, three-quarter, back, shoulder from above, wide
		scene(mc, t, idx++, true, true, CameraType.THIRD_PERSON_BACK, new Vec3(0, 1.42, 1.0), chest);
		shot(mc, t + 25, "suited_front_close");
		cam(t + 28, new Vec3(0.75, 1.5, 0.8), chest);
		shot(mc, t + 36, "suited_34_close");
		cam(t + 39, new Vec3(0, 1.45, -1.05), chest);
		shot(mc, t + 47, "suited_back_close");
		cam(t + 50, new Vec3(0.55, 2.35, 0.45), new Vec3(0, 1.45, 0));
		shot(mc, t + 58, "suited_shoulders_above");
		cam(t + 61, new Vec3(0.9, 1.5, 2.6), new Vec3(0, 1.0, 0));
		shot(mc, t + 69, "suited_wide");
		t += 80;
		// the necklace worn, suit away: nothing drawn
		scene(mc, t, idx++, true, false, CameraType.THIRD_PERSON_BACK, new Vec3(0, 1.42, 1.0), chest);
		shot(mc, t + 25, "unsuited_necklace_worn_front");
		t += 35;
		// suited first person
		scene(mc, t, idx++, true, true, CameraType.FIRST_PERSON, null, null);
		shot(mc, t + 25, "suited_first_person");
		t += 35;
		// H suit-up (the real client path), then H again: the retract
		scene(mc, t, idx++, true, false, CameraType.THIRD_PERSON_BACK, new Vec3(0.9, 1.55, 1.9), new Vec3(0, 1.1, 0));
		record(t + 18, "suit_up", 55);
		at(t + 22, () -> pressH(mc));
		shot(mc, t + 22 + 44, "after_h_suited");
		record(t + 80, "retract", 45);
		at(t + 84, () -> pressH(mc));
		t += 140;
		at(t, () -> {
			System.out.println("[HARNESS] DONE");
			mc.stop();
		});
	}
}
