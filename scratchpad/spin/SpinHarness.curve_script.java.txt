	/** Client-side flick: turn his view onto origin + (dx, dy, dz) over {@code ticks} ticks (the server takes the client's look). */
	private static void turnTo(Minecraft mc, int t, int ticks, double dx, double dy, double dz) {
		float[] from = new float[2];
		for (int k = 0; k <= ticks + 30; k++) { // (then held there, in case anything resets the view)
			int kk = k;
			at(t + k, () -> {
				var me = mc.player;
				if (me == null) {
					return;
				}
				if (kk == 0) {
					from[0] = me.getYRot();
					from[1] = me.getXRot();
				}
				Vec3 eye = me.getEyePosition();
				Vec3 d = origin.add(dx, dy, dz).subtract(eye);
				float ty = (float) Math.toDegrees(Math.atan2(-d.x, d.z));
				float tp = (float) -Math.toDegrees(Math.atan2(d.y, Math.sqrt(d.x * d.x + d.z * d.z)));
				float f = ticks == 0 ? 1f : Math.min(1f, kk / (float) ticks);
				f = f * f * (3 - 2 * f);
				float yaw = from[0] + net.minecraft.util.Mth.wrapDegrees(ty - from[0]) * f;
				float pitch = from[1] + (tp - from[1]) * f;
				me.setYRot(yaw);
				me.setXRot(pitch);
				me.setYHeadRot(yaw);
			});
		}
	}

	/** Everything back: fresh balls in both hands. */
	private static void freshBalls(Minecraft mc, int t) {
		at(t, () -> server(mc, p -> com.projecthero.mod.spin.SpinBalls.replace(p)));
	}

	private static void night(Minecraft mc, int t, boolean on) {
		at(t, () -> server(mc, p -> p.serverLevel().setDayTime(on ? 18000L : 6000L)));
	}

	private static void clean(Minecraft mc, int t) {
		at(t, () -> {
			mc.gui.clear();
			mc.gui.getChat().clearMessages(false);
			mc.getToasts().clear();
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
		});
	}

	private static void script(Minecraft mc) {
		int t = 20;
		// 1. the flick: thrown straight, then the view snaps 30 degrees onto a husk 14 blocks off -- the ball curves in
		if (want("curve_hit_fp")) {
			reset(mc, t, 0f);
			freshBalls(mc, t + 1);
			clean(mc, t + 3);
			dummy(mc, t + 1, 7.0, 12.1);
			firstPerson(mc, t + 2, true);
			record(t + 4, "curve_hit_fp", 40);
			use(mc, t + 8, SpinAbilityManager.STEEL_BALL);
			turnTo(mc, t + 9, 3, 7.0, 1.0, 12.1);
			firstPerson(mc, t + 46, false);
			t += 50;
		}
		if (want("curve_hit_side")) {
			reset(mc, t, 0f);
			freshBalls(mc, t + 1);
			clean(mc, t + 3);
			dummy(mc, t + 1, 7.0, 12.1);
			cam(t + 1, 200f, 12f, 6f, 0f, 0.0f);
			record(t + 4, "curve_hit_side", 40);
			use(mc, t + 8, SpinAbilityManager.STEEL_BALL);
			turnTo(mc, t + 9, 3, 7.0, 1.0, 12.1);
			t += 50;
		}
		// 2. too sharp: the husk is close and 60 degrees off; he only looks at it once the ball is 18 blocks out -- no curve onto it

			reset(mc, t, 0f);
			freshBalls(mc, t + 1);
			clean(mc, t + 3);
			dummy(mc, t + 1, 4.3, 2.5);
			firstPerson(mc, t + 2, true);
			record(t + 4, "curve_miss_fp", 50);
			use(mc, t + 8, SpinAbilityManager.STEEL_BALL);
			turnTo(mc, t + 23, 3, 4.3, 1.0, 2.5);
			firstPerson(mc, t + 56, false);
			t += 60;
		}
		if (want("curve_miss_side")) {
			reset(mc, t, 0f);
			freshBalls(mc, t + 1);
			clean(mc, t + 3);
			dummy(mc, t + 1, 4.3, 2.5);
			cam(t + 1, 200f, 12f, 6f, 0f, 0.0f);
			record(t + 4, "curve_miss_side", 50);

			t += 60;
		}
		// 3. holding the balls

			firstPerson(mc, t + 2, true);

		}
		if (want("hold_tp")) {
			reset(mc, t, 0f);
			freshBalls(mc, t + 1);
			clean(mc, t + 3);
			cam(t + 1, 0f, 0.2f, 2.6f, 4.5f, 1.0f);
			record(t + 4, "hold_tp", 80);
			t += 88;
		}
		// 4. a throw and the return into the same hand
		if (want("hold_throw")) {
			reset(mc, t, 0f);
			freshBalls(mc, t + 1);
			clean(mc, t + 3);
			dummy(mc, t + 1, 0, 6);
			cam(t + 1, 95f, 0.5f, 3.4f, 0f, 1.3f);
			record(t + 4, "hold_throw", 50);
			use(mc, t + 10, SpinAbilityManager.STEEL_BALL);
			t += 68;
		}
		// 5. Twin Balls with one in each hand
		if (want("hold_twin")) {
			reset(mc, t, 2f);
			freshBalls(mc, t + 1);
			clean(mc, t + 3);
			dummy(mc, t + 1, 0, 10);
			cam(t + 1, 160f, 1.6f, 4.5f, 0f, 1.1f);
			record(t + 4, "hold_twin", 60);

			t += 68;
		}
		// 6. VFX: the trails at night, the Golden Spin line, the Shield's arcs, the Quake's ring, a spun husk's swirl

		if (want("vfx_golden")) {

			freshBalls(mc, t + 1);
			clean(mc, t + 3);

			charge(mc, t + 8, SpinAbilityManager.GOLDEN_SPIN, 30);
			t += 58;
		}
		if (want("vfx_shield")) {
			reset(mc, t, 0f);
			freshBalls(mc, t + 1);
			clean(mc, t + 3);
			cam(t + 1, 70f, 1.6f, 4.5f, 0f, 1.1f);
			record(t + 4, "vfx_shield", 60);
			use(mc, t + 8, SpinAbilityManager.SPIN_SHIELD);
			t += 68;

			reset(mc, t, 0f);

			clean(mc, t + 3);

				for (Husk h : p.serverLevel().getEntitiesOfClass(Husk.class, new AABB(origin, origin).inflate(10))) {

		at(t, () -> {

			Runtime.getRuntime().halt(0);
