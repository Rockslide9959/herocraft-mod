/** TEMP-HARNESS (not shipped). Kept as scratchpad/TempleHarness.v0177.java.txt. */
public final class TempleHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks, planTick;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static File out;

	static final BlockPos DESERT = new BlockPos(0, 100, 0);
	static final BlockPos JUNGLE = new BlockPos(400, 100, 0);

	record View(String name, double cx, double cy, double cz, double lx, double ly, double lz, boolean gui, int delay) {
	}

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("temple.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(TempleHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(720_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "temple-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		mc.getToasts().clear();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			prev.run();
			r.run();
		});
	}

	static TemplePiece piece(BlockPos o, boolean desert) {
		return new TemplePiece(o, Rotation.NONE, desert);
	}

	static void view(Minecraft mc, BlockPos o, View v) {
		mc.options.hideGui = !v.gui;
		Vec3 cam = new Vec3(o.getX() + v.cx, o.getY() + v.cy, o.getZ() + v.cz);
		Vec3 d = new Vec3(o.getX() + v.lx, o.getY() + v.ly, o.getZ() + v.lz).subtract(cam);
		float yaw = (float) Math.toDegrees(Math.atan2(-d.x, d.z));
		float pitch = (float) -Math.toDegrees(Math.atan2(d.y, Math.sqrt(d.x * d.x + d.z * d.z)));
		server(mc, p -> {
			p.getAbilities().mayfly = true;
			p.getAbilities().flying = true;
			p.onUpdateAbilities();
			p.connection.teleport(cam.x, cam.y - 1.62, cam.z, yaw, pitch);
		});
	}

	/** A thin crust round the surface (rock under the platform), air everywhere else, so the underground is a floating cutaway. */
	static void prepare(ServerLevel level, BlockPos o, boolean desert) {
		BlockState top = desert ? Blocks.SAND.defaultBlockState() : Blocks.GRASS_BLOCK.defaultBlockState();
		for (int x = -45; x <= 45; x++) {
			for (int z = -55; z <= 30; z++) {
				for (int y = -3; y <= 0; y++) {
					level.setBlock(o.offset(x, y, z), y < 0 ? (desert ? Blocks.SANDSTONE : Blocks.DIRT).defaultBlockState() : top, 2 | 16);
				}
			}
		}
		if (!desert) {
			// a few jungle trees round the site
			for (int[] t : new int[][] { { -26, -30 }, { 24, -28 }, { -30, 6 }, { 28, 12 }, { -20, -44 }, { 18, -46 } }) {
				for (int y = 1; y <= 9; y++) {
					level.setBlock(o.offset(t[0], y, t[1]), Blocks.JUNGLE_LOG.defaultBlockState(), 2 | 16);
				}
				for (int dx = -3; dx <= 3; dx++) {
					for (int dz = -3; dz <= 3; dz++) {
						for (int dy = 7; dy <= 10; dy++) {
							if (dx * dx + dz * dz + (dy - 8) * (dy - 8) <= 10 && level.getBlockState(o.offset(t[0] + dx, dy, t[1] + dz)).isAir()) {
								level.setBlock(o.offset(t[0] + dx, dy, t[1] + dz), Blocks.JUNGLE_LEAVES.defaultBlockState()
										.setValue(net.minecraft.world.level.block.LeavesBlock.PERSISTENT, true), 2 | 16);
							}
						}
					}
				}
			}
		}
	}

	static void freezeAll(ServerPlayer p, boolean noAi) {
		for (Mob m : p.serverLevel().getEntitiesOfClass(Mob.class, new AABB(-100, -100, -100, 600, 300, 400))) {
			m.setNoAi(noAi);
		}
	}

	static AnkhatepBoss king(ServerPlayer p) {
		var l = p.serverLevel().getEntitiesOfClass(AnkhatepBoss.class, new AABB(-50, 0, 100, 50, 120, 260));
		return l.isEmpty() ? null : l.get(0);
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(TempleHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(80);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.renderDistance().set(8);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOFIRETICK).set(false, p.server);
				p.setGameMode(GameType.CREATIVE);
				p.server.getPlayerList().setViewDistance(8); // the options file is on the low-graphics 2 chunks
				p.addEffect(new MobEffectInstance(MobEffects.NIGHT_VISION, 20 * 60 * 30, 0, false, false));
			});
			buildPlan(mc);
		}
		if (t < 50) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		} else if (!PLAN.isEmpty()) {
			Runtime.getRuntime().halt(0);
		}
	}

		}));
		int k = t + 130;
		if (SECOND_PASS) {
			secondPass(mc, k);
			return;
		}

			at(k + v.delay, () -> shot(mc, v.name));

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("TEMPLE_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				rules, WorldDataConfiguration.DEFAULT);
