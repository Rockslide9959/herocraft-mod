/**
 * TEMP-HARNESS (not shipped, v0.17.5 The Vault): in a normal world, finds the nearest naturally generated Vault with the
 * Dungeon Locator search, flies there and photographs the guard post and the generated arena; then builds a clean copy
 * high above spawn and photographs every floor, films Warden Kross's four moves frame by frame and shows the artifact
 * tooltips. Enabled by run/vault.flag = "<out dir>".
 */
public final class VaultDungeonHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static File out;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static int planTick;
	private static BlockPos natural;
	private static BlockPos naturalHome;
	private static WardenKrossBoss filmBoss;
	private static Husk dummy;

	static final BlockPos SITE = new BlockPos(0, 230, 0);

	record View(String name, double cx, double cy, double cz, double lx, double ly, double lz) {
	}

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("vault.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(VaultDungeonHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(900_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "vault-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		if (!name.startsWith("film_")) {
			mc.particleEngine.setLevel(mc.level); // stray ambient particles in front of the lens
		}
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	static VaultPiece piece() {
		return new VaultPiece(SITE, Rotation.NONE);
	}

	static void look(Minecraft mc, Vec3 cam, Vec3 at) {
		Vec3 d = at.subtract(cam);
		float yaw = (float) Math.toDegrees(Math.atan2(-d.x, d.z));
		float pitch = (float) -Math.toDegrees(Math.atan2(d.y, Math.sqrt(d.x * d.x + d.z * d.z)));
		server(mc, p -> {
			p.getAbilities().flying = true;
			p.onUpdateAbilities();
			p.connection.teleport(cam.x, cam.y - 1.62, cam.z, yaw, pitch);
			for (net.minecraft.world.entity.Entity e : p.serverLevel().getEntities(p, p.getBoundingBox().inflate(6), e -> !(e instanceof Mob))) {
				System.out.println("[HARNESS] near camera: " + e.getType() + " " + e.getClass().getSimpleName());
			}
		});
		mc.getToasts().clear();
		if (mc.level != null && mc.player != null) {
			for (net.minecraft.world.entity.Entity e : mc.level.entitiesForRendering()) {
				if (e != mc.player && e.distanceTo(mc.player) < 8 && !(e instanceof Mob)) {
					System.out.println("[HARNESS] client near camera: " + e.getType());
				}
			}
		}
	}

	static void view(Minecraft mc, View v) {
		look(mc, new Vec3(SITE.getX() + v.cx, SITE.getY() + v.cy, SITE.getZ() + v.cz), new Vec3(SITE.getX() + v.lx, SITE.getY() + v.ly, SITE.getZ() + v.lz));
	}

	static final List<View> VIEWS = List.of(
			new View("1_surface_guard_post", -20, 12, -24, 0, 2, 0),
			new View("2_guard_post_inside_shaft", 4, 3.2, -4, -1, 0, 1),
			new View("3_cell_block", 5.5, -7.0, -17, -3, -11, 6),
			new View("4_cell_block_gallery", -5.5, -7.4, 14, -6, -9, -10),
			new View("5_security_laser_corridor", -7, -25.2, 23, -7, -27, 10),
			new View("6_security_turret_hall", 10, -23.2, 8, -4, -27, -4),
			new View("7_security_office_console", 3, -25, 12, 7, -27, 19.5),
			new View("8_lockdown_gate", 0, -25.5, 3, -0.5, -26.5, -7),
			new View("9_processing_room", -19, -40, 3.5, -24, -42.5, -8),
			new View("10_arena_warden_kross", 0, -38, -5, 0, -41, 10),
			new View("11_vault_room_sealed", 0, -41, 18.5, 0, -42.5, 24));

	static void prepare(ServerLevel level) {
		for (int x = -32; x <= 18; x++) {
			for (int z = -26; z <= 30; z++) {
				for (int y = -52; y <= 14; y++) {
					BlockState s = y < 0 ? Blocks.STONE.defaultBlockState() : y == 0 ? Blocks.GRASS_BLOCK.defaultBlockState() : Blocks.AIR.defaultBlockState();
					level.setBlock(SITE.offset(x, y, z), s, 2 | 16);
				}
			}
		}
	}

	static void freeze(ServerLevel level, BlockPos around, boolean noAi) {
		for (Mob m : level.getEntitiesOfClass(Mob.class, AABB.ofSize(Vec3.atCenterOf(around), 120, 140, 120))) {
			m.setNoAi(noAi);
		}
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(VaultDungeonHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(80);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.renderDistance().set(5);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(com.projecthero.mod.loot.finds.SupplyDrops.RULE).set(false, p.server);
				p.setGameMode(GameType.CREATIVE);
				p.addEffect(new MobEffectInstance(MobEffects.NIGHT_VISION, 20 * 60 * 30, 0, false, false));
			});
			buildPlan(mc);
		}
		if (t < 50) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		}
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			prev.run();
			r.run();
		});
	}

	private static void buildPlan(Minecraft mc) {
		// ---- 1. natural generation: the Dungeon Locator search, fly there
		at(1, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			Optional<LocatorSearch.Found> f = LocatorSearch.search(level, p.blockPosition(), 4);
			System.out.println("[HARNESS] locate the_vault from " + p.blockPosition() + " -> " + f.map(x -> x.pos() + " " + x.structure()).orElse("NONE"));
			if (f.isPresent()) {
				natural = f.get().pos();
				p.connection.teleport(natural.getX() + 0.5, 200, natural.getZ() + 0.5, 0f, 90f);
			}
		}));
		at(300, () -> server(mc, p -> {
			if (natural == null) {
				return;
			}
			ServerLevel level = p.serverLevel();
			int y = level.getHeight(Heightmap.Types.MOTION_BLOCKING, natural.getX(), natural.getZ());
			natural = new BlockPos(natural.getX(), y, natural.getZ());
			var ks = level.getEntitiesOfClass(WardenKrossBoss.class, AABB.ofSize(Vec3.atCenterOf(natural), 160, 200, 160));
			naturalHome = ks.isEmpty() ? null : ks.get(0).home();
			System.out.println("[HARNESS] natural vault surface " + natural + " kross=" + ks.size() + " home=" + naturalHome);
			freeze(level, natural, true);
		}));
		at(310, () -> {
			if (natural != null) {
				mc.options.hideGui = true;
				look(mc, Vec3.atCenterOf(natural).add(-22, 14, -22), Vec3.atCenterOf(natural).add(0, 1, 0));
			}
		});
		at(390, () -> shot(mc, "natural_1_guard_post"));
		at(400, () -> {
			if (naturalHome != null) {
				look(mc, Vec3.atCenterOf(naturalHome).add(0.3, 4, 0.3), Vec3.atCenterOf(naturalHome).add(4, 0, 4));
			}
		});
		at(470, () -> {
			if (naturalHome != null) {
				mc.options.hideGui = false;
				shot(mc, "natural_2_arena_from_inside");
			}
		});

		// ---- 2. the clean copy above spawn: every floor
		int t = 480;
		at(t, () -> look(mc, new Vec3(0, SITE.getY() + 30, -40), Vec3.atCenterOf(SITE)));
		at(t + 40, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			prepare(level);
			piece().buildAll(level);
			freeze(level, SITE, true);
			System.out.println("[HARNESS] built anchor=" + piece().anchor() + " vaults=" + piece().vaults());
		}));
		int k = t + 90;
		for (View v : VIEWS) {
			at(k, () -> {
				mc.options.hideGui = !v.name.startsWith("10_");
				mc.options.particles().set(net.minecraft.client.ParticleStatus.MINIMAL);
				view(mc, v);
			});
			at(k + 38, () -> mc.particleEngine.setLevel(mc.level));
			at(k + 40, () -> shot(mc, v.name));
			k += 46;
		}

		// ---- 3. Warden Kross's moves, filmed in the arena (fixed camera)
		int film = k + 10;
		at(film, () -> {
			mc.options.hideGui = false;
			view(mc, new View("film", -7.5, -39.8, 5.5, 0, -41.6, 5.5));
			mc.options.particles().set(net.minecraft.client.ParticleStatus.ALL);
		});
		at(film + 30, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			var ks = level.getEntitiesOfClass(WardenKrossBoss.class, AABB.ofSize(Vec3.atCenterOf(SITE), 120, 140, 120));
			if (ks.isEmpty()) {
				System.out.println("[HARNESS] no kross!");
				return;
			}
			filmBoss = ks.get(0);
			BlockPos home = filmBoss.home();
			filmBoss.moveTo(home.getX() + 0.5, home.getY(), home.getZ() + 3.5, 180f, 0f);
			filmBoss.setNoAi(false);
			dummy = EntityType.HUSK.create(level);
			dummy.moveTo(home.getX() + 0.5, home.getY(), home.getZ() - 3.5, 0f, 0f);
			dummy.setNoAi(true);
			dummy.setInvulnerable(true);
			level.addFreshEntity(dummy);
			filmBoss.setTarget(dummy);
		}));
		int[] starts = { 40, 85, 140, 205 };
		String[] moves = { "shock", "charge", "taser", "lockdown" };
		for (int i = 0; i < starts.length; i++) {
			final String m = moves[i];
			at(film + starts[i], () -> server(mc, p -> {
				if (filmBoss == null) {
					return;
				}
				filmBoss.setTarget(dummy);
				BlockPos home = filmBoss.home();
				switch (m) {
					case "shock" -> {
						dummy.moveTo(home.getX() + 0.5, home.getY(), home.getZ() - 1.5, 0f, 0f);
						filmBoss.moveTo(home.getX() + 0.5, home.getY(), home.getZ() + 3.5, 180f, 0f);
						filmBoss.startShock();
					}
					case "charge" -> {
						dummy.moveTo(home.getX() + 0.5, home.getY(), home.getZ() - 4.5, 0f, 0f);
						filmBoss.moveTo(home.getX() + 0.5, home.getY(), home.getZ() + 6.5, 180f, 0f);
						filmBoss.startCharge();
					}
					case "taser" -> {
						dummy.moveTo(home.getX() - 3.5, home.getY(), home.getZ() - 3.5, 0f, 0f);
						filmBoss.moveTo(home.getX() + 0.5, home.getY(), home.getZ() + 6.5, 180f, 0f);
						filmBoss.startTaser();
					}
					default -> {
						filmBoss.setHealth(filmBoss.getMaxHealth() * 0.5f + 2f);
						filmBoss.hurt(p.serverLevel().damageSources().playerAttack(p), 10f);
						filmBoss.startLockdown();
					}
				}
				System.out.println("[HARNESS] move " + m + " started, move=" + filmBoss.move());
			}));
		}
		int frames = 270;
		for (int f = 0; f < frames; f++) {
			final int n = f;
			at(film + 35 + f, () -> shot(mc, String.format("film_kross_%03d", n)));
		}
		// a still of the Lockdown shutters from the vault-room door
		int after = film + 35 + frames + 5;
		at(after, () -> {
			mc.options.hideGui = true;
			view(mc, new View("lockdown_shutters", -7, -41, 2, -12.5, -42, 0));
		});
		at(after + 25, () -> shot(mc, "12_arena_lockdown_shutters"));

		// ---- 4. the artifact tooltips
		int tips = after + 35;
		for (int page = 0; page < 3; page++) {
			final int pg = page;
			at(tips + page * 8, () -> {
				mc.options.hideGui = false;
				mc.setScreen(new ArtifactScreen(pg));
			});
			at(tips + page * 8 + 5, () -> shot(mc, String.format("artifact_tooltip_%d", pg)));
		}
		at(tips + 30, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	static final class ArtifactScreen extends Screen {
		private final int page;

		ArtifactScreen(int page) {
			super(Component.literal("artifacts"));
			this.page = page;
		}

		@Override
		public void render(GuiGraphics g, int mouseX, int mouseY, float partial) {
			g.fill(0, 0, width, height, 0xFF1B1E24);
			ItemStack[] l = { new ItemStack(VaultItems.WARDENS_KEYRING), new ItemStack(VaultItems.MJOLNIR_SHARD), new ItemStack(VaultItems.RIOT_CAPACITOR) };
			for (int i = 0; i < l.length; i++) {
				int x = width - 170, y = 10 + i * 20;
				g.renderItem(l[i], x, y);
				g.drawString(font, l[i].getHoverName(), x + 19, y + 4, 0xFFFFFF, true);
			}
			ItemStack tip = l[page % l.length];
			g.renderTooltip(font, Screen.getTooltipFromItem(minecraft, tip), tip.getTooltipImage(), -6, 22);
		}

		@Override
		public boolean isPauseScreen() {
			return false;
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("VAULT_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17050L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.NORMAL).value().createWorldDimensions(),
				mc.screen);
	}
}
