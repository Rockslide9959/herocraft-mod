/**
 * TEMP-HARNESS (not shipped, v0.17.5 villain hideouts): builds each of the five hideouts on a prepared site, takes an
 * outside overview, interiors and the boss room (with its mini-boss), films the Parademon Commander's Dive Strike frame
 * by frame, and shows the artifacts' tooltips. Enabled by run/hideout.flag = "<out dir>".
 */
public final class HideoutHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static File out;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static int planTick;

	/** One camera view: name, local camera position, local look-at point. */
	record View(String name, double cx, double cy, double cz, double lx, double ly, double lz) {
	}

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("hideout.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(HideoutHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(600_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "hideout-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	static BlockPos siteOrigin(HideoutFaction f) {
		return new BlockPos(f.ordinal() * 220, 100, 0);
	}

	static HideoutPiece piece(HideoutFaction f) {
		return new HideoutPiece(f, siteOrigin(f), Rotation.NONE);
	}

	static Vec3 local(HideoutFaction f, double x, double y, double z) {
		BlockPos o = siteOrigin(f);
		return new Vec3(o.getX() + x, o.getY() + y, o.getZ() + z);
	}

	/** Teleports the (flying, creative) player to a view. */
	static void view(Minecraft mc, HideoutFaction f, View v) {
		Vec3 cam = local(f, v.cx, v.cy, v.cz);
		Vec3 at = local(f, v.lx, v.ly, v.lz);
		Vec3 d = at.subtract(cam);
		float yaw = (float) Math.toDegrees(Math.atan2(-d.x, d.z));
		float pitch = (float) -Math.toDegrees(Math.atan2(d.y, Math.sqrt(d.x * d.x + d.z * d.z)));
		server(mc, p -> {
			p.getAbilities().flying = true;
			p.onUpdateAbilities();
			p.connection.teleport(cam.x, cam.y - 1.62, cam.z, yaw, pitch);
		});
	}

	static List<View> views(HideoutFaction f) {
		return switch (f) {
			case KINGPIN -> List.of(new View("outside", -24, 12, -32, 0, 3, 0), new View("shop_and_hatch", -8, 2.6, -10.5, 8, 0.5, -4.5),
					new View("warehouse", -9, 8, 10.5, 4, 1, 0), new View("arrow_trap_corridor", 7.5, -4.6, -3, 8, -6.5, 8),
					new View("boss_room", 6.5, -4.4, 10.2, 6.5, -6.5, 19));
			case ULTRON -> List.of(new View("outside", -30, 16, -44, 0, 3, -5), new View("assembly_hall", 0, 6, -10, 0, 1.5, 4),
					new View("drone_bay", -7, 3.5, 8, -13, 1, 14), new View("core_room", 9, 4, 8, 4, 2, 15));
			case APOKOLIPS -> List.of(new View("outside", -32, 20, -34, 0, 6, 0), new View("crater_and_perches", 12, 3, -16, 0, 2, 4),
					new View("ground_floor", 0, -1.8, -4.4, 0, -3.5, 3), new View("war_room", -2.5, 13.6, -1.5, 1, 12, 3.5));
			case SYMBIOTE -> List.of(new View("outside_sinkhole", -11, 9, -13, 0, -1, 0), new View("great_chamber", 9, -23.5, -8, -2, -27, 4),
					new View("nursery", 13, -26.4, 0, 22, -28, 0), new View("den", 0, -25.5, 13.5, 0, -27.5, 27));
			case FROST -> List.of(new View("outside", -28, 13, -38, 0, 7, 0), new View("hall_interior", 0, 6, -18, 0, 3, 10),
					new View("dais_and_throne", 0, 5.5, 6, 0, 4, 17));
		};
	}

	/** A clean site: rock below, the faction's ground on top, air above. */
	static void prepare(ServerLevel level, HideoutFaction f) {
		BlockPos o = siteOrigin(f);
		BlockState top = switch (f) {
			case FROST -> Blocks.SNOW_BLOCK.defaultBlockState();
			case APOKOLIPS -> Blocks.RED_SAND.defaultBlockState();
			case ULTRON -> Blocks.STONE.defaultBlockState();
			default -> Blocks.GRASS_BLOCK.defaultBlockState();
		};
		for (int x = -34; x <= 34; x++) {
			for (int z = -34; z <= 36; z++) {
				for (int y = -40; y <= 40; y++) {
					BlockState s = y < 0 ? Blocks.STONE.defaultBlockState() : y == 0 ? top : Blocks.AIR.defaultBlockState();
					if (f == HideoutFaction.ULTRON && y > 0 && y <= 6 && z > 4) {
						s = Blocks.STONE.defaultBlockState(); // a slope behind the forge so it reads as dug in
					}
					level.setBlock(o.offset(x, y, z), s, 2 | 16);
				}
			}
		}
	}

	static void freeze(ServerLevel level, HideoutFaction f, boolean noAi) {
		AABB box = AABB.ofSize(Vec3.atCenterOf(siteOrigin(f)), 90, 90, 90);
		for (Mob m : level.getEntitiesOfClass(Mob.class, box)) {
			m.setNoAi(noAi);
		}
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(HideoutHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(80);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.renderDistance().set(5);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOFIRETICK).set(false, p.server);
				p.setGameMode(GameType.CREATIVE);
				p.addEffect(new MobEffectInstance(MobEffects.NIGHT_VISION, 20 * 60 * 30, 0, false, false));
			});
			buildPlan(mc);
		}
		if (t < 50) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		}
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		PLAN.set(ticks - 1, r);
	}

	private static void buildPlan(Minecraft mc) {
		int t = 1;
		for (HideoutFaction f : HideoutFaction.values()) {
			final int start = t;
			at(start, () -> {
				view(mc, f, new View("pre", 0, 30, -40, 0, 0, 0));
			});
			at(start + 30, () -> server(mc, p -> {
				ServerLevel level = p.serverLevel();
				prepare(level, f);
				piece(f).buildAll(level);
				freeze(level, f, true);
				System.out.println("[HARNESS] built " + f + " core=" + piece(f).corePos() + " vault=" + piece(f).vaultPos());
			}));
			int k = start + 70;
			int i = 0;
			for (View v : views(f)) {
				final int idx = ++i;
				at(k, () -> {
					mc.options.hideGui = !v.name.equals("boss_room") && !v.name.equals("core_room") && !v.name.equals("war_room")
							&& !v.name.equals("den") && !v.name.equals("dais_and_throne");
					view(mc, f, v);
				});
				at(k + 40, () -> shot(mc, f.getSerializedName() + "_" + idx + "_" + v.name));
				k += 46;
			}
			t = k + 10;
		}
		// --- the Parademon Commander's Dive Strike, filmed in the open crater
		HideoutFaction ap = HideoutFaction.APOKOLIPS;
		int film = t;
		final ParademonCommanderBoss[] boss = new ParademonCommanderBoss[1];
		at(film, () -> {
			mc.options.hideGui = false;
			view(mc, ap, new View("film", 12, 3, -19, 4, 1, -11));
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
		});
		at(film + 40, () -> {
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				freeze(level, ap, true);
				var found = level.getEntitiesOfClass(ParademonCommanderBoss.class, AABB.ofSize(Vec3.atCenterOf(siteOrigin(ap)), 90, 90, 90));
				if (found.isEmpty()) {
					return;
				}
				ParademonCommanderBoss b = found.get(0);
				Vec3 spot = local(ap, 3.5, -2, -8.5);
				b.moveTo(spot.x, spot.y, spot.z, 180f, 0f);
				b.setNoAi(false);
				Husk dummy = EntityType.HUSK.create(level);
				Vec3 d = local(ap, 4.5, -1, -14.5);
				dummy.moveTo(d.x, d.y + 1, d.z, 0f, 0f);
				dummy.setNoAi(true);
				dummy.setInvulnerable(true);
				level.addFreshEntity(dummy);
				b.setTarget(dummy);
				boss[0] = b;
			});
		});
		at(film + 55, () -> server(mc, p -> {
			System.out.println("[HARNESS] commander found: " + (boss[0] != null));
			if (boss[0] != null) {
				boss[0].startDive();
			}
		}));
		for (int f = 0; f < 70; f++) {
			final int n = f;
			at(film + 56 + f, () -> shot(mc, String.format("film_commander_dive_%03d", n)));
		}
		int tips = film + 135;
		for (int page = 0; page < 5; page++) {
			final int pg = page;
			at(tips + page * 8, () -> mc.setScreen(new ArtifactScreen(pg)));
			at(tips + page * 8 + 5, () -> shot(mc, String.format("artifact_tooltip_%d", pg)));
		}
		at(tips + 48, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	/** The five artifacts with names on the right, one tooltip on the left. */
	static final class ArtifactScreen extends Screen {
		private final int page;

		ArtifactScreen(int page) {
			super(Component.literal("artifacts"));
			this.page = page;
		}

		@Override
		public void render(GuiGraphics g, int mouseX, int mouseY, float partial) {
			g.fill(0, 0, width, height, 0xFF1B1E24);
			ItemStack[] l = { new ItemStack(HideoutItems.GILDED_CANE), new ItemStack(HideoutItems.CORTEX_SHARD),
					new ItemStack(HideoutItems.MOTHER_BOX_FRAGMENT), new ItemStack(HideoutItems.HIVE_HEART), new ItemStack(HideoutItems.FROSTBRAND) };
			for (int i = 0; i < l.length; i++) {
				int x = width - 170, y = 10 + i * 20;
				g.renderItem(l[i], x, y);
				g.drawString(font, l[i].getHoverName(), x + 19, y + 4, 0xFFFFFF, true);
			}
			ItemStack tip = l[page % l.length];
			g.renderTooltip(font, Screen.getTooltipFromItem(minecraft, tip), tip.getTooltipImage(), -6, 22);
		}

		@Override
		public boolean isPauseScreen() {
			return false;
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("HIDEOUT_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17050L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
