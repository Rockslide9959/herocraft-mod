/**
 * TEMP-HARNESS (not shipped, v0.17.5 Symbiote moveset v3): the V Form wheel (held open through the real V key, then stills with a slice
 * hovered), the Thorns aura on husks pressing in and Spike Surge (real Z key), Spike Nova (Sneak + G) on a ring of husks, and the grab
 * (Sneak + C, left clicks through the real attack button, Sneak + C set-down, grab again, C throw). Every tick is a film frame. Enabled
 * by run/symv3.flag = "<out dir>". Kept as scratchpad/SymbioteMovesetV3Harness.v0175.java.txt.
 */
public final class SymbioteMovesetV3Harness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static File out;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static int planTick;
	private static BlockPos site;
	private static final List<Integer> SPAWNED = new ArrayList<>();
	public static volatile net.minecraft.world.phys.Vec3 camOffset;
	public static volatile double camLookY = 1.2;

	private static void cam(double x, double y, double z, double lookY, int fov) {
		camOffset = new net.minecraft.world.phys.Vec3(x, y, z);
		camLookY = lookY;
		Minecraft.getInstance().options.fov().set(fov);
	}

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("symv3.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readAllLines(flag).get(0).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(SymbioteMovesetV3Harness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(420_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "symv3-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		mc.getToasts().clear();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		if (!name.startsWith("film_")) {
			System.out.println("[HARNESS] shot " + name);
		}
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			prev.run();
			r.run();
		});
	}

	private static void toggle() {
		ClientPlayNetworking.send(new com.projecthero.mod.network.SpiderActionPayload(
				com.projecthero.mod.network.SpiderActionPayload.Action.TOGGLE_SYMBIOTE));
	}

	/** Key i (0 = R, 1 = G, 2 = X, 3 = Z, 4 = V, 5 = C) down at {@code t}, up at {@code t + held}. */
	private static void key(int t, int i, int held) {
		KeyMapping k = ModKeyBindings.ABILITY_SLOTS[i];
		at(t, () -> k.setDown(true));
		at(t + held, () -> k.setDown(false));
	}

	/** Sneak + key i: Sneak goes down 3 ticks before and comes up 3 ticks after. */
	private static void shiftKey(Minecraft mc, int t, int i) {
		at(t - 3, () -> mc.options.keyShift.setDown(true));
		key(t, i, 2);
		at(t + 5, () -> mc.options.keyShift.setDown(false));
	}

	/** A left click through the real attack button (SymbioteKitClient swallows it and sends the crush). */
	private static void click(int t) {
		at(t, () -> KeyMapping.click(InputConstants.Type.MOUSE.getOrCreate(0)));
	}

	private static void clearTargets(Minecraft mc) {
		server(mc, p -> {
			for (int id : SPAWNED) {
				Entity e = p.serverLevel().getEntity(id);
				if (e != null) {
					e.discard();
				}
			}
			SPAWNED.clear();
		});
	}

	/** A target {@code dz} blocks ahead (north) and {@code dx} to the side. */
	private static void spawn(Minecraft mc, EntityType<? extends Mob> type, double dx, double dz, boolean ai) {
		server(mc, p -> {
			ServerLevel level = p.serverLevel();
			Mob m = type.create(level);
			if (m == null) {
				return;
			}
			m.moveTo(site.getX() + 0.5 + dx, site.getY() + 1, site.getZ() + 0.5 - dz, (float) Math.toDegrees(Math.atan2(dx, -dz)) + 180f, 0f);
			m.setNoAi(!ai);
			m.setPersistenceRequired();
			m.getAttribute(Attributes.MAX_HEALTH).setBaseValue(150); // under 200: not a boss
			m.setHealth(150f);
			m.addEffect(new MobEffectInstance(MobEffects.FIRE_RESISTANCE, 20 * 600, 0, false, false));
			if (ai) {
				m.setTarget(p);
			}
			level.addFreshEntity(m);
			SPAWNED.add(m.getId());
		});
	}

	private static void face(Minecraft mc, double dz, float yaw, float pitch) {
		server(mc, p -> p.connection.teleport(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5 - dz, yaw, pitch));
	}

	private static void logHealth(Minecraft mc, String label) {
		server(mc, p -> {
			StringBuilder sb = new StringBuilder("[HARNESS] " + label + ":");
			for (int id : SPAWNED) {
				if (p.serverLevel().getEntity(id) instanceof LivingEntity le) {
					sb.append(String.format(" %.0f/%.0f", le.getHealth(), le.getMaxHealth()))
							.append(le.hasEffect(MobEffects.WITHER) ? "W" : "");
				}
			}
			sb.append(" grab=").append(Symbiote.state(p).tendrilGrabHeld).append(" form=").append(SymbioteVitalsManager.form(p));
			System.out.println(sb);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(SymbioteMovesetV3Harness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(70);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(com.projecthero.mod.loot.finds.SupplyDrops.RULE).set(false, p.server);
				BlockPos sp = p.blockPosition();
				int y = level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, sp.getX(), sp.getZ());
				site = new BlockPos(sp.getX(), y - 1, sp.getZ());
				for (int x = -14; x <= 14; x++) {
					for (int z = -16; z <= 12; z++) {
						for (int yy = -2; yy <= 14; yy++) {
							BlockState st = yy < 0 ? Blocks.DIRT.defaultBlockState() : yy == 0 ? Blocks.GRASS_BLOCK.defaultBlockState() : Blocks.AIR.defaultBlockState();
							level.setBlock(site.offset(x, yy, z), st, 2 | 16);
						}
					}
				}
				p.setGameMode(GameType.SURVIVAL);
				p.connection.teleport(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5, 180f, 8f);
				Symbiote.grant(p);
				SymbioteSuitColours.setPreset(p, SymbioteColourPreset.CLASSIC);
				p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 20 * 60 * 30, 4, false, false));
				p.addEffect(new MobEffectInstance(MobEffects.SATURATION, 20 * 60 * 30, 4, false, false));
			});
		}
		if (t == 46) {
			buildPlan(mc);
		}
		if (t < 50 || site == null) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		}
	}

	private static void film(Minecraft mc, String name, int from, int len) {
		for (int f = 0; f < len; f++) {
			final int n = f;
			at(from + f, () -> shot(mc, String.format("film_%s_%03d", name, n)));
		}
	}

	private static void buildPlan(Minecraft mc) {
		int t = 10;
		at(t, () -> {
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.hideGui = false;
		});
		at(t + 5, SymbioteMovesetV3Harness::toggle);

		// ---- 1. the V wheel: held open by the real V key (logged), then stills with a slice hovered
		int w = t + 95;
		at(w - 4, () -> {
			cam(-2.8, 2.0, -3.4, 1.2, 60);
			SymbioteFormWheelScreen.stayOpen = true; // the harness key is not physically down: keep the wheel the hold opened
		});
		key(w, 4, 16);
		at(w + 12, () -> System.out.println("[HARNESS] wheel open by holding V: " + (mc.screen instanceof SymbioteFormWheelScreen)));
		at(w + 14, () -> shot(mc, "01_form_wheel_by_hold"));
		at(w + 22, () -> mc.setScreen(null));
		at(w + 24, () -> {
			SymbioteFormWheelScreen.stayOpen = true;
			SymbioteFormWheelScreen.forceHover = 1;
			mc.setScreen(new SymbioteFormWheelScreen());
		});
		at(w + 32, () -> shot(mc, "02_form_wheel_blade_hovered"));
		at(w + 34, () -> SymbioteFormWheelScreen.forceHover = 5);
		at(w + 40, () -> shot(mc, "03_form_wheel_thorns_hovered"));
		at(w + 42, () -> {
			SymbioteFormWheelScreen.forceHover = -1;
			SymbioteFormWheelScreen.stayOpen = false;
			mc.setScreen(null);
			// the pick the wheel would send: Thorns
			ClientPlayNetworking.send(new com.projecthero.mod.network.SymbioteKitPayload(
					com.projecthero.mod.network.SymbioteKitPayload.PICK_FORM, SymbioteVitalsManager.FORM_THORNS));
		});
		at(w + 50, () -> logHealth(mc, "after wheel pick (form 2 = Thorns)"));

		// ---- 2. the Thorns aura on husks walking in, then Spike Surge (real Z)
		int a = w + 60;
		at(a - 8, () -> {
			face(mc, 0, 180f, 8f);
			cam(-5.0, 3.2, 3.2, 1.0, 70);
			spawn(mc, EntityType.HUSK, 0.0, 1.9, false);
			spawn(mc, EntityType.HUSK, 1.8, 0.4, false);
			spawn(mc, EntityType.HUSK, -1.0, 3.6, false);
			spawn(mc, EntityType.HUSK, 2.6, 2.6, false);
			spawn(mc, EntityType.HUSK, -3.4, 0.6, false);
		});
		film(mc, "thorns", a, 210);
		at(a + 20, () -> shot(mc, "04_thorns_aura"));
		at(a + 50, () -> logHealth(mc, "thorns aura (2 close, 3 at ~3 blocks)"));
		key(a + 60, 3, 2);
		at(a + 66, () -> shot(mc, "05_spike_surge_growing"));
		at(a + 80, () -> shot(mc, "06_spike_surge_long"));
		at(a + 120, () -> logHealth(mc, "spike surge (all 5 should be hit now)"));
		at(a + 186, () -> shot(mc, "07_spike_surge_drawing_back"));

		// ---- 3. Spike Nova on a ring of husks
		int n = a + 230;
		at(n - 12, () -> {
			clearTargets(mc);
			ClientPlayNetworking.send(new com.projecthero.mod.network.SymbioteKitPayload(
					com.projecthero.mod.network.SymbioteKitPayload.PICK_FORM, SymbioteVitalsManager.FORM_NORMAL));
			face(mc, 0, 180f, 10f);
			cam(-6.5, 4.5, 4.0, 0.8, 70);
			for (int i = 0; i < 8; i++) {
				double ang = Math.PI * 2 * i / 8 + 0.2;
				spawn(mc, i % 4 == 0 ? EntityType.IRON_GOLEM : EntityType.HUSK, Math.cos(ang) * 4.2, Math.sin(ang) * 4.2, false);
			}
		});
		film(mc, "nova", n - 4, 50);
		shiftKey(mc, n, 1);
		at(n + 2, () -> shot(mc, "08_spike_nova_burst"));
		at(n + 5, () -> shot(mc, "09_spike_nova_ring"));
		at(n + 14, () -> logHealth(mc, "spike nova ring"));

		// ---- 4. the grab: Sneak+C, three crushes (left click), Sneak+C set down; grab again, C throw
		int g = n + 70;
		at(g - 10, () -> {
			clearTargets(mc);
			face(mc, 0, 180f, 12f);
			cam(-5.0, 2.2, 0.8, 1.6, 65);
			spawn(mc, EntityType.HUSK, 0.0, 5.0, false);
		});
		film(mc, "grab", g - 4, 250);
		shiftKey(mc, g, 5);
		at(g + 14, () -> server(mc, p -> p.connection.teleport(p.getX(), p.getY(), p.getZ(), 180f, 0f)));
		at(g + 20, () -> shot(mc, "10_grab_hold"));
		at(g + 85, () -> logHealth(mc, "held 4 s (old limit 3.5 s)"));
		click(g + 90);
		at(g + 93, () -> shot(mc, "11_grab_crush"));
		click(g + 105);
		click(g + 108); // too soon: ignored
		click(g + 120);
		at(g + 123, () -> shot(mc, "12_grab_crush_again"));
		at(g + 128, () -> logHealth(mc, "after 3 crushes (+1 ignored)"));
		shiftKey(mc, g + 140, 5);
		at(g + 150, () -> shot(mc, "13_grab_set_down"));
		at(g + 152, () -> logHealth(mc, "set down"));
		at(g + 160, () -> server(mc, p -> {
			// the grab's 5 s cooldown would block the re-grab in the film: clear it
			com.projecthero.mod.symbiote.SymbioteState c = Symbiote.state(p).copy();
			c.abilityCooldowns.set(5, 0L);
			p.setAttached(com.projecthero.mod.attachment.ModAttachments.SYMBIOTE_STATE, c);
			p.connection.teleport(p.getX(), p.getY(), p.getZ(), 180f, 25f);
		}));
		shiftKey(mc, g + 175, 5);
		at(g + 186, () -> server(mc, p -> p.connection.teleport(p.getX(), p.getY(), p.getZ(), 180f, -8f)));
		key(g + 205, 5, 2);
		at(g + 208, () -> shot(mc, "14_grab_throw"));
		at(g + 225, () -> logHealth(mc, "thrown"));
		at(g + 252, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("SYMV3_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17051L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
