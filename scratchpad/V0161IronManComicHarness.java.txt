/**
 * TEMP-HARNESS (not shipped): v0.16.1 Iron Man comic rework screenshots -- the new HUD (normal, Shift held, disabled)
 * and signature moves. Flag file {@code run/v0161im.flag}; shots land in {@code run/screenshots/v0161im_*.png}.
 * Copy into src/client/java/com/projecthero/mod/client/ with the camera mixin, call {@code init()} from the client
 * initializer, run, then REMOVE before committing.
 */
public final class V0161IronManHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	public static Vec3 camOffset;
	public static Vec3 camLook = new Vec3(0, 1.0, 0);
	private static volatile long lastProgress = System.currentTimeMillis();

	private V0161IronManHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("v0161im.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(V0161IronManHarness::tick);
		Thread wd = new Thread(() -> {
			while (true) {
				try {
					Thread.sleep(5000);
				} catch (InterruptedException e) {
					return;
				}
				if (System.currentTimeMillis() - lastProgress > 180_000L) {
					System.out.println("[HARNESS] WATCHDOG: no progress for 180 s");
					Runtime.getRuntime().halt(3);
				}
			}
		}, "v0161im-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	public static boolean active() {
		return t0 >= 0;
	}

	private static ServerPlayer sp(Minecraft mc) {
		MinecraftServer srv = mc.getSingleplayerServer();
		return srv == null || srv.getPlayerList().getPlayers().isEmpty() ? null : srv.getPlayerList().getPlayers().get(0);
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				ServerPlayer p = sp(mc);
				if (p != null) {
					try {
						fn.accept(p);
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		Runnable prev = AT.get(t);
		AT.put(t, prev == null ? r : () -> {
			prev.run();
			r.run();
		});
	}

	private static void shot(String name) {
		Minecraft mc = Minecraft.getInstance();
		Screenshot.grab(mc.gameDirectory, "v0161im_" + name + ".png", mc.getMainRenderTarget(), c -> {
		});
		System.out.println("[HARNESS] shot " + name);
		lastProgress = System.currentTimeMillis();
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				lastProgress = System.currentTimeMillis();
				mc.execute(V0161IronManHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks % 10 == 0) {
			lastProgress = System.currentTimeMillis();
		}
		if (t0 >= 0) {
			mc.player.setYRot(0f);
			mc.player.yHeadRot = 0f;
			mc.player.yHeadRotO = 0f;
			mc.player.yBodyRot = 0f;
			mc.player.yBodyRotO = 0f;
			mc.gui.getChat().clearMessages(false);
		}
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.setGameMode(GameType.SURVIVAL);
				TonyStark.grant(p);
				p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 99999, 4, true, false, false));
				p.addEffect(new MobEffectInstance(MobEffects.SATURATION, 99999, 4, true, false, false));
				p.teleportTo(level, Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5, 0f, 0f);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("V0161IM_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
				Difficulty.NORMAL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, WorldOptions.defaultWithRandomSeed(),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	private static void view(Minecraft mc, String v, double scale) {
		switch (v) {
			case "q" -> {
				mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
				camOffset = new Vec3(-2.2 * scale, 1.8 * scale, -2.8 * scale);
				camLook = new Vec3(0, 1.0 * scale, 2.0);
			}
			case "s" -> {
				mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
				camOffset = new Vec3(-5.5 * scale, 2.0 * scale, 3.0);
				camLook = new Vec3(0, 1.0 * scale, 4.0);
			}
			case "far" -> {
				mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
				camOffset = new Vec3(-9.0, 4.0, 6.0);
				camLook = new Vec3(0, 0.0, 6.0);
			}
			default -> {
				camOffset = null;
				mc.options.setCameraType(CameraType.FIRST_PERSON);
			}
		}
	}

	private static void wear(ServerPlayer p, String suit) {
		EquipmentSlot[] slots = { EquipmentSlot.HEAD, EquipmentSlot.CHEST, EquipmentSlot.LEGS, EquipmentSlot.FEET };
		ArmorItem.Type[] types = { ArmorItem.Type.HELMET, ArmorItem.Type.CHESTPLATE, ArmorItem.Type.LEGGINGS, ArmorItem.Type.BOOTS };
		for (int i = 0; i < 4; i++) {
			p.setItemSlot(slots[i], new ItemStack(IronManItems.armor(suit, types[i])));
		}
		TonyStark.setActiveSuit(p, suit);
		IronManEnergy.setIntegrity(p, suit, IronManEnergy.maxIntegrity(suit));
		var st = TonyStark.state(p).copy();
		st.abilityReadyAt.clear();
		p.setAttached(com.projecthero.mod.attachment.ModAttachments.TONY_STARK_STATE, st);
	}

	private static void clear(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		for (var e : level.getEntitiesOfClass(Mob.class, p.getBoundingBox().inflate(40))) {
			e.discard();
		}
	}

	private static void dummy(ServerPlayer p, double dx, double dy, double dz) {
		var h = EntityType.HUSK.create(p.serverLevel());
		h.moveTo(p.getX() + dx, p.getY() + dy, p.getZ() + dz, 180f, 0f);
		h.setNoAi(true);
		h.getAttribute(net.minecraft.world.entity.ai.attributes.Attributes.MAX_HEALTH).setBaseValue(500.0);
		h.setHealth(500f);
		p.serverLevel().addFreshEntity(h);
	}

	private static void fire(ServerPlayer p, String suitId, int slot, boolean shift) {
		IronManSuit suit = IronManSuits.byId(suitId);
		boolean ok = IronManComicRouter.fireForTests(p, suit, slot, shift);
		System.out.println("[HARNESS] fire " + suitId + " " + IronManComicKit.move(suitId, slot, shift).id() + " -> " + ok);
	}

	private static void script(Minecraft mc) {
		int c = 10;
		// ---- HUD (first person, Mark 6, visor closed, a bit damaged)
		at(c, () -> {
			view(mc, "f", 1);
			mc.options.hideGui = false;
			server(mc, p -> {
				wear(p, "mark_6");
				IronManEnergy.setIntegrity(p, "mark_6", 1450f);
				fire(p, "mark_6", 1, false);
				fire(p, "mark_6", 4, true);
			});
		});
		at(c + 30, () -> shot("hud"));
		at(c + 32, () -> mc.options.keyShift.setDown(true));
		at(c + 38, () -> shot("hud_shift"));
		at(c + 40, () -> {
			mc.options.keyShift.setDown(false);
			server(mc, p -> IronManEnergy.setIntegrity(p, "mark_6", 0f));
		});
		at(c + 52, () -> shot("hud_disabled"));
		c += 60;
		at(c, () -> {
			mc.options.hideGui = true;
			server(mc, p -> wear(p, "mark_6"));
		});
		c += 10;

		// ---- Mark 1 Smoke Screen
		at(c, () -> {
			view(mc, "q", 1.25);
			server(mc, p -> {
				clear(p);
				wear(p, "mark_1");
				dummy(p, 1.5, 0, 3);
				dummy(p, -2, 0, 2.5);
			});
		});
		at(c + 20, () -> server(mc, p -> fire(p, "mark_1", 2, true)));
		at(c + 30, () -> shot("mk1_smoke_screen_a"));
		at(c + 60, () -> shot("mk1_smoke_screen_b"));
		c += 80;

		// ---- Mark 2 Tractor Beam (held)
		at(c, () -> {
			view(mc, "s", 1);
			server(mc, p -> {
				clear(p);
				wear(p, "mark_2");
				dummy(p, 0, 0, 6);
			});
		});
		at(c + 20, () -> server(mc, p -> {
			p.setXRot(10f);
			p.setShiftKeyDown(false);
			IronManComicRouter.trigger(p, IronManSuits.MARK_2, IronManComicKit.SLOT_G, true);
		}));
		at(c + 35, () -> shot("mk2_tractor_beam_a"));
		at(c + 50, () -> shot("mk2_tractor_beam_b"));
		at(c + 58, () -> server(mc, p -> IronManComicRouter.trigger(p, IronManSuits.MARK_2, IronManComicKit.SLOT_G, false)));
		at(c + 62, () -> shot("mk2_tractor_beam_throw"));
		c += 80;

		// ---- Mark 4 Torpedo underwater
		at(c, () -> server(mc, p -> {
			clear(p);
			ServerLevel level = p.serverLevel();
			BlockPos base = p.blockPosition();
			for (int x = -4; x <= 4; x++) {
				for (int z = -3; z <= 16; z++) {
					for (int y = 0; y <= 5; y++) {
						level.setBlockAndUpdate(base.offset(x, y, z), y == 5 ? Blocks.AIR.defaultBlockState() : Blocks.WATER.defaultBlockState());
					}
				}
			}
			wear(p, "mark_4");
			p.teleportTo(level, base.getX() + 0.5, base.getY() + 1.5, base.getZ() + 0.5, 0f, 0f);
			var d = EntityType.DROWNED.create(level);
			d.moveTo(base.getX() + 0.5, base.getY() + 1.0, base.getZ() + 11.5, 180f, 0f);
			d.setNoAi(true);
			d.getAttribute(net.minecraft.world.entity.ai.attributes.Attributes.MAX_HEALTH).setBaseValue(500.0);
			d.setHealth(500f);
			level.addFreshEntity(d);
		}));
		at(c + 2, () -> view(mc, "q", 1));
		at(c + 25, () -> server(mc, p -> fire(p, "mark_4", 2, false)));
		at(c + 29, () -> shot("mk4_torpedo_a"));
		at(c + 33, () -> shot("mk4_torpedo_b"));
		at(c + 40, () -> shot("mk4_torpedo_c"));
		at(c + 48, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			BlockPos base = p.blockPosition().below();
			for (int x = -6; x <= 6; x++) {
				for (int z = -6; z <= 20; z++) {
					for (int y = -2; y <= 7; y++) {
						BlockPos bp = base.offset(x, y, z);
						if (!level.getFluidState(bp).isEmpty()) {
							level.setBlockAndUpdate(bp, Blocks.AIR.defaultBlockState());
						}
					}
				}
			}
		}));
		c += 60;

		// ---- Mark 5 Chameleon Cloak
		at(c, () -> {
			view(mc, "q", 1);
			server(mc, p -> {
				clear(p);
				wear(p, "mark_v");
				dummy(p, 0, 0, 5);
			});
		});
		at(c + 15, () -> shot("mk5_cloak_before"));
		at(c + 18, () -> server(mc, p -> fire(p, "mark_v", 2, false)));
		at(c + 22, () -> shot("mk5_cloak_a"));
		at(c + 40, () -> shot("mk5_cloak_b"));
		c += 60;

		// ---- Mark 7 Meteor Dive
		at(c, () -> {
			view(mc, "far", 1);
			server(mc, p -> {
				clear(p);
				wear(p, "mark_vii");
				dummy(p, 2, 0, 6);
				dummy(p, -2, 0, 6.5);
				dummy(p, 0, 0, 8);
				p.teleportTo(p.serverLevel(), p.getX(), p.getY() + 22, p.getZ(), 0f, 60f);
				com.projecthero.mod.ironman.IronManFlight.setFlying(p, true);
			});
		});
		at(c + 20, () -> server(mc, p -> fire(p, "mark_vii", 3, true)));
		at(c + 23, () -> shot("mk7_meteor_dive_a"));
		at(c + 27, () -> shot("mk7_meteor_dive_b"));
		at(c + 31, () -> shot("mk7_meteor_dive_c"));
		at(c + 36, () -> shot("mk7_meteor_dive_d"));
		c += 60;

		// ---- Mark 8 Earthquake Stomp
		at(c, () -> {
			view(mc, "s", 1.45);
			server(mc, p -> {
				clear(p);
				wear(p, "mark_8");
				p.setXRot(0f);
				dummy(p, 0, 0, 5);
				dummy(p, 0.6, 0, 9);
				dummy(p, -0.6, 0, 13);
			});
		});
		at(c + 25, () -> server(mc, p -> fire(p, "mark_8", 2, true)));
		at(c + 31, () -> shot("mk8_earthquake_a"));
		at(c + 36, () -> shot("mk8_earthquake_b"));
		at(c + 42, () -> shot("mk8_earthquake_c"));
		c += 60;
		// a plain look at the Hulkbuster's size next to a husk
		at(c, () -> view(mc, "s", 1.45));
		at(c + 5, () -> shot("mk8_size"));
		c += 10;

		at(c, () -> {
			System.out.println("[HARNESS] done");
			mc.stop();
		});
	}
}
