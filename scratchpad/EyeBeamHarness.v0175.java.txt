/**
 * TEMP-HARNESS (not shipped, v0.17.5 eye beams): stills of every player eye beam (third person side / front / wide / back view,
 * first person, impact close-up; Heat Vision also crouching, looking up and flying) and films of Heat Vision start -> sweep -> stop
 * in third and first person. The beams are switched on purely client-side (the synced state each renderer reads), so nothing
 * on the server fires. Enabled by run/eyebeam.flag = "<out dir>|<before|after>"; a fixed camera via EyeBeamHarnessCameraMixin.
 */
public final class EyeBeamHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static BlockPos base;
	private static File out;
	private static String mode = "after";
	private static final Map<Integer, List<Runnable>> AT = new java.util.HashMap<>();
	private static int film0 = -1, filmEnd = -1;
	private static String filmDir = "film";
	private static boolean camOn;
	private static Vec3 camEye, camLook;
	private static String beam = "";
	private static boolean beamOn, flyingFlag;
	private static long beamStart;
	private static float yaw = 0f, pitch = 5f, prevYaw = 0f, prevPitch = 5f;
	public static volatile double[] camPose;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("eyebeam.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			String[] parts = Files.readString(flag).trim().split("\\|");
			out = new File(parts[0]);
			if (parts.length > 1) {
				mode = parts[1];
			}
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(EyeBeamHarness::tick);
		ClientTickEvents.START_CLIENT_TICK.register(mc -> {
			placeCam(mc);
			apply(mc);
		});
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(600_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "eyebeam-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		new File(out, "screenshots/" + name + ".png").getParentFile().mkdirs();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
	}

	private static void at(int t, Runnable r) {
		AT.computeIfAbsent(t, k -> new ArrayList<>()).add(r);
	}

	/** The beam state this client shows (re-asserted every tick in case a sync lands). */
	private static void apply(Minecraft mc) {
		var p = mc.player;
		if (p == null || mc.level == null) {
			return;
		}
		switch (beam) {
			case "heat" -> {
				KryptonianState s = new KryptonianState();
				s.hasPower = true;
				s.heatVision = beamOn;
				s.flying = flyingFlag;
				p.setAttached(ModAttachments.KRYPTONIAN_STATE, s);
			}
			case "laser" -> p.setAttached(ModAttachments.MUTATION_VISUALS,
					beamOn ? new MutationVisualState("p02.beam", beamStart, List.of(), Map.of()) : MutationVisualState.EMPTY);
			case "solar" -> {
				SentryState s = new SentryState();
				s.hasPower = true;
				s.beamStart = beamOn ? beamStart : 0L;
				p.setAttached(SentryAttachments.STATE, s);
			}
			case "starfire" -> {
				StarfireState s = new StarfireState();
				s.hasPower = true;
				s.beams = beamOn;
				p.setAttached(ModAttachments.STARFIRE_STATE, s);
			}
			case "vision" -> {
				VisionState s = new VisionState();
				s.hasPower = true;
				s.beam = beamOn;
				p.setAttached(VisionAttachments.STATE, s);
			}
			default -> {
			}
		}
	}

	private static void select(String b) {
		var p = Minecraft.getInstance().player;
		p.removeAttached(ModAttachments.KRYPTONIAN_STATE);
		p.removeAttached(ModAttachments.MUTATION_VISUALS);
		p.removeAttached(SentryAttachments.STATE);
		p.removeAttached(ModAttachments.STARFIRE_STATE);
		p.removeAttached(VisionAttachments.STATE);
		beam = b;
		beamOn = false;
	}

	private static void on(boolean v) {
		Minecraft mc = Minecraft.getInstance();
		if (v && !beamOn) {
			beamStart = mc.level.getGameTime();
		}
		beamOn = v;
		apply(mc);
	}

	private static void camAt(Vec3 eye, Vec3 look) {
		camEye = eye;
		camLook = look;
		camOn = true;
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = true;
		placeCam(mc);
	}

	/** Fixed camera {@code dist} out at {@code angle} degrees round the player (0 = straight ahead of him), {@code h} up. */
	private static void cam3(float angle, float dist, float h, float lookAhead, float lookH) {
		Vec3 c = new Vec3(base.getX() + 0.5, base.getY() + liftY, base.getZ() + 0.5);
		double r = Math.toRadians(angle);
		camAt(c.add(Math.sin(r) * dist, h, Math.cos(r) * dist), c.add(0, lookH, lookAhead));
	}

	private static double liftY;

	private static void fp() {
		camOn = false;
		camPose = null;
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = false;
		mc.options.setCameraType(CameraType.FIRST_PERSON);
	}

	private static void back() {
		camOn = false;
		camPose = null;
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = true;
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
	}

	private static void placeCam(Minecraft mc) {
		if (!camOn || mc.player == null || mc.level == null || base == null) {
			camPose = null;
			return;
		}
		Vec3 d = camLook.subtract(camEye);
		float cy = (float) (Math.toDegrees(Math.atan2(-d.x, d.z)));
		float cp = (float) (-Math.toDegrees(Math.atan2(d.y, d.horizontalDistance())));
		camPose = new double[] { camEye.x, camEye.y, camEye.z, cy, cp };
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
	}

	private static void face(Minecraft mc) {
		var p = mc.player;
		p.yRotO = prevYaw;
		p.xRotO = prevPitch;
		p.setYRot(yaw);
		p.setXRot(pitch);
		p.yBodyRotO = prevYaw;
		p.yBodyRot = yaw;
		p.yHeadRotO = prevYaw;
		p.yHeadRot = yaw;
		prevYaw = yaw;
		prevPitch = pitch;
	}

	private static void look(float y, float x) {
		yaw = y;
		pitch = x;
	}

	private static void crouch(boolean v) {
		KeyMapping.set(Minecraft.getInstance().options.keyShift.getDefaultKey(), v);
	}

	private static void stage(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		level.setDayTime(6000L);
		level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_DOFIRETICK).set(false, p.server);
		p.server.setDifficulty(Difficulty.EASY, true);
		p.setGameMode(GameType.CREATIVE);
		base = p.blockPosition();
		for (int x = -12; x <= 12; x++) {
			for (int z = -8; z <= 14; z++) {
				level.setBlock(base.offset(x, -1, z), (x + z) % 2 == 0 ? Blocks.SMOOTH_STONE.defaultBlockState() : Blocks.POLISHED_ANDESITE.defaultBlockState(), 3);
			}
		}
		for (int x = -8; x <= 8; x++) {
			for (int y = 0; y <= 6; y++) {
				level.setBlock(base.offset(x, y, 9), Blocks.STONE_BRICKS.defaultBlockState(), 3);
			}
		}
		p.getInventory().clearContent();
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, 5f);
	}

	private static void husk(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		Husk h = new Husk(EntityType.HUSK, level);
		h.setNoAi(true);
		h.getAttribute(Attributes.MAX_HEALTH).setBaseValue(1000);
		h.setHealth(1000);
		h.setPersistenceRequired();
		h.moveTo(base.getX() + 0.5 + 3.0, base.getY(), base.getZ() + 0.5 + 6.0, 180f, 0f);
		h.setYHeadRot(180f);
		level.addFreshEntity(h);
	}

	private static void film(int from, int to, String dir) {
		at(from, () -> {
			film0 = from;
			filmEnd = to;
			filmDir = dir;
		});
	}

	/** The standard set of stills for one beam. */
	private static int stills(Minecraft mc, int t, String b) {
		at(t, () -> {
			select(b);
			look(0f, 5f);
		});
		at(t + 2, () -> on(true));
		at(t + 3, () -> cam3(90f, 3.2f, 1.7f, 2.5f, 1.4f));
		at(t + 10, () -> shot(mc, b + "_tp_side"));
		at(t + 11, () -> cam3(-30f, 2.4f, 1.75f, 0f, 1.55f));
		at(t + 16, () -> shot(mc, b + "_tp_front"));
		at(t + 17, () -> cam3(65f, 7f, 2.4f, 4.5f, 1.1f));
		at(t + 22, () -> shot(mc, b + "_tp_wide"));
		at(t + 23, () -> back());
		at(t + 28, () -> shot(mc, b + "_tp_back"));
		at(t + 29, () -> camAt(new Vec3(base.getX() + 1.9, base.getY() + 1.5, base.getZ() + 6.6), new Vec3(base.getX() + 0.5, base.getY() + 0.85, base.getZ() + 9)));
		at(t + 36, () -> shot(mc, b + "_impact"));
		at(t + 37, () -> fp());
		at(t + 44, () -> shot(mc, b + "_fp"));
		at(t + 45, () -> look(-26.6f, 4f)); // at the husk
		at(t + 50, () -> shot(mc, b + "_fp_entity"));
		at(t + 51, () -> cam3(110f, 3.4f, 1.7f, 3f, 1.3f));
		at(t + 56, () -> shot(mc, b + "_tp_entity"));
		at(t + 57, () -> {
			on(false);
			look(0f, 5f);
		});
		return t + 64;
	}

		int t = 0;
		at(t, () -> server(mc, EyeBeamHarness::husk));

			}
			at(t, () -> {
				select("heat");

			});

			at(t + 12, () -> shot(mc, "heat_fp_husk_fov90"));

				mc.options.bobView().set(false);

				final float y = -20f + 40f * i / 40f;
				final float x = 6f + 6f * (float) Math.sin(i / 40.0 * Math.PI);
				at(t + 26 + i, () -> look(y, x));
			}
			at(t + 72, () -> on(false));
			t += 92;
			final int e2 = t;
			at(e2, () -> {
				System.out.println("[HARNESS] done");
				Runtime.getRuntime().halt(0);
			});
			return;
		}
		if (mode.startsWith("gaze")) {
			// Solar Gaze ray check: at the husk, then just beside it (the line passes ~0.2 clear of its hit-box), then over its head

			String[] names = { "gaze_at_husk", "gaze_beside_husk", "gaze_over_husk" };
			at(t, () -> select("solar"));
			at(t + 1, () -> on(true));

				final float[] a = aims[i];
				final String n = names[i];
				at(t + 2, () -> look(a[0], a[1]));
				at(t + 4, () -> cam3(-70f, 5.5f, 2.0f, 4.5f, 1.0f));
				at(t + 10, () -> shot(mc, n + "_tp_side"));
				at(t + 11, () -> cam3(150f, 3.0f, 2.6f, 4.5f, 1.0f));

			});
			look(0f, 25f);
			liftY = 3.0;
		});
		at(t + 36, () -> on(true));
		at(t + 37, () -> cam3(90f, 3.6f, 1.7f, 2.5f, 1.0f));
		at(t + 44, () -> shot(mc, "heat_flying_tp_side"));
		at(t + 45, () -> cam3(-35f, 2.6f, 1.6f, 0f, 1.3f));
		at(t + 50, () -> shot(mc, "heat_flying_tp_front"));
		at(t + 51, () -> {
			on(false);
			flyingFlag = false;
			liftY = 0;
			server(mc, p -> {
				p.getAbilities().flying = false;
				p.onUpdateAbilities();
				p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, 5f);
			});
			look(0f, 5f);
		});
		t += 64;
		if (mode.equals("after")) {
			// films: start -> sweep -> stop, third person then first person
			for (String view : new String[] { "tp", "fp" }) {
				final boolean tp = view.equals("tp");
				at(t, () -> {
					select("heat");
					look(-20f, 6f);
					if (tp) {
						cam3(70f, 6.5f, 2.3f, 4.5f, 1.1f);
					} else {
						fp();
					}
				});
				film(t + 6, t + 86, "film_heat_" + view);
				at(t + 14, () -> on(true));
				for (int i = 0; i <= 40; i++) {
					final float y = -20f + 40f * i / 40f;
					final float x = 6f + 6f * (float) Math.sin(i / 40.0 * Math.PI);
					at(t + 26 + i, () -> look(y, x));
				}
				at(t + 72, () -> on(false));
				t += 92;
			}
			// Laser Vision, a short third-person clip
			at(t, () -> {
				select("laser");
				look(10f, 6f);
				cam3(70f, 6.5f, 2.3f, 4.5f, 1.1f);

			film(t + 6, t + 56, "film_laser_tp");
			at(t + 12, () -> on(true));
			at(t + 44, () -> on(false));
			t += 60;
		}
		final int end = t;
		at(end, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(EyeBeamHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 20) {
			GLFW.glfwSetWindowSize(mc.getWindow().getWindow(), 854, 480);
		}
		if (t == 40) {
			mc.options.fov().set(70);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			mc.getToasts().clear();
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, EyeBeamHarness::stage);
		}
		if (t < 140 || base == null) {
			return;
		}
		face(mc);
		int s = t - 140;
		if (s == 0) {
			plan();
		}
		if (s % 10 == 0) {
			mc.gui.getChat().clearMessages(false);
		}
		List<Runnable> todo = AT.remove(s);
		if (todo != null) {
			todo.forEach(Runnable::run);
		}
		face(mc);
		apply(mc);
		if (s >= film0 && s < filmEnd) {
			shot(mc, String.format(filmDir + "/f%03d", s - film0));
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(GameRules.RULE_DOMOBSPAWNING).set(false, null);
		LevelSettings settings = new LevelSettings("EYEBEAM_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.EASY, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}

	private boolean detached;

	protected abstract void setRotation(float yaw, float pitch);
