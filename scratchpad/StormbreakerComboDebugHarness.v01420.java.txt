package com.projecthero.mod.client;

import java.util.HashMap;
import java.util.Map;

/** TEMP-HARNESS (not shipped): Stormbreaker / Mjolnir display + combo tour. Shots land in run/screenshots/sb_*.png. */
public final class SbDebugHarness {
	public static Vec3 camOffset;
	public static double camLookY = 1.2;
	private static boolean createdWorld;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static final String TAG = System.getenv().getOrDefault("PH_SB_TAG", "a");

	private SbDebugHarness() {
	}

	public static void init() {
		ClientTickEvents.END_CLIENT_TICK.register(SbDebugHarness::tick);
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					fn.accept(srv.getPlayerList().getPlayers().get(0));
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		Screenshot.grab(mc.gameDirectory, "sb_" + TAG + "_" + name + ".png", mc.getMainRenderTarget(), x -> {
		});
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			if (!createdWorld && mc.screen instanceof TitleScreen) {
				createdWorld = true;
				LevelSettings settings = new LevelSettings("SbDebug" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
						Difficulty.NORMAL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
				mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, WorldOptions.defaultWithRandomSeed(),
						reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
								.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
						mc.screen);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.hideGui = false;
			mc.options.bobView().set(false);
			org.lwjgl.glfw.GLFW.glfwSetWindowSize(mc.getWindow().getWindow(), 1280, 720);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.setGameMode(GameType.SURVIVAL);
				Worthiness.setScore(p, Worthiness.TEST_WORTHY_SCORE);
				face(p, 0f);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	private static void face(ServerPlayer p, float pitch) {
		p.setYRot(0);
		p.setYHeadRot(0);
		p.setYBodyRot(0);
		p.setXRot(pitch);
		p.connection.teleport(p.getX(), p.getY(), p.getZ(), 0f, pitch);
	}

	private static void hold(Minecraft mc, Item item) {
		server(mc, p -> {
			p.getInventory().selected = 0;
			p.setItemInHand(InteractionHand.MAIN_HAND, new ItemStack(item));
		});
	}

	/** Display tour for one item, starting at tick {@code s}; returns the next free tick. */
	private static int tour(Minecraft mc, int s, Item item, String n) {
		at(s, () -> { hold(mc, item); mc.player.getInventory().selected = 0; mc.options.setCameraType(CameraType.THIRD_PERSON_BACK); camOffset = new Vec3(-2.6, 1.3, 0.4); });
		at(s + 8, () -> shot(mc, n + "_tp_right"));
		at(s + 10, () -> camOffset = new Vec3(0.9, 1.4, 2.8));
		at(s + 16, () -> shot(mc, n + "_tp_front"));
		at(s + 18, () -> camOffset = new Vec3(-1.6, 1.6, -2.4));
		at(s + 24, () -> shot(mc, n + "_tp_backright"));
		at(s + 26, () -> { camOffset = null; mc.options.setCameraType(CameraType.FIRST_PERSON); });
		at(s + 34, () -> shot(mc, n + "_fp"));
		at(s + 36, () -> mc.options.setCameraType(CameraType.THIRD_PERSON_BACK));
		return s + 40;
	}

	private static Husk dummy;

	private static void hit(Minecraft mc) {
		server(mc, p -> {
			face(p, 0f);
			((com.projecthero.mod.mixin.LivingEntityAccessor) p).projecthero$setAttackStrengthTicker(100);
			if (dummy != null) {
				dummy.invulnerableTime = 0;
				dummy.setHealth(dummy.getMaxHealth());
				p.swing(InteractionHand.MAIN_HAND, true);
				p.attack(dummy);
			}
		});
	}

	/** Combo tour for one weapon, starting at tick {@code s}; returns the next free tick. */
	private static int combo(Minecraft mc, int s, Item item, String n) {
		at(s, () -> {
			hold(mc, item);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			camOffset = new Vec3(-2.4, 1.7, 2.6);
			server(mc, p -> {
				if (dummy == null) {
					dummy = EntityType.HUSK.create(p.serverLevel());
					dummy.moveTo(p.getX(), p.getY(), p.getZ() + 2.3, 180f, 0f);
					dummy.setNoAi(true);
					dummy.getAttribute(net.minecraft.world.entity.ai.attributes.Attributes.MAX_HEALTH).setBaseValue(1000.0);
					dummy.getAttribute(net.minecraft.world.entity.ai.attributes.Attributes.KNOCKBACK_RESISTANCE).setBaseValue(1.0);
					dummy.setHealth(1000f);
					p.serverLevel().addFreshEntity(dummy);
				}
			});
		});
		for (int k = 0; k < 3; k++) {
			int h = s + 10 + k * 22;
			int step = k + 1;
			at(h, () -> hit(mc));
			for (int d : new int[] { 1, 3, 5, 8 }) {
				at(h + d, () -> shot(mc, n + "_tp_hit" + step + "_t" + d));
			}
		}
		int f = s + 10 + 3 * 22 + 50;
		at(f - 6, () -> { camOffset = null; mc.options.setCameraType(CameraType.FIRST_PERSON); });
		for (int k = 0; k < 3; k++) {
			int h = f + k * 22;
			int step = k + 1;
			at(h, () -> hit(mc));
			for (int d : new int[] { 1, 3, 5, 8 }) {
				at(h + d, () -> shot(mc, n + "_fp_hit" + step + "_t" + d));
			}
		}
		at(f + 3 * 22, () -> mc.options.setCameraType(CameraType.THIRD_PERSON_BACK));
		return f + 3 * 22 + 50;
	}

	private static void script(Minecraft mc) {
		int t = 4;
		t = tour(mc, t, ModItems.STORMBREAKER, "storm");
		t = tour(mc, t, ModItems.MJOLNIR, "mjol");
		t = tour(mc, t, net.minecraft.world.item.Items.IRON_AXE, "axe");
		if (System.getenv("PH_SB_COMBO") != null) {
			t = combo(mc, t, ModItems.MJOLNIR, "mjol");
			t = combo(mc, t, ModItems.STORMBREAKER, "storm");
		}
		// ground + item frame
		final int g = t;
		at(g, () -> {
			hold(mc, net.minecraft.world.item.Items.AIR);
			server(mc, p -> {
				if (dummy != null) {
					dummy.discard();
					dummy = null;
				}
				ServerLevel level = p.serverLevel();
				ItemEntity ie = new ItemEntity(level, p.getX() + 0.8, p.getY() + 0.1, p.getZ() + 1.6, new ItemStack(ModItems.STORMBREAKER));
				ie.setDeltaMovement(Vec3.ZERO);
				ie.setPickUpDelay(32767);
				level.addFreshEntity(ie);
				BlockPos wall = BlockPos.containing(p.getX() + 2.5, p.getY() + 1.0, p.getZ() + 1.5);
				level.setBlockAndUpdate(wall, Blocks.STONE.defaultBlockState());
				ItemFrame frame = new ItemFrame(level, wall.relative(Direction.WEST), Direction.WEST);
				frame.setItem(new ItemStack(ModItems.STORMBREAKER));
				level.addFreshEntity(frame);
			});
			camOffset = new Vec3(-1.6, 2.0, 3.4);
			camLookY = 0.6;
		});
		at(g + 16, () -> shot(mc, "ground_frame"));
		// thrown
		at(g + 20, () -> { camLookY = 1.2; camOffset = new Vec3(-5.0, 1.8, 5.0); hold(mc, ModItems.STORMBREAKER); });
		at(g + 24, () -> server(mc, p -> { face(p, 0f); StormbreakerEntity.throwFrom(p); }));
		at(g + 25, () -> shot(mc, "thrown_a"));
		at(g + 26, () -> shot(mc, "thrown_b"));
		at(g + 27, () -> shot(mc, "thrown_c"));
		at(g + 60, () -> mc.stop());
	}
}
