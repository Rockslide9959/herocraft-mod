/** TEMP-HARNESS (not shipped): stills and films of the v0.17.7 Thor skin and cape. Enabled by run/thorskin.flag = "<out dir>". */
public final class ThorSkinHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static BlockPos base;
	private static File out;
	private static final java.util.Map<Integer, List<Runnable>> AT = new java.util.HashMap<>();
	private static int film0 = -1, filmEnd = -1;
	private static String filmDir = "film";
	private static float pitch = 8f;
	private static boolean walk, sprint;

	/** Camera offset from the player's (interpolated) feet: dx, dy, dz, yaw, pitch; null = the normal camera. */
	public static volatile double[] camPose;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("thorskin.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim().split("\\|")[0]);
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(ThorSkinHarness::tick);
		ClientTickEvents.START_CLIENT_TICK.register(ThorSkinHarness::keys);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(540_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "thorskin-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		new File(out, "screenshots/" + name + ".png").getParentFile().mkdirs();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
	}

	private static void at(int t, Runnable r) {
		AT.computeIfAbsent(t, k -> new ArrayList<>()).add(r);
	}

	/** Camera {@code dist} from the player at {@code angle} (0 = in front of him, 180 = behind), {@code h} up, looking at {@code lookH}. */
	private static void cam(float angle, float dist, float h, float lookH) {
		double r = Math.toRadians(angle);
		double dx = Math.sin(r) * dist, dz = Math.cos(r) * dist;
		double ldx = -dx, ldy = lookH - h, ldz = -dz;
		float yaw = (float) Math.toDegrees(Math.atan2(-ldx, ldz));
		float pt = (float) -Math.toDegrees(Math.atan2(ldy, Math.sqrt(ldx * ldx + ldz * ldz)));
		camPose = new double[] { dx, h, dz, yaw, pt };
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = true;
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
	}

	private static void keys(Minecraft mc) {
		if (mc.player == null) {
			return;
		}
		mc.options.keyUp.setDown(walk);
		mc.options.keySprint.setDown(sprint);
		if (sprint && walk) {
			mc.player.setSprinting(true);
		}
	}

	private static void face(Minecraft mc) {
		var p = mc.player;
		p.setYRot(0f);
		p.yRotO = 0f;
		p.setXRot(pitch);
		p.xRotO = pitch;
		p.yHeadRot = 0f;
		p.yHeadRotO = 0f;
	}

	private static void stage(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		level.setDayTime(6000L);
		level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
		p.setGameMode(GameType.SURVIVAL);
		base = p.blockPosition();
		p.getInventory().clearContent();
		com.projecthero.mod.worthiness.Worthiness.setScore(p, com.projecthero.mod.worthiness.Worthiness.TEST_WORTHY_SCORE);
		ItemStack hammer = new ItemStack(com.projecthero.mod.item.ModItems.MJOLNIR);
		p.getInventory().selected = 0;
		p.setItemInHand(InteractionHand.MAIN_HAND, hammer);
		com.projecthero.mod.power.ThorPowers.toggleBinding(p, hammer);
		System.out.println("[HARNESS] thor: " + com.projecthero.mod.power.ThorPassives.hasPowerOfThor(p));
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, 8f);
	}

	/** Hammer to slot 1 (empty hands) or back to slot 0. */
	private static void hands(ServerPlayer p, boolean hammer) {
		var inv = p.getInventory();
		ItemStack h = ItemStack.EMPTY;
		for (int i = 0; i < inv.getContainerSize(); i++) {
			if (inv.getItem(i).is(com.projecthero.mod.item.ModItems.MJOLNIR)) {
				h = inv.getItem(i);
				inv.setItem(i, ItemStack.EMPTY);
			}
		}
		inv.setItem(hammer ? 0 : 1, h);
		inv.selected = 0;
	}

	private static void home(ServerPlayer p) {
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, pitch);
	}

	private static void film(int from, int to, String dir) {
		at(from, () -> {
			film0 = from;
			filmEnd = to;
			filmDir = dir;
		});
	}

	private static void plan() {

			com.projecthero.mod.power.StormEnergy.set(p, com.projecthero.mod.power.StormEnergy.MAX);
			com.projecthero.mod.thorarmor.ThorArmor.toggle(p);
		}));
		t += 86;

		at(t, () -> server(mc, p -> {
			hands(p, false);
			System.out.println("[HARNESS] armour: " + com.projecthero.mod.thorarmor.ThorArmor.isWearing(p));

		at(t + 12, () -> cam(180f, 3.2f, 1.4f, 1.0f));
		at(t + 18, () -> shot(mc, "thor_back"));
		at(t + 20, () -> cam(90f, 3.2f, 1.3f, 1.0f));
		at(t + 26, () -> shot(mc, "thor_side"));
		at(t + 28, () -> cam(-140f, 3.2f, 1.6f, 1.0f));
		at(t + 34, () -> shot(mc, "thor_back_three_quarter"));
		at(t + 36, () -> cam(35f, 3.0f, 1.4f, 1.0f));

		at(t, () -> server(mc, p -> hands(p, true)));
		at(t + 4, () -> cam(30f, 3.0f, 1.3f, 1.0f));
		at(t + 10, () -> shot(mc, "thor_mjolnir_front"));
		at(t + 12, () -> cam(-150f, 3.2f, 1.5f, 1.0f));
		at(t + 18, () -> shot(mc, "thor_mjolnir_back"));
		t += 22;
		// ---- walking (camera following, side-back)

			walk = false;
			sprint = false;

		t += 66;
		// ---- flying (Thor's flight, hammer in hand): take off 10 up, hover, then fly forward level (pitch 0 -- the flight follows the look)

			home(p);

		at(t + 60, () -> shot(mc, "fp_suitup_helmet"));
		at(t + 78, () -> shot(mc, "fp_suitup_cape"));
		t += 100;
		final int end = t;
		at(end, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(ThorSkinHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 20) {
			GLFW.glfwSetWindowSize(mc.getWindow().getWindow(), 854, 480);
		}
		if (t == 40) {
			mc.options.fov().set(70);
			mc.options.fovEffectScale().set(0.0);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			mc.getToasts().clear();
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, ThorSkinHarness::stage);
		}
		if (t < 140 || base == null) {
			return;
		}
		face(mc);
		int s = t - 140;
		if (s == 0) {
			plan();
		}
		if (s % 10 == 0) {
			mc.gui.getChat().clearMessages(false);
		}
		List<Runnable> todo = AT.remove(s);
		if (todo != null) {
			todo.forEach(Runnable::run);
		}
		if (s >= film0 && s < filmEnd) {
			shot(mc, String.format(filmDir + "/f%03d", s - film0));
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(GameRules.RULE_DOMOBSPAWNING).set(false, null);
		LevelSettings settings = new LevelSettings("THORSKIN_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.PEACEFUL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}

package com.projecthero.mod.client.mixin;

import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.Shadow;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

import net.minecraft.client.Camera;
import net.minecraft.world.entity.Entity;
import net.minecraft.world.level.BlockGetter;
import net.minecraft.world.phys.Vec3;

/** TEMP-HARNESS (not shipped): the ThorSkinHarness's camera, a fixed offset from the player's interpolated position. */
@Mixin(Camera.class)
public abstract class ThorSkinHarnessCameraMixin {
	@Shadow
	private boolean detached;

	@Shadow
	protected abstract void setRotation(float yaw, float pitch);

	@Shadow
	protected abstract void setPosition(double x, double y, double z);

	@Inject(method = "setup", at = @At("TAIL"))
	private void projecthero$harnessCam(BlockGetter level, Entity entity, boolean detached, boolean mirrored, float partial, CallbackInfo ci) {
		double[] c = com.projecthero.mod.client.ThorSkinHarness.camPose;
		if (c != null && entity != null) {
			Vec3 p = entity.getPosition(partial);
			this.detached = true;
			setPosition(p.x + c[0], p.y + c[1], p.z + c[2]);
			setRotation((float) c[3], (float) c[4]);
		}
	}
}
