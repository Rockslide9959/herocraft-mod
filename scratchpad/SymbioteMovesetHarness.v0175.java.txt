/**
 * TEMP-HARNESS (not shipped, v0.17.5 Symbiote moveset v2): a suited Normal host on a flat arena runs every new move on husks / iron
 * golems through the REAL keys (the ability key mappings held down and let go, so the client sends its own press / release packets and
 * the HUD fills while a key is held): R x3 (Tendril Slam), hold R (Impaling Spear), hold G (Spike Rain), X in the air (Dive Bomb), hold
 * Z (Ground Eruption), C (Maw Bite) and hold C (Devour) as front close-ups, V (the Form cycle). Stills per move, the HUD and the
 * Left-Alt list, and a film (every tick). Enabled by run/symmove.flag = "<out dir>". Kept as scratchpad/SymbioteMovesetHarness.v0175.java.txt.
 */
public final class SymbioteMovesetHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static File out;
	private static final List<Runnable> PLAN = new ArrayList<>();
	private static int planTick;
	private static BlockPos site;
	private static final List<Integer> SPAWNED = new ArrayList<>();

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("symmove.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readAllLines(flag).get(0).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(SymbioteMovesetHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(480_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "symmove-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		mc.gui.getChat().clearMessages(false);
		mc.getToasts().clear();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
		if (!name.startsWith("film_")) {
			System.out.println("[HARNESS] shot " + name + " active=" + Symbiote.isActive(mc.player));
		}
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			prev.run();
			r.run();
		});
	}

	private static void toggle() {
		ClientPlayNetworking.send(new com.projecthero.mod.network.SpiderActionPayload(
				com.projecthero.mod.network.SpiderActionPayload.Action.TOGGLE_SYMBIOTE));
	}

	/** Key i (0 = R, 1 = G, 2 = X, 3 = Z, 4 = V, 5 = C) down at {@code t}, up at {@code t + held}. */
	private static void key(int t, int i, int held) {
		KeyMapping k = ModKeyBindings.ABILITY_SLOTS[i];
		at(t, () -> k.setDown(true));
		at(t + held, () -> k.setDown(false));
	}

	private static void clearTargets(Minecraft mc) {
		server(mc, p -> {
			for (int id : SPAWNED) {
				Entity e = p.serverLevel().getEntity(id);
				if (e != null) {
					e.discard();
				}
			}
			SPAWNED.clear();
		});
	}

	/** A target {@code dz} blocks ahead (north) and {@code dx} to the side; health / max health as given. */
	private static void spawn(Minecraft mc, EntityType<? extends Mob> type, double dx, double dz, double max, float health) {
		server(mc, p -> {
			ServerLevel level = p.serverLevel();
			Mob m = type.create(level);
			if (m == null) {
				return;
			}
			m.moveTo(site.getX() + 0.5 + dx, site.getY() + 1, site.getZ() + 0.5 - dz, 0f, 0f);
			m.setNoAi(true);
			m.setPersistenceRequired();
			m.getAttribute(Attributes.MAX_HEALTH).setBaseValue(max);
			m.setHealth(health);
			m.addEffect(new MobEffectInstance(MobEffects.FIRE_RESISTANCE, 20 * 600, 0, false, false));
			level.addFreshEntity(m);
			SPAWNED.add(m.getId());
		});
	}

	private static void face(Minecraft mc, double dz, float yaw, float pitch, double up) {
		server(mc, p -> p.connection.teleport(site.getX() + 0.5, site.getY() + 1 + up, site.getZ() + 0.5 - dz, yaw, pitch));
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(854, 480);
				mc.execute(SymbioteMovesetHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(70);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(com.projecthero.mod.loot.finds.SupplyDrops.RULE).set(false, p.server);
				BlockPos sp = p.blockPosition();
				int y = level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, sp.getX(), sp.getZ());
				site = new BlockPos(sp.getX(), y - 1, sp.getZ());
				for (int x = -14; x <= 14; x++) {
					for (int z = -16; z <= 12; z++) {
						for (int yy = -2; yy <= 14; yy++) {
							BlockState st = yy < 0 ? Blocks.DIRT.defaultBlockState() : yy == 0 ? Blocks.GRASS_BLOCK.defaultBlockState() : Blocks.AIR.defaultBlockState();
							level.setBlock(site.offset(x, yy, z), st, 2 | 16);
						}
					}
				}
				p.setGameMode(GameType.SURVIVAL);
				p.connection.teleport(site.getX() + 0.5, site.getY() + 1, site.getZ() + 0.5, 180f, 8f);
				Symbiote.grant(p);
				SymbioteSuitColours.setPreset(p, SymbioteColourPreset.CLASSIC);
				p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 20 * 60 * 30, 4, false, false));
				p.addEffect(new MobEffectInstance(MobEffects.SATURATION, 20 * 60 * 30, 4, false, false));
			});
		}
		if (t == 46) {
			buildPlan(mc);
		}
		if (t < 50 || site == null) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		}
	}

	private static void buildPlan(Minecraft mc) {
		int t = 10;
		at(t, () -> {
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.hideGui = false;
		});
		at(t + 5, SymbioteMovesetHarness::toggle);
		at(t + 90, () -> shot(mc, "01_hud_suited"));
		at(t + 92, () -> MoveListOverlay.forceHeld = true);
		at(t + 95, () -> shot(mc, "02_left_alt_move_list"));
		at(t + 97, () -> MoveListOverlay.forceHeld = false);

		// ---- the film: every tick from here
		int f0 = t + 100;
		int filmLen = 490;
		for (int f = 0; f < filmLen; f++) {
			final int n = f;
			at(f0 + f, () -> shot(mc, String.format("film_moves_%03d", n)));
		}
		// A: R x3 on an iron golem 7 blocks ahead -- the 3rd is the Tendril Slam
		int a = f0;
		at(a - 6, () -> {
			face(mc, 0, 180f, 4f, 0);
			spawn(mc, EntityType.IRON_GOLEM, 0, 7, 2000, 2000f);
		});
		key(a + 4, 0, 2);
		at(a + 9, () -> shot(mc, "03_tendril_strike"));
		key(a + 38, 0, 2);
		key(a + 72, 0, 2);
		at(a + 77, () -> shot(mc, "04_tendril_slam_yank"));
		at(a + 82, () -> shot(mc, "05_tendril_slam_smash"));
		// B: hold R -- the Impaling Spear through two husks
		int b = a + 105;
		at(b - 6, () -> {
			clearTargets(mc);
			spawn(mc, EntityType.HUSK, 0, 4, 1000, 1000f);
			spawn(mc, EntityType.HUSK, 0.2, 7.5, 1000, 1000f);
		});
		key(b, 0, 30);
		at(b + 24, () -> shot(mc, "06_impaling_spear_charging"));
		at(b + 33, () -> shot(mc, "07_impaling_spear_thrown"));
		// C: hold G -- Spike Rain on a group 9 blocks ahead
		int c = b + 55;
		at(c - 6, () -> {
			clearTargets(mc);
			face(mc, 0, 180f, 18f, 0);
			spawn(mc, EntityType.HUSK, 0, 9, 1000, 1000f);
			spawn(mc, EntityType.HUSK, -1.8, 10, 1000, 1000f);
			spawn(mc, EntityType.IRON_GOLEM, 2.0, 9.5, 2000, 2000f);
		});
		key(c, 1, 12);
		at(c + 16, () -> shot(mc, "08_spike_rain_launch"));
		at(c + 34, () -> shot(mc, "09_spike_rain_falling"));
		// D: X in the air -- Dive Bomb onto the same group
		int d = c + 75;
		at(d - 4, () -> face(mc, 6.5, 180f, 40f, 7.0));
		key(d, 2, 2);
		at(d + 2, () -> shot(mc, "10_dive_bomb"));
		at(d + 6, () -> shot(mc, "11_dive_bomb_impact"));
		// E: hold Z -- Ground Eruption down a line of husks
		int e = d + 40;
		at(e - 6, () -> {
			clearTargets(mc);
			face(mc, 0, 180f, 10f, 0);
			spawn(mc, EntityType.HUSK, 0, 3.5, 1000, 1000f);
			spawn(mc, EntityType.HUSK, 0.4, 7, 1000, 1000f);
			spawn(mc, EntityType.IRON_GOLEM, -0.3, 10.5, 2000, 2000f);
		});
		key(e, 3, 12);
		at(e + 12, () -> shot(mc, "12_ground_eruption"));
		at(e + 16, () -> shot(mc, "13_ground_eruption_launch"));
		// F: C -- Maw Bite, a front close-up (the husk off to the left in the bite cone)
		int g = e + 45;
		at(g - 8, () -> {
			clearTargets(mc);
			face(mc, 0, 180f, 6f, 0);
			spawn(mc, EntityType.HUSK, 1.2, 2.4, 1000, 1000f);
			mc.options.setCameraType(CameraType.THIRD_PERSON_FRONT);
			mc.options.fov().set(30);
		});
		key(g, 5, 2);
		at(g + 3, () -> shot(mc, "14_maw_bite_gape"));
		at(g + 6, () -> shot(mc, "15_maw_bite_snap"));
		at(g + 14, () -> shot(mc, "16_maw_bite_lick"));
		// G: hold C on a husk at 20% -- Devour (the tongue)
		int h = g + 45;
		at(h - 8, () -> {
			clearTargets(mc);
			face(mc, 0, 180f, 6f, 0);
			spawn(mc, EntityType.HUSK, 1.3, 3.0, 20, 4f);
		});
		key(h, 5, 12);
		at(h + 11, () -> shot(mc, "17_devour_tongue_out"));
		at(h + 15, () -> shot(mc, "18_devour_reeling_in"));
		at(h + 22, () -> shot(mc, "19_devour_jaws_close"));
		at(h + 28, () -> shot(mc, "20_devour_chewing"));
		// H: V -- the Form cycle
		int v = h + 55;
		at(v - 6, () -> {
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.fov().set(70);
			face(mc, 0, 150f, 8f, 0);
		});
		key(v, 4, 2);
		at(v + 14, () -> shot(mc, "21_form_blade"));
		key(v + 24, 4, 2);
		at(v + 38, () -> shot(mc, "22_form_thorns"));
		key(v + 48, 4, 2);
		at(v + 60, () -> shot(mc, "23_form_normal"));
		at(f0 + filmLen + 4, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("SYMMOVE_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(17051L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

}
