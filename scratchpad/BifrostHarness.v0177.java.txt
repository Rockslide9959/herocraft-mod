/**
 * TEMP-HARNESS (not shipped, v0.17.7 Stormbreaker clipping fix + Bifrost Travel cinematic): in-game stills of the axe up close
 * (first person both hands, third person close, the hotbar, on the ground) with the old model swapped in ("before") and the
 * new one ("after"), then films of the whole Bifrost Travel cinematic -- third person from a fixed camera at each end with a
 * wolf and a cat riding along, and first person. One frame per client tick. Enabled by run/bifrost.flag = "<out dir>|<dir of
 * the old stormbreaker*.json models>"; the camera is BifrostHarnessCameraMixin.
 */
public final class BifrostHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks, t0 = -1;
	private static File out;
	private static Path beforeModels;
	private static final Map<Integer, List<Runnable>> AT = new HashMap<>();
	private static volatile long lastProgress = System.currentTimeMillis();
	/** Camera relative to the player (stills) ... */
	public static volatile Vec3 camOffset;
	public static volatile Vec3 camLook = new Vec3(0, 1.1, 0);
	/** ... or at fixed world spots (films): the one nearer the player is used, looking at the player + lookUp. */
	public static volatile Vec3 camA, camB;
	public static volatile double lookUp = 2.0;
	public static float pitch;
	public static float yaw;
	private static String film;
	private static int frame;
	private static Vec3 origin;
	private static final int DX = 40;

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("bifrost.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			String[] parts = Files.readString(flag).trim().split("\\|");
			out = new File(parts[0]);
			beforeModels = parts.length > 1 ? Path.of(parts[1]) : null;
		} catch (Exception e) {
			out = new File("bifrost_out");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(BifrostHarness::tick);
		Thread wd = new Thread(() -> {
			long start = System.currentTimeMillis();
			while (true) {
				try {
					Thread.sleep(5000);
				} catch (InterruptedException e) {
					return;
				}
				if (System.currentTimeMillis() - lastProgress > 150_000L || System.currentTimeMillis() - start > 900_000L) {
					System.out.println("[HARNESS] WATCHDOG: halting");
					Runtime.getRuntime().halt(3);
				}
			}
		}, "bifrost-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	public static boolean active() {
		return t0 >= 0;
	}

	private static void shot(Minecraft mc, String name) {
		new File(out, "screenshots/" + name + ".png").getParentFile().mkdirs();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), c -> {
		});
		lastProgress = System.currentTimeMillis();
	}

	private static ServerPlayer sp(Minecraft mc) {
		MinecraftServer srv = mc.getSingleplayerServer();
		return srv == null || srv.getPlayerList().getPlayers().isEmpty() ? null : srv.getPlayerList().getPlayers().get(0);
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				ServerPlayer p = sp(mc);
				if (p != null) {
					try {
						fn.accept(p);
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.computeIfAbsent(t, k -> new ArrayList<>()).add(r);
	}

	private static void view(Minecraft mc, String v) {
		camA = camB = null;
		mc.options.hideGui = true;
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
		switch (v) {
			case "cq" -> { camOffset = new Vec3(-1.0, 1.3, 1.35); camLook = new Vec3(-0.25, 1.0, 0.3); }
			case "chead" -> { camOffset = new Vec3(-1.25, 1.35, 1.55); camLook = new Vec3(-0.45, 1.0, 0.7); }
			case "front" -> { camOffset = new Vec3(0.35, 1.3, 2.7); camLook = new Vec3(0, 1.05, 0); }
			case "g" -> { camOffset = new Vec3(0.0, 1.0, 0.9); camLook = new Vec3(0, 0.2, 2.2); }
			default -> {
				camOffset = null;
				mc.options.setCameraType(CameraType.FIRST_PERSON);
			}
		}
	}

	private static void hands(Minecraft mc, ItemStack main, ItemStack off) {
		server(mc, p -> {
			p.getInventory().clearContent();
			p.getInventory().selected = 0;
			p.setItemInHand(InteractionHand.MAIN_HAND, main);
			p.setItemInHand(InteractionHand.OFF_HAND, off);
		});
	}

	private static ItemStack s() {
		return new ItemStack(ModItems.STORMBREAKER);
	}

	/** Copies the old (before) or current (after) Stormbreaker models over the ones the client loads, then reloads. */
	private static void models(Minecraft mc, boolean before) {
		try {
			Path root = FabricLoader.getInstance().getModContainer("projecthero").orElseThrow().getRootPaths().stream()
					.map(r -> r.resolve("assets/projecthero/models/item")).filter(Files::isDirectory).findFirst().orElseThrow();
			Path src = before ? beforeModels : Path.of("../src/main/resources/assets/projecthero/models/item");
			for (String n : new String[] {"stormbreaker.json", "stormbreaker_thrown.json"}) {
				Files.copy(src.resolve(n), root.resolve(n), StandardCopyOption.REPLACE_EXISTING);
			}
			System.out.println("[HARNESS] models " + (before ? "before" : "after") + " -> " + root);
		} catch (Exception e) {
			System.out.println("[HARNESS] model swap failed: " + e);
		}
		mc.reloadResourcePacks();
	}

	private static int stills(Minecraft mc, int c, String tag) {
		at(c, () -> {
			hands(mc, s(), ItemStack.EMPTY);
			pitch = 0;
		});
		c += 24;
		for (String v : new String[] {"fp", "cq", "chead", "front"}) {
			at(c, () -> view(mc, v));
			at(c + 5, () -> shot(mc, tag + "_right_" + v));
			c += 7;
		}
		at(c, () -> hands(mc, ItemStack.EMPTY, s()));
		c += 24;
		at(c, () -> view(mc, "fp"));
		at(c + 5, () -> shot(mc, tag + "_left_fp"));
		c += 8;
		// on the ground (the resting / thrown model) and in the hotbar
		at(c, () -> {
			view(mc, "g");
			server(mc, p -> {
				p.getInventory().clearContent();
				p.getInventory().setItem(0, s());
				p.getInventory().selected = 4;
				ItemEntity a = new ItemEntity(p.serverLevel(), p.getX(), p.getY() + 0.4, p.getZ() + 2.2, s(), 0, 0, 0);
				p.serverLevel().addFreshEntity(a);
			});
		});
		at(c + 40, () -> shot(mc, tag + "_ground"));
		at(c + 41, () -> {
			view(mc, "fp");
			mc.options.hideGui = false;
			pitch = 50f;
		});
		at(c + 46, () -> shot(mc, tag + "_hotbar"));
		at(c + 47, () -> {
			pitch = 0f;
			server(mc, p -> {
				for (var e : p.serverLevel().getEntities((net.minecraft.world.entity.Entity) null, p.getBoundingBox().inflate(8),
						e -> e instanceof ItemEntity || e instanceof com.projecthero.mod.stormbreaker.StormbreakerEntity)) {
					e.discard();
				}
			});
		});
		return c + 50;
	}

	private static void pets(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		Wolf wolf = EntityType.WOLF.create(level);
		wolf.moveTo(p.getX() + 1.6, p.getY(), p.getZ() + 1.1, 0, 0);
		wolf.tame(p);
		wolf.setOrderedToSit(false);
		level.addFreshEntity(wolf);
		Cat cat = EntityType.CAT.create(level);
		cat.moveTo(p.getX() - 1.4, p.getY(), p.getZ() + 0.9, 0, 0);
		cat.tame(p);
		cat.setOrderedToSit(false);
		level.addFreshEntity(cat);
	}

	/** Builds a small stone-brick plaza (a landmark) at {@code c}. */
	private static void plaza(ServerLevel level, BlockPos c, boolean pillars) {
		for (int x = -5; x <= 5; x++) {
			for (int z = -5; z <= 5; z++) {
				level.setBlock(c.offset(x, -1, z), (Math.abs(x) == 5 || Math.abs(z) == 5 ? Blocks.CHISELED_STONE_BRICKS : Blocks.STONE_BRICKS).defaultBlockState(), 3);
			}
		}
		if (pillars) {
			for (int[] o : new int[][] {{-5, -5}, {5, -5}, {-5, 5}, {5, 5}}) {
				for (int y = 0; y < 4; y++) {
					level.setBlock(c.offset(o[0], y, o[1]), Blocks.POLISHED_DEEPSLATE.defaultBlockState(), 3);
				}
				level.setBlock(c.offset(o[0], 4, o[1]), Blocks.LANTERN.defaultBlockState(), 3);
			}
		}
	}

	private static void filmTrip(Minecraft mc, int c, String name, boolean firstPerson, int toDx) {
		at(c, () -> {
			hands(mc, s(), ItemStack.EMPTY);
			server(mc, p -> {
				Bifrost.resetCooldownForTests(p);
				pets(p);
			});
			pitch = firstPerson ? -8f : 0f;
			server(mc, p -> p.serverLevel().setDayTime(13000L)); // dusk: the rainbow against a darkening sky
			if (firstPerson) {
				view(mc, "fp");
			} else {
				mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
				camOffset = null;
				double from = toDx > 0 ? 0 : DX;
				double to = toDx > 0 ? DX : 0;
				mc.options.hideGui = true;
				camA = origin.add(from - 8.0, 1.8, 2.5);
				camB = origin.add(to - 8.0, 1.8, 2.5);
				lookUp = 2.2;
			}
		});
		c += 40; // pets settle beside him
		at(c, () -> {
			film = name;
			frame = 0;
		});
		at(c + 6, () -> server(mc, p -> {
			BlockPos to = BlockPos.containing(origin.add(toDx, 0, 0));
			Bifrost.Result r = Bifrost.travelTo(p, to.getX(), to.getY(), to.getZ());
			System.out.println("[HARNESS] travel " + r);
		}));
		if (firstPerson) {
			// look up at the axe and the bridge, down as the ground falls away, ahead as you come down
			for (int k = 0; k <= BifrostTravel.LAND + 10; k++) {
				final int kk = k;
				at(c + 6 + k, () -> pitch = firstPersonPitch(kk));
			}
		}
		at(c + 6 + BifrostTravel.LAND + 26, () -> film = null);
	}

	private static float firstPersonPitch(int t) {
		if (t < BifrostTravel.STRIKE + 4) {
			return -8f - 30f * Math.min(1f, t / 12f); // up to the raised axe and the sky
		}
		if (t < BifrostTravel.TRANSPORT) {
			float k = Math.min(1f, (t - BifrostTravel.STRIKE - 4) / 30f);
			return -38f + 95f * k * k * (3 - 2 * k) * 0.8f; // down at the ground falling away
		}
		float k = Math.min(1f, (t - BifrostTravel.TRANSPORT) / 20f);
		return 38f - 30f * k;
	}

	private static void script(Minecraft mc) {
		at(2, () -> server(mc, p -> {
			p.setGameMode(GameType.SURVIVAL);
			Worthiness.setScore(p, Worthiness.TEST_WORTHY_SCORE);
			p.setItemInHand(InteractionHand.MAIN_HAND, s());
			p.getAbilities().invulnerable = true;
			p.onUpdateAbilities();
			origin = p.position();
			plaza(p.serverLevel(), p.blockPosition(), false);
			plaza(p.serverLevel(), p.blockPosition().offset(DX, 0, 0), true);
		}));
		int c = 20;
		if (beforeModels != null) {
			at(c, () -> models(mc, true));
			c = stills(mc, c + 40, "clip_before");
			at(c, () -> models(mc, false));
			c += 40;
		}
		c = stills(mc, c, "clip_after");
		filmTrip(mc, c, "film_bifrost_third_person", false, DX);
		c += 40 + 6 + BifrostTravel.LAND + 40;
		at(c, () -> server(mc, p -> {
			for (var e : p.serverLevel().getEntities((net.minecraft.world.entity.Entity) null, p.getBoundingBox().inflate(16),
					e -> e instanceof Wolf || e instanceof Cat)) {
				e.discard();
			}
		}));
		filmTrip(mc, c + 4, "film_bifrost_first_person", true, -DX);
		c += 4 + 40 + 6 + BifrostTravel.LAND + 40;
		at(c, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				lastProgress = System.currentTimeMillis();
				mc.execute(BifrostHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks % 10 == 0) {
			lastProgress = System.currentTimeMillis();
		}
		if (inWorldTicks == 20) {
			GLFW.glfwSetWindowSize(mc.getWindow().getWindow(), 854, 480);
		}
		mc.getToasts().clear();
		if (mc.getOverlay() != null) {
			if (t0 >= 0) {
				t0++; // hold the script while a resource reload overlay is up
			}
			return;
		}
		if (film != null) {
			shot(mc, film + String.format("/f%03d", frame++));
		}
		if (t0 >= 0) {
			mc.player.setYRot(yaw);
			mc.player.setXRot(pitch);
			mc.player.yRotO = yaw;
			mc.player.xRotO = pitch;
			mc.player.yHeadRot = yaw;
			mc.player.yHeadRotO = yaw;
			mc.player.yBodyRot = yaw;
			mc.player.yBodyRotO = yaw;
			if (film != null) {
				mc.options.hideGui = !mc.options.getCameraType().isFirstPerson();
			}
			mc.gui.getChat().clearMessages(false);
		}
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			mc.getToasts().clear();
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.setWeatherParameters(6000, 0, false, false);
				p.teleportTo(level, Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5, 0f, 0f);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			List<Runnable> r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.forEach(Runnable::run);
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("BIFROST_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false,
				Difficulty.PEACEFUL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}

package com.projecthero.mod.client.mixin;

import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.Shadow;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

import com.projecthero.mod.client.BifrostHarness;

import net.minecraft.client.Camera;
import net.minecraft.world.entity.Entity;
import net.minecraft.world.level.BlockGetter;
import net.minecraft.world.phys.Vec3;

/** TEMP-HARNESS (not shipped): BifrostHarness's camera -- relative to the player, or at the nearer of two fixed world spots. */
@Mixin(Camera.class)
public abstract class BifrostHarnessCameraMixin {
	@Shadow
	protected abstract void setPosition(Vec3 pos);

	@Shadow
	protected abstract void setRotation(float yRot, float xRot);

	@Inject(method = "setup", at = @At("TAIL"))
	private void projecthero$bifrostHarnessCam(BlockGetter level, Entity entity, boolean detached, boolean reverse, float partialTick, CallbackInfo ci) {
		if (!detached || !BifrostHarness.active()) {
			return;
		}
		Vec3 base = entity.getPosition(partialTick);
		Vec3 pos;
		Vec3 look;
		Vec3 a = BifrostHarness.camA, b = BifrostHarness.camB;
		if (a != null && b != null) {
			pos = a.distanceToSqr(base) <= b.distanceToSqr(base) ? a : b;
			look = base.add(0, BifrostHarness.lookUp, 0);
		} else if (BifrostHarness.camOffset != null) {
			pos = base.add(BifrostHarness.camOffset);
			look = base.add(BifrostHarness.camLook);
		} else {
			return;
		}
		Vec3 d = look.subtract(pos);
		float yaw = (float) Math.toDegrees(Math.atan2(-d.x, d.z));
		float pitch = (float) -Math.toDegrees(Math.atan2(d.y, Math.sqrt(d.x * d.x + d.z * d.z)));
		setRotation(yaw, pitch);
		setPosition(pos);
	}
}
