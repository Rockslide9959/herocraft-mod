/** TEMP-HARNESS (not shipped): v0.17.1 Wildmutt screenshots. Enabled by run/alienlook.flag. */
// TEMP-HARNESS
public final class AlienLookHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static Vec3 origin;
	private static ServerPlayer ben;
	private static Zombie dummy;
	/** Runs on the server every tick (a walking fake, a camera that follows). */
	private static volatile java.util.function.Consumer<ServerPlayer> mover;
	private static String only = System.getProperty("wm.only", "");

	private AlienLookHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("alienlook.flag"))) {
			return;
		}
		try {
			only = Files.readString(Path.of("wildmutt.flag")).trim();
		} catch (java.io.IOException e) {
			only = "";
		}
		ClientTickEvents.END_CLIENT_TICK.register(AlienLookHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(600_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "alienlook-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (x, y) -> () -> {
			x.run();
			y.run();
		});
	}

	private static void shot(int t, String name) {
		at(t - 1, () -> {
			if (!name.startsWith("quills") && !name.startsWith("slam") && !name.startsWith("frenzy")) {
				try {
					var m = net.minecraft.client.particle.ParticleEngine.class.getDeclaredMethod("clearParticles");
					m.setAccessible(true);
					m.invoke(Minecraft.getInstance().particleEngine);
				} catch (ReflectiveOperationException e) {
					System.out.println("[HARNESS] " + e);
				}
			}
		});
		at(t, () -> {
			Minecraft mc = Minecraft.getInstance();
			Screenshot.grab(mc.gameDirectory, "omnitrix_look_" + name + ".png", mc.getMainRenderTarget(), m -> {
			});
			System.out.println("[HARNESS] shot " + name);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.options.getSoundSourceOptionInstance(net.minecraft.sounds.SoundSource.MASTER).get() > 0.05) {
			mc.options.getSoundSourceOptionInstance(net.minecraft.sounds.SoundSource.MASTER).set(0.05);
		}
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(AlienLookHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			mc.options.chatVisibility().set(net.minecraft.world.entity.player.ChatVisiblity.HIDDEN);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.EASY, true);
				p.setGameMode(GameType.CREATIVE);
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			AT.put(3, () -> script(mc));
		}
		if (t0 >= 0) {
			server(mc, p -> {
				if (ben != null) {
					ben.tickCount++;
					Omnitrix.tick(ben);
				}
				var m = mover;
				if (m != null) {
					m.accept(p);
				}
				if (p.tickCount % 5 == 0) {
					for (var e : p.serverLevel().getEntitiesOfClass(net.minecraft.world.entity.monster.Slime.class, p.getBoundingBox().inflate(120))) {
						e.discard();
					}
				}
			});
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("WM_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.EASY, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}

	private static void camera(ServerPlayer p, Vec3 eye, Vec3 target) {
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		p.teleportTo(p.serverLevel(), eye.x, eye.y - p.getEyeHeight(), eye.z, yaw, pit);
	}

	private static void face(ServerPlayer f, Vec3 target) {
		Vec3 eye = f.getEyePosition();
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		f.teleportTo(f.serverLevel(), f.getX(), f.getY(), f.getZ(), yaw, pit);
		f.setYHeadRot(yaw);
		f.yBodyRot = yaw;
		f.yBodyRotO = yaw;
		f.swing(net.minecraft.world.InteractionHand.MAIN_HAND, true);
	}

	private static ServerPlayer fake(ServerPlayer near, String name, Vec3 at, float yaw) {
		ServerLevel level = near.serverLevel();
		GameProfile profile = new GameProfile(java.util.UUID.randomUUID(), name);
		CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
		ServerPlayer f = new ServerPlayer(level.getServer(), level, profile, cookie.clientInformation());
		Connection conn = new Connection(PacketFlow.SERVERBOUND);
		new io.netty.channel.embedded.EmbeddedChannel(conn);
		level.getServer().getPlayerList().placeNewPlayer(conn, f, cookie);
		f.setGameMode(GameType.SURVIVAL);
		f.teleportTo(level, at.x, at.y, at.z, yaw, 0f);
		f.setYHeadRot(yaw);
		f.yBodyRot = yaw;
		f.yBodyRotO = yaw;
		return f;
	}

	private static void gui(Minecraft mc, boolean on) {
		mc.execute(() -> mc.options.hideGui = !on);
	}

	private static void become(ServerPlayer f, String alien) {
		if (Omnitrix.state(f).transformed()) {
			Omnitrix.revert(f, Omnitrix.Reason.REVOKE);
		}
		var n = Omnitrix.state(f).copy();
		n.rechargeUntil = 0;
		n.selected = alien;
		Omnitrix.save(f, n);
		Omnitrix.transform(f);
		f.setOnGround(true);
	}


	private static Zombie dummy(ServerPlayer near, Vec3 at) {
		if (dummy != null) {
			dummy.discard();
		}
		ServerLevel l = near.serverLevel();
		Zombie z = EntityType.ZOMBIE.create(l);
		z.moveTo(at.x, at.y, at.z, 0f, 0f);
		z.setNoAi(true);
		z.setPersistenceRequired();
		z.getAttribute(net.minecraft.world.entity.ai.attributes.Attributes.MAX_HEALTH).setBaseValue(500);
		z.setHealth(500);
		z.setItemSlot(net.minecraft.world.entity.EquipmentSlot.HEAD, new net.minecraft.world.item.ItemStack(net.minecraft.world.item.Items.LEATHER_HELMET));
		l.addFreshEntity(z);
		dummy = z;
		return z;
	}

	private static void info(Minecraft mc, String scrollTo) {
		PowerInfoScreen s = new PowerInfoScreen();
		mc.setScreen(s);
		if (scrollTo != null && !s.scrollTo(scrollTo)) {
			System.out.println("[HARNESS] scrollTo miss: " + scrollTo);
		}
	}

	private static boolean want(String part) {
		return only.isEmpty() || only.contains(part);
	}

	private static void script(Minecraft mc) {
		int t = 10;
		at(t, () -> server(mc, p -> {
			p.teleportTo(p.serverLevel(), origin.x - 12, origin.y + 2, origin.z - 12, 200f, 10f);
			ben = fake(p, "Ben", origin, 180f);
			UniquePowers.forced(() -> Omnitrix.grant(ben));
			UniquePowers.forced(() -> Omnitrix.grant(p));
		}));
		t += 40;
		String[] ids = { "four_arms", "heatblast", "diamondhead" };
		int[][] moves = { { 13, 5, 10 }, { 2, 8, 1 }, { 4, 3, 1 } };
		for (int k = 0; k < ids.length; k++) {
			final String id = ids[k];
			if (!want(id)) {
				continue;
			}
			at(t, () -> server(mc, p -> become(ben, id)));
			// front, three-quarter, side, back
			double[][] cams = { { 0, 1.5, -2.6 }, { 1.9, 1.7, -1.9 }, { 2.6, 1.4, 0 }, { 0.6, 1.9, 2.6 } };
			String[] names = { "front", "three_quarter", "side", "back" };
			for (int c = 0; c < cams.length; c++) {
				final double[] cm = cams[c];
				at(t + 40 + c * 8, () -> server(mc, p -> camera(p, origin.add(cm[0], cm[1], cm[2]), origin.add(0, 1.15, 0))));
				shot(t + 46 + c * 8, id + "_" + names[c]);
			}
			int tt = t + 80;
			for (int m = 0; m < moves[k].length; m++) {
				final int anim = moves[k][m];
				at(tt, () -> server(mc, p -> {
					camera(p, origin.add(1.8, 1.6, -2.1), origin.add(0, 1.15, 0));
					Aliens.kit(id).anim(ben, anim);
				}));
				shot(tt + 5, id + "_move" + anim);
				tt += 30;
			}
			// first person on the real player
			at(tt, () -> server(mc, p -> {
				p.setGameMode(GameType.SURVIVAL);
				p.getAbilities().mayfly = true;
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				become(p, id);
				p.teleportTo(p.serverLevel(), origin.x + 6, origin.y + 1, origin.z, 90f, 20f);
			}));
			shot(tt + 40, id + "_fp");
			at(tt + 44, () -> server(mc, p -> {
				Omnitrix.revert(p, Omnitrix.Reason.REVOKE);
				p.setGameMode(GameType.CREATIVE);
			}));
			t = tt + 60;
		}
		if (want("timein")) {
			// raised while timed out: no dial; it times in while held -- red to green, then the dial pops into the alien screen
			gui(mc, true);
			at(t, () -> server(mc, p -> {
				p.teleportTo(p.serverLevel(), origin.x + 4, origin.y, origin.z + 4, 135f, 10f);
				var n = Omnitrix.state(p).copy();
				n.rechargeUntil = p.level().getGameTime() + 40;
				Omnitrix.save(p, n);
				Omnitrix.activate(p);
			}));
			for (int i = 0; i < 30; i++) {
				final int k = i;
				at(t + 2 + i * 3, () -> server(mc, p -> {
					var st = Omnitrix.state(p);
					long now = p.level().getGameTime();
					System.out.println("[HARNESS] timein " + k + " recharge=" + (st.rechargeUntil - now) + " selecting=" + st.selecting + " coreUp=" + Omnitrix.coreUp(st, now)
							+ " popAt=" + (Omnitrix.popAt(st) - now));
				}));
				shot(t + 2 + i * 3, "timein_" + String.format("%02d", i));
			}
			at(t + 95, () -> server(mc, Omnitrix::deactivate));
			t += 110;
		}
		if (want("badge")) {
			// every badge state: running, the warning flash, the five power-down beeps, the timeout, back to human
			at(t, () -> server(mc, p -> {
				become(ben, "four_arms");
				camera(p, origin.add(0, 1.4, -1.25), origin.add(0, 1.3, 0));
			}));
			at(t + 30, () -> server(mc, p -> {
				var n = Omnitrix.state(ben).copy();
				n.endsAt = ben.level().getGameTime() + 24;
				Omnitrix.save(ben, n);
			}));
			for (int i = 0; i < 44; i++) {
				final int k = i;
				at(t + 20 + i * 3, () -> server(mc, p -> {
					var st = Omnitrix.state(ben);
					System.out.println("[HARNESS] badge " + k + " transformed=" + st.transformed() + " left=" + (st.endsAt - ben.level().getGameTime()) + " powerDown="
							+ Omnitrix.poweringDown(st) + " anim=" + st.animId + " active=" + st.active);
				}));
				shot(t + 20 + i * 3, "badge_" + String.format("%02d", i));
			}
			t += 20 + 44 * 3 + 20;
		}
		at(t, () -> {
			System.out.println("[HARNESS] DONE");
			Runtime.getRuntime().halt(0);
		});
	}
}
