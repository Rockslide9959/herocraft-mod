/**
 * TEMP-HARNESS (not shipped): v0.16.1 Frost Giant real-model screenshots. Enabled by run/v0161fm.flag. Needs
 * V0161FrostCamMixin in projecthero.client.mixins.json and V0161FrostModelHarness.init() in onInitializeClient.
 * Shots go to run/screenshots/v0161fm_*.png.
 */
public final class V0161FrostModelHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	public static Vec3 camPos;
	public static Vec3 camAt;
	private static BlockPos C;
	private static volatile long lastProgress = System.currentTimeMillis();

	private V0161FrostModelHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("v0161fm.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(V0161FrostModelHarness::tick);
		Thread wd = new Thread(() -> {
			while (true) {
				try {
					Thread.sleep(5000);
				} catch (InterruptedException e) {
					return;
				}
				if (System.currentTimeMillis() - lastProgress > 120_000L) {
					System.out.println("[HARNESS] WATCHDOG: no progress for 120 s");
					Runtime.getRuntime().halt(3);
				}
			}
		}, "v0161fm-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static ServerPlayer sp(Minecraft mc) {
		MinecraftServer srv = mc.getSingleplayerServer();
		return srv == null || srv.getPlayerList().getPlayers().isEmpty() ? null : srv.getPlayerList().getPlayers().get(0);
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				ServerPlayer p = sp(mc);
				if (p != null) {
					try {
						fn.accept(p);
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		lastProgress = System.currentTimeMillis();
		Screenshot.grab(mc.gameDirectory, "v0161fm_" + name + ".png", mc.getMainRenderTarget(), c -> {
		});
		System.out.println("[HARNESS] shot " + name);
	}

	private static void at(int t, Runnable r) {
		Runnable prev = AT.get(t);
		AT.put(t, prev == null ? r : () -> {
			prev.run();
			r.run();
		});
	}

	private static void cam(double x, double y, double z, double ax, double ay, double az) {
		camPos = Vec3.atBottomCenterOf(C).add(x, y, z);
		camAt = Vec3.atBottomCenterOf(C).add(ax, ay, az);
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0) {
				lastProgress = System.currentTimeMillis();
				if (!createdWorld && mc.getOverlay() == null && mc.screen != null && !(mc.screen instanceof TitleScreen)) {
					mc.options.onboardAccessibility = false;
					mc.setScreen(new TitleScreen());
				}
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				lastProgress = System.currentTimeMillis();
				mc.execute(() -> {
					LevelSettings settings = new LevelSettings("V0161FM_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
							Difficulty.PEACEFUL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
					mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, WorldOptions.defaultWithRandomSeed(),
							reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
									.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
							mc.screen);
				});
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks % 20 == 0) {
			lastProgress = System.currentTimeMillis();
		}
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(60);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.getServer().setDifficulty(Difficulty.NORMAL, true);
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				C = p.blockPosition();
				p.addEffect(new MobEffectInstance(MobEffects.INVISIBILITY, 999999, 0, false, false));
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				lastProgress = System.currentTimeMillis();
				try {
					r.run();
				} catch (Throwable e) {
					System.out.println("[HARNESS] error " + e);
					e.printStackTrace();
				}
			}
		}
	}

	private static BlockPos ground(ServerLevel level, BlockPos at) {
		return new BlockPos(at.getX(), level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, at.getX(), at.getZ()), at.getZ());
	}

	private static <T extends Mob> T spawn(ServerLevel level, EntityType<T> type, BlockPos at, float yaw) {
		T m = type.create(level);
		BlockPos g = ground(level, at);
		m.moveTo(g.getX() + 0.5, g.getY(), g.getZ() + 0.5, yaw, 0f);
		m.finalizeSpawn(level, level.getCurrentDifficultyAt(g), MobSpawnType.COMMAND, null);
		m.setNoAi(true);
		m.setPersistenceRequired();
		level.addFreshEntity(m);
		m.setYRot(yaw);
		m.setYHeadRot(yaw);
		m.setYBodyRot(yaw);
		return m;
	}

	private static void clear(ServerPlayer p) {
		for (Entity e : p.serverLevel().getEntities((Entity) null, new AABB(C).inflate(80), e -> e instanceof FrostGiantEntity || e instanceof YmirArmEntity)) {
			e.discard();
		}
	}

	private static final Object[][] UNITS = {
			// type, height, clips to show at their hit tick (null = mid-clip)
			{ FrostGiantEntityTypes.HOUND, 1.6f, new Clip[] { FrostGiantAnims.HOUND_WALK, FrostGiantAnims.HOUND_POUNCE, FrostGiantAnims.HOUND_HOWL } },
			{ FrostGiantEntityTypes.RAIDER, 4.5f, new Clip[] { FrostGiantAnims.RAIDER_WALK, FrostGiantAnims.RAIDER_SPEAR_THRUST, FrostGiantAnims.RAIDER_WAR_CRY } },
			{ FrostGiantEntityTypes.SHAMAN, 5f, new Clip[] { FrostGiantAnims.SHAMAN_WALK, FrostGiantAnims.SHAMAN_ICE_WALL, FrostGiantAnims.SHAMAN_FROST_NOVA } },
			{ FrostGiantEntityTypes.BRUTE, 7f, new Clip[] { FrostGiantAnims.BRUTE_WALK, FrostGiantAnims.BRUTE_GLACIER_SPIKES, FrostGiantAnims.BRUTE_WAR_CRY } },
			{ FrostGiantEntityTypes.WARLORD, 11f, new Clip[] { FrostGiantAnims.WARLORD_WALK, FrostGiantAnims.WARLORD_CLEAVE, FrostGiantAnims.WARLORD_RALLY } },
			{ FrostGiantEntityTypes.LAUFEY, 14f, new Clip[] { FrostGiantAnims.LAUFEY_WALK, FrostGiantAnims.LAUFEY_CASKET_OPEN, FrostGiantAnims.LAUFEY_SPEAR_FAN } },
	};

	@SuppressWarnings("unchecked")
	private static void script(Minecraft mc) {
		int t = 20;
		// the lineup, idle, at true scale
		at(t, () -> server(mc, p -> {
			ServerLevel l = p.serverLevel();
			int x = -26;
			for (Object[] u : UNITS) {
				spawn(l, (EntityType<? extends Mob>) u[0], C.offset(x, 0, 0), 0f);
				x += (int) Math.max(4, (float) u[1] * 0.7f) + 2;
			}
			p.teleportTo(l, C.getX(), C.getY() + 6, C.getZ() + 40, 0f, 0f);
		}));
		at(t + 4, () -> cam(-6, 8, 44, -6, 6, 0));
		at(t + 60, () -> shot(mc, "00_lineup"));
		t += 70;
		at(t, () -> server(mc, V0161FrostModelHarness::clear));
		t += 10;
		int i = 0;
		for (Object[] u : UNITS) {
			final float h = (float) u[1];
			final EntityType<? extends Mob> type = (EntityType<? extends Mob>) u[0];
			final String name = String.format("%02d_%s", ++i, EntityType.getKey(type).getPath());
			final Mob[] m = new Mob[1];
			final double d = Math.max(4.0, h * 1.45);
			at(t, () -> server(mc, p -> {
				m[0] = spawn(p.serverLevel(), type, C, 0f);
				p.teleportTo(p.serverLevel(), C.getX(), C.getY() + h, C.getZ() + d, 0f, 0f);
			}));
			at(t + 4, () -> cam(0, h * 0.55, d, 0, h * 0.5, 0));
			at(t + 40, () -> shot(mc, name + "_a_front"));
			at(t + 42, () -> cam(d * 0.72, h * 0.62, d * 0.72, 0, h * 0.5, 0));
			at(t + 52, () -> shot(mc, name + "_b_threequarter"));
			at(t + 54, () -> cam(d, h * 0.55, 0.01, 0, h * 0.5, 0));
			at(t + 64, () -> shot(mc, name + "_c_side"));
			int tt = t + 66;
			for (Clip c : (Clip[]) u[2]) {
				final Clip clip = c;
				final int fire = c.loops() ? c.length() / 4 + 6 : Math.max(c.hit(), 6) + 1;
				at(tt, () -> cam(d * 0.72, h * 0.62, d * 0.72, 0, h * 0.5, 0));
				at(tt + 2, () -> server(mc, p -> ((FrostGiantEntity) m[0]).setAnim(clip)));
				at(tt + 2 + fire, () -> shot(mc, name + "_d_" + clip.name()));
				tt += fire + 30;
			}
			final int end = tt;
			at(end, () -> server(mc, V0161FrostModelHarness::clear));
			t = end + 10;
		}
		// Ymir's arm: the palm slam, seen from the side and from the front, with a raider for scale
		final YmirArmEntity[] arm = new YmirArmEntity[1];
		at(t, () -> server(mc, p -> {
			ServerLevel l = p.serverLevel();
			arm[0] = FrostGiantEntityTypes.YMIR_ARM.create(l);
			BlockPos g = ground(l, C);
			arm[0].moveTo(g.getX() + 0.5, g.getY(), g.getZ() + 0.5, 0f, 0f);
			l.addFreshEntity(arm[0]);
			arm[0].play(FrostGiantAnims.YMIR_IDLE);
			spawn(l, FrostGiantEntityTypes.RAIDER, C.offset(10, 0, 6), 300f);
			p.teleportTo(l, C.getX(), C.getY() + 20, C.getZ() + 60, 0f, 0f);
		}));
		at(t + 4, () -> cam(30, 18, 50, 0, 18, 0));
		at(t + 40, () -> shot(mc, "07_ymir_a_idle"));
		at(t + 42, () -> server(mc, p -> arm[0].play(FrostGiantAnims.YMIR_PALM_SLAM)));
		at(t + 42 + 26, () -> shot(mc, "07_ymir_b_palm_slam_raised"));
		at(t + 42 + 41, () -> shot(mc, "07_ymir_c_palm_slam_hit"));
		at(t + 100, () -> cam(55, 14, 2, 0, 16, 0));
		at(t + 102, () -> server(mc, p -> arm[0].play(FrostGiantAnims.YMIR_PALM_SLAM)));
		at(t + 102 + 41, () -> shot(mc, "07_ymir_d_palm_slam_side"));
		at(t + 160, () -> server(mc, p -> arm[0].play(FrostGiantAnims.YMIR_GLACIAL_SWEEP)));
		at(t + 160 + 62, () -> shot(mc, "07_ymir_e_glacial_sweep"));
		at(t + 250, () -> {
			System.out.println("[HARNESS] DONE");
			mc.stop();
		});
	}
}
