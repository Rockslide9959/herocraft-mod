package com.projecthero.mod.client.raidharness;

import java.util.ArrayList;
import java.util.List;

import com.projecthero.mod.event.entity.PillagerSpy;

import net.minecraft.core.component.DataComponents;
import net.minecraft.world.InteractionHand;
import net.minecraft.world.entity.EquipmentSlot;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.item.Items;
import net.minecraft.world.item.component.ChargedProjectiles;

/** TEMP-HARNESS: the moments shot for the raid mobs on their player-rig skins (v0.16.1). */
final class RaidHarnessActs {
	private RaidHarnessActs() {
	}

	private static RaidModelHarness.Act act(String label, java.util.function.BiConsumer<net.minecraft.server.level.ServerPlayer, net.minecraft.world.entity.Mob> fire, int after) {
		return new RaidModelHarness.Act(label, fire, after);
	}

	static List<RaidModelHarness.Act> actsFor(String name) {
		List<RaidModelHarness.Act> a = new ArrayList<>();
		if (name.startsWith("empowered")) {
			return a;
		}
		if (name.equals("pillager_spy")) {
			a.add(act("crossbow_charge", (p, m) -> ((PillagerSpy) m).setChargingCrossbow(true), 10));
			a.add(act("crossbow_aim", (p, m) -> {
				((PillagerSpy) m).setChargingCrossbow(false);
				ItemStack cb = new ItemStack(Items.CROSSBOW);
				cb.set(DataComponents.CHARGED_PROJECTILES, ChargedProjectiles.of(new ItemStack(Items.ARROW)));
				m.setItemSlot(EquipmentSlot.MAINHAND, cb);
				m.setAggressive(true);
			}, 8));
		} else {
			a.add(act("aggressive", (p, m) -> m.setAggressive(true), 6));
			a.add(act("attack", (p, m) -> m.swing(InteractionHand.MAIN_HAND), 3));
			a.add(act("attack_follow", (p, m) -> m.swing(InteractionHand.MAIN_HAND), 6));
		}
		a.add(act("hurt", (p, m) -> m.hurt(m.damageSources().generic(), 1.0f), 2));
		return a;
	}

	static List<RaidModelHarness.Unit> extra() {
		return new ArrayList<>();
	}
}
