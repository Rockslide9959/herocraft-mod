	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	static void at(int ticks, Runnable r) {
		while (PLAN.size() < ticks) {
			PLAN.add(() -> {
			});
		}
		Runnable prev = PLAN.get(ticks - 1);
		PLAN.set(ticks - 1, () -> {
			prev.run();
			r.run();
		});
	}

	static float[] yawPitch(Vec3 cam, Vec3 at) {
		Vec3 d = at.subtract(cam);
		float yaw = (float) Math.toDegrees(Math.atan2(-d.x, d.z));
		float pitch = (float) -Math.toDegrees(Math.atan2(d.y, Math.sqrt(d.x * d.x + d.z * d.z)));
		return new float[] { yaw, pitch };
	}

	/** Creative flying camera at {@code cam} looking at {@code at}. */
	static void look(Minecraft mc, Vec3 cam, Vec3 at, boolean gui) {
		mc.options.hideGui = !gui;
		mc.options.setCameraType(CameraType.FIRST_PERSON);
		float[] yp = yawPitch(cam, at);
		server(mc, p -> {
			p.setGameMode(GameType.CREATIVE);
			p.getAbilities().mayfly = true;
			p.getAbilities().flying = true;
			p.onUpdateAbilities();
			p.connection.teleport(cam.x, cam.y - 1.62, cam.z, yp[0], yp[1]);
		});
	}

	/** Stands the player on the ground at {@code xz} (y from the heightmap). */
	static Vec3 groundAt(ServerPlayer p, Vec3 xz) {
		int y = p.serverLevel().getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, (int) Math.floor(xz.x), (int) Math.floor(xz.z));
		return new Vec3(xz.x, y, xz.z);
	}

	/** A survival player standing at {@code feet} facing {@code at}. */
	static void stand(ServerPlayer p, Vec3 feet, Vec3 at) {
		feet = groundAt(p, feet);
		float[] yp = yawPitch(feet.add(0, 1.62, 0), at);
		p.setGameMode(GameType.SURVIVAL);
		p.getAbilities().flying = false;
		p.getAbilities().mayfly = false;
		p.onUpdateAbilities();
		p.connection.teleport(feet.x, feet.y, feet.z, yp[0], yp[1]);
	}

	static Vec3 c(double dx, double dy, double dz) {
		return new Vec3(cx + 0.5 + dx, ground + dy, cz + 0.5 + dz);
	}

	static MjolnirEntity hammer(ServerPlayer p) {
		var l = p.serverLevel().getEntitiesOfClass(MjolnirEntity.class, p.getBoundingBox().inflate(80),
				m -> m.getState() == MjolnirEntity.State.RESTING);
		return l.isEmpty() ? null : l.get(0);
	}

	static StormwardenBoss boss(ServerPlayer p) {
		var l = p.serverLevel().getEntitiesOfClass(StormwardenBoss.class, p.getBoundingBox().inflate(80));
		return l.isEmpty() ? null : l.get(0);
	}

	static void film(Minecraft mc, int start, String name, int frames, int every) {
		film(mc, start, name, frames, every, null);
	}

	/** Frames every {@code every} ticks; {@code cam} (if set) is forced on every frame. */
	static void film(Minecraft mc, int start, String name, int frames, int every, CameraType cam) {
		for (int i = 0; i < frames; i++) {
			final int n = i;
			if (cam != null) {
				at(Math.max(1, start + i * every - 1), () -> mc.options.setCameraType(cam));
			}
			at(start + i * every, () -> shot(mc, String.format("film_%s_%03d", name, n)));
		}
	}

	private static int[] findSpot(ServerLevel lvl, int x0, int z0) {
		var gen = lvl.getChunkSource().getGenerator();
		var rs = lvl.getChunkSource().randomState();
		for (int dz = 0; dz < 3000; dz += 48) {
			for (int x = x0; x < x0 + 400; x += 16) {
				int z = z0 + dz;
				int s = gen.getFirstOccupiedHeight(x, z, Heightmap.Types.WORLD_SURFACE_WG, lvl, rs);
				if (s != gen.getFirstOccupiedHeight(x, z, Heightmap.Types.OCEAN_FLOOR_WG, lvl, rs) || s < 64) {
					continue;
				}
				var biome = gen.getBiomeSource().getNoiseBiome(x >> 2, s >> 2, z >> 2, rs.sampler());
				if (!(biome.is(net.minecraft.world.level.biome.Biomes.PLAINS) || biome.is(net.minecraft.world.level.biome.Biomes.MEADOW)
						|| biome.is(net.minecraft.world.level.biome.Biomes.SAVANNA) || biome.is(net.minecraft.world.level.biome.Biomes.SUNFLOWER_PLAINS))) {
					continue;
				}
				int min = s, max = s;
				for (int[] o : new int[][] { { -14, -14 }, { 14, -14 }, { -14, 14 }, { 14, 14 }, { 0, 14 }, { 14, 0 }, { -14, 0 }, { 0, -14 }, { 7, 7 }, { -7, -7 }, { 7, -7 }, { -7, 7 } }) {
					int y = gen.getFirstOccupiedHeight(x + o[0], z + o[1], Heightmap.Types.OCEAN_FLOOR_WG, lvl, rs);
					min = Math.min(min, y);
					max = Math.max(max, y);
				}
				if (max - min <= 1) {
					return new int[] { x, z };
				}
			}
		}
		return new int[] { x0, z0 };
	}

	private static void clearTrees(ServerLevel level, int x, int y, int z, int r) {
		for (BlockPos q : BlockPos.betweenClosed(x - r, y - 3, z - r, x + r, y + 26, z + r)) {
			var s = level.getBlockState(q);
			if (s.is(net.minecraft.tags.BlockTags.LEAVES) || s.is(net.minecraft.tags.BlockTags.LOGS) || s.is(Blocks.BAMBOO)
					|| s.is(Blocks.TALL_GRASS) || s.is(Blocks.SHORT_GRASS)) {
				level.setBlock(q, Blocks.AIR.defaultBlockState(), 2);
			}
		}
	}

			for (int x = cx - 22; x <= cx + 22; x++) {
				for (int z = cz - 22; z <= cz + 22; z++) {
					for (int y = ground - 12; y <= ground + 16; y++) {
						BlockPos q = new BlockPos(x, y, z);
						if (y >= ground) {
							level.setBlock(q, Blocks.AIR.defaultBlockState(), 2);
						} else if (y == ground - 1) {
							level.setBlock(q, Blocks.GRASS_BLOCK.defaultBlockState(), 2);
						} else if (y >= ground - 4) {
							level.setBlock(q, Blocks.DIRT.defaultBlockState(), 2);
						} else if (level.getBlockState(q).isAir() || !level.getFluidState(q).isEmpty()) {
							level.setBlock(q, Blocks.STONE.defaultBlockState(), 2);
						}
					}
				}
			}

			spot = MjolnirCraterPiece.buildOnTerrain(level, cx, cz);

				}

				for (int dz = -1; dz <= 1; dz++) {
					if ((dx != 0 || dz != 0) && n++ < 5) {
						level.setBlockAndUpdate(forge[0].offset(dx, 0, dz), (dx + dz) % 2 == 0 ? Blocks.GILDED_BLACKSTONE.defaultBlockState()
								: Blocks.POLISHED_BLACKSTONE_BRICKS.defaultBlockState());
					}
				}
			}
			for (var m : level.getEntitiesOfClass(net.minecraft.world.entity.Mob.class, p.getBoundingBox().inflate(60), m -> m instanceof net.minecraft.world.entity.monster.Enemy)) {
				m.discard();
			}
			p.getInventory().clearContent();
			p.getAbilities().invulnerable = false;
			p.onUpdateAbilities();
			p.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 20 * 120, 2, false, false));
			p.addEffect(new MobEffectInstance(MobEffects.REGENERATION, 20 * 120, 3, false, false));
		}));
		at(k + 4, () -> server(mc, p -> {
			mc.options.hideGui = false;
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			stand(p, Vec3.atBottomCenterOf(forge[0]).add(0, 0, 2.6), Vec3.atCenterOf(forge[0]));
		}));
		at(k + 24, () -> server(mc, p -> {
			p.setItemInHand(InteractionHand.MAIN_HAND, new ItemStack(ModItems.UNFORGED_STORMBREAKER));
			StarForge.use(p.serverLevel(), forge[0], (StarForgeBlockEntity) p.serverLevel().getBlockEntity(forge[0]), p, p.getMainHandItem());
		}));
		at(k + 28, () -> shot(mc, "10_forge_ring_incomplete"));
		at(k + 32, () -> server(mc, p -> {
			ServerLevel level = p.serverLevel();
			for (BlockPos m : StarForge.missingRing(level, forge[0])) {
				level.setBlockAndUpdate(m, Blocks.POLISHED_BLACKSTONE_BRICKS.defaultBlockState());
			}
		}));
		at(k + 36, () -> look(mc, Vec3.atCenterOf(forge[0]).add(3.6, 2.6, 3.6), Vec3.atCenterOf(forge[0]), false));
		at(k + 60, () -> shot(mc, "11_forge_multiblock"));
		k += 66;
		// ---- film: the channel (the hero is Thor now, Mjolnir in hand)
		at(k, () -> server(mc, p -> {
			mc.options.hideGui = true;
			p.getInventory().clearContent();
			ItemStack head = new ItemStack(ModItems.UNFORGED_STORMBREAKER);
			StarForgeBlockEntity be = (StarForgeBlockEntity) p.serverLevel().getBlockEntity(forge[0]);
			StarForge.use(p.serverLevel(), forge[0], be, p, head);
			ItemStack mj = new ItemStack(ModItems.MJOLNIR);
			p.setItemInHand(InteractionHand.MAIN_HAND, mj);
			com.projecthero.mod.worthiness.Worthiness.setScore(p, com.projecthero.mod.worthiness.Worthiness.TEST_WORTHY_SCORE);
			stand(p, Vec3.atBottomCenterOf(forge[0]).add(0.4, 0, 2.3), Vec3.atCenterOf(forge[0]).add(0, 0.4, 0));
		}));
		at(k + 2, () -> mc.options.setCameraType(CameraType.THIRD_PERSON_BACK));
		k += 12;
		for (int i = 0; i <= StarForge.CHANNEL_TICKS + 8; i += 4) {
			at(k + i, () -> server(mc, p -> {
				StarForgeBlockEntity be = (StarForgeBlockEntity) p.serverLevel().getBlockEntity(forge[0]);
				p.swing(InteractionHand.MAIN_HAND);
				StarForge.ping(p.serverLevel(), forge[0], be, p);
			}));
		}
		film(mc, k, "channel", 110, 2, CameraType.FIRST_PERSON);
		k += StarForge.CHANNEL_TICKS + 30;
		// ---- film: the Living Branch finishes Stormbreaker
		at(k, () -> server(mc, p -> {
			StarForgeBlockEntity be = (StarForgeBlockEntity) p.serverLevel().getBlockEntity(forge[0]);
			if (!StarForge.isForged(be.head())) {
				StarForge.insertForTest(be, StarForge.forgedHead());
			}
			stand(p, Vec3.atBottomCenterOf(forge[0]).add(0.6, 0, 3.6), Vec3.atCenterOf(forge[0]).add(0, 1.2, 0));
		}));
		at(k + 10, () -> server(mc, p -> {
			StarForgeBlockEntity be = (StarForgeBlockEntity) p.serverLevel().getBlockEntity(forge[0]);
			StarForge.use(p.serverLevel(), forge[0], be, p, new ItemStack(ThorAcquire.LIVING_BRANCH));
		}));
		film(mc, k + 8, "finish", 90, 1, CameraType.FIRST_PERSON);
		k += 110;
		at(k, () -> System.out.println("[HARNESS] done"));
	}

	private static void openGuide(Minecraft mc, String heading) {
		try {
			var screen = new com.projecthero.mod.client.gui.HeroPackGuideScreen();
			mc.setScreen(screen);
			int ch = HeroPackGuide.chapters().indexOf(HeroPackGuide.thorChapter());
			var lines = HeroPackGuide.chapterAt(ch).lines();
			int line = 0;
			for (int i = 0; i < lines.size(); i++) {
				if (lines.get(i).getString().trim().equals(heading)) {
					line = i;
					break;
				}
			}
			var m = com.projecthero.mod.client.gui.HeroPackGuideScreen.class.getDeclaredMethod("openChapter", int.class, int.class);
			m.setAccessible(true);
			m.invoke(screen, ch, line);
		} catch (Throwable e) {
			System.out.println("[HARNESS] guide error " + e);
		}
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.getWindow().setWindowed(960, 540);
				mc.execute(ThorAcquireHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 30) {
			mc.options.fov().set(75);
			mc.options.tutorialStep = net.minecraft.client.tutorial.TutorialSteps.NONE;
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.renderDistance().set(10);
			mc.options.entityDistanceScaling().set(2.0);
			if (mc.getSingleplayerServer() != null) {
				mc.getSingleplayerServer().getPlayerList().setViewDistance(10);
			}
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(5000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.setWeatherParameters(24000, 0, false, false);
				p.setGameMode(GameType.CREATIVE);
				p.addEffect(new MobEffectInstance(MobEffects.NIGHT_VISION, 20 * 60 * 30, 0, false, false));
			});
			buildPlan(mc);
		}
		if (t < 50) {
			return;
		}
		if (waiting) {
			return;
		}
		if (planTick < PLAN.size()) {
			PLAN.get(planTick++).run();
		} else if (!PLAN.isEmpty()) {
			Runtime.getRuntime().halt(0);
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		LevelSettings settings = new LevelSettings("THOR_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.NORMAL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(5175L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.NORMAL).value().createWorldDimensions(),
				mc.screen);
	}
