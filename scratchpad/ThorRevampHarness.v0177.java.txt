	@Override
	public void onInitializeClient() {
		Path flag = Path.of("thorrevamp.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			String[] parts = Files.readString(flag).trim().split("\\|");
			out = new File(parts[0]);
			if (parts.length > 1) {
				mode = parts[1];
			}
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(ThorRevampHarness::tick);
		ClientTickEvents.START_CLIENT_TICK.register(mc -> placeCam(mc));
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(1_500_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "thorrevamp-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.computeIfAbsent(t, k -> new ArrayList<>()).add(r);

	private static void cam3(float angle, float dist, float h, float lookAhead, float lookH) {
		camAngle = angle;
		camDist = dist;
		camH = h;
		camLookAhead = lookAhead;
		camLookH = lookH;
		camOn = true;
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = true;
		placeCam(mc);
	}

	private static void fp() {
		camOn = false;
		camPose = null;
		Minecraft mc = Minecraft.getInstance();
		mc.options.hideGui = false;
		mc.options.setCameraType(CameraType.FIRST_PERSON);
		if (mc.player != null) {
			mc.setCameraEntity(mc.player);
		}
	}

	public static volatile double[] camPose;

	private static void placeCam(Minecraft mc) {
		if (!camOn || mc.player == null || mc.level == null || base == null) {
			camPose = null;
			return;
		}
		Vec3 c = new Vec3(base.getX() + 0.5, base.getY(), base.getZ() + 0.5);
		double r = Math.toRadians(camAngle);
		Vec3 eye = c.add(Math.sin(r) * camDist, camH, Math.cos(r) * camDist);
		Vec3 look = c.add(0, camLookH, camLookAhead);
		Vec3 d = look.subtract(eye);
		float yaw = (float) (Math.toDegrees(Math.atan2(-d.x, d.z)));
		float pitch = (float) (-Math.toDegrees(Math.atan2(d.y, d.horizontalDistance())));
		camPose = new double[] { eye.x, eye.y, eye.z, yaw, pitch };
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
	}

	private static float pitch = 6f;

	private static void face(Minecraft mc) {
		var p = mc.player;
		p.setYRot(0f);
		p.yRotO = 0f;
		p.setXRot(pitch);
		p.xRotO = pitch;
		p.yHeadRot = 0f;
		p.yHeadRotO = 0f;
	}

	private static void stage(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		level.setDayTime(6000L);
		level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
		level.setWeatherParameters(6000, 0, false, false);
		p.server.setDifficulty(Difficulty.EASY, true);
		p.setGameMode(GameType.SURVIVAL);
		base = p.blockPosition();
		for (int x = -20; x <= 20; x++) {
			for (int z = -12; z <= 34; z++) {
				level.setBlock(base.offset(x, -1, z), (x + z) % 2 == 0 ? Blocks.SMOOTH_STONE.defaultBlockState() : Blocks.POLISHED_ANDESITE.defaultBlockState(), 3);
			}
		}
		p.getInventory().clearContent();
		System.out.println("[HARNESS] grant: " + com.projecthero.mod.hero.PowerGrants.grantHero(p, "thor").message().getString());
		ItemStack hammer = new ItemStack(ModItems.MJOLNIR);
		com.projecthero.mod.power.ThorPowers.bindOnAscend(p, hammer);
		com.projecthero.mod.power.ThorPassives.reconcile(p);
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, pitch);
	}

	/** Back to the start: position, health, no cooldowns, full Storm Energy, hands as given, every old mob / thrown weapon gone. */
	private static void reset(ServerPlayer p, Item main, Item off) {
		ServerLevel level = p.serverLevel();
		for (Mob m : level.getEntitiesOfClass(Mob.class, new AABB(base).inflate(60))) {
			m.discard();
		}
		for (Entity e : level.getEntitiesOfClass(Entity.class, new AABB(base).inflate(80),
				e -> e instanceof com.projecthero.mod.entity.MjolnirEntity || e instanceof com.projecthero.mod.stormbreaker.StormbreakerEntity
						|| e instanceof net.minecraft.world.entity.item.ItemEntity)) {
			e.discard();
		}
		p.removeAllEffects();
		p.setHealth(p.getMaxHealth());
		p.setAttached(com.projecthero.mod.attachment.ModAttachments.COOLDOWNS, new com.projecthero.mod.attachment.CooldownState());
		p.setAttached(com.projecthero.mod.attachment.ModAttachments.STORM_CALL_STATE, new com.projecthero.mod.attachment.StormCallState());
		com.projecthero.mod.power.StormEnergy.set(p, com.projecthero.mod.power.StormEnergy.MAX);
		p.getInventory().clearContent();
		p.getInventory().selected = 0;
		p.getInventory().setItem(0, main == null ? ItemStack.EMPTY : new ItemStack(main));
		p.setItemInHand(InteractionHand.OFF_HAND, off == null ? ItemStack.EMPTY : new ItemStack(off));
		p.setDeltaMovement(Vec3.ZERO);
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, pitch);
	}

	private static Husk husk(ServerPlayer p, double x, double z) {
		ServerLevel level = p.serverLevel();
		Husk h = new Husk(EntityType.HUSK, level);
		h.setNoAi(true);
		h.getAttribute(Attributes.MAX_HEALTH).setBaseValue(1000);
		h.setHealth(1000);
		h.setPersistenceRequired();
		h.moveTo(base.getX() + 0.5 + x, base.getY(), base.getZ() + 0.5 + z, 180f, 0f);
		h.setYHeadRot(180f);
		level.addFreshEntity(h);
		return h;
	}

	private static void crowd(ServerPlayer p) {
		husk(p, 0.0, 3.5);
		husk(p, -2.2, 5.0);

		husk(p, -3.5, 8.5);
		husk(p, 3.2, 9.5);

	}

	}

	private static void film(int from, int to, String dir) {
		at(from, () -> {
			film0 = from;
			filmEnd = to;
			filmDir = dir;
		});
	}

	/** One move in a state film: reset, crowd, fire {slot, shift} (and release after {@code holdTicks} if > 0), with a still. */
	private static int move(Minecraft mc, int t, Item main, Item off, int slot, boolean shift, int holdTicks, int len, String still) {
		at(t, () -> server(mc, p -> {
			reset(p, main, off);
			crowd(p);
		}));
		at(t + 6, () -> server(mc, p -> press(p, slot, shift)));
		if (holdTicks > 0) {
			at(t + 6 + holdTicks, () -> server(mc, p -> release(p, slot)));
		} else {
			at(t + 7, () -> server(mc, p -> release(p, slot)));
		}
		if (still != null) {
			at(t + 6 + Math.max(4, Math.min(len - 8, holdTicks > 0 ? holdTicks - 6 : 8)), () -> shot(mc, still));
		}
		return t + len;
	}

	/** One melee swing (the combo) at the nearest husk in front, with the attack ticker reset so it lands at full strength. */
	private static void swingAt(ServerPlayer p) {
		var hs = p.serverLevel().getEntitiesOfClass(Husk.class, new AABB(p.blockPosition()).inflate(4));
		if (!hs.isEmpty()) {
			p.swing(InteractionHand.MAIN_HAND, true);
			p.attack(hs.get(0));
			p.resetAttackStrengthTicker();
		}
	}

	/** mode "axe": close looks at every Stormbreaker / dual pose with the 1.5-block greataxe (side, front, first person). */
	private static void planAxe() {
		Minecraft mc = Minecraft.getInstance();
		final Item MJ = ModItems.MJOLNIR, SB = ModItems.STORMBREAKER;
		int t = 0;
		final int R = 1, G = 2, X = 3, Z = 4, C = 6;
		float[][] cams = { { 90f, 4.2f, 1.5f, 0.6f, 1.2f }, { 200f, 4.4f, 1.6f, 0.5f, 1.3f } };
		String[] camName = { "side", "front" };
		for (int ci = 0; ci < 2; ci++) {
			final float[] c = cams[ci];
			final String cn = camName[ci];
			for (int w = 0; w < 2; w++) {
				final Item off = w == 0 ? null : MJ;
				final String wn = w == 0 ? "stormbreaker" : "twin";
				at(t, () -> cam3(c[0], c[1], c[2], c[3], c[4]));
				int s0 = t;
				at(t, () -> server(mc, p -> {
					reset(p, SB, off);
					husk(p, 0.0, 1.8);
				}));
				for (int k = 0; k < (w == 0 ? 3 : 4); k++) {
					at(t + 8 + k * 30, () -> server(mc, ThorRevampHarness::swingAt));
				}
				t += 8 + (w == 0 ? 3 : 4) * 30 + 10;
				film(s0 + 6, t, "film_axe_combo_" + wn + "_" + cn);
				int s1 = t;
				if (w == 0) {
					t = move(mc, t, SB, null, G, false, 0, 30, null);
					t = move(mc, t, SB, null, C, false, 0, 26, null);
					t = move(mc, t, SB, null, R, true, 0, 22, null);
					t = move(mc, t, SB, null, Z, false, 0, 50, null);
				} else {
					t = move(mc, t, SB, MJ, G, false, 0, 32, null);
					t = move(mc, t, SB, MJ, R, true, 0, 26, null);
					t = move(mc, t, SB, MJ, C, true, 24, 30, null);
					t = move(mc, t, SB, MJ, Z, false, 0, 40, null);
				}
				film(s1 + 4, t - 2, "film_axe_moves_" + wn + "_" + cn);
			}
		}
		at(t, () -> fp());
		int f0 = t;
		at(t, () -> server(mc, p -> {
			reset(p, SB, null);
			husk(p, 0.0, 2.2);
		}));
		for (int k = 0; k < 3; k++) {
			at(t + 8 + k * 30, () -> server(mc, ThorRevampHarness::swingAt));
		}
		t += 8 + 90 + 6;
		t = move(mc, t, SB, null, G, false, 0, 26, null);
		t = move(mc, t, SB, null, C, false, 0, 24, null);
		film(f0 + 4, t - 2, "film_axe_fp");
		final int end = t + 4;
		at(end, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

		film(u2 + 4, t - 2, "film_ult_bifrost_judgement");
		int u3 = t;
		at(t, () -> cam3(55f, 14.0f, 5.5f, 6.0f, 3.0f));
		t = move(mc, t, MJ, SB, Z, true, 0, 110, "ult_ragnarok_rise");
		at(u3 + 6 + 40, () -> shot(mc, "ult_ragnarok_rain"));
		at(u3 + 6 + 73, () -> shot(mc, "ult_ragnarok_impact"));

			Runtime.getRuntime().halt(0);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(ThorRevampHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 20) {
			GLFW.glfwSetWindowSize(mc.getWindow().getWindow(), 854, 480);
		}
		if (t == 40) {
			mc.options.fov().set(70);
			mc.options.fovEffectScale().set(0.0);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			mc.getToasts().clear();
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, ThorRevampHarness::stage);
		}
		if (t < 140 || base == null) {
			return;
		}
		face(mc);
		int s = t - 140;
		if (s == 0) {
			plan();
		}
		if (s % 10 == 0) {
			mc.gui.getChat().clearMessages(false);
		}
		List<Runnable> todo = AT.remove(s);
		if (todo != null) {
			todo.forEach(Runnable::run);
		}
		if (s >= film0 && s < filmEnd) {
			shot(mc, String.format(filmDir + "/f%03d", s - film0));
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(GameRules.RULE_DOMOBSPAWNING).set(false, null);
		LevelSettings settings = new LevelSettings("THORREVAMP_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.EASY, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
