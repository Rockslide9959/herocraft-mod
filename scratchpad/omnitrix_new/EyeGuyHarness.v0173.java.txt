package com.projecthero.mod.client;

import java.nio.file.Files;
import java.nio.file.Path;

import com.projecthero.mod.omnitrix.Omnitrix;
import com.projecthero.mod.omnitrix.alien.Aliens;
import com.projecthero.mod.omnitrix.alien.eyeguy.EyeGuyKit;

import net.fabricmc.fabric.api.client.event.lifecycle.v1.ClientTickEvents;
import net.minecraft.client.CameraType;
import net.minecraft.client.Minecraft;
import net.minecraft.client.Screenshot;
import net.minecraft.client.gui.screens.TitleScreen;
import net.minecraft.core.BlockPos;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.Difficulty;
import net.minecraft.world.entity.EntityType;
import net.minecraft.world.entity.LivingEntity;
import net.minecraft.world.entity.Mob;
import net.minecraft.world.level.GameRules;
import net.minecraft.world.level.GameType;
import net.minecraft.world.level.LevelSettings;
import net.minecraft.world.level.WorldDataConfiguration;
import net.minecraft.world.level.block.Blocks;
import net.minecraft.world.level.levelgen.WorldOptions;
import net.minecraft.world.level.levelgen.presets.WorldPresets;
import net.minecraft.world.phys.Vec3;

/** TEMP-HARNESS (not shipped): Eye Guy -- a turntable and every move, one frame per tick. Enabled by run/eyeguy.flag. */
public final class EyeGuyHarness {
	/** The camera (EyeGuyHarnessCameraMixin): offset from the player, looking at the player + camTarget. */
	public static volatile Vec3 camOffset, camTarget = new Vec3(0, 1.0, 0);

	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static int frame;
	private static String seg;
	private static Vec3 home;
	private static Mob golem;
	private static final Mob[] zombies = new Mob[3];

	private EyeGuyHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("eyeguy.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(EyeGuyHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(300_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "eyeguy-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static EyeGuyKit kit() {
		return (EyeGuyKit) Aliens.kit(EyeGuyKit.ID);
	}

	private static Mob mob(ServerPlayer p, EntityType<? extends Mob> type, double dx, double dz) {
		Mob m = type.create(p.serverLevel());
		m.moveTo(home.x + dx, home.y, home.z + dz, 180f, 0f);
		m.setNoAi(true);
		m.setPersistenceRequired();
		p.serverLevel().addFreshEntity(m);
		return m;
	}

	private static void face(ServerPlayer p, Vec3 at) {
		Vec3 d = at.subtract(p.getEyePosition());
		float yaw = (float) (Math.atan2(-d.x, d.z) * 180 / Math.PI);
		float pitch = (float) (-Math.atan2(d.y, Math.sqrt(d.x * d.x + d.z * d.z)) * 180 / Math.PI);
		p.connection.teleport(p.getX(), p.getY(), p.getZ(), yaw, pitch);
		p.setYRot(yaw);
		p.setXRot(pitch);
		p.setYHeadRot(yaw);
	}

	private static Vec3 chest(LivingEntity e) {
		return e.position().add(0, e.getBbHeight() * 0.6, 0);
	}

	private static void ready(ServerPlayer p) {
		for (var m : kit().moves()) {
			kit().cooldown(p, m.id(), 0);
		}
		if (golem != null) {
			golem.setHealth(golem.getMaxHealth());
			golem.clearFire();
			golem.setTicksFrozen(0);
			golem.removeAllEffects();
		}
	}

	private static void cam(double x, double y, double z) {
		camTarget = new Vec3(0, 1.0, 0);
		camOffset = new Vec3(x, y, z);
	}

	/** Starts a filmed segment: frames eye_guy_<name>_NNN.png until the next segment (or null). */
	private static void segment(String name) {
		seg = name;
		frame = 0;
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(EyeGuyHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (seg != null) {
			Screenshot.grab(mc.gameDirectory, String.format("eye_guy_%s_%03d.png", seg, frame++), mc.getMainRenderTarget(), m -> {
			});
		}
		if (t == 60) {
			mc.getWindow().setWindowed(854, 480);
		}
		if (t == 80) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.bobView().set(false);
			server(mc, p -> {
				p.serverLevel().setDayTime(6000L);
				p.serverLevel().getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				p.serverLevel().getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.serverLevel().getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				p.setGameMode(GameType.SURVIVAL);
				Omnitrix.grant(p);
				Omnitrix.noGlitch(p);
				Aliens.discover(p, EyeGuyKit.ID, true);
			});
		}
		if (t == 100) {
			server(mc, p -> Omnitrix.activate(p));
		}
		if (t == 115) {
			server(mc, p -> Omnitrix.select(p, EyeGuyKit.ID));
		}
		if (t == 120) {
			server(mc, p -> {
				home = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
				p.connection.teleport(home.x, home.y, home.z, 0f, 0f);
				Omnitrix.select(p, EyeGuyKit.ID);
				System.out.println("[HARNESS] transform=" + Omnitrix.transform(p));
			});
		}
		// ---- the turntable (he faces +z): one full orbit, 120 frames
		if (t >= 140 && t < 260) {
			if (t == 140) {
				segment("turntable");
			}
			double a = (t - 140) / 120.0 * Math.PI * 2;
			camTarget = new Vec3(0, 1.0, 0);
			camOffset = new Vec3(Math.sin(a) * 2.2, 1.3, Math.cos(a) * 2.2);
		}
		// ---- a close look at the badge and the chest eye
		if (t == 260) {
			segment("closeup");
			camTarget = new Vec3(0, 1.1, 0);
			camOffset = new Vec3(0.5, 1.2, 1.3);
		}
		// ---- the combat stage: a golem 7 ahead
		if (t == 262) {
			server(mc, p -> {
				p.connection.teleport(home.x, home.y, home.z, 0f, 0f);
				golem = mob(p, EntityType.IRON_GOLEM, 0, 7);
			});
		}
		// R: Eye Laser (twice)
		if (t == 290) {
			segment("laser");
			cam(4.6, 2.0, 3.2);
			camTarget = new Vec3(0, 1.1, 3.6);
		}
		if (t == 298 || t == 312) {
			server(mc, p -> {
				ready(p);
				face(p, chest(golem));
				System.out.println("[HARNESS] laser=" + kit().laser(p));
			});
		}
		// Shift+R: Laser Volley at three zombies
		if (t == 330) {
			segment("volley");
			cam(3.4, 2.2, -0.8);
			camTarget = new Vec3(0, 1.0, 3.0);
			server(mc, p -> {
				golem.discard();
				golem = null;
				zombies[0] = mob(p, EntityType.ZOMBIE, -2.5, 6);
				zombies[1] = mob(p, EntityType.ZOMBIE, 0, 7);
				zombies[2] = mob(p, EntityType.ZOMBIE, 2.5, 6);
				for (Mob z : zombies) {
					z.setItemSlot(net.minecraft.world.entity.EquipmentSlot.HEAD, new net.minecraft.world.item.ItemStack(net.minecraft.world.item.Items.CARVED_PUMPKIN));
				}
			});
		}
		if (t == 340) {
			server(mc, p -> {
				ready(p);
				for (Mob z : zombies) {
					z.setHealth(z.getMaxHealth());
				}
				face(p, home.add(0, 1.5, 7));
				System.out.println("[HARNESS] volley=" + kit().volley(p));
			});
		}
		// G: Freeze Beam
		if (t == 370) {
			segment("freeze");
			cam(4.6, 2.0, 3.2);
			camTarget = new Vec3(0, 1.1, 3.6);
			server(mc, p -> {
				for (Mob z : zombies) {
					z.discard();
				}
				golem = mob(p, EntityType.IRON_GOLEM, 0, 7);
			});
		}
		if (t == 378) {
			server(mc, p -> {
				ready(p);
				face(p, chest(golem));
				System.out.println("[HARNESS] freeze=" + kit().freezeBeam(p));
			});
		}
		// Shift+G: Fire Beam
		if (t == 412) {
			segment("fire");
		}
		if (t == 418) {
			server(mc, p -> {
				ready(p);
				face(p, chest(golem));
				System.out.println("[HARNESS] fire=" + kit().fireBeam(p));
			});
		}
		// Z: Chest Eye Beam (charge + fire)
		if (t == 452) {
			segment("chest_beam");
			cam(5.2, 2.2, 3.4);
			camTarget = new Vec3(0, 1.1, 3.6);
			server(mc, p -> System.out.println("[HARNESS] golems=" + p.serverLevel().getEntitiesOfClass(net.minecraft.world.entity.animal.IronGolem.class, p.getBoundingBox().inflate(30)).stream().map(g -> g.blockPosition().toShortString()).toList()));
		}
		if (t == 458) {
			server(mc, p -> {
				ready(p);
				face(p, chest(golem));
				p.setXRot(0f);
				p.connection.teleport(p.getX(), p.getY(), p.getZ(), p.getYRot(), 0f);
				System.out.println("[HARNESS] chest=" + kit().chestBeam(p));
			});
		}
		// C: Eye Flash (from the front)
		if (t == 520) {
			segment("flash");
			cam(-1.8, 1.5, 3.0);
			camTarget = new Vec3(0, 1.1, 0.5);
			server(mc, p -> {
				ready(p);
				golem.moveTo(home.x + 1.5, home.y, home.z + 3.5, 180f, 0f);
			});
		}
		if (t == 528) {
			server(mc, p -> System.out.println("[HARNESS] flash=" + kit().flash(p)));
		}
		// V: All-Seeing -- a stone wall between him and the golem; it shows through
		if (t == 560) {
			server(mc, p -> {
				ready(p);
				golem.moveTo(home.x, home.y, home.z + 7, 180f, 0f);
				for (int x = -3; x <= 3; x++) {
					for (int y = 0; y <= 3; y++) {
						p.serverLevel().setBlockAndUpdate(BlockPos.containing(home.x + x, home.y + y, home.z + 4), Blocks.STONE.defaultBlockState());
					}
				}
				face(p, home.add(0, 1.6, 7));
			});
		}
		if (t == 566) {
			segment("all_seeing");
			cam(0.9, 1.9, -2.6);
			camTarget = new Vec3(0, 1.4, 3.0);
		}
		if (t == 574) {
			server(mc, p -> System.out.println("[HARNESS] sight=" + kit().allSeeing(p)));
		}
		// X: Eye Dodge (standing: straight back), from the side
		if (t == 620) {
			segment("dodge");
			cam(3.0, 1.3, 0.4);
			camTarget = new Vec3(0, 0.9, -1.0);
			server(mc, p -> {
				ready(p);
				for (int x = -3; x <= 3; x++) {
					for (int y = 0; y <= 3; y++) {
						p.serverLevel().setBlockAndUpdate(BlockPos.containing(home.x + x, home.y + y, home.z + 4), Blocks.AIR.defaultBlockState());
					}
				}
			});
		}
		if (t == 628) {
			server(mc, p -> System.out.println("[HARNESS] dodge=" + kit().dodge(p)));
		}
		if (t == 652) {
			server(mc, p -> {
				ready(p);
				System.out.println("[HARNESS] dodge2=" + kit().dodge(p));
			});
		}
		if (t == 680) {
			seg = null;
			System.out.println("[HARNESS] done");
		}
		if (t == 690) {
			Runtime.getRuntime().halt(0);
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("EG_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.EASY, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}
}
