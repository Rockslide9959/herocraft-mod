package com.projecthero.mod.client;

import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;
import java.util.function.Consumer;

import com.projecthero.mod.omnitrix.Omnitrix;
import com.projecthero.mod.omnitrix.alien.Aliens;
import com.projecthero.mod.omnitrix.alien.frankenstrike.FrankenstrikeKit;

/** TEMP-HARNESS (not shipped): Frankenstrike -- a turntable, a bolt close-up and one clip per move. Enabled by run/frankenstrike.flag. */
public final class FrankenstrikeHarness {
	/** The camera (FrankenstrikeHarnessCameraMixin): offset from the player, looking at the player + camTarget. */
	public static volatile Vec3 camOffset, camTarget = new Vec3(0, 1.25, 0);

	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static Vec3 home;
	private static final List<Mob> MOBS = new ArrayList<>();

	/** One filmed clip: frames {@code prefix_NNN.png} for {@code length} ticks from {@code start}. */
	private record Seg(String name, int length, Consumer<Integer> each) {
	}

	private static final List<Seg> SEGS = new ArrayList<>();
	private static int segStart = 200, segIndex, frame;

	private FrankenstrikeHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("frankenstrike.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(FrankenstrikeHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(360_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "frankenstrike-watchdog");
		wd.setDaemon(true);
		wd.start();
		build();
	}

	private static void server(Consumer<ServerPlayer> fn) {
		MinecraftServer srv = Minecraft.getInstance().getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static FrankenstrikeKit kit() {
		return (FrankenstrikeKit) Aliens.kit(FrankenstrikeKit.ID);
	}

	private static void clearMobs() {
		for (Mob m : MOBS) {
			m.discard();
		}
		MOBS.clear();
	}

	private static Mob zombie(ServerPlayer p, double dx, double dz) {
		Mob m = EntityType.ZOMBIE.create(p.serverLevel());
		m.moveTo(home.x + dx, home.y, home.z + dz, 180f, 0f);
		m.setNoAi(true);
		m.setPersistenceRequired();
		m.getAttribute(Attributes.MAX_HEALTH).setBaseValue(120); // (under 150: not a "boss", so Grab & Shock lifts it)
		m.setHealth(120);
		m.setItemSlot(EquipmentSlot.HEAD, new ItemStack(Items.CARVED_PUMPKIN));
		p.serverLevel().addFreshEntity(m);
		MOBS.add(m);
		return m;
	}

	private static void face(ServerPlayer p, Vec3 at) {
		Vec3 d = at.subtract(p.getEyePosition());
		float yaw = (float) (Math.atan2(-d.x, d.z) * 180 / Math.PI);
		float pitch = (float) (-Math.atan2(d.y, Math.sqrt(d.x * d.x + d.z * d.z)) * 180 / Math.PI);
		p.connection.teleport(p.getX(), p.getY(), p.getZ(), yaw, pitch);
		p.setYRot(yaw);
		p.setXRot(pitch);
		p.setYHeadRot(yaw);
	}

	private static Vec3 chest(LivingEntity e) {
		return e.position().add(0, e.getBbHeight() * 0.6, 0);
	}

	private static void reset(ServerPlayer p, int charge) {
		clearMobs();
		p.connection.teleport(home.x, home.y, home.z, 0f, 0f);
		p.setYRot(0f);
		p.setYHeadRot(0f);
		p.setHealth(p.getMaxHealth());
		for (var m : kit().moves()) {
			kit().cooldown(p, m.id(), 0);
		}
		kit().fill(p, charge);
	}

	private static void cam(double x, double y, double z, double tx, double ty, double tz) {
		camOffset = new Vec3(x, y, z);
		camTarget = new Vec3(tx, ty, tz);
	}

	/** The side-on fight camera: from his front-right, looking a little ahead of him. */
	private static void fightCam() {
		cam(-4.6, 2.2, 1.6, 0, 1.2, 2.4);
	}

	private static void build() {
		// the turntable: one full orbit
		SEGS.add(new Seg("turntable", 140, i -> {
			if (i == 0) {
				server(p -> reset(p, 60));
			}
			double a = Math.PI * 2 * i / 140.0;
			cam(Math.sin(a) * 2.8, 1.6, Math.cos(a) * 2.8, 0, 1.3, 0);
		}));
		// the bolts on his back, the arc crackling between them (full Charge: always on)
		SEGS.add(new Seg("bolts", 60, i -> {
			if (i == 0) {
				server(p -> reset(p, 100));
			}
			double a = Math.PI * (0.75 + 0.5 * i / 60.0);
			cam(Math.sin(a) * 1.9, 2.6, Math.cos(a) * 1.9, 0, 2.1, 0);
		}));
		SEGS.add(new Seg("shock_punch", 45, i -> {
			if (i == 0) {
				fightCam();
				server(p -> {
					reset(p, 20);
					zombie(p, 0, 2.6);
					zombie(p, 2.6, 5.0);
				});
			}
			if (i == 12) {
				server(p -> {
					face(p, chest(MOBS.get(0)));
					System.out.println("[HARNESS] punch=" + kit().shockPunch(p) + " charge=" + FrankenstrikeKit.charge(p));
				});
			}
		}));
		SEGS.add(new Seg("lightning_bolt", 45, i -> {
			if (i == 0) {
				cam(-3.6, 2.4, -1.4, 0, 1.4, 4.0);
				server(p -> {
					reset(p, 40);
					zombie(p, 0.5, 10);
				});
			}
			if (i == 10) {
				server(p -> {
					face(p, chest(MOBS.get(0)));
					System.out.println("[HARNESS] bolt=" + kit().lightningBolt(p) + " charge=" + FrankenstrikeKit.charge(p));
				});
			}
		}));
		SEGS.add(new Seg("conductor_slam", 55, i -> {
			if (i == 0) {
				cam(-5.0, 3.2, 3.0, 0, 0.8, 0.8);
				server(p -> {
					reset(p, 40);
					zombie(p, 0, 3.0);
					zombie(p, 3.0, 1.0);
					zombie(p, -2.5, -2.0);
				});
			}
			if (i == 10) {
				server(p -> System.out.println("[HARNESS] slam=" + kit().conductorSlam(p)));
			}
		}));
		SEGS.add(new Seg("electro_leap", 70, i -> {
			if (i == 0) {
				cam(-6.5, 2.6, 2.0, 0, 1.2, 1.0);
				server(p -> {
					reset(p, 40);
					zombie(p, 0, 9.5);
					zombie(p, 1.8, 10.5);
				});
			}
			if (i == 10) {
				server(p -> {
					face(p, home.add(0, 2.0, 6));
					System.out.println("[HARNESS] leap=" + kit().electroLeap(p));
				});
			}
		}));
		SEGS.add(new Seg("overcharge_storm", 85, i -> {
			if (i == 0) {
				cam(-8.0, 5.0, -3.0, 0, 1.2, 2.0);
				server(p -> {
					reset(p, 100);
					zombie(p, 0, 5);
					zombie(p, 4, 3);
					zombie(p, -4, 4);
					zombie(p, 3, -3);
				});
			}
			if (i == 10) {
				server(p -> System.out.println("[HARNESS] storm=" + kit().overchargeStorm(p)));
			}
		}));
		SEGS.add(new Seg("grab_and_shock", 70, i -> {
			if (i == 0) {
				fightCam();
				server(p -> {
					reset(p, 20);
					zombie(p, 0, 2.4);
				});
			}
			if (i == 8) {
				server(p -> {
					face(p, chest(MOBS.get(0)));
					System.out.println("[HARNESS] grab=" + kit().grab(p));
				});
			}
		}));
		SEGS.add(new Seg("recharge", 75, i -> {
			if (i == 0) {
				cam(-3.4, 1.9, 2.6, 0, 1.4, 0);
				server(p -> reset(p, 0));
			}
			if (i == 8) {
				server(p -> System.out.println("[HARNESS] quick=" + kit().quickRecharge(p) + " charge=" + FrankenstrikeKit.charge(p)));
			}
			if (i == 22) {
				server(p -> System.out.println("[HARNESS] recharge=" + kit().startRecharge(p)));
			}
			if (i == 66) {
				server(p -> {
					kit().endRecharge(p);
					System.out.println("[HARNESS] recharge end charge=" + FrankenstrikeKit.charge(p));
				});
			}
		}));
		SEGS.add(new Seg("lightning_rod", 50, i -> {
			if (i == 0) {
				cam(-4.6, 2.4, 3.0, 0, 1.4, 0);
				server(p -> {
					reset(p, 0);
					zombie(p, 1.8, 1.2);
				});
			}
			if (i == 10) {
				server(p -> System.out.println("[HARNESS] rod=" + kit().lightningRod(p) + " charge=" + FrankenstrikeKit.charge(p)));
			}
		}));
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(FrankenstrikeHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 60) {
			mc.getWindow().setWindowed(854, 480);
		}
		if (t == 80) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.bobView().set(false);
			server(p -> {
				p.serverLevel().setDayTime(6000L);
				p.serverLevel().getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				p.serverLevel().getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.serverLevel().getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				p.setGameMode(GameType.SURVIVAL);
				Omnitrix.grant(p);
				Omnitrix.noGlitch(p);
				Aliens.discover(p, FrankenstrikeKit.ID, true);
			});
		}
		if (t == 100) {
			server(p -> Omnitrix.activate(p));
		}
		if (t == 115) {
			server(p -> Omnitrix.select(p, FrankenstrikeKit.ID));
		}
		if (t == 120) {
			server(p -> {
				home = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
				p.connection.teleport(home.x, home.y, home.z, 0f, 0f);
				Omnitrix.select(p, FrankenstrikeKit.ID);
				System.out.println("[HARNESS] transform=" + Omnitrix.transform(p));
			});
		}
		if (t == 140) {
			cam(0, 1.5, 3.4, 0, 1.3, 0);
		}
		if (t >= segStart && segIndex < SEGS.size()) {
			Seg s = SEGS.get(segIndex);
			int i = t - segStart;
			s.each.accept(i);
			if (i >= 2) { // (the reset's teleport lands first)
				Screenshot.grab(mc.gameDirectory, String.format("fs_%s_%03d.png", s.name, frame++), mc.getMainRenderTarget(), m -> {
				});
			}
			if (i + 1 >= s.length) {
				segIndex++;
				segStart = t + 1;
				frame = 0;
			}
		}
		if (segIndex >= SEGS.size() && t == segStart + 2) {
			camOffset = null;
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.hideGui = false;
			server(p -> reset(p, 64));
		}
		if (segIndex >= SEGS.size() && t == segStart + 25) {
			Screenshot.grab(mc.gameDirectory, "fs_firstperson_hud.png", mc.getMainRenderTarget(), m -> {
			});
		}
		if (segIndex >= SEGS.size() && t == segStart + 30) {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("FS_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.EASY, true,
				new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value()
						.createWorldDimensions(),
				mc.screen);
	}
}
