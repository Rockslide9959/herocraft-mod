package com.projecthero.mod.client;

import java.io.File;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;
import java.util.function.Consumer;

import net.fabricmc.api.ClientModInitializer;
import net.fabricmc.fabric.api.client.event.lifecycle.v1.ClientTickEvents;
import net.minecraft.client.CameraType;
import net.minecraft.client.Minecraft;
import net.minecraft.client.Screenshot;
import net.minecraft.client.gui.screens.TitleScreen;
import net.minecraft.core.BlockPos;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.Difficulty;
import net.minecraft.world.level.GameRules;
import net.minecraft.world.level.GameType;
import net.minecraft.world.level.LevelSettings;
import net.minecraft.world.level.WorldDataConfiguration;
import net.minecraft.world.level.block.Blocks;
import net.minecraft.world.level.levelgen.WorldOptions;
import net.minecraft.world.level.levelgen.presets.WorldPresets;

import org.lwjgl.glfw.GLFW;

import com.projecthero.mod.titanshifter.TitanCharges;
import com.projecthero.mod.titanshifter.TitanShifter;
import com.projecthero.mod.titanshifter.TitanType;

/**
 * TEMP-HARNESS (not shipped, v0.17.5 Titan charges): HUD stills with full / partial / empty charges for the Attack and the Female
 * Titan, and a film of shifting twice in a row on two charges. Enabled by run/titancharges.flag = "<out dir>".
 */
public final class TitanChargesHarness implements ClientModInitializer {
	private static boolean createdWorld;
	private static int waitTicks, inWorldTicks;
	private static BlockPos base;
	private static File out;
	private static final java.util.Map<Integer, List<Runnable>> AT = new java.util.HashMap<>();
	private static int film0 = -1, filmEnd = -1;
	private static String filmDir = "film";

	@Override
	public void onInitializeClient() {
		Path flag = Path.of("titancharges.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			out = new File(Files.readString(flag).trim());
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(TitanChargesHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(600_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "titancharges-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	private static void server(Minecraft mc, Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		new File(out, "screenshots/" + name + ".png").getParentFile().mkdirs();
		Screenshot.grab(out, name + ".png", mc.getMainRenderTarget(), m -> {
		});
	}

	private static void at(int t, Runnable r) {
		AT.computeIfAbsent(t, k -> new ArrayList<>()).add(r);
	}

	private static void film(int from, int to, String dir) {
		at(from, () -> {
			film0 = from;
			filmEnd = to;
			filmDir = dir;
		});
	}

	private static void type(ServerPlayer p, TitanType type) {
		var s = TitanShifter.state(p).copy();
		s.typeId = type.id;
		TitanCharges.fill(s);
		p.setAttached(com.projecthero.mod.attachment.ModAttachments.TITAN_SHIFTER_STATE, s);
	}

	private static void charges(Minecraft mc, int n, int secondsLeft) {
		server(mc, p -> {
			TitanShifter.setCharges(p, n);
			TitanShifter.setChargeTicksLeft(p, secondsLeft * 20);
			System.out.println("[HARNESS] charges " + TitanShifter.charges(p) + " progress " + TitanShifter.state(p).chargeProgress);
		});
	}

	private static void stage(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		level.setDayTime(6000L);
		level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
		p.server.setDifficulty(Difficulty.PEACEFUL, true);
		p.setGameMode(GameType.SURVIVAL);
		base = p.blockPosition();
		for (int x = -14; x <= 14; x++) {
			for (int z = -14; z <= 14; z++) {
				level.setBlock(base.offset(x, -1, z), (x + z) % 2 == 0 ? Blocks.SMOOTH_STONE.defaultBlockState() : Blocks.POLISHED_ANDESITE.defaultBlockState(), 3);
			}
		}
		p.getInventory().clearContent();
		System.out.println("[HARNESS] grant: " + TitanShifter.grant(p));
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, 10f);
	}

	private static void plan() {
		Minecraft mc = Minecraft.getInstance();
		int t = 0;
		at(t, () -> {
			mc.options.hideGui = false;
			mc.options.setCameraType(CameraType.FIRST_PERSON);
		});
		// Attack Titan: full / partial / empty
		at(t + 6, () -> shot(mc, "attack_full"));
		at(t + 8, () -> charges(mc, 2, 35));
		at(t + 16, () -> shot(mc, "attack_partial"));
		at(t + 18, () -> charges(mc, 0, 48));
		at(t + 26, () -> shot(mc, "attack_empty"));
		// Female Titan: full / partial / empty
		at(t + 28, () -> server(mc, p -> type(p, TitanType.FEMALE_TITAN)));
		at(t + 36, () -> shot(mc, "female_full"));
		at(t + 38, () -> charges(mc, 2, 21));
		at(t + 46, () -> shot(mc, "female_partial"));
		at(t + 48, () -> charges(mc, 0, 57));
		at(t + 56, () -> shot(mc, "female_empty"));
		t += 60;
		// back to the Attack Titan with all three: shift, change back, shift again at once
		at(t, () -> server(mc, p -> type(p, TitanType.GENERIC_TITAN)));
		at(t + 2, () -> mc.options.setCameraType(CameraType.THIRD_PERSON_BACK));
		film(t + 6, t + 280, "film_two_shifts");
		at(t + 12, () -> server(mc, TitanShifter::requestToggle)); // H: 3 -> 2
		at(t + 105, () -> shot(mc, "attack_titan_form_hud"));
		at(t + 110, () -> server(mc, TitanShifter::requestToggle)); // H: change back
		at(t + 200, () -> server(mc, TitanShifter::requestToggle)); // H again at once: 2 -> 1
		at(t + 275, () -> shot(mc, "attack_second_shift_hud"));
		t += 284;
		at(t, () -> {
			System.out.println("[HARNESS] done");
			Runtime.getRuntime().halt(0);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(TitanChargesHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 20) {
			GLFW.glfwSetWindowSize(mc.getWindow().getWindow(), 854, 480);
		}
		if (t == 40) {
			mc.options.fov().set(70);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			mc.getToasts().clear();
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			server(mc, TitanChargesHarness::stage);
		}
		if (t < 140 || base == null) {
			return;
		}
		int s = t - 140;
		if (s == 0) {
			plan();
		}
		if (s % 10 == 0) {
			mc.gui.getChat().clearMessages(false);
		}
		List<Runnable> todo = AT.remove(s);
		if (todo != null) {
			todo.forEach(Runnable::run);
		}
		if (s >= film0 && s < filmEnd) {
			shot(mc, String.format(filmDir + "/f%03d", s - film0));
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(GameRules.RULE_DOMOBSPAWNING).set(false, null);
		LevelSettings settings = new LevelSettings("TITANCHG_" + System.currentTimeMillis() % 100000, GameType.SURVIVAL, false, Difficulty.PEACEFUL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
}
