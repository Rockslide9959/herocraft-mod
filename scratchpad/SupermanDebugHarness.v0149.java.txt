package com.projecthero.mod.client;

import java.util.HashMap;
import java.util.Map;

import com.projecthero.mod.kryptonian.Kryptonian;
import com.projecthero.mod.kryptonian.KryptonianFlight;
import com.projecthero.mod.kryptonian.SupermanSuit;

import net.fabricmc.fabric.api.client.event.lifecycle.v1.ClientTickEvents;

import net.minecraft.client.CameraType;
import net.minecraft.client.Minecraft;
import net.minecraft.client.Screenshot;
import net.minecraft.client.gui.screens.TitleScreen;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.Difficulty;
import net.minecraft.world.entity.EquipmentSlot;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.GameRules;
import net.minecraft.world.level.GameType;
import net.minecraft.world.level.LevelSettings;
import net.minecraft.world.level.WorldDataConfiguration;
import net.minecraft.world.level.levelgen.WorldOptions;
import net.minecraft.world.level.levelgen.presets.WorldPresets;
import net.minecraft.world.phys.Vec3;

/** TEMP-HARNESS (not shipped): v0.14.9 Superman Suit + cape screenshot tour. Env PROJECTHERO_SUPERMAN_DEBUG=1. */
public final class SupermanDebugHarness {
	public static Vec3 camOffset;
	public static double camLookY = 1.1;
	private static boolean createdWorld;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static boolean sprint;
	private static final Map<Integer, Runnable> AT = new HashMap<>();

	private SupermanDebugHarness() {
	}

	public static void init() {
		if (!"1".equals(System.getenv("PROJECTHERO_SUPERMAN_DEBUG"))) {
			return;
		}
		System.out.println("[SM-HARNESS] armed");
		ClientTickEvents.END_CLIENT_TICK.register(SupermanDebugHarness::tick);
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					fn.accept(srv.getPlayerList().getPlayers().get(0));
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		Screenshot.grab(mc.gameDirectory, "sm_" + name + ".png", mc.getMainRenderTarget(), x -> {
		});
		if (mc.player != null) {
			System.out.println("[SM-HARNESS] " + name + " v=" + mc.player.getDeltaMovement() + " flying=" + Kryptonian.isFlying(mc.player)
					+ " sprint=" + mc.player.isSprinting() + " lean=" + FlightPoseHelper.lean(mc.player, 0f)
					+ " cape=" + SupermanSuit.wearsCape(mc.player));
		}
	}

	private static void at(int t, Runnable r) {
		AT.put(t, r);
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			if (!createdWorld && mc.screen != null && !(mc.screen instanceof TitleScreen)
					&& mc.screen.getClass().getSimpleName().contains("Onboarding")) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
				return;
			}
			if (!createdWorld && mc.screen instanceof TitleScreen) {
				createdWorld = true;
				LevelSettings settings = new LevelSettings("SmDebug" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
						Difficulty.PEACEFUL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
				mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, WorldOptions.defaultWithRandomSeed(),
						reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
								.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
						mc.screen);
			}
			return;
		}
		if (sprint && mc.player != null) {
			mc.player.setSprinting(true);
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.hideGui = true;
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				p.setGameMode(GameType.SURVIVAL);
				Kryptonian.grant(p);
				p.setItemSlot(EquipmentSlot.HEAD, new ItemStack(SupermanSuit.HELMET));
				p.setItemSlot(EquipmentSlot.CHEST, new ItemStack(SupermanSuit.CHESTPLATE));
				p.setItemSlot(EquipmentSlot.LEGS, new ItemStack(SupermanSuit.LEGGINGS));
				p.setItemSlot(EquipmentSlot.FEET, new ItemStack(SupermanSuit.BOOTS));
				p.setYRot(0);
				p.setYHeadRot(0);
				p.setYBodyRot(0);
				p.setXRot(0);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	private static void script(Minecraft mc) {
		Vec3 front = new Vec3(1.2, 1.3, 3.2);
		Vec3 back = new Vec3(-1.2, 1.5, -3.2);
		Vec3 side = new Vec3(3.6, 1.1, 0.0);
		at(10, () -> camOffset = front);
		at(30, () -> shot(mc, "a01_front"));
		at(32, () -> camOffset = back);
		at(36, () -> shot(mc, "a02_back"));
		at(38, () -> camOffset = side);
		at(42, () -> shot(mc, "a03_side"));
		at(44, () -> camOffset = new Vec3(-0.4, 1.2, -2.2));
		at(48, () -> shot(mc, "a04_back_close"));
		// walking
		at(50, () -> {
			camOffset = side;
			mc.options.keyUp.setDown(true);
		});
		at(62, () -> shot(mc, "a05_walk_side"));
		at(64, () -> mc.options.keyUp.setDown(false));
		// flight: hover, cruise, super-speed
		at(70, () -> server(mc, p -> {
			p.teleportTo(p.getX(), p.getY() + 8, p.getZ());
			KryptonianFlight.start(p);
		}));
		at(85, () -> camOffset = side);
		at(90, () -> shot(mc, "b01_hover_side"));
		at(92, () -> mc.options.keyUp.setDown(true));
		at(112, () -> shot(mc, "b02_cruise_side"));
		at(114, () -> camOffset = new Vec3(-1.5, 1.8, -3.5));
		at(118, () -> shot(mc, "b03_cruise_behind"));
		at(120, () -> {
			camOffset = side;
			sprint = true;
			mc.options.keySprint.setDown(true);
		});
		at(140, () -> shot(mc, "b04_superspeed_side"));
		at(142, () -> camOffset = new Vec3(-1.0, 2.6, -3.5));
		at(146, () -> shot(mc, "b05_superspeed_behind_above"));
		at(148, () -> camOffset = new Vec3(2.5, 0.2, 2.5));
		at(152, () -> shot(mc, "b06_superspeed_front_low"));
		at(154, () -> {
			sprint = false;
			mc.options.keySprint.setDown(false);
			mc.options.keyUp.setDown(false);
			mc.options.keyJump.setDown(true);
			camOffset = side;
		});
		at(170, () -> shot(mc, "b07_climbing_side"));
		at(172, () -> mc.options.keyJump.setDown(false));
		// lose the power: the suit pops off
		at(180, () -> server(mc, p -> Kryptonian.revoke(p)));
		at(190, () -> camOffset = front);
		at(200, () -> shot(mc, "c01_power_lost"));
		at(205, () -> mc.stop());
	}
}
