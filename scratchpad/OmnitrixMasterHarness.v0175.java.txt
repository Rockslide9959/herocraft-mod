	@Override
	public void onInitializeClient() {
		Path flag = Path.of("omnimaster.flag");
		if (!Files.exists(flag)) {
			return;
		}
		try {
			String[] parts = Files.readString(flag).trim().split("\\|");
			out = new File(parts[0]);
			if (parts.length > 1) {
				mode = parts[1];
			}
		} catch (Exception e) {
			out = new File(".");
		}
		out.mkdirs();
		new File(out, "screenshots/film").mkdirs();
		new File(out, "screenshots/film3").mkdirs();
		ClientTickEvents.END_CLIENT_TICK.register(OmnitrixMasterHarness::tick);
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(600_000L);
			} catch (InterruptedException e) {
				return;
			}
			Runtime.getRuntime().halt(3);
		}, "omnimaster-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	/** A tap of R (raise / lower the watch). */

	private static void middle(int t) {

	/** Types the code {@code c} supplies (read when the time comes) from slots showing {@code from}; {@code gap} ticks per notch. */
	private static int type(int t0, java.util.function.Supplier<String> c, int gap, String from) {
		return type(t0, c, gap, () -> from);
	}

	private static int type(int t0, java.util.function.Supplier<String> c, int gap, java.util.function.Supplier<String> fromS) {
		int t = t0;
		for (int i = 0; i < 6; i++) {
			final int slot = i;
			for (int k = 0; k < 9; k++) {
				final int kk = k;
				at(t + k * gap, () -> {
					String s = c.get();
					int d = s.charAt(slot) - '0';
					String from = fromS.get();
					int f = from == null ? 0 : from.charAt(slot) - '0';
					if (kk < Math.floorMod(d - f, 10)) {
						wheel(1.0);
					}
				});
			}
			t += 9 * gap + 4;
			middle(t);
			t += 6;
		}
		return t;
	}

	/** The camera on an invisible client-side stand {@code dist} blocks out at {@code angle} degrees round the player (0 = in front), eye {@code h} up. */
	private static void cam(float angle, float dist, float h) {
		Minecraft mc = Minecraft.getInstance();

		camOffset = new net.minecraft.world.phys.Vec3(-Math.sin(r) * dist, h, Math.cos(r) * dist);
		mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
	}

	private static void stage(ServerPlayer p) {
		ServerLevel level = p.serverLevel();
		level.setDayTime(6000L);
		level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
		level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
		p.server.setDifficulty(Difficulty.PEACEFUL, true);
		p.setGameMode(GameType.SURVIVAL);
		base = p.blockPosition();
		for (int x = -6; x <= 6; x++) {
			for (int z = -6; z <= 6; z++) {
				level.setBlock(base.offset(x, -1, z), (x + z) % 2 == 0 ? Blocks.SMOOTH_STONE.defaultBlockState() : Blocks.STONE.defaultBlockState(), 3);
			}
		}
		p.getInventory().clearContent();
		Omnitrix.grant(p, false);
		p.connection.teleport(base.getX() + 0.5, base.getY(), base.getZ() + 0.5, 0f, 10f);
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (mc.screen instanceof net.minecraft.client.gui.screens.BackupConfirmScreen bs && waitTicks % 20 == 0) {
				for (var c : bs.children()) {
					if (c instanceof net.minecraft.client.gui.components.Button b && b.getMessage().getString().toLowerCase().contains("without")) {
						b.onPress();
						break;
					}
				}
			}
			if (waitTicks % 100 == 0) {
				System.out.println("[HARNESS] waiting, screen=" + (mc.screen == null ? null : mc.screen.getClass().getName()));
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(mode.equals("user") ? () -> mc.createWorldOpenFlows().openWorld("usertest7", () -> {}) : OmnitrixMasterHarness::createWorld);
			}
			return;
		}
		int t = ++inWorldTicks;
		if (t == 40) {
			mc.options.fov().set(70);
			mc.getTutorial().setStep(net.minecraft.client.tutorial.TutorialSteps.NONE);
			mc.getToasts().clear();
			mc.options.bobView().set(false);
			mc.options.guiScale().set(2);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			server(mc, mode.equals("user") ? p -> base = p.blockPosition() : OmnitrixMasterHarness::stage);
		}
		if (t < 170 || base == null) {
			return;
		}
		int s = t - 170;
		if (s == 0) {
			plan();
		}
		if (s % 10 == 0 && mc.screen == null && filmEnd > 0) {
			mc.gui.getChat().clearMessages(false);
		}
		List<Runnable> todo = AT.remove(s);
		if (todo != null) {
			todo.forEach(Runnable::run);
		}
		if (s >= film0 && s < filmEnd) {
			shot(mc, String.format(filmDir + "/f%03d", s - film0));
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		GameRules rules = new GameRules();
		rules.getRule(com.projecthero.mod.calling.SpawnCallings.RULE).set(false, null);
		LevelSettings settings = new LevelSettings("OMNIMASTER_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false, Difficulty.PEACEFUL, true,
				rules, WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, true, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}
