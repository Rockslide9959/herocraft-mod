/**
 * TEMP-HARNESS (not shipped): v0.16.7 Engine quirk screenshots. Fake players hold Engine; the local player is the camera.
 * A stand-in costume layer draws textures/entity/quirk/engine_*.png (the core agent owns the real one). Shots go to
 * run/screenshots/en_&lt;name&gt;.png. Enabled by run/engine.flag.
 */
public final class EngineHarness {
	private static boolean createdWorld;
	private static int waitTicks;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();
	private static Vec3 origin;
	private static ServerPlayer a, b;
	private static double runA = 0;
	private static boolean running;

	private EngineHarness() {
	}

	public static void init() {
		if (!Files.exists(Path.of("engine.flag"))) {
			return;
		}
		ClientTickEvents.END_CLIENT_TICK.register(EngineHarness::tick);
		LivingEntityFeatureRendererRegistrationCallback.EVENT.register((entityType, entityRenderer, helper, context) -> {
			if (entityRenderer instanceof PlayerRenderer pr) {
				helper.register(new Costume(pr));
			}
		});
		Thread wd = new Thread(() -> {
			try {
				Thread.sleep(420_000L);
			} catch (InterruptedException e) {
				return;
			}
			System.out.println("[HARNESS] WATCHDOG timeout");
			Runtime.getRuntime().halt(3);
		}, "en-harness-watchdog");
		wd.setDaemon(true);
		wd.start();
	}

	static final class Costume extends RenderLayer<AbstractClientPlayer, PlayerModel<AbstractClientPlayer>> {
		Costume(RenderLayerParent<AbstractClientPlayer, PlayerModel<AbstractClientPlayer>> parent) {
			super(parent);
		}

		@Override
		public void render(PoseStack pose, MultiBufferSource buffers, int light, AbstractClientPlayer p, float limb, float limbAmount, float partial,
				float age, float yaw, float pitch) {
			if (!Quirks.holds(p, Quirk.ENGINE) || !Quirks.costumeOn(p) || p.isInvisible()) {
				return;
			}
			boolean slim = p.getSkin().model() == net.minecraft.client.resources.PlayerSkin.Model.SLIM;
			ResourceLocation tex = ProjectHeroMod.id("textures/entity/quirk/engine_" + (slim ? "slim" : "wide") + ".png");
			PlayerModel<AbstractClientPlayer> m = getParentModel();
			m.renderToBuffer(pose, buffers.getBuffer(RenderType.entityTranslucent(tex)), light, LivingEntityRenderer.getOverlayCoords(p, 0f));
		}
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					try {
						fn.accept(srv.getPlayerList().getPlayers().get(0));
					} catch (Throwable e) {
						System.out.println("[HARNESS] error: " + e);
						e.printStackTrace();
					}
				}
			});
		}
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (x, y) -> () -> {
			x.run();
			y.run();
		});
	}

	private static void shot(int t, String name) {
		at(t, () -> {
			Minecraft mc = Minecraft.getInstance();
			Screenshot.grab(mc.gameDirectory, "en_" + name + ".png", mc.getMainRenderTarget(), m -> {
			});
			System.out.println("[HARNESS] shot " + name);
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			waitTicks++;
			if (waitTicks % 100 == 0 && !createdWorld && mc.getOverlay() == null && !(mc.screen instanceof TitleScreen)) {
				mc.options.onboardAccessibility = false;
				mc.setScreen(new TitleScreen());
			}
			if (!createdWorld && mc.getOverlay() == null && mc.screen instanceof TitleScreen && waitTicks > 40) {
				createdWorld = true;
				mc.execute(EngineHarness::createWorld);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.hideGui = true;
			mc.options.fov().set(70);
			mc.options.setCameraType(CameraType.FIRST_PERSON);
			mc.options.bobView().set(false);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_WEATHER_CYCLE).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.server.setDifficulty(Difficulty.PEACEFUL, true);
				p.setGameMode(GameType.CREATIVE);
				p.getAbilities().flying = true;
				p.onUpdateAbilities();
				origin = new Vec3(Math.floor(p.getX()) + 0.5, p.getY(), Math.floor(p.getZ()) + 0.5);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
			server(mc, p -> {
				if (running && a != null) {
					runA += 0.45;
					a.teleportTo(p.serverLevel(), origin.x + runA, origin.y, origin.z + 3, -90f, 0f);
					a.setYHeadRot(-90f);
					a.yBodyRot = -90f;
					a.setSprinting(true);
				}
				for (ServerPlayer f : new ServerPlayer[] { a, b }) {
					if (f != null) {
						Quirks.tick(f);
					}
				}
			});
		}
	}

	private static void createWorld() {
		Minecraft mc = Minecraft.getInstance();
		LevelSettings settings = new LevelSettings("EN_" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
				Difficulty.PEACEFUL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
		mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, new WorldOptions(424242L, false, false),
				reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
						.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
				mc.screen);
	}

	private static java.util.UUID skinUuid(boolean slim) {
		for (int i = 0; i < 1000; i++) {
			java.util.UUID id = java.util.UUID.randomUUID();
			boolean s = net.minecraft.client.resources.DefaultPlayerSkin.get(id).model() == net.minecraft.client.resources.PlayerSkin.Model.SLIM;
			if (s == slim) {
				return id;
			}
		}
		return java.util.UUID.randomUUID();
	}

	private static ServerPlayer fake(ServerPlayer near, String name, Vec3 at, java.util.UUID id) {
		ServerLevel level = near.serverLevel();
		GameProfile profile = new GameProfile(id, name);
		CommonListenerCookie cookie = CommonListenerCookie.createInitial(profile, false);
		ServerPlayer f = new ServerPlayer(level.getServer(), level, profile, cookie.clientInformation());
		Connection conn = new Connection(PacketFlow.SERVERBOUND);
		new io.netty.channel.embedded.EmbeddedChannel(conn);
		level.getServer().getPlayerList().placeNewPlayer(conn, f, cookie);
		f.setGameMode(GameType.SURVIVAL);
		f.teleportTo(level, at.x, at.y, at.z, 0f, 0f);
		f.setYHeadRot(0f);
		f.yBodyRot = 0f;
		return f;
	}

	private static void camera(ServerPlayer p, Vec3 eye, Vec3 target) {
		double dx = target.x - eye.x, dz = target.z - eye.z, dy = target.y - eye.y;
		float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
		float pit = (float) -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
		p.teleportTo(p.serverLevel(), eye.x, eye.y - 1.62, eye.z, yaw, pit);
	}

	private static void ready(ServerPlayer f) {
		Quirks.edit(f, n -> n.abilityReadyAt.clear());
	}

	private static void script(Minecraft mc) {
		int t = 10;
		at(t, () -> server(mc, p -> {
			a = fake(p, "Tenya", origin, skinUuid(false));
			b = fake(p, "Tensei", origin.add(1.6, 0, 0), skinUuid(true));
			Quirks.grant(a, Quirk.ENGINE);
			Quirks.grant(b, Quirk.ENGINE);
			camera(p, origin.add(0.8, 1.6, 3.6), origin.add(0.8, 1.0, 0));
		}));
		shot(t + 30, "suit_front");
		at(t + 32, () -> server(mc, p -> camera(p, origin.add(0.8, 1.6, -3.6), origin.add(0.8, 1.0, 0))));
		shot(t + 40, "suit_back");
		at(t + 42, () -> server(mc, p -> camera(p, origin.add(-2.6, 1.3, -1.6), origin.add(0, 0.7, 0))));
		shot(t + 50, "suit_side_back_calves");
		at(t + 52, () -> server(mc, p -> camera(p, origin.add(-0.6, 0.9, -1.4), origin.add(0, 0.45, 0))));
		shot(t + 60, "calf_pipes_close");
		t += 70;
		// Recipro Kick + Afterburner on Tensei (B)
		at(t, () -> server(mc, p -> {
			camera(p, origin.add(4.2, 1.4, 1.2), origin.add(1.6, 0.9, 0));
			ready(b);
			EngineMoves.kick(b);
		}));
		shot(t + 2, "recipro_kick_1");
		shot(t + 6, "recipro_kick_2");
		at(t + 20, () -> server(mc, p -> {
			ready(b);
			EngineMoves.afterburner(b);
		}));
		shot(t + 24, "afterburner_kick");
		at(t + 36, () -> server(mc, p -> {
			ready(b);
			EngineMoves.spin(b);
		}));
		shot(t + 40, "spin_kick");
		t += 50;
		// Rescue Carry: a pig on Tensei's shoulders
		at(t, () -> server(mc, p -> {
			Pig pig = EntityType.PIG.create(p.serverLevel());
			pig.moveTo(b.getX(), b.getY(), b.getZ() + 1.4, 0f, 0f);
			pig.setNoAi(true);
			p.serverLevel().addFreshEntity(pig);
			ready(b);
			b.setXRot(30f);
			EngineMoves.carry(b);
			b.setXRot(0f);
			camera(p, origin.add(3.6, 1.8, 3.2), origin.add(1.6, 1.4, 0));
		}));
		shot(t + 12, "rescue_carry");
		at(t + 14, () -> server(mc, p -> {
			ready(b);
			EngineMoves.carry(b);
		}));
		t += 20;
		// Recipro Burst running (Tenya runs along +X past the camera), then Turbo
		at(t, () -> server(mc, p -> {
			ready(a);
			EngineMoves.burst(a);
			runA = -6;
			running = true;
			camera(p, origin.add(0, 1.4, 7.2), origin.add(0, 0.9, 3));
		}));
		shot(t + 13, "recipro_burst_run");
		shot(t + 15, "recipro_burst_run_b");
		at(t + 30, () -> server(mc, p -> {
			ready(a);
			EngineMoves.turbo(a);
			runA = -6;
		}));
		shot(t + 42, "recipro_turbo_run");
		at(t + 44, () -> server(mc, p -> {
			running = false;
			a.teleportTo(p.serverLevel(), origin.x, origin.y, origin.z + 3, 0f, 0f);
			ready(a);
			EngineMoves.coolDown(a);
			camera(p, origin.add(1.5, 1.5, 6.0), origin.add(0, 0.9, 3));
		}));
		shot(t + 52, "cool_down_steam");
		at(t + 60, () -> {
			System.out.println("[HARNESS] DONE");
			Runtime.getRuntime().halt(0);
		});
	}
}
