/** TEMP-HARNESS (not shipped): v0.14.3 Green Lantern tour. Shots land in run/screenshots/gl_*.png; the client quits. */
public final class GreenLanternDebugHarness {
	public static Vec3 camOffset;
	public static Vec3 camTarget = new Vec3(0, 1.1, 0);
	private static boolean createdWorld;
	private static int inWorldTicks;
	private static int t0 = -1;
	private static final Map<Integer, Runnable> AT = new HashMap<>();

	private GreenLanternDebugHarness() {
	}

	public static void init() {
		ClientTickEvents.END_CLIENT_TICK.register(GreenLanternDebugHarness::tick);
	}

	private static void server(Minecraft mc, java.util.function.Consumer<ServerPlayer> fn) {
		MinecraftServer srv = mc.getSingleplayerServer();
		if (srv != null) {
			srv.execute(() -> {
				if (!srv.getPlayerList().getPlayers().isEmpty()) {
					fn.accept(srv.getPlayerList().getPlayers().get(0));
				}
			});
		}
	}

	private static void shot(Minecraft mc, String name) {
		Screenshot.grab(mc.gameDirectory, "gl_" + name + ".png", mc.getMainRenderTarget(), x -> {
		});
	}

	private static void at(int t, Runnable r) {
		AT.merge(t, r, (a, b) -> () -> {
			a.run();
			b.run();
		});
	}

	private static void tick(Minecraft mc) {
		mc.options.pauseOnLostFocus = false;
		if (mc.level == null || mc.player == null) {
			if (!createdWorld && mc.screen instanceof TitleScreen) {
				createdWorld = true;
				LevelSettings settings = new LevelSettings("GlDebug" + System.currentTimeMillis() % 100000, GameType.CREATIVE, false,
						Difficulty.NORMAL, true, new GameRules(), WorldDataConfiguration.DEFAULT);
				mc.createWorldOpenFlows().createFreshLevel(settings.levelName(), settings, WorldOptions.defaultWithRandomSeed(),
						reg -> reg.registryOrThrow(net.minecraft.core.registries.Registries.WORLD_PRESET)
								.getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions(),
						mc.screen);
			}
			return;
		}
		inWorldTicks++;
		if (inWorldTicks == 100) {
			mc.options.setCameraType(CameraType.THIRD_PERSON_BACK);
			mc.options.hideGui = false;
			org.lwjgl.glfw.GLFW.glfwSetWindowSize(mc.getWindow().getWindow(), 1280, 720);
			server(mc, p -> {
				ServerLevel level = p.serverLevel();
				level.setDayTime(6000L);
				level.getGameRules().getRule(GameRules.RULE_DAYLIGHT).set(false, p.server);
				level.getGameRules().getRule(GameRules.RULE_DOMOBSPAWNING).set(false, p.server);
				p.setGameMode(GameType.SURVIVAL);
				face(p, 0f);
			});
			t0 = inWorldTicks;
			script(mc);
		}
		if (t0 >= 0) {
			Runnable r = AT.remove(inWorldTicks - t0);
			if (r != null) {
				r.run();
			}
		}
	}

	private static void face(ServerPlayer p, float pitch) {
		p.setYRot(0);
		p.setYHeadRot(0);
		p.setYBodyRot(0);
		p.setXRot(pitch);
		p.connection.teleport(p.getX(), p.getY(), p.getZ(), 0f, pitch);
	}

	private static void charge(ServerPlayer p) {
		GreenLanternState s = GreenLantern.state(p).copy();
		s.ringCharge = GreenLanternConfig.MAX_RING_CHARGE * 0.87f;
		s.abilityReadyAt.clear();
		GreenLantern.save(p, s);
	}

	private static void husks(ServerPlayer p, double z) {
		for (int i = -1; i <= 1; i++) {
			Husk h = EntityType.HUSK.create(p.serverLevel());
			h.moveTo(p.getX() + i * 2.5, p.getY(), p.getZ() + z, 180f, 0f);
			h.setNoAi(true);
			h.setHealth(h.getMaxHealth());
			h.setInvulnerable(true);
			p.serverLevel().addFreshEntity(h);
		}
	}

	private static void press(ServerPlayer p, int slot, boolean shift, boolean down) {
		p.setShiftKeyDown(shift);
		GreenLanternAbilityManager.handle(p, AbilitySlot.byNumber(slot), down);
		p.setShiftKeyDown(false);
	}
